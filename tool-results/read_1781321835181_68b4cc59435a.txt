   141→async def get_or_create_user(user_id: int, username: str = "", first_name: str = "",
   142→                              last_name: str = "", language_code: str = "ru") -> Dict[str, Any]:
   143→    """Get or create a user record. Returns user dict.
   144→    
   145→    Handles race condition where two concurrent requests try to INSERT
   146→    the same user_id — uses INSERT OR IGNORE + SELECT instead of
   147→    INSERT + catch IntegrityError.
   148→    """
   149→    now = time.time()
   150→    async with aiosqlite.connect(DB_PATH) as db:
   151→        db.row_factory = aiosqlite.Row
   152→        async with db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) as cursor:
   153→            row = await cursor.fetchone()
   154→
   155→        if row is None:
   156→            try:
   157→                await db.execute(
   158→                    """INSERT INTO users (user_id, username, first_name, last_name, language_code,
   159→                       first_seen, last_seen, message_count)
   160→                       VALUES (?, ?, ?, ?, ?, ?, ?, 1)""",
   161→                    (user_id, username, first_name, last_name, language_code, now, now),
   162→                )
   163→                await db.commit()
   164→                return {
   165→                    "user_id": user_id, "username": username, "first_name": first_name,
   166→                    "last_name": last_name, "language_code": language_code,
   167→                    "is_blocked": 0, "is_admin": 0, "first_seen": now,
   168→                    "last_seen": now, "message_count": 1, "chat_mode": "normal",
   169→                }
   170→            except aiosqlite.IntegrityError:
   171→                # Race condition: another request already inserted this user
   172→                # Re-fetch the existing row
   173→                async with db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) as cursor:
   174→                    row = await cursor.fetchone()
   175→                if row:
   176→                    # Still update last_seen
   177→                    await db.execute(
   178→                        """UPDATE users SET username=?, first_name=?, last_name=?,
   179→                           last_seen=?, message_count=message_count+1 WHERE user_id=?""",
   180→                        (username, first_name, last_name, now, user_id),
   181→                    )
   182→                    await db.commit()
   183→                    return dict(row)
   184→                # Shouldn't happen, but fallback
   185→                return {
   186→                    "user_id": user_id, "username": username, "first_name": first_name,
   187→                    "last_name": last_name, "language_code": language_code,
   188→                    "is_blocked": 0, "is_admin": 0, "first_seen": now,
   189→                    "last_seen": now, "message_count": 1, "chat_mode": "normal",
   190→                }
   191→        else:
   192→            await db.execute(
   193→                """UPDATE users SET username=?, first_name=?, last_name=?,
   194→                   last_seen=?, message_count=message_count+1 WHERE user_id=?""",
   195→                (username, first_name, last_name, now, user_id),
   196→            )
   197→            await db.commit()
   198→            return dict(row)

   294→async def add_news_item(source: str, title: str, url: str, summary: str = "",
   295→                         published: float = 0, category: str = "auto", lang: str = "ru",
   296→                         image_urls: list = None) -> bool:
   297→    """Add a news item. Returns True if new, False if duplicate.
   298→    
   299→    image_urls: list of image URL strings extracted from the RSS feed.
   300→    """
   301→    now = time.time()
   302→    image_urls_json = json.dumps(image_urls or [], ensure_ascii=False)
   303→    try:
   304→        async with aiosqlite.connect(DB_PATH) as db:
   305→            await db.execute(
   306→                """INSERT INTO news_items (source, title, url, summary, published, fetched_at, category, lang, image_urls)
   307→                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
   308→                (source, title, url, summary, published, now, category, lang, image_urls_json),
   309→            )
   310→            await db.commit()
   311→            return True
   312→    except aiosqlite.IntegrityError:
   313→        # If already exists, update image_urls if we have new ones
   314→        if image_urls:
   315→            try:
   316→                async with aiosqlite.connect(DB_PATH) as db:
   317→                    await db.execute(
   318→                        "UPDATE news_items SET image_urls = ? WHERE url = ? AND image_urls = '[]'",
   319→                        (image_urls_json, url),
   320→                    )
   321→                    await db.commit()
   322→            except Exception:
   323→                pass
   324→        return False

   327→async def get_unposted_news(limit: int = 10, category: str = "") -> List[Dict[str, Any]]:
   328→    """Get unposted news items, ordered by publish date.
   329→    
   330→    Parses image_urls JSON field back into a Python list.
   331→    """
   332→    async with aiosqlite.connect(DB_PATH) as db:
   333→        db.row_factory = aiosqlite.Row
   334→        if category:
   335→            async with db.execute(
   336→                """SELECT * FROM news_items WHERE is_posted = 0 AND category = ?
   337→                   ORDER BY published DESC LIMIT ?""",
   338→                (category, limit),
   339→            ) as cursor:
   340→                rows = await cursor.fetchall()
   341→        else:
   342→            async with db.execute(
   343→                """SELECT * FROM news_items WHERE is_posted = 0
   344→                   ORDER BY published DESC LIMIT ?""",
   345→                (limit,),
   346→            ) as cursor:
   347→                rows = await cursor.fetchall()
   348→        result = []
   349→        for r in rows:
   350→            item = dict(r)
   351→            # Parse image_urls from JSON string to list
   352→            if "image_urls" in item and isinstance(item["image_urls"], str):
   353→                try:
   354→                    item["image_urls"] = json.loads(item["image_urls"])
   355→                except (json.JSONDecodeError, TypeError):
   356→                    item["image_urls"] = []
   357→            elif "image_urls" not in item:
   358→                item["image_urls"] = []
   359→            result.append(item)
   360→        return result

   513→# ── Rate Limiting ────────────────────────────────────────────────────────────────
   514→
   515→_user_message_times: Dict[int, List[float]] = {}
   516→RATE_LIMIT_MESSAGES = 10  # Max messages per minute
   517→RATE_LIMIT_WINDOW = 60    # 1 minute window
   518→
   519→
   520→def check_rate_limit(user_id: int) -> bool:
   521→    """Check if user is within rate limits. Returns True if allowed."""
   522→    now = time.time()
   523→    if user_id not in _user_message_times:
   524→        _user_message_times[user_id] = [now]
   525→        return True
   526→
   527→    # Clean old timestamps
   528→    times = _user_message_times[user_id]
   529→    times = [t for t in times if now - t < RATE_LIMIT_WINDOW]
   530→    times.append(now)
   531→    _user_message_times[user_id] = times
   532→
   533→    return len(times) <= RATE_LIMIT_MESSAGES

   557→async def is_duplicate_post(title: str, content: str = "", hours: int = 48,
   558→                            source_url: str = "") -> bool:
   559→    """Check if a post with similar title or content was recently posted.
   560→
   561→    Checks (in order):
   562→    1. Source URL match (same article from different RSS feeds)
   563→    2. Exact title hash match (same title, possibly different source)
   564→    3. Title prefix match (first 30 chars — catches reworded titles about same topic)
   565→    4. Keyword overlap match (extract key nouns — catches paraphrased titles about same subject)
   566→       Also normalizes car brand names (BMW/Bayerische, LADA/ВАЗ, etc.)
   567→    5. Content hash match (same content, possibly different title)
   568→    6. NORMALIZED CORE MATCH — extract core words (brand + model + event) and compare
   569→       This catches "BMW представила новый X5" vs "Новый BMW X5 2026: первые детали"
   570→
   571→    Args:
   572→        title: News item title to check
   573→        content: Post content to check (optional)
   574→        hours: How many hours back to check (default 48h)
   575→        source_url: Original article URL for URL-based dedup (optional)
   576→
   577→    Returns:
   578→        True if a similar post was found (DUPLICATE), False if unique
   579→    """
   580→    cutoff = time.time() - (hours * 3600)
   581→    title_hash = _make_title_hash(title)
   582→    title_prefix = _normalize_text(title)[:30]
   583→
   584→    async with aiosqlite.connect(DB_PATH) as db:
   585→        # Check 1: Source URL match — same article from different RSS feeds
   586→        # Compares the path component of URLs (ignores www/http/https differences)
   587→        if source_url:
   588→            url_fingerprint = _make_url_fingerprint(source_url)
   589→            if url_fingerprint:
   590→                async with db.execute(
   591→                    "SELECT source_url FROM channel_posts WHERE created_at >= ?",
   592→                    (cutoff,),
   593→                ) as cursor:
   594→                    rows = await cursor.fetchall()
   595→                    for row in rows:
   596→                        existing_url = row[0] if row else ""
   597→                        if existing_url and _make_url_fingerprint(existing_url) == url_fingerprint:
   598→                            return True
   599→
   600→        # Check 2: Exact title hash match
   601→        async with db.execute(
   602→            "SELECT COUNT(*) FROM post_fingerprints WHERE title_hash = ? AND created_at >= ?",
   603→            (title_hash, cutoff),
   604→        ) as cursor:
   605→            row = await cursor.fetchone()
   606→            if row and row[0] > 0:
   607→                return True
   608→
   609→        # Check 3: Title prefix match (catches reworded titles about same topic)
   610→        if len(title_prefix) >= 10:
   611→            async with db.execute(
   612→                "SELECT COUNT(*) FROM post_fingerprints WHERE title_prefix = ? AND created_at >= ?",
   613→                (title_prefix, cutoff),
   614→            ) as cursor:
   615→                row = await cursor.fetchone()
   616→                if row and row[0] > 0:
   617→                    return True
   618→
   619→        # Check 4: Keyword overlap — extract significant words from title and compare
   620→        # Also normalizes car brand names (BMW/Bayerische, LADA/ВАЗ, etc.)
   621→        title_keywords = _extract_title_keywords(title)
   622→        title_keywords_normalized = _normalize_brand_keywords(title_keywords)
   623→        if len(title_keywords_normalized) >= 2:
   624→            async with db.execute(
   625→                "SELECT title_prefix FROM post_fingerprints WHERE created_at >= ?",
   626→                (cutoff,),
   627→            ) as cursor:
   628→                rows = await cursor.fetchall()
   629→                for row in rows:
   630→                    existing_prefix = row[0] if row else ""
   631→                    if not existing_prefix:
   632→                        continue
   633→                    existing_keywords = _extract_title_keywords(existing_prefix)
   634→                    existing_keywords_normalized = _normalize_brand_keywords(existing_keywords)
   635→                    if not existing_keywords_normalized:
   636→                        continue
   637→                    # Calculate keyword overlap ratio (using normalized brands)
   638→                    common = title_keywords_normalized & existing_keywords_normalized
   639→                    if len(common) >= 2:
   640→                        overlap_ratio = len(common) / min(len(title_keywords_normalized), len(existing_keywords_normalized))
   641→                        if overlap_ratio >= 0.50:
   642→                            return True
   643→                    # Also check overlap between raw keywords for better matching
   644→                    raw_common = title_keywords & existing_keywords
   645→                    if len(raw_common) >= 3:
   646→                        raw_overlap_ratio = len(raw_common) / min(len(title_keywords), len(existing_keywords)) if min(len(title_keywords), len(existing_keywords)) > 0 else 0
   647→                        if raw_overlap_ratio >= 0.45:
   648→                            return True
   649→
   650→        # ── Check 6: NORMALIZED CORE MATCH ──
   651→        # Extract core content words (brands, models, numbers, events) and compare
   652→        # This catches cases where titles are completely reworded but about same event
   653→        core_words = _extract_core_words(title)
   654→        if len(core_words) >= 2:
   655→            async with db.execute(
   656→                "SELECT title_prefix FROM post_fingerprints WHERE created_at >= ?",
   657→                (cutoff,),
   658→            ) as cursor:
   659→                rows = await cursor.fetchall()
   660→                for row in rows:
   661→                    existing_prefix = row[0] if row else ""
   662→                    if not existing_prefix:
   663→                        continue
   664→                    existing_core = _extract_core_words(existing_prefix)
   665→                    if len(existing_core) < 2:
   666→                        continue
   667→                    core_common = core_words & existing_core
   668→                    # If 2+ core words match (e.g., "BMW" + "X5" or "Tesla" + "recalls"),
   669→                    # it's almost certainly the same event
   670→                    if len(core_common) >= 2:
   671→                        return True
   672→
   673→        # Check 5: Content hash match (if content provided)
   674→        if content:
   675→            content_hash = _make_content_hash(content)
   676→            async with db.execute(
   677→                "SELECT COUNT(*) FROM post_fingerprints WHERE content_hash = ? AND created_at >= ?",
   678→                (content_hash, cutoff),
   679→            ) as cursor:
   680→                row = await cursor.fetchone()
   681→                if row and row[0] > 0:
   682→                    return True
   683→
   684→    return False

   912→def _make_url_fingerprint(url: str) -> str:
   913→    """Create a fingerprint of a URL for deduplication.
   914→    
   915→    Strips protocol, www prefix, trailing slashes, and query parameters
   916→    so that https://www.example.com/article/123 and http://example.com/article/123/
   917→    are treated as the same URL.
   918→    """
   919→    from urllib.parse import urlparse
   920→    try:
   921→        parsed = urlparse(url.strip())
   922→        # Use path only (ignore protocol, www, query, fragment)
   923→        path = parsed.path.rstrip('/')
   924→        domain = parsed.netloc.lower()
   925→        # Remove www prefix
   926→        if domain.startswith('www.'):
   927→            domain = domain[4:]
   928→        fingerprint_str = f"{domain}{path}"
   929→        if not fingerprint_str or len(fingerprint_str) < 5:
   930→            return ""
   931→        return hashlib.sha256(fingerprint_str.encode('utf-8')).hexdigest()
   932→    except Exception:
   933→        return ""

   982→async def load_topic_registry() -> Dict:
   983→    """Load topic registry from DB into memory. Returns dict of entity_key -> {first_seen, last_posted, post_count, titles}.
   984→
   985→    Only loads topics that are NOT expired (within 72h of last_posted).
   986→    This is called at startup to restore the registry after restart.
   987→    """
   988→    max_age = 48 * 3600  # 48 hours — faster topic cycling
   989→    cutoff = time.time() - max_age
   990→    registry = {}
   991→    try:
   992→        async with aiosqlite.connect(DB_PATH) as db:
   993→            async with db.execute(
   994→                "SELECT entity_key, first_seen, last_posted, post_count, titles FROM topic_registry WHERE last_posted >= ?",
   995→                (cutoff,),
   996→            ) as cursor:
   997→                rows = await cursor.fetchall()
   998→                for row in rows:
   999→                    entity_key, first_seen, last_posted, post_count, titles_json = row
  1000→                    try:
  1001→                        titles = json.loads(titles_json) if titles_json else []
  1002→                    except (json.JSONDecodeError, TypeError):
  1003→                        titles = []
  1004→                    registry[entity_key] = {
  1005→                        "first_seen": first_seen,
  1006→                        "last_posted": last_posted,
  1007→                        "post_count": post_count,
  1008→                        "titles": titles,
  1009→                    }
  1010→    except Exception as e:
  1011→        # Table might not exist yet on first run — that's OK
  1012→        import logging
  1013→        logging.getLogger("asya.database").debug(f"Could not load topic registry: {e}")
  1014→    return registry

  1018→                                  post_count: int, titles: list) -> None:
  1019→    """Save or update a single topic in the DB registry."""
  1020→    if not entity_key:
  1021→        return
  1022→    titles_json = json.dumps(titles[-20:], ensure_ascii=False)  # Keep last 20 titles
  1023→    try:
  1024→        async with aiosqlite.connect(DB_PATH) as db:
  1025→            await db.execute(
  1026→                """INSERT INTO topic_registry (entity_key, first_seen, last_posted, post_count, titles)
  1027→                   VALUES (?, ?, ?, ?, ?)
  1028→                   ON CONFLICT(entity_key) DO UPDATE SET
  1029→                       last_posted = excluded.last_posted,
  1030→                       post_count = excluded.post_count,
  1031→                       titles = excluded.titles""",
  1032→                (entity_key, first_seen, last_posted, post_count, titles_json),
  1033→            )
  1034→            await db.commit()
  1035→    except Exception as e:
  1036→        import logging
  1037→        logging.getLogger("asya.database").debug(f"Could not save topic to registry: {e}")

  1040→async def cleanup_topic_registry(max_age_hours: int = 48) -> int:
  1041→    """Remove expired topics from the DB registry. Returns count of removed rows."""
  1042→    cutoff = time.time() - (max_age_hours * 3600)
  1043→    try:
  1044→        async with aiosqlite.connect(DB_PATH) as db:
  1045→            cursor = await db.execute(
  1046→                "DELETE FROM topic_registry WHERE last_posted < ?", (cutoff,)
  1047→            )
  1048→            await db.commit()
  1049→            return cursor.rowcount
  1050→    except Exception:
  1051→        return 0
