    25→def _compute_fingerprint(title: str) -> str:
    26→    """Compute a simplified fingerprint for dedup.
    27→    Remove articles, lowercase, take first 5 words.
    28→    """
    29→    import re
    30→    # Remove common articles/prepositions
    31→    cleaned = re.sub(r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b', '', title.lower())
    32→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
    33→    words = cleaned.split()
    34→    return ' '.join(words[:5])

    37→def _fingerprint_matches_existing(fingerprint: str) -> bool:
    38→    """Check if a fingerprint matches any recently used fingerprint (edit distance on first 4 words)."""
    39→    fp_words = fingerprint.split()[:4]
    40→    for existing in _recent_fingerprints:
    41→        ex_words = existing.split()[:4]
    42→        # Count matching words
    43→        matches = sum(1 for w in fp_words if w in ex_words)
    44→        if matches >= 3:
    45→            return True
    46→    return False

   274→# ── RSS Fetcher ────────────────────────────────────────────────────────────────
   275→
   276→async def fetch_rss(source: NewsSource) -> List[Dict]:
   277→    """
   278→    Fetch and parse an RSS feed.
   279→    Returns list of news items.
   280→    """
   281→    items = []
   282→    try:
   283→        async with httpx.AsyncClient(
   284→            timeout=20.0,
   285→            follow_redirects=True,
   286→            headers={
   287→                "User-Agent": "Mozilla/5.0 (compatible; AsyaBot/1.0; +https://t.me/asiaexp_bot)",
   288→                "Accept": "application/rss+xml, application/xml, text/xml, application/atom+xml, */*",
   289→            },
   290→        ) as client:
   291→            response = await client.get(source.url)
   292→
   293→            if response.status_code != 200:
   294→                logger.warning(f"RSS {source.name} returned {response.status_code}")
   295→                return items
   296→
   297→            # feedparser can parse from string
   298→            feed = feedparser.parse(response.text)
   299→
   300→            for entry in feed.entries[:20]:  # Limit per source
   301→                title = entry.get("title", "").strip()
   302→                url = entry.get("link", "").strip()
   303→
   304→                if not title or not url:
   305→                    continue
   306→
   307→                # Get summary/description
   308→                summary = ""
   309→                if entry.get("summary"):
   310→                    summary = _clean_html(entry["summary"])
   311→                elif entry.get("description"):
   312→                    summary = _clean_html(entry["description"])
   313→
   314→                # Truncate summary
   315→                if len(summary) > 500:
   316→                    summary = summary[:500] + "..."
   317→
   318→                # Get publish date
   319→                published = 0.0
   320→                if entry.get("published_parsed"):
   321→                    try:
   322→                        published = time.mktime(entry["published_parsed"])
   323→                    except (ValueError, TypeError, OverflowError):
   324→                        published = time.time()
   325→                elif entry.get("updated_parsed"):
   326→                    try:
   327→                        published = time.mktime(entry["updated_parsed"])
   328→                    except (ValueError, TypeError, OverflowError):
   329→                        published = time.time()
   330→                else:
   331→                    published = time.time()
   332→
   333→                # Extract image URLs from RSS entry
   334→                image_urls = _extract_entry_images(entry)
   335→
   336→                items.append({
   337→                    "source": source.name,
   338→                    "title": title,
   339→                    "url": url,
   340→                    "summary": summary,
   341→                    "published": published,
   342→                    "category": source.category,
   343→                    "lang": source.lang,
   344→                    "image_urls": image_urls,
   345→                    "rss_entry": entry,  # Raw feedparser entry for content:encoded image extraction
   346→                })
   347→
   348→            logger.info(f"Fetched {len(items)} items from {source.name}")
   349→
   350→    except httpx.TimeoutException:
   351→        logger.warning(f"RSS timeout: {source.name}")
   352→    except Exception as e:
   353→        logger.error(f"RSS fetch error for {source.name}: {e}")
   354→
   355→    return items

   360→def is_auto_relevant(item: Dict, lang: str = "ru") -> bool:
   361→    """Check if a news item is relevant to the auto topic."""
   362→    text = f"{item.get('title', '')} {item.get('summary', '')}".lower()
   363→
   364→    if lang == "ru" or lang == "":
   365→        keywords = AUTO_KEYWORDS_RU
   366→    elif lang == "en":
   367→        keywords = AUTO_KEYWORDS_EN
   368→    elif lang == "de":
   369→        keywords = AUTO_KEYWORDS_DE
   370→    else:
   371→        keywords = AUTO_KEYWORDS_RU + AUTO_KEYWORDS_EN
   372→
   373→    # Check if any keyword matches
   374→    for kw in keywords:
   375→        if kw.lower() in text:
   376→            return True
   377→
   378→    return False

   508→    Fetch news — WEB SEARCH FIRST, then RSS as fallback/supplement.

   510→    
   511→    Phase 1: Web search for automotive news (broad queries, primary source)
   512→    Phase 2: RSS feeds (fallback/supplement)
   513→    Phase 3: Global RSS sources (additional international coverage)
   514→    Phase 4: Google News RSS batch (broad query-based coverage)
   515→    
   516→    All existing filtering is preserved (auto-relevance, political block, dedup).
   517→    """
   518→    total_new = 0
   519→
   520→    # DO NOT reset fingerprint set every cycle!
   521→    # Previously this caused the same news to be re-added across cycles.
   522→    # The _recent_fingerprints set now persists across fetch cycles within the process.
   523→    # DB-level dedup (is_duplicate_post) provides cross-restart protection.
   524→
   525→    # ── Phase 1: Web Search FIRST (primary source) ──
   526→    logger.info("Phase 1: Web search for automotive news")
   527→    web_search_queries = [
   528→        "automotive industry news today",
   529→        "new car models 2026",
   530→        "electric vehicle updates",

   728→# ── Image extraction from RSS entries ──────────────────────────────────────────
   729→
   730→# Image extensions that are valid for Telegram posts
   731→_VALID_IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".gif"}
   732→# Domains to skip (icons, trackers, tiny thumbnails)
   733→_SKIP_IMAGE_DOMAINS = {
   734→    "feeds.feedburner.com", "feedburner.google.com",
   735→    "pixel.wp.com", "stats.wordpress.com",
   736→    "i0.wp.com", "i1.wp.com", "i2.wp.com",
   737→}
   738→# Minimum image URL length (filter out tiny 1x1 tracking pixels etc.)
   739→_MIN_IMAGE_URL_LEN = 30
   740→
   741→
   742→def _extract_entry_images(entry) -> List[str]:
   743→    """Extract image URLs from a feedparser entry.
   744→    
   745→    Checks multiple locations where RSS/Atom feeds store images:
   746→    - media_content (Media RSS)
   747→    - enclosures
   748→    - media_thumbnail
   749→    - links with image type
   750→    - content/summary HTML <img> tags
   751→    
   752→    Returns list of image URLs (up to 10).
   753→    """
   754→    images = []
   755→    seen = set()
   756→
   757→    def _add_image(url: str):
   758→        """Add image URL if valid and not already seen."""
   759→        if not url or len(url) < _MIN_IMAGE_URL_LEN:
   760→            return
   761→        # Normalize URL
   762→        url = url.strip()
   763→        if url.startswith("//"):
   764→            url = "https:" + url
   765→        # Skip tracking pixels and tiny icons
   766→        from urllib.parse import urlparse
   767→        try:
   768→            parsed = urlparse(url)
   769→            domain = parsed.netloc.lower()
   770→            if any(skip in domain for skip in _SKIP_IMAGE_DOMAINS):
   771→                return
   772→            # Skip very small-looking URLs (often icons/pixels)
   773→            path_lower = parsed.path.lower()
   774→            if any(kw in path_lower for kw in ["icon", "avatar", "logo", "favicon", "pixel", "badge", "button", "banner"]):
   775→                return
   776→        except Exception:
   777→            return
   778→        if url not in seen:
   779→            seen.add(url)
   780→            images.append(url)
   781→
   782→    # 1. media_content (Media RSS extension — most reliable)
   783→    media_content = entry.get("media_content", [])
   784→    for media in media_content:
   785→        url = media.get("url", "")
   786→        media_type = media.get("type", "")
   787→        if url and ("image" in media_type or _has_image_ext(url)):
   788→            _add_image(url)
   789→
   790→    # 2. enclosures
   791→    for enclosure in entry.get("enclosures", []):
   792→        url = enclosure.get("href", "") or enclosure.get("url", "")
   793→        enc_type = enclosure.get("type", "")
   794→        if url and ("image" in enc_type or _has_image_ext(url)):
   795→            _add_image(url)
   796→
   797→    # 3. media_thumbnail
   798→    for thumb in entry.get("media_thumbnail", []):
   799→        url = thumb.get("url", "")
   800→        if url:
   801→            _add_image(url)
   802→
   803→    # 4. links with image rel or type
   804→    for link in entry.get("links", []):
   805→        link_type = link.get("type", "")
   806→        link_rel = link.get("rel", "")
   807→        url = link.get("href", "")
   808→        if url and ("image" in link_type or link_rel == "enclosure"):
   809→            _add_image(url)
   810→
   811→    # 5. Extract <img> tags from content/summary HTML
   812→    for field_name in ["content", "summary", "description"]:
   813→        content_value = entry.get(field_name, [])
   814→        if isinstance(content_value, list):
   815→            # feedparser sometimes returns list of dicts
   816→            for item in content_value:
   817→                html = item.get("value", "") if isinstance(item, dict) else str(item)
   818→                _extract_img_urls(html, _add_image)
   819→        elif isinstance(content_value, str):
   820→            _extract_img_urls(content_value, _add_image)
   821→
   822→    return images[:10]  # Up to 10 candidate images per news item (Telegram mediagroup limit)
   823→
   824→
   825→def _has_image_ext(url: str) -> bool:
   826→    """Check if URL ends with an image extension."""
   827→    from urllib.parse import urlparse
   828→    try:
   829→        path = urlparse(url).path.lower()
   830→        # Remove query-like suffixes
   831→        path = path.split("?")[0].split("#")[0]
   832→        return any(path.endswith(ext) for ext in _VALID_IMAGE_EXTENSIONS)
   833→    except Exception:
   834→        return False
   835→
   836→
   837→def _extract_img_urls(html: str, callback):
   838→    """Extract src attributes from <img> tags in HTML."""
   839→    if not html:
   840→        return
   841→    for match in re.finditer(r'<img[^>]+src=["\']([^"\']+)["\']', html, re.IGNORECASE):
   842→        url = match.group(1)
   843→        if url and len(url) >= _MIN_IMAGE_URL_LEN:
   844→            callback(url)
   845→

   847→# ── Utility ────────────────────────────────────────────────────────────────────
   848→
   849→def _clean_html(text: str) -> str:
   850→    """Remove HTML tags and decode entities."""
   851→    text = re.sub(r'<[^>]+>', '', text)
   852→    text = text.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>')
   853→    text = text.replace('&quot;', '"').replace('&#39;', "'").replace('&nbsp;', ' ')
   854→    text = re.sub(r'\s+', ' ', text).strip()
   855→    return text
