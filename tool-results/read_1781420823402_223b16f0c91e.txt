    40→class PartnerProgram:
    41→    """Single partner program from admitad."""

    43→    def __init__(self, data: Dict):
    44→        self.id = str(data.get("id", ""))

    58→        self.category = data.get("category", "")

    64→    def has_region(self, region: str = DEFAULT_REGION) -> bool:
    65→        """Check if program is available in a region."""
    66→        if not self.allowed_regions:
    67→            return True  # No region info = available everywhere
    68→        region_upper = region.upper()
    69→        # "00" means worldwide
    70→        if "00" in self.allowed_regions:
    71→            return True
    72→        return region_upper in [r.upper() for r in self.allowed_regions]
    73→
    74→    def has_category(self, category: str) -> bool:
    75→        """Check if program belongs to a category."""
    76→        cat_lower = category.lower()
    77→        if cat_lower == self.category.lower():
    78→            return True
    79→        if cat_lower in self.category_name.lower():
    80→            return True
    81→        return False
    82→
    83→    def matches_text(self, text: str) -> bool:
    84→        """Check if text contains keywords related to this program."""
    85→        text_lower = text.lower()
    86→        # Check program name words
    87→        name_words = [w.lower() for w in self.name.split() if len(w) > 3]
    88→        for word in name_words:
    89→            if word in text_lower:
    90→                return True
    91→        # Check category name words
    92→        cat_words = [w.lower() for w in self.category_name.split() if len(w) > 3]
    93→        for word in cat_words:
    94→            if word in text_lower:
    95→                return True
    96→        # Check site_url domain
    97→        if self.site_url:
    98→            domain = urlparse(self.site_url).netloc.replace("www.", "")
    99→            if domain and domain in text_lower:
   100→                return True
   101→        return False
   102→
   103→    def get_search_url(self, query: str) -> str:
   104→        """Get a search URL for this partner, using the goto_link as base.
   105→
   106→        If the goto_link has a ulp parameter (redirect URL), we modify it
   107→        to include the search path. Otherwise, returns the goto_link as-is.
   108→        """
   109→        if not self.goto_link:
   110→            return ""
   111→        if not query:
   112→            return self.goto_link
   113→
   114→        # Try to modify the ulp parameter to include search
   115→        try:
   116→            parsed = urlparse(self.goto_link)
   117→            params = parse_qs(parsed.query)
   118→
   119→            if "ulp" in params and params["ulp"]:
   120→                original_ulp = params["ulp"][0]
   121→                # Build search URL based on site_url patterns
   122→                search_url = self._build_search_url(original_ulp, query)
   123→                if search_url != original_ulp:
   124→                    # Replace ulp parameter
   125→                    new_params = {}
   126→                    for k, v_list in params.items():
   127→                        if k == "ulp":
   128→                            new_params[k] = search_url
   129→                        else:
   130→                            new_params[k] = v_list[0] if len(v_list) == 1 else v_list
   131→
   132→                    # Rebuild URL with new ulp
   133→                    new_query = urlencode(new_params, doseq=True)
   134→                    return urlunparse(parsed._replace(query=new_query))
   135→        except Exception as e:
   136→            logger.debug(f"Error modifying goto_link for search: {e}")
   137→
   138→        # If we can't modify, return as-is
   139→        return self.goto_link

   141→    def _build_search_url(self, original_ulp: str, query: str) -> str:
   142→        """Build a search URL by modifying the original redirect URL."""
   143→        site_url = self.site_url.rstrip("/")
   144→        query_encoded = quote_plus(query)

   149→            "autopiter.ru": f"{site_url}/search?querystr={query_encoded}",
   150→            "autopiter.kz": f"{site_url}/search?querystr={query_encoded}",
   151→            "exist.ru": f"{site_url}/Price/?p={query_encoded}",
   152→            "emex.ru": f"{site_url}/products?search={query_encoded}",
   153→            "autodoc.ru": f"{site_url}/search?keyword={query_encoded}",
   154→            "zzap.ru": f"{site_url}/search/?q={query_encoded}",
   155→            "avtoall.ru": f"{site_url}/search/?q={query_encoded}",
   156→            "aliexpress.ru": f"{site_url}/wholesale?SearchText={query_encoded}",
   157→            "aliexpress.com": f"{site_url}/wholesale?SearchText={query_encoded}",
   158→            "hyperauto.ru": f"{site_url}/search/?q={query_encoded}",
   159→            "euro-diski.ru": f"{site_url}/search/?q={query_encoded}",
   160→            "bs-tyres.ru": f"{site_url}/search/?q={query_encoded}",
   161→            "koleso.ru": f"{site_url}/search/?q={query_encoded}",
   162→            "avtocod.ru": f"{site_url}/search/?q={query_encoded}",
   163→            "petrolplus.ru": f"{site_url}/search/?q={query_encoded}",
   164→            "globaldrive.ru": f"{site_url}/search/?q={query_encoded}",
   165→            "mirdvornikov.ru": f"{site_url}/search/?q={query_encoded}",
   166→            "lukoil-shop.com": f"{site_url}/search/?q={query_encoded}",
   167→        }
   168→
   169→        # Check if site_url matches any pattern
   170→        for domain, pattern in search_patterns.items():
   171→            if domain in self.site_url:
   172→                return quote_plus(pattern)
   173→
   174→        # Generic fallback: just add search parameter
   175→        return original_ulp

   177→    def format_link(self, with_description: bool = True) -> str:
   178→        """Format this partner's link for display.
   179→
   180→        Uses goto_link EXACTLY as-is from the file.
   181→        No subid additions — the link is ready to use!

   183→        if not self.goto_link:

   187→            return f"{self.name} ({self.category_name}): {self.goto_link}"
   188→        return f"{self.name}: {self.goto_link}"

   190→    def format_link_with_search(self, query: str) -> str:
   191→        """Format this partner's link with a search query.
   192→
   193→        Modifies the ulp parameter in goto_link to include search.

   196→        search_url = self.get_search_url(query)

   201→            # Add helpful description based on category
   202→            desc = self._get_category_description()

   204→        return f"{self.name}: {search_url}"

   221→class PartnerManager:
   222→    """Manages all partner programs — loading, matching, posting.
   223→
   224→    v3.0: Downloads admitad_ads.json from remote URL, auto-refreshes.
   225→    Uses goto_link EXACTLY as-is — NO subid additions!
   226→    """
   227→
   228→    def __init__(self):
   229→        self.programs: List[PartnerProgram] = []
   230→        self._loaded = False
   231→        self._last_load_time: float = 0
   232→        self._last_post_time: float = 0
   233→        self._posted_today: int = 0
   234→        self._day_start: float = 0
   235→        # Site URL → PartnerProgram mapping for fast lookup
   236→        self._site_map: Dict[str, PartnerProgram] = {}
   237→
   238→    async def load_async(self) -> int:
   239→        """Load partner programs — try remote first, then local cache."""
   240→        count = await self._load_from_remote()
   241→        if count > 0:
   242→            return count
   243→        # Fallback to local cache
   244→        return self._load_from_local()
   245→
   246→    # Alias for backward compatibility
   247→    load_admitad_async = load_async
   248→
   249→    async def _load_from_remote(self) -> int:
   250→        """Download admitad_ads.json from GitHub."""
   251→        try:
   252→            import httpx
   253→            async with httpx.AsyncClient(timeout=30.0, follow_redirects=True) as client:
   254→                response = await client.get(ADMITAD_JSON_URL)
   255→                if response.status_code == 200:
   256→                    data = response.json()
   257→                    count = self._parse_programs(data)
   258→                    if count > 0:
   259→                        # Save to local cache
   260→                        self._save_cache(data)
   261→                        self._loaded = True
   262→                        self._last_load_time = time.time()
   263→                        logger.info(f"Loaded {count} partner programs from remote URL")
   264→                        return count
   265→        except Exception as e:
   266→            logger.warning(f"Failed to load admitad_ads.json from remote: {e}")
   267→        return 0
   268→
   269→    def _load_from_local(self) -> int:
   270→        """Load from local cache file."""
   271→        # Try data/ directory first, then root
   272→        for filepath in [ADMITAD_LOCAL_CACHE, "admitad_ads.json"]:
   273→            path = Path(filepath)
   274→            if path.exists():
   275→                try:
   276→                    with open(path, "r", encoding="utf-8") as f:
   277→                        data = json.load(f)
   278→                    count = self._parse_programs(data)
   279→                    self._loaded = True
   280→                    self._last_load_time = time.time()
   281→                    logger.info(f"Loaded {count} partner programs from local cache: {filepath}")
   282→                    return count
   283→                except Exception as e:
   284→                    logger.error(f"Error loading local admitad cache: {e}")
   285→        logger.warning("No admitad_ads.json found locally or remotely")
   286→        self._loaded = True
   287→        return 0
   288→
   289→    def load(self, filepath: str = "") -> int:
   290→        """Synchronous load from local file only."""
   291→        filepath = filepath or config.ADMITAD_ADS_FILE
   292→        path = Path(filepath)
   293→        if not path.exists():
   294→            path = Path(ADMITAD_LOCAL_CACHE)
   295→        if not path.exists():
   296→            path = Path("admitad_ads.json")
   297→
   298→        if not path.exists():
   299→            logger.warning(f"Partner ads file not found")
   300→            self._loaded = True
   301→            return 0
   302→
   303→        try:
   304→            with open(path, "r", encoding="utf-8") as f:
   305→                data = json.load(f)
   306→            count = self._parse_programs(data)
   307→            self._loaded = True
   308→            self._last_load_time = time.time()
   309→            logger.info(f"Loaded {count} partner programs from {path}")
   310→            return count
   311→        except Exception as e:
   312→            logger.error(f"Error loading partner ads: {e}")
   313→            self._loaded = True
   314→            return 0
   315→
   316→    def _parse_programs(self, data) -> int:
   317→        """Parse programs from JSON data."""
   318→        self.programs = []
   319→        self._site_map = {}
   320→
   321→        items = []
   322→        if isinstance(data, list):
   323→            items = data
   324→        elif isinstance(data, dict):
   325→            items = data.get("programs", data.get("items", data.get("results", [])))
   326→            if not isinstance(items, list):
   327→                items = []
   328→
   329→        for item in items:
   330→            prog = PartnerProgram(item)
   331→            if prog.goto_link:
   332→                self.programs.append(prog)
   333→                # Build site URL mapping
   334→                if prog.site_url:
   335→                    domain = urlparse(prog.site_url).netloc.replace("www.", "")
   336→                    self._site_map[domain] = prog
   337→
   338→        return len(self.programs)
   339→
   340→    def _save_cache(self, data) -> None:
   341→        """Save data to local cache."""
   342→        try:
   343→            cache_path = Path(ADMITAD_LOCAL_CACHE)
   344→            cache_path.parent.mkdir(parents=True, exist_ok=True)
   345→            with open(cache_path, "w", encoding="utf-8") as f:
   346→                json.dump(data, f, ensure_ascii=False)
   347→        except Exception as e:
   348→            logger.warning(f"Failed to save admitad cache: {e}")
   349→
   350→    def ensure_loaded(self) -> None:
   351→        """Load partner programs if not yet loaded."""
   352→        if not self._loaded:
   353→            self.load()
   354→
   355→    async def maybe_refresh(self) -> None:
   356→        """Refresh from remote if enough time has passed."""
   357→        if not self._loaded or (time.time() - self._last_load_time > ADMITAD_REFRESH_INTERVAL):
   358→            await self.load_async()

   360→    def get_by_category(self, category: str, region: str = DEFAULT_REGION) -> List[PartnerProgram]:
   361→        """Get programs in a specific category and region."""
   362→        self.ensure_loaded()
   363→        return [p for p in self.programs if p.has_category(category) and p.has_region(region)]

   365→    def get_by_site(self, site_url: str) -> Optional[PartnerProgram]:
   366→        """Find a partner program by its site URL or domain."""
   367→        self.ensure_loaded()
   368→        if not site_url:
   369→            return None
   370→        
   371→        # Try parsing as URL first
   372→        domain = urlparse(site_url).netloc.replace("www.", "") if site_url else ""
   373→        
   374→        # If urlparse didn't extract a netloc (bare domain like "rossko.ru"),
   375→        # treat the input itself as the domain
   376→        if not domain and site_url:
   377→            domain = site_url.replace("www.", "").rstrip("/")
   378→        
   379→        # Direct lookup
   380→        result = self._site_map.get(domain)
   381→        if result:
   382→            return result
   383→        
   384→        # Fallback: partial match on domain keys
   385→        for key, prog in self._site_map.items():
   386→            if domain in key or key in domain:
   387→                return prog
   388→        
   389→        return None

   450→    def generate_partner_context(self, text: str, max_programs: int = 3, region: str = DEFAULT_REGION) -> str:
   451→        """
   452→        Generate context about matching partner programs for AI to reference
   453→        naturally in its response.
   454→
   455→        v3: Uses goto_link from admitad_ads.json EXACTLY as-is.
   456→        No subid additions — the link is ready!
   457→        """
   458→        self.ensure_loaded()
   459→
   460→        # Find matching programs
   461→        matches = self.find_matching_programs(text, region)
   462→        if not matches:
   463→            # Default: show autoparts programs for auto-related queries
   464→            text_lower = text.lower()
   465→            auto_keywords = ["запчаст", "деталь", "артикул", "купить", "подобрать",
   466→                             "масло", "фильтр", "колодки", "ремонт", "сто"]
   467→            if any(kw in text_lower for kw in auto_keywords):
   468→                matches = self.get_by_category("autoparts", region)
   469→
   470→        if not matches:
   471→            return ""
   472→
   473→        # Check if there's an article number for search-specific links
   474→        article_match = re.search(r'\b([A-Z0-9]{4,}[-/]?[A-Z0-9]*)\b', text.upper())
   475→        article = article_match.group(1) if article_match else ""
   476→
   477→        lines = [
   478→            "Партнёрские ссылки (Вставь ЕСТЕСТВЕННО в ответ — как рекомендация, не как реклама!):",
   479→        ]
   480→
   481→        for p in matches[:max_programs]:
   482→            if article:
   483→                # For article searches, try to get a search-specific URL
   484→                link = p.format_link_with_search(article)
   485→            else:
   486→                link = p.format_link(with_description=True)
   487→
   488→            if link:
   489→                lines.append(f"- {link}")
   490→
   491→        # Also add Rossko specifically for auto parts queries
   492→        text_lower = text.lower()
   493→        parts_keywords = ["запчаст", "деталь", "артикул", "купить запчас", "подбор",
   494→                          "фильтр", "колодки", "свечи", "ремень", "масло"]
   495→        if any(kw in text_lower for kw in parts_keywords):
   496→            rossko = self.get_by_site("rossko.ru")
   497→            if rossko and rossko not in matches[:max_programs]:
   498→                if article:
   499→                    lines.append(f"- {rossko.format_link_with_search(article)}")
   500→                else:
   501→                    lines.append(f"- {rossko.format_link(with_description=True)}")
   502→
   503→        if len(lines) <= 1:
   504→            return ""
   505→
   506→        lines.append("")
   507→        lines.append("ВАЖНО: Ссылки выше — ПАРТНЁРСКИЕ (goto_link из admitad_ads.json). Используй их КАК ЕСТЬ, ничего не добавляй и не меняй!")
   508→
   509→        return "\n".join(lines)

   511→    def get_primary_parts_links(self, region: str = DEFAULT_REGION) -> List[Dict[str, str]]:
   512→        """Get the THREE primary partner links for auto parts in strict order.
   513→
   514→        Order: 1) Rossko, 2) Autopiter (RU), 3) AvtoALL

   715→    def should_post_partner(self) -> bool:
   716→        """Check if it's time to post a partner message to channel."""
   717→        now = time.time()
   718→        if now - self._day_start > 86400:
   719→            self._day_start = now
   720→            self._posted_today = 0
   721→
   722→        if self._posted_today >= config.PARTNER_DAILY_LIMIT:
   723→            return False
   724→
   725→        min_interval = config.PARTNER_POST_INTERVAL_HOURS * 3600
   726→        if now - self._last_post_time < min_interval:
   727→            return False
   728→
   729→        return True
