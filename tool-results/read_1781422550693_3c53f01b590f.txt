    50→# ── VIN / Body number detection ───────────────────────────────────────────────
    51→
    52→_VIN_PATTERN = re.compile(r'\b[A-HJ-NPR-Z0-9]{17}\b', re.IGNORECASE)
    53→# Also match VINs with spaces or dashes (users often type them separated)
    54→_VIN_FLEX_PATTERN = re.compile(
    55→    r'(?:VIN[-:]?\s*|вин[-:]?\s*|вин-код[-:]?\s*)?([A-HJ-NPR-Z0-9](?:[A-HJ-NPR-Z0-9\s\-]{14,22})[A-HJ-NPR-Z0-9])',
    56→    re.IGNORECASE
    57→)
    58→_BODY_NUMBER_PATTERN = re.compile(
    59→    r'(?:номер\s+кузова|кузовн?ой\s+номер|body\s*number|кузов)\s*[:\s]*([A-Z0-9\-/]{5,20})',
    60→    re.IGNORECASE
    61→)
    62→
    63→
    64→def _detect_vin(text: str) -> Optional[str]:
    65→    """Detect a VIN code (17 chars) in text."""
    66→    match = _VIN_PATTERN.search(text.upper())
    67→    if match:
    68→        vin = match.group(0)
    69→        # Validate check digit position (9th char)
    70→        if len(vin) == 17 and vin[8] in '0123456789X':
    71→            return vin
    72→    return None
    73→
    74→
    75→def _detect_body_number(text: str) -> Optional[str]:
    76→    """Detect a body number reference in text."""
    77→    match = _BODY_NUMBER_PATTERN.search(text)
    78→    if match:
    79→        return match.group(1)
    80→    return None

   118→def _guess_gender(first_name: str) -> str:
   119→    """Guess gender from Russian first name. Returns 'male', 'female', or 'unknown'."""
   120→    if not first_name:
   121→        return "unknown"
   122→
   123→    name_lower = first_name.lower().strip()
   124→
   125→    if name_lower in COMMON_MALE_NAMES:
   126→        return "male"
   127→    if name_lower in COMMON_FEMALE_NAMES:
   128→        return "female"
   129→
   130→    if name_lower.endswith(FEMALE_NAME_ENDINGS):
   131→        if name_lower.endswith("ь"):
   132→            pass
   133→        else:
   134→            return "female"
   135→
   136→    if name_lower.endswith("й") or name_lower.endswith("ь"):
   137→        return "male"
   138→
   139→    return "unknown"

   486→    In GROUPS/SUPERGROUPS: use LOCAL-ONLY routing (comments mode).

   488→    """
   489→    if not await _check_user(message):
   490→        return
   491→
   492→    # Check if we're in a group/supergroup — use LOCAL-ONLY for comments
   493→    is_group = message.chat.type in ("group", "supergroup")
   494→
   495→    # In groups: Ася should just comment on local model ONLY, not use cloud vision
   496→    if is_group:
   497→        # LOCAL MODEL ONLY for group comments — no cloud API waste!
   498→        # Use singleton from ai_router — no reloading the model every time!
   499→        from ai.router import ai_router
   500→        local_provider = ai_router._local
   501→        if local_provider and await local_provider.is_available():

  1167→def _smart_truncate(text: str, max_chars: int) -> str:
  1168→    """Truncate text to max_chars, trying not to cut in the middle of a sentence.
  1169→    
  1170→    Looks for the last sentence boundary (. ! ? …) within the limit.
  1171→    Falls back to last newline, then last space, then hard cut.
  1172→    Always appends "..." if truncated.
  1173→    """
  1174→    if len(text) <= max_chars:
  1175→        return text
  1176→
  1177→    # Reserve 3 chars for "..."
  1178→    limit = max_chars - 3
  1179→    if limit <= 0:
  1180→        return "..."
  1181→
  1182→    # Try to find last sentence boundary within limit
  1183→    sentence_endings = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
  1184→    best_pos = -1
  1185→    best_ending_len = 0
  1186→    for ending in sentence_endings:
  1187→        pos = text.rfind(ending, 0, limit + 1)
  1188→        if pos > best_pos:
  1189→            best_pos = pos
  1190→            best_ending_len = len(ending)
  1191→
  1192→    if best_pos > limit // 2:
  1193→        # Found a good sentence boundary — cut after the punctuation mark
  1194→        return text[:best_pos + best_ending_len].rstrip() + "..."
  1195→
  1196→    # Try last newline
  1197→    nl_pos = text.rfind('\n', 0, limit + 1)
  1198→    if nl_pos > limit // 2:
  1199→        return text[:nl_pos].rstrip() + "..."
  1200→
  1201→    # Try last space
  1202→    sp_pos = text.rfind(' ', 0, limit + 1)
  1203→    if sp_pos > limit // 2:
  1204→        return text[:sp_pos].rstrip() + "..."
  1205→
  1206→    # Hard cut
  1207→    return text[:limit].rstrip() + "..."

  1322→def _clean_raw_partner_urls(text: str, partner_links: list) -> str:
  1323→    """Remove raw affiliate URLs that AI may have dumped into the response.
  1324→    
  1325→    If the AI already included the full affiliate URLs in its response text,
  1326→    this function removes them to avoid duplication with the clean formatted
  1327→    section that will be appended by _format_partner_links_section.
  1328→    Also removes shop name + URL patterns like "Росско: https://...".
  1329→    """
  1330→    if not partner_links:
  1331→        return text
  1332→    
  1333→    for name, url in partner_links:
  1334→        url_escaped = re.escape(url)
  1335→        # Remove standalone URL lines
  1336→        text = re.sub(
  1337→            rf'^[\s\-—]*{url_escaped}[\s]*$',
  1338→            '',
  1339→            text,
  1340→            flags=re.MULTILINE
  1341→        )
  1342→        # Remove "name: URL" or "name — URL" patterns on their own lines
  1343→        text = re.sub(
  1344→            rf'^[\s\-—]*{re.escape(name)}\s*[:\-—]\s*{url_escaped}[\s]*$',
  1345→            '',
  1346→            text,
  1347→            flags=re.MULTILINE
  1348→        )
  1349→        # Remove any line containing the full raw affiliate URL (even mid-sentence)
  1350→        # This catches cases where AI writes: "Посмотри на Росско: https://ujhjj.com/g/..."
  1351→        text = re.sub(
  1352→            rf'[^\n]*{url_escaped}[^\n]*',
  1353→            '',
  1354→            text,
  1355→            flags=re.MULTILINE
  1356→        )
  1357→        # Remove bare URL anywhere in text (affiliate URLs are very long, easy to detect)
  1358→        if len(url) > 50:  # Only long tracking URLs
  1359→            # Match the base tracking domain pattern
  1360→            url_base = re.escape(url.split('?')[0])  # Before query params
  1361→            text = re.sub(
  1362→                rf'{url_base}\?[^\s\n"]+',
  1363→                '',
  1364→                text
  1365→            )
  1366→    
  1367→    text = re.sub(r'\n{3,}', '\n\n', text)
  1368→    return text.strip()

  1394→def _replace_plain_urls_with_affiliate(text: str) -> str:
  1395→    """Replace any plain partner domain URLs with affiliate goto_link equivalents.
  1396→    
  1397→    When the AI generates responses containing plain URLs like rossko.ru or 
  1398→    autopiter.ru instead of the affiliate tracking links from admitad_ads.json,
  1399→    this function detects them and replaces with the proper goto_link.
  1400→    
  1401→    This handles cases where:
  1402→    - AI ignores the system prompt and invents plain URLs
  1403→    - AI uses domain names without the affiliate wrapper
  1404→    - Photo handler bypasses the normal link injection pipeline
  1405→    """
  1406→    try:
  1407→        partner_manager.ensure_loaded()
  1408→    except Exception:
  1409→        return text
  1410→    
  1411→    for domain, display_name in _PARTNER_DOMAINS_MAP.items():
  1412→        prog = partner_manager.get_by_site(domain)
  1413→        if not prog or not prog.goto_link:
  1414→            continue
  1415→        
  1416→        affiliate_url = prog.goto_link
  1417→        
  1418→        # Pattern 1: Full URLs with paths — https://rossko.ru/search?text=abc
  1419→        # Replace the entire URL with the affiliate link
  1420→        pattern = rf'https?://{re.escape(domain)}[^\s<>)\]"\']*'
  1421→        matches = re.findall(pattern, text)
  1422→        for plain_url in matches:
  1423→            # Try to extract search query and build affiliate link with search
  1424→            search_query = ""
  1425→            if "search" in plain_url or "querystr" in plain_url or "q=" in plain_url:
  1426→                try:
  1427→                    from urllib.parse import urlparse, parse_qs
  1428→                    parsed = urlparse(plain_url)
  1429→                    params = parse_qs(parsed.query)
  1430→                    for key in ("text", "querystr", "q", "query", "SearchText", "keyword", "p"):
  1431→                        if key in params:
  1432→                            search_query = params[key][0]
  1433→                            break
  1434→                except Exception:
  1435→                    pass
  1436→            
  1437→            if search_query:
  1438→                replacement = prog.get_search_url(search_query)
  1439→            else:
  1440→                replacement = affiliate_url
  1441→            
  1442→            text = text.replace(plain_url, replacement)
  1443→        
  1444→        # Pattern 2: Bare domain mentions — rossko.ru or www.rossko.ru (not already part of a longer URL)
  1445→        # Only replace if it's NOT already part of an affiliate URL (which would be much longer)
  1446→        bare_pattern = rf'(?<![/\w.-])(?:www\.)?{re.escape(domain)}(?![/\w.-])'
  1447→        bare_matches = re.findall(bare_pattern, text)

  1461→def _clean_markdown(text: str) -> str:
  1462→    """Remove markdown formatting that Asya shouldn't use.
  1463→    
  1464→    Markdown links [text](url) are converted to plain text: text — url.
  1465→    Other markdown is stripped. No HTML — plain text for chat responses.
  1466→    """
  1467→    # Convert markdown links [text](url) → text — url (plain text format)
  1468→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1 — \2', text)
  1469→    
  1470→    # Remove bold (after escaping, ** won't be in &lt; form)
  1471→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
  1472→    # Remove italic
  1473→    text = re.sub(r'\*(.+?)\*', r'\1', text)
  1474→    # Remove code blocks
  1475→    text = re.sub(r'```[\s\S]*?```', lambda m: m.group(0).strip('`').strip(), text)
  1476→    # Remove inline code
  1477→    text = re.sub(r'`(.+?)`', r'\1', text)
  1478→    # Remove headers
  1479→    text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1480→    # Remove bullet points (convert to simple text)
  1481→    text = re.sub(r'^[-*]\s+', '— ', text, flags=re.MULTILINE)
  1482→    return text

  1485→def _split_message(text: str, max_length: int = 4096) -> list:
  1486→    """Split a long message into chunks at paragraph boundaries."""
  1487→    if len(text) <= max_length:
  1488→        return [text]
  1489→
  1490→    chunks = []
  1491→    while text:
  1492→        if len(text) <= max_length:
  1493→            chunks.append(text)
  1494→            break
  1495→
  1496→        # Try to split at newline near the limit
  1497→        split_pos = text.rfind('\n', 0, max_length)
  1498→        if split_pos < max_length // 2:
  1499→            # Try splitting at space
  1500→            split_pos = text.rfind(' ', 0, max_length)
  1501→        if split_pos < max_length // 2:
  1502→            # Hard split
  1503→            split_pos = max_length
  1504→
  1505→        chunks.append(text[:split_pos].strip())
  1506→        text = text[split_pos:].strip()
  1507→
  1508→    return chunks
