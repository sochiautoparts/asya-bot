    25→def _compute_fingerprint(title: str) -> str:
    26→    """Compute a simplified fingerprint for dedup.
    27→    Remove articles, lowercase, take first 5 words.
    28→    """
    29→    import re
    30→    # Remove common articles/prepositions
    31→    cleaned = re.sub(r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b', '', title.lower())
    32→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
    33→    words = cleaned.split()
    34→    return ' '.join(words[:5])

    37→def _fingerprint_matches_existing(fingerprint: str) -> bool:
    38→    """Check if a fingerprint matches any recently used fingerprint (edit distance on first 4 words)."""
    39→    fp_words = fingerprint.split()[:4]
    40→    for existing in _recent_fingerprints:
    41→        ex_words = existing.split()[:4]
    42→        # Count matching words
    43→        matches = sum(1 for w in fp_words if w in ex_words)
    44→        if matches >= 3:
    45→            return True
    46→    return False

   274→# ── RSS Fetcher ────────────────────────────────────────────────────────────────
   275→
   276→async def fetch_rss(source: NewsSource) -> List[Dict]:
   277→    """
   278→    Fetch and parse an RSS feed.
   279→    Returns list of news items.
   280→    """
   281→    items = []
   282→    try:
   283→        async with httpx.AsyncClient(
   284→            timeout=20.0,
   285→            follow_redirects=True,
   286→            headers={
   287→                "User-Agent": "Mozilla/5.0 (compatible; AsyaBot/1.0; +https://t.me/asiaexp_bot)",
   288→                "Accept": "application/rss+xml, application/xml, text/xml, application/atom+xml, */*",
   289→            },
   290→        ) as client:
   291→            response = await client.get(source.url)
   292→
   293→            if response.status_code != 200:
   294→                logger.warning(f"RSS {source.name} returned {response.status_code}")
   295→                return items
   296→
   297→            # feedparser can parse from string
   298→            feed = feedparser.parse(response.text)
   299→
   300→            for entry in feed.entries[:20]:  # Limit per source
   301→                title = entry.get("title", "").strip()
   302→                url = entry.get("link", "").strip()
   303→
   304→                if not title or not url:
   305→                    continue
   306→
   307→                # Get summary/description
   308→                summary = ""
   309→                if entry.get("summary"):
   310→                    summary = _clean_html(entry["summary"])
   311→                elif entry.get("description"):
   312→                    summary = _clean_html(entry["description"])
   313→
   314→                # Truncate summary
   315→                if len(summary) > 500:
   316→                    summary = summary[:500] + "..."
   317→
   318→                # Get publish date
   319→                published = 0.0
   320→                if entry.get("published_parsed"):
   321→                    try:
   322→                        published = time.mktime(entry["published_parsed"])
   323→                    except (ValueError, TypeError, OverflowError):
   324→                        published = time.time()
   325→                elif entry.get("updated_parsed"):
   326→                    try:
   327→                        published = time.mktime(entry["updated_parsed"])
   328→                    except (ValueError, TypeError, OverflowError):
   329→                        published = time.time()
   330→                else:
   331→                    published = time.time()
   332→
   333→                # Extract image URLs from RSS entry
   334→                image_urls = _extract_entry_images(entry)
   335→
   336→                items.append({
   337→                    "source": source.name,
   338→                    "title": title,
   339→                    "url": url,
   340→                    "summary": summary,
   341→                    "published": published,
   342→                    "category": source.category,
   343→                    "lang": source.lang,
   344→                    "image_urls": image_urls,
   345→                })
   346→
   347→            logger.info(f"Fetched {len(items)} items from {source.name}")
   348→
   349→    except httpx.TimeoutException:
   350→        logger.warning(f"RSS timeout: {source.name}")
   351→    except Exception as e:
   352→        logger.error(f"RSS fetch error for {source.name}: {e}")
   353→
   354→    return items

   359→def is_auto_relevant(item: Dict, lang: str = "ru") -> bool:
   360→    """Check if a news item is relevant to the auto topic."""
   361→    text = f"{item.get('title', '')} {item.get('summary', '')}".lower()
   362→
   363→    if lang == "ru" or lang == "":
   364→        keywords = AUTO_KEYWORDS_RU
   365→    elif lang == "en":
   366→        keywords = AUTO_KEYWORDS_EN
   367→    elif lang == "de":
   368→        keywords = AUTO_KEYWORDS_DE
   369→    else:
   370→        keywords = AUTO_KEYWORDS_RU + AUTO_KEYWORDS_EN
   371→
   372→    # Check if any keyword matches
   373→    for kw in keywords:
   374→        if kw.lower() in text:
   375→            return True
   376→
   377→    return False

   745→_JUNK_IMAGE_STEMS = {
   746→    "business_card", "placeholder", "default-image", "default_image",

   748→    "spacer", "blank", "transparent", "pixel", "tracker",

   753→_TINY_SIZE_PATTERNS = [
   754→    r'[?&]width=(?:1\d\d|80|100|120)(?:&|$)',   # width=80..199
   755→    r'[?&]height=(?:1\d\d|80|100|120)(?:&|$)',   # height=80..199
   756→    r'[?&]crop=1:1[,&]',                         # square crop = icon/thumb
   757→    r'/\d+x\d+/',                                 # /108x108/ in path

   759→]

   765→def _normalize_image_url(url: str) -> str:
   766→    """Normalize an image URL: decode HTML entities, fix scheme, clean up."""
   767→    if not url:
   768→        return ""
   769→    url = url.strip()
   770→    # Decode HTML entities: &amp; → &, &#038; → &, &lt; → <, etc.
   771→    url = _html_unescape(url)
   772→    # Fix scheme-relative URLs
   773→    if url.startswith("//"):
   774→        url = "https:" + url
   775→    return url

   778→def _is_junk_image_url(url: str) -> bool:
   779→    """Check if an image URL is junk: tracker, icon, tiny thumbnail, default image."""
   780→    from urllib.parse import urlparse, parse_qs
   781→    try:
   782→        url_lower = url.lower()
   783→        parsed = urlparse(url)
   784→        domain = parsed.netloc.lower()
   785→        path_lower = parsed.path.lower()
   786→
   787→        # Skip tracker/analytics domains
   788→        if any(skip in domain for skip in _SKIP_IMAGE_DOMAINS):
   789→            return True
   790→
   791→        # Skip icon/logo/badge/button filenames in path
   792→        if any(kw in path_lower for kw in [
   793→            "icon", "avatar", "logo", "favicon", "badge", "button", "banner",
   794→        ]):
   795→            return True
   796→
   797→        # Skip /feed/ URLs (CarScoops bug)
   798→        if _FEED_URL_PATTERN.search(url_lower):
   799→            return True
   800→
   801→        # Skip generic/default image filenames (also prefix-match: business_card-1)
   802→        stem = path_lower.rsplit("/", 1)[-1].split(".")[0].replace("-", "_")
   803→        if stem in _JUNK_IMAGE_STEMS:
   804→            return True
   805→        for junk_stem in _JUNK_IMAGE_STEMS:
   806→            if stem.startswith(junk_stem):
   807→                return True
   808→
   809→        # Skip tiny thumbnail dimensions in URL
   810→        for pattern in _TINY_SIZE_PATTERNS:
   811→            if re.search(pattern, url_lower):
   812→                return True
   813→
   814→    except Exception:
   815→        return True  # If we can't parse it, skip it
   816→    return False

   819→def _upgrade_thumbnail_url(url: str) -> str:
   820→    """Upgrade low-quality thumbnail URLs to higher resolution versions.
   821→
   822→    Known CDN patterns:
   823→    - BBC: /240/ → /640/ in path
   824→    - Autosport/Motorsport.com: /s6/ → /s12/ in path
   825→    - Reddit: width=140 → width=640 in query
   826→    """
   827→    # BBC: /240/cpsprodpb/ → /640/cpsprodpb/
   828→    if "bbci.co.uk" in url or "bbc.co.uk" in url:
   829→        url = re.sub(r'/240/', '/640/', url, count=1)
   830→
   831→    # Autosport / Motorsport.com: /s6/ → /s12/
   832→    if "motorsport.com" in url or "autosport.com" in url:
   833→        url = re.sub(r'/s6/', '/s12/', url, count=1)
   834→
   835→    # Reddit preview: width=140 → width=640
   836→    if "preview.redd.it" in url or "external-preview.redd.it" in url:
   837→        url = re.sub(r'width=140', 'width=640', url, count=1)
   838→        url = re.sub(r'height=140', 'height=640', url, count=1)
   839→
   840→    return url

   843→def _extract_entry_images(entry) -> List[str]:
   844→    """Extract image URLs from a feedparser entry.
   845→
   846→    Priority order (highest quality first):
   847→    1. media_content (Media RSS — editors pick these, usually hero images)
   848→    2. enclosures (often full-size article images)
   849→    3. media_thumbnail (small but from structured metadata)
   850→    4. links with image type
   851→    5. <img> tags in content/summary HTML (last — often thumbnails + junk)
   852→
   853→    All URLs are HTML-entity-decoded, deduplicated (normalized),
   854→    junk-filtered, and thumbnail-upgraded before returning.
   855→    Returns list of clean image URLs (up to 10).
   856→    """
   857→    images = []
   858→    seen_normalized = set()  # Dedup by normalized URL (after unescape)
   859→
   860→    def _add_image(raw_url: str):
   861→        """Add image URL if valid and not already seen (after normalization)."""
   862→        url = _normalize_image_url(raw_url)
   863→        if not url or len(url) < _MIN_IMAGE_URL_LEN:
   864→            return
   865→        if _is_junk_image_url(url):
   866→            return
   867→        # Upgrade low-quality thumbnails to bigger versions
   868→        url = _upgrade_thumbnail_url(url)
   869→        # Dedup by normalized form (lowercase for comparison)
   870→        norm_key = url.lower().rstrip("/")
   871→        if norm_key in seen_normalized:
   872→            return
   873→        seen_normalized.add(norm_key)
   874→        images.append(url)
   875→
   876→    # 1. media_content (Media RSS extension — most reliable, editor-chosen)
   877→    for media in entry.get("media_content", []) or []:
   878→        url = media.get("url", "")
   879→        media_type = media.get("type", "").lower()
   880→        medium = media.get("medium", "").lower()
   881→        if url and (medium == "image" or "image" in media_type or _has_image_ext(url)):
   882→            _add_image(url)
   883→
   884→    # 2. enclosures (often full-size article images)
   885→    for enclosure in entry.get("enclosures", []) or []:
   886→        url = enclosure.get("href", "") or enclosure.get("url", "")
   887→        enc_type = enclosure.get("type", "").lower()
   888→        if url and ("image" in enc_type or _has_image_ext(url)):
   889→            _add_image(url)
   890→
   891→    # 3. media_thumbnail (small but structured metadata)
   892→    for thumb in entry.get("media_thumbnail", []) or []:
   893→        url = thumb.get("url", "")
   894→        if url:
   895→            _add_image(url)
   896→
   897→    # 4. links with image rel or type
   898→    for link in entry.get("links", []) or []:
   899→        link_type = link.get("type", "").lower()
   900→        link_rel = link.get("rel", "").lower()
   901→        url = link.get("href", "")
   902→        if url and ("image" in link_type or link_rel == "enclosure"):
   903→            _add_image(url)
   904→
   905→    # 5. Extract <img> tags from content/summary HTML (last resort — noisy)
   906→    for field_name in ["content", "summary", "description"]:
   907→        content_value = entry.get(field_name)
   908→        if isinstance(content_value, list):
   909→            for item in content_value:
   910→                html = item.get("value", "") if isinstance(item, dict) else str(item)
   911→                _extract_img_urls(html, _add_image)
   912→        elif isinstance(content_value, str):
   913→            _extract_img_urls(content_value, _add_image)
   914→
   915→    return images[:10]

   918→def _has_image_ext(url: str) -> bool:
   919→    """Check if URL ends with an image extension."""
   920→    from urllib.parse import urlparse
   921→    try:
   922→        path = urlparse(url).path.lower().split("?")[0].split("#")[0]
   923→        return any(path.endswith(ext) for ext in _VALID_IMAGE_EXTENSIONS)
   924→    except Exception:
   925→        return False

   928→def _extract_img_urls(html: str, callback):
   929→    """Extract src attributes from <img> tags in HTML.
   930→
   931→    Also checks data-src and data-lazy-src for lazy-loaded images.
   932→    Decodes HTML entities before passing to callback.
   933→    """
   934→    if not html:
   935→        return
   936→    for attr in ["src", "data-src", "data-lazy-src"]:
   937→        for match in re.finditer(
   938→            rf'<img[^>]+{attr}=["\']([^"\']+)["\']', html, re.IGNORECASE
   939→        ):
   940→            url = match.group(1)
   941→            if url and len(url) >= _MIN_IMAGE_URL_LEN:
   942→                callback(url)
   943→

   947→def _clean_html(text: str) -> str:
   948→    """Remove HTML tags and decode entities."""
   949→    text = re.sub(r'<[^>]+>', '', text)
   950→    text = text.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>')
   951→    text = text.replace('&quot;', '"').replace('&#39;', "'").replace('&nbsp;', ' ')
   952→    text = re.sub(r'\s+', ' ', text).strip()
   953→    return text
