   236→class PollinationsProvider(BaseAIProvider):
   237→    """Pollinations AI provider v5.0 — DUAL KEY + FREE API FALLBACK + 60+ MODELS!
   238→
   239→    Uses gen.pollinations.ai/v1/chat/completions (OpenAI-compatible).
   240→    Falls back to text.pollinations.ai (FREE, no auth) when keys are depleted.
   241→
   242→    FAILOVER CHAIN (within provider):
   243→      1. Try KEY1 first (config.POLLINATIONS_API_KEY)
   244→      2. On 402/401 from KEY1 -> switch to KEY2 (config.POLLINATIONS_API_KEY_2)
   245→      3. On 402/401 from KEY2 -> try FREE API (text.pollinations.ai, no auth)
   246→      4. Depleted keys auto-retry after 600 seconds cooldown
   247→      5. Free API always available but rate-limited and slower
   248→
   249→    IMPORTANT: Models are NEVER removed when unavailable.
   250→    Pollinations.ai rotates model availability — a failure today
   251→    doesn't mean the model is gone. Circuit breaking (5-min cooldown)
   252→    handles temporary failures.
   253→    """
   254→
   255→    def __init__(self):
   256→        super().__init__(
   257→            name="pollinations",
   258→            api_key=config.POLLINATIONS_API_KEY,
   259→            base_url=config.POLLINATIONS_BASE_URL,
   260→        )
   261→        # ── Dual API key storage ──
   262→        # KEY1 = primary key from config
   263→        self._api_key_1: str = config.POLLINATIONS_API_KEY
   264→        self._api_key_2: str = config.POLLINATIONS_API_KEY_2
   265→        # ── Per-key balance depletion tracking ──
   266→        # 0 = active/never depleted; >0 = timestamp when depleted
   267→        self._key1_depleted_at: float = 0.0
   268→        self._key2_depleted_at: float = 0.0
   269→
   270→        # ── Free API endpoints (no auth required) ──
   271→        self._free_text_url: str = config.POLLINATIONS_FREE_TEXT_URL
   272→        self._free_image_url: str = config.POLLINATIONS_FREE_IMAGE_URL
   273→        self._free_api_available: bool = True  # Assume available until proven otherwise
   274→        self._free_api_cooldown_until: float = 0.0  # Timestamp when free API cooldown ends
   275→
   276→    # ── API Key Management ──────────────────────────────────────
   277→
   278→    def _is_key_available(self, key_index: int) -> bool:
   279→        """Check if an API key is available (not depleted or cooldown expired)."""
   280→        if key_index == 1:
   281→            depleted_at = self._key1_depleted_at
   282→            key_val = self._api_key_1
   283→        elif key_index == 2:
   284→            depleted_at = self._key2_depleted_at
   285→            key_val = self._api_key_2
   286→        else:
   287→            return False  # Only KEY1 and KEY2 supported
   288→
   289→        if not key_val:
   290→            return False
   291→
   292→        if depleted_at == 0:
   293→            return True
   294→
   295→        elapsed = time.time() - depleted_at
   296→        if elapsed >= KEY_COOLDOWN:
   297→            if key_index == 1:
   298→                self._key1_depleted_at = 0
   299→            else:
   300→                self._key2_depleted_at = 0
   301→            logger.info(f"API KEY{key_index} cooldown expired after {elapsed:.0f}s — retrying")
   302→            return True
   303→
   304→        return False
   305→
   306→    def _mark_key_depleted(self, key_index: int) -> None:
   307→        """Mark an API key as depleted (balance exhausted)."""
   308→        if key_index == 1:
   309→            self._key1_depleted_at = time.time()
   310→        else:
   311→            self._key2_depleted_at = time.time()
   312→        logger.warning(
   313→            f"API KEY{key_index} depleted (402/401). "
   314→            f"Will auto-retry after {KEY_COOLDOWN}s cooldown."
   315→        )
   316→
   317→    def _get_active_key_tier(self) -> Tuple[str, int]:
   318→        """Determine which key/tier to use for the next request."""
   319→        if self._is_key_available(1):
   320→            return self._api_key_1, 1
   321→        if self._is_key_available(2):
   322→            return self._api_key_2, 2
   323→        return "", 0
   324→
   325→    def _get_key_status_summary(self) -> str:
   326→        """Get a human-readable summary of key statuses."""
   327→        parts = []
   328→        for idx, (key_val, depleted_at) in enumerate([
   329→            (self._api_key_1, self._key1_depleted_at),
   330→            (self._api_key_2, self._key2_depleted_at),
   331→        ], start=1):
   332→            if not key_val:
   333→                parts.append(f"KEY{idx}=not_set")
   334→            elif depleted_at == 0:
   335→                parts.append(f"KEY{idx}=active")
   336→            else:
   337→                remaining = KEY_COOLDOWN - (time.time() - depleted_at)
   338→                parts.append(f"KEY{idx}=depleted({remaining:.0f}s)")
   339→        return ", ".join(parts)
   340→
   341→    def _build_key_tier_list(self) -> List[Tuple[str, int]]:
   342→        """Build ordered list of available key tiers to try."""
   343→        tiers: List[Tuple[str, int]] = []
   344→        if self._is_key_available(1):
   345→            tiers.append((self._api_key_1, 1))
   346→        if self._is_key_available(2):
   347→            tiers.append((self._api_key_2, 2))
   348→        return tiers
   349→
   350→    async def chat(
   351→        self,
   352→        messages: List[Dict[str, str]],
   353→        model: str = "",
   354→        temperature: float = 0.7,
   355→        max_tokens: int = 2048,
   356→        **kwargs,
   357→    ) -> AIResponse:
   358→        """Send a chat completion request to Pollinations API with DUAL-KEY FAILOVER.
   359→
   360→        Strategy:
   361→        1. Try with KEY1 first (if balance available)
   362→        2. On 402/401 from KEY1 -> switch to KEY2
   363→        3. On 402/401 from KEY2 -> return error (router decides fallback)
   364→
   365→        Empty content handling:
   366→        - If a model returns empty content, check reasoning_content field
   367→        - If both are empty, mark as temporary failure (circuit break)
   368→        - Models in EMPTY_CONTENT_MODELS get special logging
   369→        - Models are NEVER removed from lists
   370→        """
   371→        model = model or DEFAULT_MODEL
   372→
   373→        # Check if model is in cooldown from recent failures
   374→        if self._is_model_in_cooldown(model):
   375→            alt = self._get_available_model()
   376→            if alt:
   377→                logger.info(f"Model {model} in cooldown, using {alt}")
   378→                model = alt
   379→
   380→        # Build key tier list: active keys first
   381→        tiers_to_try = self._build_key_tier_list()
   382→
   383→        # If no keys available at all, check cooldown
   384→        if not tiers_to_try:
   385→            return AIResponse(
   386→                text="",
   387→                model=model,
   388→                provider=self.name,
   389→                error=True,
   390→                error_message=f"All API keys depleted [{self._get_key_status_summary()}]",
   391→            )
   392→
   393→        payload = {
   394→            "model": model,
   395→            "messages": messages,
   396→            "temperature": temperature,
   397→            "max_tokens": max_tokens,
   398→        }
   399→
   400→        if kwargs.get("top_p"):
   401→            payload["top_p"] = kwargs["top_p"]
   402→        if kwargs.get("frequency_penalty"):
   403→            payload["frequency_penalty"] = kwargs["frequency_penalty"]
   404→        if kwargs.get("presence_penalty"):
   405→            payload["presence_penalty"] = kwargs["presence_penalty"]
   406→
   407→        last_error = None
   408→        for api_key, tier_index in tiers_to_try:
   409→            try:
   410→                headers = {"Content-Type": "application/json"}
   411→                if api_key:
   412→                    headers["Authorization"] = f"Bearer {api_key}"
   413→
   414→                # Use 15s timeout for chat requests — fast response is critical for user experience.
   415→                # The overall chat timeout in _process_text_message is 45s, and we need to
   416→                # try multiple key tiers and models before that deadline.
   417→                async with httpx.AsyncClient(timeout=15.0) as client:
   418→                    start_time = time.time()
   419→                    url = f"{self.base_url}/v1/chat/completions"
   420→                    response = await client.post(url, headers=headers, json=payload)
   421→                    elapsed = time.time() - start_time
   422→
   423→                    if response.status_code == 200:
   424→                        data = response.json()
   425→                        text = ""
   426→                        choices = data.get("choices", [])
   427→                        if choices:
   428→                            msg = choices[0].get("message", {})
   429→                            # Primary content
   430→                            text = msg.get("content", "") or ""
   431→                            # Fallback: reasoning_content (some reasoning models
   432→                            # put their output here instead of content)
   433→                            if not text:
   434→                                text = msg.get("reasoning_content", "") or ""
   435→
   436→                        usage = data.get("usage", {})
   437→                        tokens_used = usage.get("total_tokens", 0)
   438→                        actual_model = data.get("model", model)
   439→                        tier_label = f"KEY{tier_index}"
   440→
   441→                        # Handle empty content gracefully
   442→                        if not text:
   443→                            # Check if this is a known empty-content model
   444→                            if model in EMPTY_CONTENT_MODELS:
   445→                                logger.warning(
   446→                                    f"Empty response from {model} (known empty-content model) "
   447→                                    f"via {tier_label} — marking as temporary failure"
   448→                                )
   449→                            else:
   450→                                logger.warning(
   451→                                    f"Empty response from {model} via {tier_label} — "
   452→                                    f"marking as temporary failure"
   453→                                )
   454→                            # Circuit break the model, but do NOT remove it
   455→                            _model_failures[model] = time.time()
   456→                            last_error = f"Empty response from {model}"
   457→                            continue  # Try next key tier
   458→
   459→                        logger.info(
   460→                            f"Pollinations response ({tier_label}): model={actual_model}, "
   461→                            f"tokens={tokens_used}, time={elapsed:.1f}s, "
   462→                            f"length={len(text)}"
   463→                        )
   464→
   465→                        return AIResponse(
   466→                            text=text,
   467→                            model=actual_model,
   468→                            provider=self.name,
   469→                            tokens_used=tokens_used,
   470→                        )
   471→
   472→                    elif response.status_code in (401, 402):
   473→                        # Balance depleted or unauthorized → switch key
   474→                        self._mark_key_depleted(tier_index)
   475→                        last_error = f"HTTP {response.status_code} from KEY{tier_index}"
   476→                        logger.warning(
   477→                            f"HTTP {response.status_code} from {model} via KEY{tier_index} — "
   478→                            f"switching to next key tier"
   479→                        )
   480→                        continue  # Try next key tier
   481→
   482→                    elif response.status_code == 429:
   483→                        # Rate limited — short cooldown
   484→                        logger.warning(f"Rate limited (429) on KEY{tier_index}, model={model}")
   485→                        _model_failures[model] = time.time()
   486→                        # Try next key tier
   487→                        continue
   488→
   489→                    else:
   490→                        error_text = response.text[:500]
   491→                        logger.error(
   492→                            f"Pollinations error: status={response.status_code}, "
   493→                            f"model={model}, error={error_text}"
   494→                        )
   495→                        _model_failures[model] = time.time()
   496→
   497→                        return AIResponse(
   498→                            text="",
   499→                            model=model,
   500→                            provider=self.name,
   501→                            error=True,
   502→                            error_message=f"HTTP {response.status_code}: {error_text}",
   503→                        )
   504→
   505→            except httpx.TimeoutException:
   506→                logger.error(f"Pollinations timeout: model={model}, tier=KEY{tier_index}")
   507→                _model_failures[model] = time.time()
   508→                last_error = f"Timeout from KEY{tier_index}"
   509→                # Mark the key as potentially depleted on timeout too —
   510→                # if it times out, trying the same key again is unlikely to help
   511→                if tier_index == 1 and self._key1_depleted_at == 0:
   512→                    self._key1_depleted_at = time.time() - KEY_COOLDOWN + 60  # Retry in 60s instead of 600s
   513→                elif tier_index == 2 and self._key2_depleted_at == 0:
   514→                    self._key2_depleted_at = time.time() - KEY_COOLDOWN + 60
   515→                continue  # Try next key tier
   516→
   517→            except Exception as e:
   518→                logger.error(f"Pollinations exception: model={model}, tier=KEY{tier_index}, error={e}")
   519→                _model_failures[model] = time.time()
   520→                last_error = str(e)
   521→                continue
   522→
   523→        # ── All key tiers failed → return error, let router decide ──
   524→        key_status = self._get_key_status_summary()
   525→        logger.error(
   526→            f"All API key tiers failed for model {model}. "
   527→            f"Key status: [{key_status}]. Last error: {last_error}. "
   528→            f"Router should decide fallback."
   529→        )
   530→
   531→        return AIResponse(
   532→            text="",
   533→            model=model,
   534→            provider=self.name,
   535→            error=True,
   536→            error_message=f"All API keys depleted/unavailable [{key_status}]. {last_error}",
   537→        )

   539→    async def generate_image(self, prompt: str, model: str = "flux") -> Optional[bytes]:
   540→        """Generate an image using Pollinations image API.
   541→        Uses dual-key failover: KEY1 -> KEY2.
   542→        Returns image bytes or None on failure.
   543→        """
   544→        # Build key tier list
   545→        tiers_to_try = self._build_key_tier_list()
   546→
   547→        if not tiers_to_try:
   548→            logger.warning("No API keys available for image generation")
   549→            return None
   550→
   551→        for api_key, tier_index in tiers_to_try:
   552→            try:
   553→                url = f"{self.base_url}/v1/images/generations"
   554→                headers = {
   555→                    "Content-Type": "application/json",
   556→                    "Authorization": f"Bearer {api_key}",
   557→                }
   558→                payload = {
   559→                    "model": model,
   560→                    "prompt": prompt,
   561→                    "size": "1344x768",  # Good for Telegram channel posts
   562→                }
   563→
   564→                async with httpx.AsyncClient(timeout=90.0) as client:
   565→                    response = await client.post(url, headers=headers, json=payload)
   566→
   567→                    if response.status_code == 200:
   568→                        data = response.json()
   569→                        for item in data.get("data", []):
   570→                            if item.get("b64_json"):
   571→                                return base64.b64decode(item["b64_json"])
   572→                            elif item.get("url"):
   573→                                img_resp = await client.get(item["url"])
   574→                                if img_resp.status_code == 200:
   575→                                    return img_resp.content
   576→                    elif response.status_code in (401, 402):
   577→                        self._mark_key_depleted(tier_index)
   578→                        logger.warning(f"Image gen HTTP {response.status_code} via KEY{tier_index}, trying next tier...")
   579→                        continue
   580→                    else:
   581→                        logger.error(f"Image generation error: {response.status_code} {response.text[:300]}")
   582→                        return None
   583→
   584→            except Exception as e:
   585→                logger.error(f"Image generation exception: {e}")
   586→                continue
   587→
   588→        return None

   590→    async def analyze_image(
   591→        self,
   592→        image_url: str = "",
   593→        image_base64: str = "",
   594→        prompt: str = "Опиши подробно что ты видишь на этом изображении.",
   595→        model: str = "openai",
   596→        system_prompt: str = "",
   597→        max_tokens: int = 600,
   598→        temperature: float = 0.7,
   599→    ) -> AIResponse:
   600→        """Analyze an image using vision-capable models.
   601→        Uses dual-key failover for each vision model attempt.
   602→        """
   603→        # Build the content array for vision
   604→        content = [
   605→            {"type": "text", "text": prompt}
   606→        ]
   607→
   608→        if image_url:
   609→            content.append({
   610→                "type": "image_url",
   611→                "image_url": {"url": image_url}
   612→            })
   613→        elif image_base64:
   614→            media_type = "image/jpeg"
   615→            if image_base64.startswith("data:"):
   616→                header, image_base64 = image_base64.split(",", 1)
   617→                if "png" in header:
   618→                    media_type = "image/png"
   619→                elif "webp" in header:
   620→                    media_type = "image/webp"
   621→                elif "gif" in header:
   622→                    media_type = "image/gif"
   623→
   624→            content.append({
   625→                "type": "image_url",
   626→                "image_url": {
   627→                    "url": f"data:{media_type};base64,{image_base64}"
   628→                }
   629→            })
   630→
   631→        messages = []
   632→        if system_prompt:
   633→            messages.append({"role": "system", "content": system_prompt})
   634→
   635→        messages.append({"role": "user", "content": content})
   636→
   637→        # Try primary vision model, then fallbacks
   638→        vision_models_to_try = [model]
   639→        for fallback in VISION_MODELS:
   640→            if fallback != model and not self._is_model_in_cooldown(fallback):
   641→                vision_models_to_try.append(fallback)
   642→
   643→        for vision_model in vision_models_to_try[:5]:  # Try up to 5 models
   644→            try:
   645→                result = await self.chat(
   646→                    messages=messages,
   647→                    model=vision_model,
   648→                    temperature=temperature,
   649→                    max_tokens=max_tokens,
   650→                )
   651→                if not result.error and result.text:
   652→                    return result
   653→                logger.warning(f"Vision model {vision_model} failed, trying next...")
   654→            except Exception as e:
   655→                logger.error(f"Vision error with {vision_model}: {e}")
   656→                continue
   657→
   658→        return AIResponse(
   659→            text="",
   660→            model=model,
   661→            provider=self.name,
   662→            error=True,
   663→            error_message="All vision models failed",
   664→        )

   697→    async def chat_free(
   698→        self,
   699→        messages: List[Dict[str, str]],
   700→        model: str = "openai",
   701→        temperature: float = 0.7,
   702→        max_tokens: int = 2048,
   703→    ) -> AIResponse:
   704→        """Send a chat completion request using FREE Pollinations API.
   705→
   706→        Uses text.pollinations.ai WITHOUT Authorization header.
   707→        This is the fallback when all API keys are depleted.
   708→        Free API may be rate-limited and slower.
   709→        """
   710→        if not self._is_free_api_available():
   711→            return AIResponse(
   712→                text="",
   713→                model=model,
   714→                provider=f"{self.name}-free",
   715→                error=True,
   716→                error_message="Free API unavailable or on cooldown",
   717→            )
   718→
   719→        model = model or DEFAULT_MODEL
   720→
   721→        payload = {
   722→            "model": model,
   723→            "messages": messages,
   724→            "temperature": temperature,
   725→            "max_tokens": max_tokens,
   726→        }
   727→
   728→        try:
   729→            headers = {"Content-Type": "application/json"}
   730→            # NO Authorization header — this is the free API
   731→
   732→            # Use 12s timeout for free API — it's a fallback, should be fast or fail fast.
   733→            # User is already waiting after the paid API attempt failed.
   734→            async with httpx.AsyncClient(timeout=12.0) as client:
   735→                start_time = time.time()
   736→                url = f"{self._free_text_url}/openai/chat/completions"
   737→                response = await client.post(url, headers=headers, json=payload)
   738→                elapsed = time.time() - start_time
   739→
   740→                if response.status_code == 200:
   741→                    data = response.json()
   742→                    text = ""
   743→                    choices = data.get("choices", [])
   744→                    if choices:
   745→                        msg = choices[0].get("message", {})
   746→                        text = msg.get("content", "") or ""
   747→                        if not text:
   748→                            text = msg.get("reasoning_content", "") or ""
   749→
   750→                    if not text:
   751→                        logger.warning(f"Free API: empty response from {model}")
   752→                        return AIResponse(
   753→                            text="",
   754→                            model=model,
   755→                            provider=f"{self.name}-free",
   756→                            error=True,
   757→                            error_message="Empty response from free API",
   758→                        )
   759→
   760→                    logger.info(
   761→                        f"Free API response: model={model}, "
   762→                        f"time={elapsed:.1f}s, length={len(text)}"
   763→                    )
   764→
   765→                    return AIResponse(
   766→                        text=text,
   767→                        model=model,
   768→                        provider=f"{self.name}-free",
   769→                    )
   770→
   771→                elif response.status_code == 429:
   772→                    logger.warning(f"Free API rate limited (429)")
   773→                    self._mark_free_api_cooldown(120.0)  # 2 min cooldown
   774→                    return AIResponse(
   775→                        text="",
   776→                        model=model,
   777→                        provider=f"{self.name}-free",
   778→                        error=True,
   779→                        error_message="Free API rate limited",
   780→                    )
   781→
   782→                else:
   783→                    error_text = response.text[:300]
   784→                    logger.error(f"Free API error: {response.status_code}: {error_text}")
   785→                    self._mark_free_api_cooldown(60.0)
   786→                    return AIResponse(
   787→                        text="",
   788→                        model=model,
   789→                        provider=f"{self.name}-free",
   790→                        error=True,
   791→                        error_message=f"Free API HTTP {response.status_code}: {error_text}",
   792→                    )
   793→
   794→        except httpx.TimeoutException:
   795→            logger.error(f"Free API timeout: model={model}")
   796→            self._mark_free_api_cooldown(30.0)
   797→            return AIResponse(
   798→                text="",
   799→                model=model,
   800→                provider=f"{self.name}-free",
   801→                error=True,
   802→                error_message="Free API timeout",
   803→            )
   804→
   805→        except Exception as e:
   806→            logger.error(f"Free API exception: {e}")
   807→            self._mark_free_api_cooldown(60.0)
   808→            return AIResponse(
   809→                text="",
   810→                model=model,
   811→                provider=f"{self.name}-free",
   812→                error=True,
   813→                error_message=f"Free API exception: {e}",
   814→            )

   918→    def _is_model_in_cooldown(self, model: str) -> bool:
   919→        """Check if a model has recently failed and is in cooldown.
   920→
   921→        IMPORTANT: We do NOT remove models from lists when they fail.
   922→        Pollinations.ai rotates model availability — a failure today
   923→        doesn't mean the model is permanently unavailable.
   924→        """
   925→        if model not in _model_failures:
   926→            return False
   927→        return time.time() - _model_failures[model] < _FAILURE_COOLDOWN

   929→    def _get_available_model(self) -> Optional[str]:
   930→        """Get an available model that's not in cooldown."""
   931→        if not self._is_model_in_cooldown(DEFAULT_MODEL):
   932→            return DEFAULT_MODEL
   933→        for model in FALLBACK_MODELS:
   934→            if not self._is_model_in_cooldown(model):
   935→                return model
   936→        if _model_failures:
   937→            oldest = min(_model_failures, key=_model_failures.get)
   938→            return oldest
   939→        return DEFAULT_MODEL
