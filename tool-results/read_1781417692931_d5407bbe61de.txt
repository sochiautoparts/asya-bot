   163→class ImgTagExtractor(HTMLParser):
   164→    """Extract <img> src URLs from HTML content."""
   165→    def __init__(self):
   166→        super().__init__()
   167→        self.urls: List[str] = []
   168→
   169→    def handle_starttag(self, tag, attrs):
   170→        if tag == "img":
   171→            for attr_name, attr_value in attrs:
   172→                if attr_name == "src" and attr_value:
   173→                    self.urls.append(attr_value)

   176→def _safe_get_dict(obj, key, default=""):
   177→    """Safely get a value from something that might be a dict or not."""
   178→    if isinstance(obj, dict):
   179→        return obj.get(key, default)
   180→    return default

   183→def extract_rss_images(entry) -> List[str]:
   184→    """Extract image URLs from an RSS entry (media_content, enclosures, <img> tags)."""
   185→    urls: List[str] = []
   186→
   187→    # 1. media_content (most RSS feeds use this)
   188→    media_list = getattr(entry, "media_content", [])
   189→    if media_list:
   190→        for media in media_list:
   191→            if isinstance(media, str):
   192→                if _is_image_url(media):
   193→                    urls.append(media)
   194→            elif isinstance(media, dict):
   195→                url = media.get("url", "")
   196→                if url and _is_image_url(url):
   197→                    urls.append(url)
   198→
   199→    # 2. media_thumbnail
   200→    thumbnails = getattr(entry, "media_thumbnail", [])
   201→    if thumbnails:
   202→        for thumb in thumbnails:
   203→            if isinstance(thumb, str):
   204→                if _is_image_url(thumb):
   205→                    urls.append(thumb)
   206→            elif isinstance(thumb, dict):
   207→                url = thumb.get("url", "")
   208→                if url and _is_image_url(url):
   209→                    urls.append(url)
   210→
   211→    # 3. enclosures
   212→    for enclosure in getattr(entry, "enclosures", []):
   213→        if isinstance(enclosure, str):
   214→            if _is_image_url(enclosure):
   215→                urls.append(enclosure)
   216→        elif isinstance(enclosure, dict):
   217→            href = enclosure.get("href", "")
   218→            enc_type = enclosure.get("type", "")
   219→            if href and (enc_type.startswith("image/") or _is_image_url(href)):
   220→                urls.append(href)
   221→
   222→    # 4. Parse HTML in summary/content for <img> tags
   223→    for field_name in ("summary", "description", "content"):
   224→        field_value = getattr(entry, field_name, None)
   225→        if not field_value:
   226→            continue
   227→        html_text = ""
   228→        if isinstance(field_value, str):
   229→            html_text = field_value
   230→        elif isinstance(field_value, dict):
   231→            html_text = field_value.get("value", "")
   232→        elif isinstance(field_value, list):
   233→            # feedparser sometimes returns content as a list of dicts with 'value' key
   234→            for item in field_value:
   235→                if isinstance(item, dict):
   236→                    html_text += item.get("value", "")
   237→                elif isinstance(item, str):
   238→                    html_text += item
   239→        if html_text:
   240→            extractor = ImgTagExtractor()
   241→            try:
   242→                extractor.feed(html_text)
   243→                urls.extend(extractor.urls)
   244→            except Exception:
   245→                pass
   246→
   247→    return _dedupe_urls(urls)
   248→
   249→
   250→def _is_image_url(url: str) -> bool:
   251→    """Check if URL looks like an image."""
   252→    lower = url.lower().split("?")[0]
   253→    return any(lower.endswith(ext) for ext in (
   254→        ".jpg", ".jpeg", ".png", ".webp", ".avif"
   255→    )) or "image" in url.lower() or "/img/" in url.lower() or "/image/" in url.lower()
   256→
   257→
   258→def _dedupe_urls(urls: List[str]) -> List[str]:
   259→    """Remove duplicate URLs while preserving order."""
   260→    seen: Set[str] = set()
   261→    result: List[str] = []
   262→    for url in urls:
   263→        if url not in seen:
   264→            seen.add(url)
   265→            result.append(url)
   266→    return result

   290→def extract_article_images(html: str, base_url: str) -> List[str]:
   291→    """Extract quality image URLs from an article HTML page."""
   292→    urls: List[str] = []
   293→
   294→    # 1. og:image from meta tags
   295→    og_match = re.search(r'<meta[^>]+property=["\']og:image["\'][^>]+content=["\']([^"\']+)', html, re.IGNORECASE)
   296→    if og_match:
   297→        urls.append(og_match.group(1))
   298→
   299→    # 2. twitter:image
   300→    tw_match = re.search(r'<meta[^>]+(?:name|property)=["\']twitter:image["\'][^>]+content=["\']([^"\']+)', html, re.IGNORECASE)
   301→    if tw_match:
   302→        urls.append(tw_match.group(1))
   303→
   304→    # 3. JSON-LD images
   305→    for ld_match in re.finditer(r'<script[^>]+type=["\']application/ld\+json["\'][^>]*>(.*?)</script>', html, re.IGNORECASE | re.DOTALL):
   306→        try:
   307→            ld_data = json.loads(ld_match.group(1))
   308→            for img in _extract_jsonld_images(ld_data):
   309→                urls.append(img)
   310→        except (json.JSONDecodeError, TypeError):
   311→            pass
   312→
   313→    # 4. <img> tags with src attributes
   314→    extractor = ImgTagExtractor()
   315→    try:
   316→        extractor.feed(html)
   317→        urls.extend(extractor.urls)
   318→    except Exception:
   319→        pass
   320→
   321→    # Filter and normalize
   322→    filtered = []
   323→    for url in urls:
   324→        url = _normalize_url(url, base_url)
   325→        if url and not _is_junk_image(url):
   326→            filtered.append(url)
   327→
   328→    return _dedupe_urls(filtered)

   330→
   331→def _extract_jsonld_images(data, depth=0) -> List[str]:
   332→    """Recursively extract image URLs from JSON-LD data."""
   333→    if depth > 5:
   334→        return []
   335→    images = []
   336→    if isinstance(data, dict):
   337→        for key in ("image", "images", "thumbnailUrl", "contentUrl"):
   338→            val = data.get(key)
   339→            if isinstance(val, str):
   340→                images.append(val)
   341→            elif isinstance(val, dict):
   342→                images.extend(_extract_jsonld_images(val, depth + 1))
   343→            elif isinstance(val, list):
   344→                for item in val:
   345→                    if isinstance(item, str):
   346→                        images.append(item)
   347→                    elif isinstance(item, dict):
   348→                        images.extend(_extract_jsonld_images(item, depth + 1))
   349→    elif isinstance(data, list):
   350→        for item in data:
   351→            images.extend(_extract_jsonld_images(item, depth + 1))
   352→    return images

   355→def _normalize_url(url: str, base_url: str) -> Optional[str]:
   356→    """Normalize a URL to absolute form and validate it."""
   357→    if not url or url.startswith("data:") or url.startswith("javascript:"):
   358→        return None
   359→    # Make relative URLs absolute
   360→    if not url.startswith(("http://", "https://")):
   361→        url = urljoin(base_url, url)
   362→    # Clean URL
   363→    url = url.split("#")[0]  # Remove fragment
   364→    # Validate
   365→    parsed = urlparse(url)
   366→    if parsed.scheme not in ("http", "https"):
   367→        return None
   368→    if not parsed.netloc:
   369→        return None
   370→    return url

   373→def _is_junk_image(url: str) -> bool:
   374→    """Check if an image URL is a junk/decorative/icon image."""
   375→    full_url = url.lower()
   376→    path = urlparse(url).path.lower()
   377→    for pattern in JUNK_PATTERNS:
   378→        if pattern.search(full_url) or pattern.search(path):
   379→            return True
   380→    # Too small dimensions in URL
   381→    dim_match = re.search(r"[_\-x](\d{1,4})x(\d{1,4})", full_url)
   382→    if dim_match:
   383→        w, h = int(dim_match.group(1)), int(dim_match.group(2))
   384→        if w < IMAGE_MIN_WIDTH or h < IMAGE_MIN_HEIGHT:
   385→            return True
   386→    return False

   407→def compute_fingerprint(title: str) -> str:
   408→    """Compute a simplified fingerprint for dedup (same as Asya bot)."""
   409→    cleaned = re.sub(
   410→        r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b',
   411→        '', title.lower()
   412→    )
   413→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
   414→    words = cleaned.split()
   415→    return ' '.join(words[:5])

   417→
   418→def fingerprint_matches(fp: str, existing_fps: Set[str]) -> bool:
   419→    """Check if fingerprint matches any existing fingerprint (3+ of first 4 words match)."""
   420→    fp_words = fp.split()[:4]
   421→    for existing in existing_fps:
   422→        ex_words = existing.split()[:4]
   423→        matches = sum(1 for w in fp_words if w in ex_words)
   424→        if matches >= 3:
   425→            return True
   426→    return False

   431→def load_seen_urls() -> Set[str]:
   432→    """Load previously seen URLs from JSON file."""

   434→        try:
   435→            with open(DEDUP_FILE, "r", encoding="utf-8") as f:

   439→                return {url for url, ts in data.items() if ts > cutoff}
   440→        except Exception:
   441→            pass
   442→    return set()

   464→def fetch_rss_feed(client: httpx.Client, source: Dict) -> List[Dict]:
   465→    """Fetch and parse a single RSS feed."""
   466→    name = source["name"]
   467→    url = source["url"]
   468→    lang = source.get("lang", "en")
   469→
   470→    try:
   471→        resp = client.get(url, timeout=15.0)
   472→        if resp.status_code != 200:
   473→            logger.warning(f"[{name}] HTTP {resp.status_code} for {url}")
   474→            return []
   475→
   476→        feed = feedparser.parse(resp.text)
   477→        if feed.bozo and not feed.entries:
   478→            logger.warning(f"[{name}] Parse error: {feed.bozo_exception}")
   479→            return []
   480→
   481→        items = []
   482→        for entry in feed.entries[:25]:  # Take latest 25 per source
   483→            title = getattr(entry, "title", "").strip()
   484→            link = getattr(entry, "link", "").strip()
   485→            summary = getattr(entry, "summary", "") or getattr(entry, "description", "")
   486→            published = getattr(entry, "published", "") or getattr(entry, "updated", "")
   487→
   488→            if not title or not link:
   489→                continue
   490→
   491→            # Convert summary from list/dict to string if needed
   492→            if isinstance(summary, list):
   493→                summary = " ".join(
   494→                    item.get("value", "") if isinstance(item, dict) else str(item)
   495→                    for item in summary
   496→                )
   497→            elif isinstance(summary, dict):
   498→                summary = summary.get("value", "")
   499→
   500→            # Clean summary - strip HTML
   501→            clean_summary = re.sub(r'<[^>]+>', '', str(summary)).strip()[:500]
   502→
   503→            # Extract images from RSS entry
   504→            rss_images = extract_rss_images(entry)
   505→
   506→            items.append({
   507→                "title": title,
   508→                "url": link,
   509→                "summary": clean_summary,
   510→                "published": published,
   511→                "source_name": name,
   512→                "lang": lang,
   513→                "rss_images": rss_images,
   514→            })
   515→
   516→        logger.info(f"[{name}] Fetched {len(items)} items")
   517→        return items
   518→
   519→    except Exception as e:
   520→        logger.warning(f"[{name}] Error: {e}")
   521→        return []

   524→def fetch_article_images(client: httpx.Client, url: str) -> List[str]:
   525→    """Fetch article page and extract images from it."""
   526→    try:
   527→        # Resolve Google News redirect
   528→        real_url = resolve_google_news_url(client, url)
   529→
   530→        resp = client.get(real_url, timeout=REQUEST_TIMEOUT)
   531→        if resp.status_code != 200:
   532→            return []
   533→
   534→        html = resp.text
   535→        if len(html) > 2_000_000:  # Skip huge pages
   536→            html = html[:2_000_000]
   537→
   538→        images = extract_article_images(html, real_url)
   539→        return images[:MAX_PHOTOS_PER_NEWS * 2]  # Get extra, filter later
   540→
   541→    except Exception as e:
   542→        logger.debug(f"Error fetching article images from {url}: {e}")
   543→        return []

   546→def main():
   547→    """Main entry point for the news parser."""
   548→    logger.info("=" * 60)
   549→    logger.info("Auto News Parser — starting...")
   550→    logger.info(f"Max news per run: {MAX_NEWS_PER_RUN}")
   551→    logger.info(f"Max photos per news: {MAX_PHOTOS_PER_NEWS}")
   552→    logger.info("=" * 60)
   553→
   554→    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
   555→
   556→    seen_urls = load_seen_urls()
   557→    existing_news = load_existing_news()
   558→    existing_fingerprints = {compute_fingerprint(item.get("title", "")) for item in existing_news}
   559→
   560→    all_new_items: List[Dict] = []
   561→
   562→    with get_client() as client:
   563→        # ── Phase 1: Fetch all RSS feeds ───────────────────────────────────────
   564→        logger.info("Phase 1: Fetching RSS feeds...")
   565→
   566→        for source in RSS_SOURCES:
   567→            items = fetch_rss_feed(client, source)
   568→            for item in items:
   569→                url = item["url"]
   570→                if url in seen_urls:
   571→                    continue
   572→                fp = compute_fingerprint(item["title"])
   573→                if fingerprint_matches(fp, existing_fingerprints):
   574→                    continue
   575→                all_new_items.append(item)
   576→                seen_urls.add(url)
   577→                existing_fingerprints.add(fp)
   578→            time.sleep(FETCH_DELAY)
   579→
   580→        # ── Phase 2: Google News RSS ──────────────────────────────────────────
   581→        logger.info("Phase 2: Fetching Google News RSS...")
   582→
   583→        for query in GOOGLE_NEWS_QUERIES:
   584→            lang = "ru" if any(ord(c) > 127 for c in query) else "en"
   585→            gl = "RU" if lang == "ru" else "US"
   586→            gn_url = f"https://news.google.com/rss/search?q={quote_plus(query)}&hl={lang}&gl={gl}&ceid={gl}:{lang}"
   587→
   588→            source = {"name": f"GoogleNews_{query[:20]}", "url": gn_url, "lang": lang}
   589→            items = fetch_rss_feed(client, source)
   590→            for item in items:
   591→                url = item["url"]
   592→                if url in seen_urls:
   593→                    continue
   594→                fp = compute_fingerprint(item["title"])
   595→                if fingerprint_matches(fp, existing_fingerprints):
   596→                    continue
   597→                all_new_items.append(item)
   598→                seen_urls.add(url)
   599→                existing_fingerprints.add(fp)
   600→            time.sleep(FETCH_DELAY * 2)  # Extra delay for Google
   601→
   602→        logger.info(f"Total new items after dedup: {len(all_new_items)}")
   603→
   604→        # ── Phase 3: Fetch article images ─────────────────────────────────────
   605→        logger.info("Phase 3: Fetching article images...")
   606→
   607→        processed_items: List[Dict] = []
   608→        for i, item in enumerate(all_new_items[:MAX_NEWS_PER_RUN]):
   609→            article_images = []
   610→
   611→            # Try fetching images from the article page
   612→            if item["url"]:
   613→                article_images = fetch_article_images(client, item["url"])
   614→                time.sleep(FETCH_DELAY)
   615→
   616→            # Combine: RSS images first, then article images
   617→            all_images = item.get("rss_images", []) + article_images
   618→            all_images = _dedupe_urls(all_images)
   619→            all_images = all_images[:MAX_PHOTOS_PER_NEWS]
   620→
   621→            if not all_images:
   622→                logger.info(f"  [{i+1}/{min(len(all_new_items), MAX_NEWS_PER_RUN)}] {item['title'][:60]}... — NO IMAGES")
   623→            else:
   624→                logger.info(f"  [{i+1}/{min(len(all_new_items), MAX_NEWS_PER_RUN)}] {item['title'][:60]}... — {len(all_images)} images")
   625→
   626→            news_item = {
   627→                "title": item["title"],
   628→                "url": item["url"],
   629→                "summary": item.get("summary", ""),
   630→                "published": item.get("published", ""),
   631→                "source": item.get("source_name", ""),
   632→                "lang": item.get("lang", ""),
   633→                "images": all_images,
   634→                "fetched_at": datetime.now(timezone.utc).isoformat(),
   635→            }
   636→            processed_items.append(news_item)
   637→
   638→    # ── Phase 4: Save results ─────────────────────────────────────────────────
   639→    logger.info("Phase 4: Saving results...")
   640→
   641→    # Merge with existing news (newest first), keep max 1000
   642→    merged = processed_items + existing_news
   643→    merged = merged[:1000]
   644→
   645→    with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
   646→        json.dump(merged, f, ensure_ascii=False, indent=2)
   647→
   648→    logger.info(f"Saved {len(processed_items)} new items, total {len(merged)} in {OUTPUT_FILE}")
   649→
   650→    # Save dedup state
   651→    save_seen_urls(seen_urls)
   652→
   653→    # ── Summary ───────────────────────────────────────────────────────────────
   654→    with_photos = sum(1 for item in processed_items if item.get("images"))
   655→    without_photos = len(processed_items) - with_photos
   656→    total_images = sum(len(item.get("images", [])) for item in processed_items)
   657→
   658→    logger.info("=" * 60)
   659→    logger.info(f"PARSING COMPLETE")
   660→    logger.info(f"  New items: {len(processed_items)}")
   661→    logger.info(f"  With photos: {with_photos}")
   662→    logger.info(f"  Without photos: {without_photos}")
   663→    logger.info(f"  Total images: {total_images}")
   664→    logger.info(f"  Total in JSON: {len(merged)}")
   665→    logger.info("=" * 60)
   666→
   667→
   668→if __name__ == "__main__":
   669→    main()
