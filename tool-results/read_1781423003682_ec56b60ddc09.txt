   236→class PollinationsProvider(BaseAIProvider):
   237→    """Pollinations AI provider v5.0 — DUAL KEY + FREE API FALLBACK + 60+ MODELS!
   238→
   239→    Uses gen.pollinations.ai/v1/chat/completions (OpenAI-compatible).
   240→    Falls back to text.pollinations.ai (FREE, no auth) when keys are depleted.
   241→
   242→    FAILOVER CHAIN (within provider):
   243→      1. Try KEY1 first (config.POLLINATIONS_API_KEY)
   244→      2. On 402/401 from KEY1 -> switch to KEY2 (config.POLLINATIONS_API_KEY_2)
   245→      3. On 402/401 from KEY2 -> try FREE API (text.pollinations.ai, no auth)
   246→      4. Depleted keys auto-retry after 600 seconds cooldown
   247→      5. Free API always available but rate-limited and slower
   248→
   249→    IMPORTANT: Models are NEVER removed when unavailable.
   250→    Pollinations.ai rotates model availability — a failure today
   251→    doesn't mean the model is gone. Circuit breaking (5-min cooldown)
   252→    handles temporary failures.
   253→    """
   254→
   255→    def __init__(self):
   256→        super().__init__(
   257→            name="pollinations",
   258→            api_key=config.POLLINATIONS_API_KEY,
   259→            base_url=config.POLLINATIONS_BASE_URL,
   260→        )
   261→        # ── Dual API key storage ──
   262→        # KEY1 = primary key from config
   263→        self._api_key_1: str = config.POLLINATIONS_API_KEY
   264→        self._api_key_2: str = config.POLLINATIONS_API_KEY_2
   265→        # ── Per-key balance depletion tracking ──
   266→        # 0 = active/never depleted; >0 = timestamp when depleted
   267→        self._key1_depleted_at: float = 0.0
   268→        self._key2_depleted_at: float = 0.0
   269→
   270→        # ── Free API endpoints (no auth required) ──
   271→        self._free_text_url: str = config.POLLINATIONS_FREE_TEXT_URL
   272→        self._free_image_url: str = config.POLLINATIONS_FREE_IMAGE_URL
   273→        self._free_api_available: bool = True  # Assume available until proven otherwise
   274→        self._free_api_cooldown_until: float = 0.0  # Timestamp when free API cooldown ends
   275→
   276→    # ── API Key Management ──────────────────────────────────────
   277→
   278→    def _is_key_available(self, key_index: int) -> bool:
   279→        """Check if an API key is available (not depleted or cooldown expired)."""
   280→        if key_index == 1:
   281→            depleted_at = self._key1_depleted_at
   282→            key_val = self._api_key_1
   283→        elif key_index == 2:
   284→            depleted_at = self._key2_depleted_at
   285→            key_val = self._api_key_2
   286→        else:
   287→            return False  # Only KEY1 and KEY2 supported
   288→
   289→        if not key_val:
   290→            return False
   291→
   292→        if depleted_at == 0:
   293→            return True
   294→
   295→        elapsed = time.time() - depleted_at
   296→        if elapsed >= KEY_COOLDOWN:
   297→            if key_index == 1:
   298→                self._key1_depleted_at = 0
   299→            else:
   300→                self._key2_depleted_at = 0
   301→            logger.info(f"API KEY{key_index} cooldown expired after {elapsed:.0f}s — retrying")
   302→            return True
   303→
   304→        return False
   305→
   306→    def _mark_key_depleted(self, key_index: int) -> None:
   307→        """Mark an API key as depleted (balance exhausted)."""
   308→        if key_index == 1:
   309→            self._key1_depleted_at = time.time()
   310→        else:
   311→            self._key2_depleted_at = time.time()
   312→        logger.warning(
   313→            f"API KEY{key_index} depleted (402/401). "
   314→            f"Will auto-retry after {KEY_COOLDOWN}s cooldown."
   315→        )
   316→
   317→    def _get_active_key_tier(self) -> Tuple[str, int]:
   318→        """Determine which key/tier to use for the next request."""
   319→        if self._is_key_available(1):
   320→            return self._api_key_1, 1
   321→        if self._is_key_available(2):
   322→            return self._api_key_2, 2
   323→        return "", 0
   324→
   325→    def _get_key_status_summary(self) -> str:
   326→        """Get a human-readable summary of key statuses."""
   327→        parts = []
   328→        for idx, (key_val, depleted_at) in enumerate([
   329→            (self._api_key_1, self._key1_depleted_at),
   330→            (self._api_key_2, self._key2_depleted_at),
   331→        ], start=1):
   332→            if not key_val:
   333→                parts.append(f"KEY{idx}=not_set")
   334→            elif depleted_at == 0:
   335→                parts.append(f"KEY{idx}=active")
   336→            else:
   337→                remaining = KEY_COOLDOWN - (time.time() - depleted_at)
   338→                parts.append(f"KEY{idx}=depleted({remaining:.0f}s)")
   339→        return ", ".join(parts)
   340→
   341→    def _build_key_tier_list(self) -> List[Tuple[str, int]]:
   342→        """Build ordered list of available key tiers to try."""
   343→        tiers: List[Tuple[str, int]] = []
   344→        if self._is_key_available(1):
   345→            tiers.append((self._api_key_1, 1))
   346→        if self._is_key_available(2):
   347→            tiers.append((self._api_key_2, 2))
   348→        return tiers
   349→
   350→    async def chat(
   351→        self,
   352→        messages: List[Dict[str, str]],
   353→        model: str = "",
   354→        temperature: float = 0.7,
   355→        max_tokens: int = 2048,
   356→        **kwargs,
   357→    ) -> AIResponse:
   358→        """Send a chat completion request to Pollinations API with DUAL-KEY FAILOVER.
   359→
   360→        Strategy:
   361→        1. Try with KEY1 first (if balance available)
   362→        2. On 402/401 from KEY1 -> switch to KEY2
   363→        3. On 402/401 from KEY2 -> return error (router decides fallback)
   364→
   365→        Empty content handling:
   366→        - If a model returns empty content, check reasoning_content field
   367→        - If both are empty, mark as temporary failure (circuit break)
   368→        - Models in EMPTY_CONTENT_MODELS get special logging
   369→        - Models are NEVER removed from lists
   370→        """
   371→        model = model or DEFAULT_MODEL
   372→
   373→        # Check if model is in cooldown from recent failures
   374→        if self._is_model_in_cooldown(model):
   375→            alt = self._get_available_model()
   376→            if alt:
   377→                logger.info(f"Model {model} in cooldown, using {alt}")
   378→                model = alt
   379→
   380→        # Build key tier list: active keys first
   381→        tiers_to_try = self._build_key_tier_list()
   382→
   383→        # If no keys available at all, check cooldown
   384→        if not tiers_to_try:
   385→            return AIResponse(
   386→                text="",
   387→                model=model,
   388→                provider=self.name,
   389→                error=True,
   390→                error_message=f"All API keys depleted [{self._get_key_status_summary()}]",
   391→            )
   392→
   393→        payload = {
   394→            "model": model,
   395→            "messages": messages,
   396→            "temperature": temperature,
   397→            "max_tokens": max_tokens,
   398→        }
   399→
   400→        if kwargs.get("top_p"):
   401→            payload["top_p"] = kwargs["top_p"]
   402→        if kwargs.get("frequency_penalty"):
   403→            payload["frequency_penalty"] = kwargs["frequency_penalty"]
   404→        if kwargs.get("presence_penalty"):
   405→            payload["presence_penalty"] = kwargs["presence_penalty"]
   406→
   407→        last_error = None
   408→        for api_key, tier_index in tiers_to_try:
   409→            try:
   410→                headers = {"Content-Type": "application/json"}
   411→                if api_key:
   412→                    headers["Authorization"] = f"Bearer {api_key}"
   413→
   414→                # Use 10s timeout for chat requests — fast response is critical for user experience.
   415→                # The overall chat timeout in _process_text_message is 45s, and we need to
   416→                # try multiple key tiers and models before that deadline.
   417→                # REDUCED from 15s to 10s — if Pollinations doesn't respond in 10s, 
   418→                # Cloudflare (concurrent) will likely have responded already.
   419→                async with httpx.AsyncClient(timeout=10.0) as client:
   420→                    start_time = time.time()
   421→                    url = f"{self.base_url}/v1/chat/completions"
   422→                    response = await client.post(url, headers=headers, json=payload)
   423→                    elapsed = time.time() - start_time
   424→
   425→                    if response.status_code == 200:
   426→                        data = response.json()
   427→                        text = ""
   428→                        choices = data.get("choices", [])
   429→                        if choices:
   430→                            msg = choices[0].get("message", {})
   431→                            # Primary content
   432→                            text = msg.get("content", "") or ""
   433→                            # Fallback: reasoning_content (some reasoning models
   434→                            # put their output here instead of content)
   435→                            if not text:
   436→                                text = msg.get("reasoning_content", "") or ""
   437→
   438→                        usage = data.get("usage", {})
   439→                        tokens_used = usage.get("total_tokens", 0)
   440→                        actual_model = data.get("model", model)
   441→                        tier_label = f"KEY{tier_index}"
   442→
   443→                        # Handle empty content gracefully
   444→                        if not text:
   445→                            # Check if this is a known empty-content model
   446→                            if model in EMPTY_CONTENT_MODELS:
   447→                                logger.warning(
   448→                                    f"Empty response from {model} (known empty-content model) "
   449→                                    f"via {tier_label} — marking as temporary failure"
   450→                                )
   451→                            else:
   452→                                logger.warning(
   453→                                    f"Empty response from {model} via {tier_label} — "
   454→                                    f"marking as temporary failure"
   455→                                )
   456→                            # Circuit break the model, but do NOT remove it
   457→                            _model_failures[model] = time.time()
   458→                            last_error = f"Empty response from {model}"
   459→                            continue  # Try next key tier
   460→
   461→                        logger.info(
   462→                            f"Pollinations response ({tier_label}): model={actual_model}, "
   463→                            f"tokens={tokens_used}, time={elapsed:.1f}s, "
   464→                            f"length={len(text)}"
   465→                        )
   466→
   467→                        return AIResponse(
   468→                            text=text,
   469→                            model=actual_model,
   470→                            provider=self.name,
   471→                            tokens_used=tokens_used,
   472→                        )
   473→
   474→                    elif response.status_code in (401, 402):
   475→                        # Balance depleted or unauthorized → switch key
   476→                        self._mark_key_depleted(tier_index)
   477→                        last_error = f"HTTP {response.status_code} from KEY{tier_index}"
   478→                        logger.warning(
   479→                            f"HTTP {response.status_code} from {model} via KEY{tier_index} — "
   480→                            f"switching to next key tier"
   481→                        )
   482→                        continue  # Try next key tier
   483→
   484→                    elif response.status_code == 429:
   485→                        # Rate limited — short cooldown
   486→                        logger.warning(f"Rate limited (429) on KEY{tier_index}, model={model}")
   487→                        _model_failures[model] = time.time()
   488→                        # Try next key tier
   489→                        continue
   490→
   491→                    else:
   492→                        error_text = response.text[:500]
   493→                        logger.error(
   494→                            f"Pollinations error: status={response.status_code}, "
   495→                            f"model={model}, error={error_text}"
   496→                        )
   497→                        _model_failures[model] = time.time()
   498→
   499→                        return AIResponse(
   500→                            text="",
   501→                            model=model,
   502→                            provider=self.name,
   503→                            error=True,
   504→                            error_message=f"HTTP {response.status_code}: {error_text}",
   505→                        )
   506→
   507→            except httpx.TimeoutException:
   508→                logger.error(f"Pollinations timeout: model={model}, tier=KEY{tier_index}")
   509→                _model_failures[model] = time.time()
   510→                last_error = f"Timeout from KEY{tier_index}"
   511→                # Mark the key as potentially depleted on timeout too —
   512→                # if it times out, trying the same key again is unlikely to help
   513→                if tier_index == 1 and self._key1_depleted_at == 0:
   514→                    self._key1_depleted_at = time.time() - KEY_COOLDOWN + 60  # Retry in 60s instead of 600s
   515→                elif tier_index == 2 and self._key2_depleted_at == 0:
   516→                    self._key2_depleted_at = time.time() - KEY_COOLDOWN + 60
   517→                continue  # Try next key tier
   518→
   519→            except Exception as e:
   520→                logger.error(f"Pollinations exception: model={model}, tier=KEY{tier_index}, error={e}")
   521→                _model_failures[model] = time.time()
   522→                last_error = str(e)
   523→                continue
   524→
   525→        # ── All key tiers failed → return error, let router decide ──
   526→        key_status = self._get_key_status_summary()
   527→        logger.error(
   528→            f"All API key tiers failed for model {model}. "
   529→            f"Key status: [{key_status}]. Last error: {last_error}. "
   530→            f"Router should decide fallback."
   531→        )
   532→
   533→        return AIResponse(
   534→            text="",
   535→            model=model,
   536→            provider=self.name,
   537→            error=True,
   538→            error_message=f"All API keys depleted/unavailable [{key_status}]. {last_error}",
   539→        )

   541→    async def generate_image(self, prompt: str, model: str = "flux") -> Optional[bytes]:
   542→        """Generate an image using Pollinations image API.
   543→        Uses dual-key failover: KEY1 -> KEY2.
   544→        Returns image bytes or None on failure.
   545→        """
   546→        # Build key tier list
   547→        tiers_to_try = self._build_key_tier_list()
   548→
   549→        if not tiers_to_try:
   550→            logger.warning("No API keys available for image generation")
   551→            return None
   552→
   553→        for api_key, tier_index in tiers_to_try:
   554→            try:
   555→                url = f"{self.base_url}/v1/images/generations"
   556→                headers = {
   557→                    "Content-Type": "application/json",
   558→                    "Authorization": f"Bearer {api_key}",
   559→                }
   560→                payload = {
   561→                    "model": model,
   562→                    "prompt": prompt,
   563→                    "size": "1344x768",  # Good for Telegram channel posts
   564→                }
   565→
   566→                async with httpx.AsyncClient(timeout=90.0) as client:
   567→                    response = await client.post(url, headers=headers, json=payload)
   568→
   569→                    if response.status_code == 200:
   570→                        data = response.json()
   571→                        for item in data.get("data", []):
   572→                            if item.get("b64_json"):
   573→                                return base64.b64decode(item["b64_json"])
   574→                            elif item.get("url"):
   575→                                img_resp = await client.get(item["url"])
   576→                                if img_resp.status_code == 200:
   577→                                    return img_resp.content
   578→                    elif response.status_code in (401, 402):
   579→                        self._mark_key_depleted(tier_index)
   580→                        logger.warning(f"Image gen HTTP {response.status_code} via KEY{tier_index}, trying next tier...")
   581→                        continue
   582→                    else:
   583→                        logger.error(f"Image generation error: {response.status_code} {response.text[:300]}")
   584→                        return None
   585→
   586→            except Exception as e:
   587→                logger.error(f"Image generation exception: {e}")
   588→                continue
   589→
   590→        return None

   592→    async def analyze_image(
   593→        self,
   594→        image_url: str = "",
   595→        image_base64: str = "",
   596→        prompt: str = "Опиши подробно что ты видишь на этом изображении.",
   597→        model: str = "openai",
   598→        system_prompt: str = "",
   599→        max_tokens: int = 600,
   600→        temperature: float = 0.7,
   601→    ) -> AIResponse:
   602→        """Analyze an image using vision-capable models.
   603→        Uses dual-key failover for each vision model attempt.
   604→        """
   605→        # Build the content array for vision
   606→        content = [
   607→            {"type": "text", "text": prompt}
   608→        ]
   609→
   610→        if image_url:
   611→            content.append({
   612→                "type": "image_url",
   613→                "image_url": {"url": image_url}
   614→            })
   615→        elif image_base64:
   616→            media_type = "image/jpeg"
   617→            if image_base64.startswith("data:"):
   618→                header, image_base64 = image_base64.split(",", 1)
   619→                if "png" in header:
   620→                    media_type = "image/png"
   621→                elif "webp" in header:
   622→                    media_type = "image/webp"
   623→                elif "gif" in header:
   624→                    media_type = "image/gif"
   625→
   626→            content.append({
   627→                "type": "image_url",
   628→                "image_url": {
   629→                    "url": f"data:{media_type};base64,{image_base64}"
   630→                }
   631→            })
   632→
   633→        messages = []
   634→        if system_prompt:
   635→            messages.append({"role": "system", "content": system_prompt})
   636→
   637→        messages.append({"role": "user", "content": content})
   638→
   639→        # Try primary vision model, then fallbacks
   640→        vision_models_to_try = [model]
   641→        for fallback in VISION_MODELS:
   642→            if fallback != model and not self._is_model_in_cooldown(fallback):
   643→                vision_models_to_try.append(fallback)
   644→
   645→        for vision_model in vision_models_to_try[:5]:  # Try up to 5 models
   646→            try:
   647→                result = await self.chat(
   648→                    messages=messages,
   649→                    model=vision_model,
   650→                    temperature=temperature,
   651→                    max_tokens=max_tokens,
   652→                )
   653→                if not result.error and result.text:
   654→                    return result
   655→                logger.warning(f"Vision model {vision_model} failed, trying next...")
   656→            except Exception as e:
   657→                logger.error(f"Vision error with {vision_model}: {e}")
   658→                continue
   659→
   660→        return AIResponse(
   661→            text="",
   662→            model=model,
   663→            provider=self.name,
   664→            error=True,
   665→            error_message="All vision models failed",
   666→        )

   699→    async def chat_free(
   700→        self,
   701→        messages: List[Dict[str, str]],
   702→        model: str = "openai",
   703→        temperature: float = 0.7,
   704→        max_tokens: int = 2048,
   705→    ) -> AIResponse:
   706→        """Send a chat completion request using FREE Pollinations API.
   707→
   708→        Uses text.pollinations.ai WITHOUT Authorization header.
   709→        This is the fallback when all API keys are depleted.
   710→        Free API may be rate-limited and slower.
   711→        """
   712→        if not self._is_free_api_available():
   713→            return AIResponse(
   714→                text="",
   715→                model=model,
   716→                provider=f"{self.name}-free",
   717→                error=True,
   718→                error_message="Free API unavailable or on cooldown",
   719→            )
   720→
   721→        model = model or DEFAULT_MODEL
   722→
   723→        payload = {
   724→            "model": model,
   725→            "messages": messages,
   726→            "temperature": temperature,
   727→            "max_tokens": max_tokens,
   728→        }
   729→
   730→        try:
   731→            headers = {"Content-Type": "application/json"}
   732→            # NO Authorization header — this is the free API
   733→
   734→            # Use 12s timeout for free API — it's a fallback, should be fast or fail fast.
   735→            # User is already waiting after the paid API attempt failed.
   736→            async with httpx.AsyncClient(timeout=12.0) as client:
   737→                start_time = time.time()
   738→                url = f"{self._free_text_url}/openai/chat/completions"
   739→                response = await client.post(url, headers=headers, json=payload)
   740→                elapsed = time.time() - start_time
   741→
   742→                if response.status_code == 200:
   743→                    data = response.json()
   744→                    text = ""
   745→                    choices = data.get("choices", [])
   746→                    if choices:
   747→                        msg = choices[0].get("message", {})
   748→                        text = msg.get("content", "") or ""
   749→                        if not text:
   750→                            text = msg.get("reasoning_content", "") or ""
   751→
   752→                    if not text:
   753→                        logger.warning(f"Free API: empty response from {model}")
   754→                        return AIResponse(
   755→                            text="",
   756→                            model=model,
   757→                            provider=f"{self.name}-free",
   758→                            error=True,
   759→                            error_message="Empty response from free API",
   760→                        )
   761→
   762→                    logger.info(
   763→                        f"Free API response: model={model}, "
   764→                        f"time={elapsed:.1f}s, length={len(text)}"
   765→                    )
   766→
   767→                    return AIResponse(
   768→                        text=text,
   769→                        model=model,
   770→                        provider=f"{self.name}-free",
   771→                    )
   772→
   773→                elif response.status_code == 429:
   774→                    logger.warning(f"Free API rate limited (429)")
   775→                    self._mark_free_api_cooldown(120.0)  # 2 min cooldown
   776→                    return AIResponse(
   777→                        text="",
   778→                        model=model,
   779→                        provider=f"{self.name}-free",
   780→                        error=True,
   781→                        error_message="Free API rate limited",
   782→                    )
   783→
   784→                else:
   785→                    error_text = response.text[:300]
   786→                    logger.error(f"Free API error: {response.status_code}: {error_text}")
   787→                    self._mark_free_api_cooldown(60.0)
   788→                    return AIResponse(
   789→                        text="",
   790→                        model=model,
   791→                        provider=f"{self.name}-free",
   792→                        error=True,
   793→                        error_message=f"Free API HTTP {response.status_code}: {error_text}",
   794→                    )
   795→
   796→        except httpx.TimeoutException:
   797→            logger.error(f"Free API timeout: model={model}")
   798→            self._mark_free_api_cooldown(30.0)
   799→            return AIResponse(
   800→                text="",
   801→                model=model,
   802→                provider=f"{self.name}-free",
   803→                error=True,
   804→                error_message="Free API timeout",
   805→            )
   806→
   807→        except Exception as e:
   808→            logger.error(f"Free API exception: {e}")
   809→            self._mark_free_api_cooldown(60.0)
   810→            return AIResponse(
   811→                text="",
   812→                model=model,
   813→                provider=f"{self.name}-free",
   814→                error=True,
   815→                error_message=f"Free API exception: {e}",
   816→            )

   920→    def _is_model_in_cooldown(self, model: str) -> bool:
   921→        """Check if a model has recently failed and is in cooldown.
   922→
   923→        IMPORTANT: We do NOT remove models from lists when they fail.
   924→        Pollinations.ai rotates model availability — a failure today
   925→        doesn't mean the model is permanently unavailable.
   926→        """
   927→        if model not in _model_failures:
   928→            return False
   929→        return time.time() - _model_failures[model] < _FAILURE_COOLDOWN

   931→    def _get_available_model(self) -> Optional[str]:
   932→        """Get an available model that's not in cooldown."""
   933→        if not self._is_model_in_cooldown(DEFAULT_MODEL):
   934→            return DEFAULT_MODEL
   935→        for model in FALLBACK_MODELS:
   936→            if not self._is_model_in_cooldown(model):
   937→                return model
   938→        if _model_failures:
   939→            oldest = min(_model_failures, key=_model_failures.get)
   940→            return oldest
   941→        return DEFAULT_MODEL
