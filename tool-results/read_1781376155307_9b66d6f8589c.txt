    15→DB_PATH = config.DB_PATH
    16→
    17→# ── Database connection helper ──────────────────────────────────────────────────
    18→# WAL mode + busy_timeout fix "database is locked" errors when multiple
    19→# async tasks (news fetcher, poster, bot handlers) access the DB concurrently.
    20→
    21→from contextlib import asynccontextmanager
    22→
    23→# Database lock retry settings
    24→_DB_BUSY_TIMEOUT = 10000  # 10 seconds — wait longer for locks
    25→_DB_MAX_RETRIES = 3
    26→_DB_RETRY_DELAY = 0.5  # seconds between retries
    27→
    28→@asynccontextmanager
    29→async def _connect_db():
    30→    """Open a DB connection with WAL mode and busy_timeout.
    31→
    32→    WAL mode allows concurrent reads while a write is in progress.
    33→    busy_timeout=10000 makes SQLite wait up to 10 seconds for a lock
    34→    instead of raising OperationalError immediately.
    35→
    36→    Usage:  async with _connect_db() as db:  ...
    37→    """
    38→    db = await aiosqlite.connect(DB_PATH)
    39→    await db.execute("PRAGMA journal_mode=WAL")
    40→    await db.execute(f"PRAGMA busy_timeout={_DB_BUSY_TIMEOUT}")
    41→    await db.execute("PRAGMA synchronous=NORMAL")
    42→    await db.execute("PRAGMA cache_size=-64000")  # 64MB cache for better performance
    43→    db.row_factory = aiosqlite.Row
    44→    try:
    45→        yield db
    46→    finally:
    47→        await db.close()
    48→

   192→async def get_or_create_user(user_id: int, username: str = "", first_name: str = "",
   193→                              last_name: str = "", language_code: str = "ru") -> Dict[str, Any]:
   194→    """Get or create a user record. Returns user dict.
   195→    
   196→    Handles race condition where two concurrent requests try to INSERT
   197→    the same user_id — uses INSERT OR IGNORE + SELECT instead of
   198→    INSERT + catch IntegrityError.
   199→    """
   200→    now = time.time()
   201→    async with _connect_db() as db:
   202→        db.row_factory = aiosqlite.Row
   203→        async with db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) as cursor:
   204→            row = await cursor.fetchone()
   205→
   206→        if row is None:
   207→            try:
   208→                await db.execute(
   209→                    """INSERT INTO users (user_id, username, first_name, last_name, language_code,
   210→                       first_seen, last_seen, message_count)
   211→                       VALUES (?, ?, ?, ?, ?, ?, ?, 1)""",
   212→                    (user_id, username, first_name, last_name, language_code, now, now),
   213→                )
   214→                await db.commit()
   215→                return {
   216→                    "user_id": user_id, "username": username, "first_name": first_name,
   217→                    "last_name": last_name, "language_code": language_code,
   218→                    "is_blocked": 0, "is_admin": 0, "first_seen": now,
   219→                    "last_seen": now, "message_count": 1, "chat_mode": "normal",
   220→                }
   221→            except aiosqlite.IntegrityError:
   222→                # Race condition: another request already inserted this user
   223→                # Re-fetch the existing row
   224→                async with db.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) as cursor:
   225→                    row = await cursor.fetchone()
   226→                if row:
   227→                    # Still update last_seen
   228→                    await db.execute(
   229→                        """UPDATE users SET username=?, first_name=?, last_name=?,
   230→                           last_seen=?, message_count=message_count+1 WHERE user_id=?""",
   231→                        (username, first_name, last_name, now, user_id),
   232→                    )
   233→                    await db.commit()
   234→                    return dict(row)
   235→                # Shouldn't happen, but fallback
   236→                return {
   237→                    "user_id": user_id, "username": username, "first_name": first_name,
   238→                    "last_name": last_name, "language_code": language_code,
   239→                    "is_blocked": 0, "is_admin": 0, "first_seen": now,
   240→                    "last_seen": now, "message_count": 1, "chat_mode": "normal",
   241→                }
   242→        else:
   243→            await db.execute(
   244→                """UPDATE users SET username=?, first_name=?, last_name=?,
   245→                   last_seen=?, message_count=message_count+1 WHERE user_id=?""",
   246→                (username, first_name, last_name, now, user_id),
   247→            )
   248→            await db.commit()
   249→            return dict(row)

   345→async def add_news_item(source: str, title: str, url: str, summary: str = "",
   346→                         published: float = 0, category: str = "auto", lang: str = "ru",
   347→                         image_urls: list = None, full_text: str = "", resolved_url: str = "") -> bool:
   348→    """Add a news item. Returns True if new, False if duplicate.
   349→    
   350→    image_urls: list of image URL strings extracted from the RSS feed.
   351→    full_text: full article text (from article_fetcher or RSS content field).
   352→    resolved_url: real article URL after Google News redirect resolution.
   353→    """
   354→    now = time.time()
   355→    image_urls_json = json.dumps(image_urls or [], ensure_ascii=False)
   356→    try:
   357→        async with _connect_db() as db:
   358→            await db.execute(
   359→                """INSERT INTO news_items (source, title, url, summary, published, fetched_at, category, lang, image_urls, full_text, resolved_url)
   360→                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
   361→                (source, title, url, summary, published, now, category, lang, image_urls_json, full_text, resolved_url),
   362→            )
   363→            await db.commit()
   364→            return True
   365→    except aiosqlite.IntegrityError:
   366→        # If already exists, update image_urls and full_text if we have new ones
   367→        updates = []
   368→        params = []
   369→        if image_urls:
   370→            updates.append("image_urls = ?")
   371→            params.append(image_urls_json)
   372→        if full_text:
   373→            updates.append("full_text = ?")
   374→            params.append(full_text)
   375→        if resolved_url:
   376→            updates.append("resolved_url = ?")
   377→            params.append(resolved_url)
   378→        if updates:
   379→            try:
   380→                async with _connect_db() as db:
   381→                    params.append(url)
   382→                    await db.execute(
   383→                        f"UPDATE news_items SET {', '.join(updates)} WHERE url = ? AND (image_urls = '[]' OR full_text = '' OR resolved_url = '')",
   384→                        params,
   385→                    )
   386→                    await db.commit()
   387→            except Exception:
   388→                pass
   389→        return False

   392→async def get_unposted_news(limit: int = 10, category: str = "") -> List[Dict[str, Any]]:
   393→    """Get unposted news items, ordered by publish date.
   394→    
   395→    Parses image_urls JSON field back into a Python list.
   396→    """
   397→    async with _connect_db() as db:
   398→        db.row_factory = aiosqlite.Row
   399→        if category:
   400→            async with db.execute(
   401→                """SELECT * FROM news_items WHERE is_posted = 0 AND category = ?
   402→                   ORDER BY published DESC LIMIT ?""",
   403→                (category, limit),
   404→            ) as cursor:
   405→                rows = await cursor.fetchall()
   406→        else:
   407→            async with db.execute(
   408→                """SELECT * FROM news_items WHERE is_posted = 0
   409→                   ORDER BY published DESC LIMIT ?""",
   410→                (limit,),
   411→            ) as cursor:
   412→                rows = await cursor.fetchall()
   413→        result = []
   414→        for r in rows:
   415→            item = dict(r)
   416→            # Parse image_urls from JSON string to list
   417→            if "image_urls" in item and isinstance(item["image_urls"], str):
   418→                try:
   419→                    item["image_urls"] = json.loads(item["image_urls"])
   420→                except (json.JSONDecodeError, TypeError):
   421→                    item["image_urls"] = []
   422→            elif "image_urls" not in item:
   423→                item["image_urls"] = []
   424→            result.append(item)
   425→        return result

   622→async def is_duplicate_post(title: str, content: str = "", hours: int = 48,
   623→                            source_url: str = "") -> bool:
   624→    """Check if a post with similar title or content was recently posted.
   625→
   626→    Checks (in order):
   627→    1. Source URL match (same article from different RSS feeds)
   628→    2. Exact title hash match (same title, possibly different source)
   629→    3. Title prefix match (first 30 chars — catches reworded titles about same topic)
   630→    4. Keyword overlap match (extract key nouns — catches paraphrased titles about same subject)
   631→       Also normalizes car brand names (BMW/Bayerische, LADA/ВАЗ, etc.)
   632→    5. Content hash match (same content, possibly different title)
   633→    6. NORMALIZED CORE MATCH — extract core words (brand + model + event) and compare
   634→       This catches "BMW представила новый X5" vs "Новый BMW X5 2026: первые детали"
   635→
   636→    Args:
   637→        title: News item title to check
   638→        content: Post content to check (optional)
   639→        hours: How many hours back to check (default 48h)
   640→        source_url: Original article URL for URL-based dedup (optional)
   641→
   642→    Returns:
   643→        True if a similar post was found (DUPLICATE), False if unique
   644→    """
   645→    cutoff = time.time() - (hours * 3600)
   646→    title_hash = _make_title_hash(title)
   647→    title_prefix = _normalize_text(title)[:30]
   648→
   649→    async with _connect_db() as db:
   650→        # Check 1: Source URL match — same article from different RSS feeds
   651→        # Compares the path component of URLs (ignores www/http/https differences)
   652→        if source_url:
   653→            url_fingerprint = _make_url_fingerprint(source_url)
   654→            if url_fingerprint:
   655→                async with db.execute(
   656→                    "SELECT source_url FROM channel_posts WHERE created_at >= ?",
   657→                    (cutoff,),
   658→                ) as cursor:
   659→                    rows = await cursor.fetchall()
   660→                    for row in rows:
   661→                        existing_url = row[0] if row else ""
   662→                        if existing_url and _make_url_fingerprint(existing_url) == url_fingerprint:
   663→                            return True
   664→
   665→        # Check 2: Exact title hash match
   666→        async with db.execute(
   667→            "SELECT COUNT(*) FROM post_fingerprints WHERE title_hash = ? AND created_at >= ?",
   668→            (title_hash, cutoff),
   669→        ) as cursor:
   670→            row = await cursor.fetchone()
   671→            if row and row[0] > 0:
   672→                return True
   673→
   674→        # Check 3: Title prefix match (catches reworded titles about same topic)
   675→        if len(title_prefix) >= 10:
   676→            async with db.execute(
   677→                "SELECT COUNT(*) FROM post_fingerprints WHERE title_prefix = ? AND created_at >= ?",
   678→                (title_prefix, cutoff),
   679→            ) as cursor:
   680→                row = await cursor.fetchone()
   681→                if row and row[0] > 0:
   682→                    return True
   683→
   684→        # Check 4: Keyword overlap — extract significant words from title and compare
   685→        # Also normalizes car brand names (BMW/Bayerische, LADA/ВАЗ, etc.)
   686→        title_keywords = _extract_title_keywords(title)
   687→        title_keywords_normalized = _normalize_brand_keywords(title_keywords)
   688→        if len(title_keywords_normalized) >= 2:
   689→            async with db.execute(
   690→                "SELECT title_prefix FROM post_fingerprints WHERE created_at >= ?",
   691→                (cutoff,),
   692→            ) as cursor:
   693→                rows = await cursor.fetchall()
   694→                for row in rows:
   695→                    existing_prefix = row[0] if row else ""
   696→                    if not existing_prefix:
   697→                        continue
   698→                    existing_keywords = _extract_title_keywords(existing_prefix)
   699→                    existing_keywords_normalized = _normalize_brand_keywords(existing_keywords)
   700→                    if not existing_keywords_normalized:
   701→                        continue
   702→                    # Calculate keyword overlap ratio (using normalized brands)
   703→                    common = title_keywords_normalized & existing_keywords_normalized
   704→                    if len(common) >= 2:
   705→                        overlap_ratio = len(common) / min(len(title_keywords_normalized), len(existing_keywords_normalized))
   706→                        if overlap_ratio >= 0.50:
   707→                            return True
   708→                    # Also check overlap between raw keywords for better matching
   709→                    raw_common = title_keywords & existing_keywords
   710→                    if len(raw_common) >= 3:
   711→                        raw_overlap_ratio = len(raw_common) / min(len(title_keywords), len(existing_keywords)) if min(len(title_keywords), len(existing_keywords)) > 0 else 0
   712→                        if raw_overlap_ratio >= 0.45:
   713→                            return True
   714→
   715→        # ── Check 6: NORMALIZED CORE MATCH ──
   716→        # Extract core content words (brands, models, numbers, events) and compare
   717→        # This catches cases where titles are completely reworded but about same event
   718→        core_words = _extract_core_words(title)
   719→        if len(core_words) >= 2:
   720→            async with db.execute(
   721→                "SELECT title_prefix FROM post_fingerprints WHERE created_at >= ?",
   722→                (cutoff,),
   723→            ) as cursor:
   724→                rows = await cursor.fetchall()
   725→                for row in rows:
   726→                    existing_prefix = row[0] if row else ""
   727→                    if not existing_prefix:
   728→                        continue
   729→                    existing_core = _extract_core_words(existing_prefix)
   730→                    if len(existing_core) < 2:
   731→                        continue
   732→                    core_common = core_words & existing_core
   733→                    # If 2+ core words match (e.g., "BMW" + "X5" or "Tesla" + "recalls"),
   734→                    # it's almost certainly the same event
   735→                    if len(core_common) >= 2:
   736→                        return True
   737→
   738→        # Check 5: Content hash match (if content provided)
   739→        if content:
   740→            content_hash = _make_content_hash(content)
   741→            async with db.execute(
   742→                "SELECT COUNT(*) FROM post_fingerprints WHERE content_hash = ? AND created_at >= ?",
   743→                (content_hash, cutoff),
   744→            ) as cursor:
   745→                row = await cursor.fetchone()
   746→                if row and row[0] > 0:
   747→                    return True
   748→
   749→    return False

   977→def _make_url_fingerprint(url: str) -> str:
   978→    """Create a fingerprint of a URL for deduplication.
   979→    
   980→    Strips protocol, www prefix, trailing slashes, and query parameters
   981→    so that https://www.example.com/article/123 and http://example.com/article/123/
   982→    are treated as the same URL.
   983→    """
   984→    from urllib.parse import urlparse
   985→    try:
   986→        parsed = urlparse(url.strip())
   987→        # Use path only (ignore protocol, www, query, fragment)
   988→        path = parsed.path.rstrip('/')
   989→        domain = parsed.netloc.lower()
   990→        # Remove www prefix
   991→        if domain.startswith('www.'):
   992→            domain = domain[4:]
   993→        fingerprint_str = f"{domain}{path}"
   994→        if not fingerprint_str or len(fingerprint_str) < 5:
   995→            return ""

  1047→async def load_topic_registry() -> Dict:
  1048→    """Load topic registry from DB into memory. Returns dict of entity_key -> {first_seen, last_posted, post_count, titles}.
  1049→
  1050→    Only loads topics that are NOT expired (within 72h of last_posted).
  1051→    This is called at startup to restore the registry after restart.
  1052→    """
  1053→    max_age = 48 * 3600  # 48 hours — faster topic cycling
  1054→    cutoff = time.time() - max_age
  1055→    registry = {}
  1056→    try:
  1057→        async with _connect_db() as db:
  1058→            async with db.execute(
  1059→                "SELECT entity_key, first_seen, last_posted, post_count, titles FROM topic_registry WHERE last_posted >= ?",
  1060→                (cutoff,),
  1061→            ) as cursor:
  1062→                rows = await cursor.fetchall()
  1063→                for row in rows:
  1064→                    entity_key, first_seen, last_posted, post_count, titles_json = row
  1065→                    try:
  1066→                        titles = json.loads(titles_json) if titles_json else []
  1067→                    except (json.JSONDecodeError, TypeError):
  1068→                        titles = []
  1069→                    registry[entity_key] = {
  1070→                        "first_seen": first_seen,
  1071→                        "last_posted": last_posted,
  1072→                        "post_count": post_count,
  1073→                        "titles": titles,
  1074→                    }
  1075→    except Exception as e:
  1076→        # Table might not exist yet on first run — that's OK
  1077→        import logging
  1078→        logging.getLogger("asya.database").debug(f"Could not load topic registry: {e}")
  1079→    return registry

  1083→                                  post_count: int, titles: list) -> None:
  1084→    """Save or update a single topic in the DB registry."""
  1085→    if not entity_key:
  1086→        return
  1087→    titles_json = json.dumps(titles[-20:], ensure_ascii=False)  # Keep last 20 titles

  1090→            await db.execute(
  1091→                """INSERT INTO topic_registry (entity_key, first_seen, last_posted, post_count, titles)
  1092→                   VALUES (?, ?, ?, ?, ?)
  1093→                   ON CONFLICT(entity_key) DO UPDATE SET
  1094→                       last_posted = excluded.last_posted,
  1095→                       post_count = excluded.post_count,
  1096→                       titles = excluded.titles""",
  1097→                (entity_key, first_seen, last_posted, post_count, titles_json),
  1098→            )
  1099→            await db.commit()
  1100→    except Exception as e:
  1101→        import logging
  1102→        logging.getLogger("asya.database").debug(f"Could not save topic to registry: {e}")
