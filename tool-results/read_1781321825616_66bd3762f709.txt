    21→# Stores simplified title fingerprints to prevent similar news from being added
    22→_recent_fingerprints: set = set()
    23→
    24→
    25→def _compute_fingerprint(title: str) -> str:
    26→    """Compute a simplified fingerprint for dedup.
    27→    Remove articles, lowercase, take first 5 words.
    28→    """
    29→    import re
    30→    # Remove common articles/prepositions
    31→    cleaned = re.sub(r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b', '', title.lower())
    32→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
    33→    words = cleaned.split()
    34→    return ' '.join(words[:5])
    35→
    36→
    37→def _fingerprint_matches_existing(fingerprint: str) -> bool:
    38→    """Check if a fingerprint matches any recently used fingerprint (edit distance on first 4 words)."""
    39→    fp_words = fingerprint.split()[:4]
    40→    for existing in _recent_fingerprints:
    41→        ex_words = existing.split()[:4]
    42→        # Count matching words
    43→        matches = sum(1 for w in fp_words if w in ex_words)
    44→        if matches >= 3:
    45→            return True
    46→    return False

   251→# ── RSS Fetcher ────────────────────────────────────────────────────────────────
   252→
   253→async def fetch_rss(source: NewsSource) -> List[Dict]:
   254→    """
   255→    Fetch and parse an RSS feed.
   256→    Returns list of news items.
   257→    """
   258→    items = []
   259→    try:
   260→        async with httpx.AsyncClient(
   261→            timeout=20.0,
   262→            follow_redirects=True,
   263→            headers={
   264→                "User-Agent": "Mozilla/5.0 (compatible; AsyaBot/1.0; +https://t.me/asiaexp_bot)",
   265→                "Accept": "application/rss+xml, application/xml, text/xml, application/atom+xml, */*",
   266→            },
   267→        ) as client:
   268→            response = await client.get(source.url)
   269→
   270→            if response.status_code != 200:
   271→                logger.warning(f"RSS {source.name} returned {response.status_code}")
   272→                return items
   273→
   274→            # feedparser can parse from string
   275→            feed = feedparser.parse(response.text)
   276→
   277→            for entry in feed.entries[:20]:  # Limit per source
   278→                title = entry.get("title", "").strip()
   279→                url = entry.get("link", "").strip()
   280→
   281→                if not title or not url:
   282→                    continue
   283→
   284→                # Get summary/description
   285→                summary = ""
   286→                if entry.get("summary"):
   287→                    summary = _clean_html(entry["summary"])
   288→                elif entry.get("description"):
   289→                    summary = _clean_html(entry["description"])
   290→
   291→                # Truncate summary
   292→                if len(summary) > 500:
   293→                    summary = summary[:500] + "..."
   294→
   295→                # Get publish date
   296→                published = 0.0
   297→                if entry.get("published_parsed"):
   298→                    try:
   299→                        published = time.mktime(entry["published_parsed"])
   300→                    except (ValueError, TypeError, OverflowError):
   301→                        published = time.time()
   302→                elif entry.get("updated_parsed"):
   303→                    try:
   304→                        published = time.mktime(entry["updated_parsed"])
   305→                    except (ValueError, TypeError, OverflowError):
   306→                        published = time.time()
   307→                else:
   308→                    published = time.time()
   309→
   310→                # Extract image URLs from RSS entry
   311→                image_urls = _extract_entry_images(entry)
   312→
   313→                items.append({
   314→                    "source": source.name,
   315→                    "title": title,
   316→                    "url": url,
   317→                    "summary": summary,
   318→                    "published": published,
   319→                    "category": source.category,
   320→                    "lang": source.lang,
   321→                    "image_urls": image_urls,
   322→                    "rss_entry": entry,  # Raw feedparser entry for content:encoded image extraction
   323→                })
   324→
   325→            logger.info(f"Fetched {len(items)} items from {source.name}")
   326→
   327→    except httpx.TimeoutException:
   328→        logger.warning(f"RSS timeout: {source.name}")
   329→    except Exception as e:
   330→        logger.error(f"RSS fetch error for {source.name}: {e}")
   331→
   332→    return items

   337→def is_auto_relevant(item: Dict, lang: str = "ru") -> bool:
   338→    """Check if a news item is relevant to the auto topic."""
   339→    text = f"{item.get('title', '')} {item.get('summary', '')}".lower()
   340→
   341→    if lang == "ru" or lang == "":
   342→        keywords = AUTO_KEYWORDS_RU
   343→    elif lang == "en":
   344→        keywords = AUTO_KEYWORDS_EN
   345→    elif lang == "de":
   346→        keywords = AUTO_KEYWORDS_DE
   347→    else:
   348→        keywords = AUTO_KEYWORDS_RU + AUTO_KEYWORDS_EN
   349→
   350→    # Check if any keyword matches
   351→    for kw in keywords:
   352→        if kw.lower() in text:
   353→            return True
   354→
   355→    return False

   358→def is_blocked_topic(item: Dict, lang: str = "ru") -> bool:
   359→    """Check if a news item contains blocked topics (politics, war, etc.).
   360→    
   361→    Returns True if the item should be REJECTED.
   362→    This is a hard filter — items matching these keywords never enter the DB.
   363→    """
   364→    text = f"{item.get('title', '')} {item.get('summary', '')}".lower()
   365→    
   366→    # Check Russian blocklist (always checked for Russian and general sources)
   367→    for kw in BLOCK_KEYWORDS_RU:
   368→        if kw.lower() in text:
   369→            logger.info(f"Blocked news (RU keyword '{kw}'): {item.get('title', '')[:60]}")
   370→            return True
   371→    
   372→    # Check English blocklist (for international sources)
   373→    if lang == "en":
   374→        for kw in BLOCK_KEYWORDS_EN:
   375→            if kw.lower() in text:
   376→                logger.info(f"Blocked news (EN keyword '{kw}'): {item.get('title', '')[:60]}")
   377→                return True
   378→    
   379→    return False

   483→async def fetch_all_news() -> int:
   484→    """
   485→    Fetch news — WEB SEARCH FIRST, then RSS as fallback/supplement.
   486→    Returns the number of new items added.
   487→    
   488→    Phase 1: Web search for automotive news (broad queries, primary source)
   489→    Phase 2: RSS feeds (fallback/supplement)
   490→    Phase 3: Global RSS sources (additional international coverage)
   491→    Phase 4: Google News RSS batch (broad query-based coverage)
   492→    
   493→    All existing filtering is preserved (auto-relevance, political block, dedup).
   494→    """
   495→    total_new = 0
   496→
   497→    # DO NOT reset fingerprint set every cycle!
   498→    # Previously this caused the same news to be re-added across cycles.
   499→    # The _recent_fingerprints set now persists across fetch cycles within the process.
   500→    # DB-level dedup (is_duplicate_post) provides cross-restart protection.
   501→
   502→    # ── Phase 1: Web Search FIRST (primary source) ──
   503→    logger.info("Phase 1: Web search for automotive news")
   504→    web_search_queries = [
   505→        "automotive industry news today",
   506→        "new car models 2026",
   507→        "electric vehicle updates",
   508→        "автомобильные новости России",
   509→        "новые модели авто 2026",
   510→        "car recalls and safety",
   511→        "auto show reveals 2026",
   512→        "автоновости сегодня",
   513→        "electric vehicle news latest",
   514→        "car industry updates",
   515→    ]
   516→    try:
   517→        from bot.content_engine import search_auto_news
   518→        # search_auto_news already uses multiple rotated queries
   519→        search_items = await search_auto_news()
   520→        for item in search_items:
   521→            # Apply same filters as RSS
   522→            if is_blocked_topic(item, item.get("lang", "en")):
   523→                continue
   524→            if not is_auto_relevant(item, item.get("lang", "en")):
   525→                continue
   526→            if await is_duplicate_post(item["title"], hours=48):
   527→                continue
   528→            fp = _compute_fingerprint(item["title"])
   529→            if _fingerprint_matches_existing(fp):
   530→                continue
   531→
   532→            added = await add_news_item(
   533→                source=item["source"],
   534→                title=item["title"],
   535→                url=item["url"],
   536→                summary=item.get("summary", ""),
   537→                published=item.get("published", time.time()),
   538→                category=item.get("category", "auto"),
   539→                lang=item.get("lang", "en"),
   540→                image_urls=item.get("image_urls", []),
   541→            )
   542→            if added:
   543→                total_new += 1
   544→                _recent_fingerprints.add(fp)
   545→                logger.info(f"Web search added: {item['title'][:60]}")
   546→    except Exception as e:
   547→        logger.warning(f"Web search phase failed: {e}")
   548→
   549→    logger.info(f"Phase 1 (web search) complete: {total_new} new items")
   550→
   551→    # ── Phase 2: RSS Feeds (fallback/supplement) ──

   556→

   560→                    continue
   561→
   562→                # Filter for auto relevance (ALL sources checked — even "auto" feeds can carry non-auto items)
   563→                if not is_auto_relevant(item, source.lang):
   564→                    continue
   565→
   566→                # Filter out news with titles similar to recently posted items
   567→                if await is_duplicate_post(item["title"], hours=48):
   568→                    logger.debug(f"Skipping duplicate news title: {item['title'][:60]}")
   569→                    continue
   570→
   571→                # Fingerprint-based dedup: skip if similar to recently added news
   572→                fp = _compute_fingerprint(item["title"])
   573→                if _fingerprint_matches_existing(fp):
   574→                    logger.debug(f"Skipping similar news (fingerprint match): {item['title'][:60]}")
   575→                    continue
   576→
   577→                # Add to database (dedup by URL)

   579→                    source=item["source"],

   611→                source=item["source"],

   614→                summary=item.get("summary", ""),
   615→                published=item.get("published", time.time()),

   618→                image_urls=item.get("image_urls", []),
   619→            )
   620→            if added:
   621→                total_new += 1
   622→                _recent_fingerprints.add(fp)

   705→# ── Image extraction from RSS entries ──────────────────────────────────────────
   706→
   707→# Image extensions that are valid for Telegram posts
   708→_VALID_IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".gif"}
   709→# Domains to skip (icons, trackers, tiny thumbnails)
   710→_SKIP_IMAGE_DOMAINS = {
   711→    "feeds.feedburner.com", "feedburner.google.com",
   712→    "pixel.wp.com", "stats.wordpress.com",
   713→    "i0.wp.com", "i1.wp.com", "i2.wp.com",
   714→}
   715→# Minimum image URL length (filter out tiny 1x1 tracking pixels etc.)
   716→_MIN_IMAGE_URL_LEN = 30
   717→
   718→
   719→def _extract_entry_images(entry) -> List[str]:
   720→    """Extract image URLs from a feedparser entry.
   721→    
   722→    Checks multiple locations where RSS/Atom feeds store images:
   723→    - media_content (Media RSS)
   724→    - enclosures
   725→    - media_thumbnail
   726→    - links with image type
   727→    - content/summary HTML <img> tags
   728→    
   729→    Returns list of image URLs (up to 10).
   730→    """
   731→    images = []
   732→    seen = set()
   733→
   734→    def _add_image(url: str):
   735→        """Add image URL if valid and not already seen."""
   736→        if not url or len(url) < _MIN_IMAGE_URL_LEN:
   737→            return
   738→        # Normalize URL
   739→        url = url.strip()
   740→        if url.startswith("//"):
   741→            url = "https:" + url
   742→        # Skip tracking pixels and tiny icons
   743→        from urllib.parse import urlparse
   744→        try:
   745→            parsed = urlparse(url)
   746→            domain = parsed.netloc.lower()
   747→            if any(skip in domain for skip in _SKIP_IMAGE_DOMAINS):
   748→                return
   749→            # Skip very small-looking URLs (often icons/pixels)
   750→            path_lower = parsed.path.lower()
   751→            if any(kw in path_lower for kw in ["icon", "avatar", "logo", "favicon", "pixel", "badge", "button", "banner"]):
   752→                return
   753→        except Exception:
   754→            return
   755→        if url not in seen:
   756→            seen.add(url)
   757→            images.append(url)
   758→
   759→    # 1. media_content (Media RSS extension — most reliable)
   760→    media_content = entry.get("media_content", [])
   761→    for media in media_content:
   762→        url = media.get("url", "")
   763→        media_type = media.get("type", "")
   764→        if url and ("image" in media_type or _has_image_ext(url)):
   765→            _add_image(url)
   766→
   767→    # 2. enclosures
   768→    for enclosure in entry.get("enclosures", []):
   769→        url = enclosure.get("href", "") or enclosure.get("url", "")
   770→        enc_type = enclosure.get("type", "")
   771→        if url and ("image" in enc_type or _has_image_ext(url)):
   772→            _add_image(url)
   773→
   774→    # 3. media_thumbnail
   775→    for thumb in entry.get("media_thumbnail", []):
   776→        url = thumb.get("url", "")
   777→        if url:
   778→            _add_image(url)
   779→
   780→    # 4. links with image rel or type
   781→    for link in entry.get("links", []):
   782→        link_type = link.get("type", "")
   783→        link_rel = link.get("rel", "")
   784→        url = link.get("href", "")
   785→        if url and ("image" in link_type or link_rel == "enclosure"):
   786→            _add_image(url)
   787→
   788→    # 5. Extract <img> tags from content/summary HTML
   789→    for field_name in ["content", "summary", "description"]:
   790→        content_value = entry.get(field_name, [])
   791→        if isinstance(content_value, list):
   792→            # feedparser sometimes returns list of dicts
   793→            for item in content_value:
   794→                html = item.get("value", "") if isinstance(item, dict) else str(item)
   795→                _extract_img_urls(html, _add_image)
   796→        elif isinstance(content_value, str):
   797→            _extract_img_urls(content_value, _add_image)
   798→
   799→    return images[:10]  # Up to 10 candidate images per news item (Telegram mediagroup limit)

   802→def _has_image_ext(url: str) -> bool:
   803→    """Check if URL ends with an image extension."""
   804→    from urllib.parse import urlparse
   805→    try:
   806→        path = urlparse(url).path.lower()
   807→        # Remove query-like suffixes
   808→        path = path.split("?")[0].split("#")[0]
   809→        return any(path.endswith(ext) for ext in _VALID_IMAGE_EXTENSIONS)
   810→    except Exception:
   811→        return False

   814→def _extract_img_urls(html: str, callback):
   815→    """Extract src attributes from <img> tags in HTML."""
   816→    if not html:
   817→        return
   818→    for match in re.finditer(r'<img[^>]+src=["\']([^"\']+)["\']', html, re.IGNORECASE):
   819→        url = match.group(1)
   820→        if url and len(url) >= _MIN_IMAGE_URL_LEN:
   821→            callback(url)

   824→# ── Utility ────────────────────────────────────────────────────────────────────
   825→
   826→def _clean_html(text: str) -> str:
   827→    """Remove HTML tags and decode entities."""
   828→    text = re.sub(r'<[^>]+>', '', text)
   829→    text = text.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>')
   830→    text = text.replace('&quot;', '"').replace('&#39;', "'").replace('&nbsp;', ' ')
   831→    text = re.sub(r'\s+', ' ', text).strip()
   832→    return text
