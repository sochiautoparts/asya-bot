   163→   163→class ImgTagExtractor(HTMLParser):
   164→   164→    """Extract <img> src URLs from HTML content."""
   165→   165→    def __init__(self):

   169→   169→    def handle_starttag(self, tag, attrs):
   170→   170→        if tag == "img":
   171→   171→            for attr_name, attr_value in attrs:
   172→   172→                if attr_name == "src" and attr_value:
   173→   173→                    self.urls.append(attr_value)

   183→   183→def extract_rss_images(entry) -> List[str]:
   184→   184→    """Extract image URLs from an RSS entry (media_content, enclosures, <img> tags)."""
   185→   185→    urls: List[str] = []
   186→   186→
   187→   187→    # 1. media_content (most RSS feeds use this)
   188→   188→    media_list = getattr(entry, "media_content", [])
   189→   189→    if media_list:
   190→   190→        for media in media_list:
   191→   191→            if isinstance(media, str):
   192→   192→                if _is_image_url(media):
   193→   193→                    urls.append(media)
   194→   194→            elif isinstance(media, dict):
   195→   195→                url = media.get("url", "")
   196→   196→                if url and _is_image_url(url):
   197→   197→                    urls.append(url)
   198→   198→
   199→   199→    # 2. media_thumbnail
   200→   200→    thumbnails = getattr(entry, "media_thumbnail", [])
   201→   201→    if thumbnails:
   202→   202→        for thumb in thumbnails:
   203→   203→            if isinstance(thumb, str):
   204→   204→                if _is_image_url(thumb):
   205→   205→                    urls.append(thumb)
   206→   206→            elif isinstance(thumb, dict):
   207→   207→                url = thumb.get("url", "")
   208→   208→                if url and _is_image_url(url):
   209→   209→                    urls.append(url)
   210→   210→
   211→   211→    # 3. enclosures
   212→   212→    for enclosure in getattr(entry, "enclosures", []):
   213→   213→        if isinstance(enclosure, str):
   214→   214→            if _is_image_url(enclosure):
   215→   215→                urls.append(enclosure)
   216→   216→        elif isinstance(enclosure, dict):
   217→   217→            href = enclosure.get("href", "")
   218→   218→            enc_type = enclosure.get("type", "")
   219→   219→            if href and (enc_type.startswith("image/") or _is_image_url(href)):
   220→   220→                urls.append(href)
   221→   221→
   222→   222→    # 4. Parse HTML in summary/content for <img> tags
   223→   223→    for field_name in ("summary", "description", "content"):
   224→   224→        field_value = getattr(entry, field_name, None)
   225→   225→        if not field_value:
   226→   226→            continue
   227→   227→        html_text = ""
   228→   228→        if isinstance(field_value, str):
   229→   229→            html_text = field_value
   230→   230→        elif isinstance(field_value, dict):
   231→   231→            html_text = field_value.get("value", "")
   232→   232→        elif isinstance(field_value, list):
   233→   233→            # feedparser sometimes returns content as a list of dicts with 'value' key
   234→   234→            for item in field_value:
   235→   235→                if isinstance(item, dict):
   236→   236→                    html_text += item.get("value", "")
   237→   237→                elif isinstance(item, str):
   238→   238→                    html_text += item
   239→   239→        if html_text:
   240→   240→            extractor = ImgTagExtractor()
   241→   241→            try:
   242→   242→                extractor.feed(html_text)
   243→   243→                urls.extend(extractor.urls)
   244→   244→            except Exception:
   245→   245→                pass
   246→   246→
   247→   247→    return _dedupe_urls(urls)

   250→   250→def _is_image_url(url: str) -> bool:
   251→   251→    """Check if URL looks like an image."""
   252→   252→    lower = url.lower().split("?")[0]
   253→   253→    return any(lower.endswith(ext) for ext in (
   254→   254→        ".jpg", ".jpeg", ".png", ".webp", ".avif"
   255→   255→    )) or "image" in url.lower() or "/img/" in url.lower() or "/image/" in url.lower()

   258→   258→def _dedupe_urls(urls: List[str]) -> List[str]:
   259→   259→    """Remove duplicate URLs while preserving order."""
   260→   260→    seen: Set[str] = set()
   261→   261→    result: List[str] = []
   262→   262→    for url in urls:
   263→   263→        if url not in seen:
   264→   264→            seen.add(url)
   265→   265→            result.append(url)
   266→   266→    return result

   290→   290→def extract_article_images(html: str, base_url: str) -> List[str]:
   291→   291→    """Extract quality image URLs from an article HTML page."""
   292→   292→    urls: List[str] = []
   293→   293→
   294→   294→    # 1. og:image from meta tags
   295→   295→    og_match = re.search(r'<meta[^>]+property=["\']og:image["\'][^>]+content=["\']([^"\']+)', html, re.IGNORECASE)
   296→   296→    if og_match:
   297→   297→        urls.append(og_match.group(1))
   298→   298→
   299→   299→    # 2. twitter:image
   300→   300→    tw_match = re.search(r'<meta[^>]+(?:name|property)=["\']twitter:image["\'][^>]+content=["\']([^"\']+)', html, re.IGNORECASE)
   301→   301→    if tw_match:
   302→   302→        urls.append(tw_match.group(1))
   303→   303→
   304→   304→    # 3. JSON-LD images
   305→   305→    for ld_match in re.finditer(r'<script[^>]+type=["\']application/ld\+json["\'][^>]*>(.*?)</script>', html, re.IGNORECASE | re.DOTALL):
   306→   306→        try:
   307→   307→            ld_data = json.loads(ld_match.group(1))
   308→   308→            for img in _extract_jsonld_images(ld_data):
   309→   309→                urls.append(img)
   310→   310→        except (json.JSONDecodeError, TypeError):
   311→   311→            pass
   312→   312→
   313→   313→    # 4. <img> tags with src attributes
   314→   314→    extractor = ImgTagExtractor()
   315→   315→    try:
   316→   316→        extractor.feed(html)
   317→   317→        urls.extend(extractor.urls)
   318→   318→    except Exception:
   319→   319→        pass
   320→   320→
   321→   321→    # Filter and normalize
   322→   322→    filtered = []
   323→   323→    for url in urls:
   324→   324→        url = _normalize_url(url, base_url)
   325→   325→        if url and not _is_junk_image(url):
   326→   326→            filtered.append(url)
   327→   327→
   328→   328→    return _dedupe_urls(filtered)

   331→   331→def _extract_jsonld_images(data, depth=0) -> List[str]:
   332→   332→    """Recursively extract image URLs from JSON-LD data."""
   333→   333→    if depth > 5:
   334→   334→        return []
   335→   335→    images = []
   336→   336→    if isinstance(data, dict):
   337→   337→        for key in ("image", "images", "thumbnailUrl", "contentUrl"):
   338→   338→            val = data.get(key)
   339→   339→            if isinstance(val, str):
   340→   340→                images.append(val)
   341→   341→            elif isinstance(val, dict):
   342→   342→                images.extend(_extract_jsonld_images(val, depth + 1))
   343→   343→            elif isinstance(val, list):
   344→   344→                for item in val:
   345→   345→                    if isinstance(item, str):
   346→   346→                        images.append(item)
   347→   347→                    elif isinstance(item, dict):
   348→   348→                        images.extend(_extract_jsonld_images(item, depth + 1))
   349→   349→    elif isinstance(data, list):
   350→   350→        for item in data:
   351→   351→            images.extend(_extract_jsonld_images(item, depth + 1))
   352→   352→    return images

   355→   355→def _normalize_url(url: str, base_url: str) -> Optional[str]:
   356→   356→    """Normalize a URL to absolute form and validate it."""
   357→   357→    if not url or url.startswith("data:") or url.startswith("javascript:"):
   358→   358→        return None
   359→   359→    # Make relative URLs absolute
   360→   360→    if not url.startswith(("http://", "https://")):
   361→   361→        url = urljoin(base_url, url)
   362→   362→    # Clean URL
   363→   363→    url = url.split("#")[0]  # Remove fragment
   364→   364→    # Validate
   365→   365→    parsed = urlparse(url)
   366→   366→    if parsed.scheme not in ("http", "https"):
   367→   367→        return None
   368→   368→    if not parsed.netloc:
   369→   369→        return None
   370→   370→    return url

   373→   373→def _is_junk_image(url: str) -> bool:
   374→   374→    """Check if an image URL is a junk/decorative/icon image."""
   375→   375→    full_url = url.lower()
   376→   376→    path = urlparse(url).path.lower()
   377→   377→    for pattern in JUNK_PATTERNS:
   378→   378→        if pattern.search(full_url) or pattern.search(path):
   379→   379→            return True
   380→   380→    # Too small dimensions in URL
   381→   381→    dim_match = re.search(r"[_\-x](\d{1,4})x(\d{1,4})", full_url)
   382→   382→    if dim_match:
   383→   383→        w, h = int(dim_match.group(1)), int(dim_match.group(2))
   384→   384→        if w < IMAGE_MIN_WIDTH or h < IMAGE_MIN_HEIGHT:
   385→   385→            return True
   386→   386→    return False

   394→   394→def resolve_google_news_url(client: httpx.Client, url: str) -> str:
   395→   395→    """Follow Google News redirect to get the real article URL."""
   396→   396→    if not _GOOGLE_NEWS_RE.search(url):

   399→   399→        resp = client.get(url, timeout=10.0)
   400→   400→        return str(resp.url)
   401→   401→    except Exception:
   402→   402→        return url

   405→   405→# ── Deduplication ──────────────────────────────────────────────────────────────
   406→   406→
   407→   407→def compute_fingerprint(title: str) -> str:
   408→   408→    """Compute a simplified fingerprint for dedup (same as Asya bot)."""
   409→   409→    cleaned = re.sub(
   410→   410→        r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b',
   411→   411→        '', title.lower()
   412→   412→    )
   413→   413→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
   414→   414→    words = cleaned.split()
   415→   415→    return ' '.join(words[:5])
   416→   416→
   417→   417→
   418→   418→def fingerprint_matches(fp: str, existing_fps: Set[str]) -> bool:
   419→   419→    """Check if fingerprint matches any existing fingerprint (3+ of first 4 words match)."""
   420→   420→    fp_words = fp.split()[:4]
   421→   421→    for existing in existing_fps:
   422→   422→        ex_words = existing.split()[:4]
   423→   423→        matches = sum(1 for w in fp_words if w in ex_words)
   424→   424→        if matches >= 3:
   425→   425→            return True
   426→   426→    return False

   431→   431→def load_seen_urls() -> Set[str]:
   432→   432→    """Load previously seen URLs from JSON file."""
   433→   433→    if DEDUP_FILE.exists():
   434→   434→        try:
   435→   435→            with open(DEDUP_FILE, "r", encoding="utf-8") as f:
   436→   436→                data = json.load(f)
   437→   437→                # Keep only URLs from last 72 hours
   438→   438→                cutoff = time.time() - 72 * 3600
   439→   439→                return {url for url, ts in data.items() if ts > cutoff}
   440→   440→        except Exception:
   441→   441→            pass
   442→   442→    return set()

   464→   464→def fetch_rss_feed(client: httpx.Client, source: Dict) -> List[Dict]:
   465→   465→    """Fetch and parse a single RSS feed."""
   466→   466→    name = source["name"]
   467→   467→    url = source["url"]
   468→   468→    lang = source.get("lang", "en")
   469→   469→
   470→   470→    try:
   471→   471→        resp = client.get(url, timeout=15.0)
   472→   472→        if resp.status_code != 200:
   473→   473→            logger.warning(f"[{name}] HTTP {resp.status_code} for {url}")
   474→   474→            return []
   475→   475→
   476→   476→        feed = feedparser.parse(resp.text)
   477→   477→        if feed.bozo and not feed.entries:
   478→   478→            logger.warning(f"[{name}] Parse error: {feed.bozo_exception}")
   479→   479→            return []
   480→   480→
   481→   481→        items = []
   482→   482→        for entry in feed.entries[:25]:  # Take latest 25 per source
   483→   483→            title = getattr(entry, "title", "").strip()
   484→   484→            link = getattr(entry, "link", "").strip()
   485→   485→            summary = getattr(entry, "summary", "") or getattr(entry, "description", "")
   486→   486→            published = getattr(entry, "published", "") or getattr(entry, "updated", "")
   487→   487→
   488→   488→            if not title or not link:
   489→   489→                continue
   490→   490→
   491→   491→            # Convert summary from list/dict to string if needed
   492→   492→            if isinstance(summary, list):
   493→   493→                summary = " ".join(
   494→   494→                    item.get("value", "") if isinstance(item, dict) else str(item)
   495→   495→                    for item in summary
   496→   496→                )
   497→   497→            elif isinstance(summary, dict):
   498→   498→                summary = summary.get("value", "")
   499→   499→
   500→   500→            # Clean summary - strip HTML
   501→   501→            clean_summary = re.sub(r'<[^>]+>', '', str(summary)).strip()[:500]
   502→   502→
   503→   503→            # Extract images from RSS entry
   504→   504→            rss_images = extract_rss_images(entry)
   505→   505→
   506→   506→            items.append({
   507→   507→                "title": title,
   508→   508→                "url": link,
   509→   509→                "summary": clean_summary,
   510→   510→                "published": published,
   511→   511→                "source_name": name,
   512→   512→                "lang": lang,
   513→   513→                "rss_images": rss_images,
   514→   514→            })
   515→   515→
   516→   516→        logger.info(f"[{name}] Fetched {len(items)} items")
   517→   517→        return items
   518→   518→
   519→   519→    except Exception as e:
   520→   520→        logger.warning(f"[{name}] Error: {e}")
   521→   521→        return []

   524→   524→def fetch_article_images(client: httpx.Client, url: str) -> List[str]:
   525→   525→    """Fetch article page and extract images from it."""
   526→   526→    try:
   527→   527→        # Resolve Google News redirect
   528→   528→        real_url = resolve_google_news_url(client, url)
   529→   529→
   530→   530→        resp = client.get(real_url, timeout=REQUEST_TIMEOUT)
   531→   531→        if resp.status_code != 200:
   532→   532→            return []
   533→   533→
   534→   534→        html = resp.text
   535→   535→        if len(html) > 2_000_000:  # Skip huge pages
   536→   536→            html = html[:2_000_000]
   537→   537→
   538→   538→        images = extract_article_images(html, real_url)
   539→   539→        return images[:MAX_PHOTOS_PER_NEWS * 2]  # Get extra, filter later
   540→   540→
   541→   541→    except Exception as e:
   542→   542→        logger.debug(f"Error fetching article images from {url}: {e}")
   543→   543→        return []

   546→   546→def main():
   547→   547→    """Main entry point for the news parser."""
   548→   548→    logger.info("=" * 60)
   549→   549→    logger.info("Auto News Parser — starting...")
   550→   550→    logger.info(f"Max news per run: {MAX_NEWS_PER_RUN}")
   551→   551→    logger.info(f"Max photos per news: {MAX_PHOTOS_PER_NEWS}")
   552→   552→    logger.info("=" * 60)
   553→   553→
   554→   554→    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
   555→   555→
   556→   556→    seen_urls = load_seen_urls()
   557→   557→    existing_news = load_existing_news()
   558→   558→    existing_fingerprints = {compute_fingerprint(item.get("title", "")) for item in existing_news}
   559→   559→
   560→   560→    all_new_items: List[Dict] = []
   561→   561→
   562→   562→    with get_client() as client:
   563→   563→        # ── Phase 1: Fetch all RSS feeds ───────────────────────────────────────
   564→   564→        logger.info("Phase 1: Fetching RSS feeds...")
   565→   565→
   566→   566→        for source in RSS_SOURCES:
   567→   567→            items = fetch_rss_feed(client, source)
   568→   568→            for item in items:
   569→   569→                url = item["url"]
   570→   570→                if url in seen_urls:
   571→   571→                    continue
   572→   572→                fp = compute_fingerprint(item["title"])
   573→   573→                if fingerprint_matches(fp, existing_fingerprints):
   574→   574→                    continue
   575→   575→                all_new_items.append(item)
   576→   576→                seen_urls.add(url)
   577→   577→                existing_fingerprints.add(fp)
   578→   578→            time.sleep(FETCH_DELAY)
   579→   579→
   580→   580→        # ── Phase 2: Google News RSS ──────────────────────────────────────────
   581→   581→        logger.info("Phase 2: Fetching Google News RSS...")
   582→   582→
   583→   583→        for query in GOOGLE_NEWS_QUERIES:
   584→   584→            lang = "ru" if any(ord(c) > 127 for c in query) else "en"
   585→   585→            gl = "RU" if lang == "ru" else "US"
   586→   586→            gn_url = f"https://news.google.com/rss/search?q={quote_plus(query)}&hl={lang}&gl={gl}&ceid={gl}:{lang}"
   587→   587→
   588→   588→            source = {"name": f"GoogleNews_{query[:20]}", "url": gn_url, "lang": lang}
   589→   589→            items = fetch_rss_feed(client, source)
   590→   590→            for item in items:
   591→   591→                url = item["url"]
   592→   592→                if url in seen_urls:
   593→   593→                    continue
   594→   594→                fp = compute_fingerprint(item["title"])
   595→   595→                if fingerprint_matches(fp, existing_fingerprints):
   596→   596→                    continue
   597→   597→                all_new_items.append(item)
   598→   598→                seen_urls.add(url)
   599→   599→                existing_fingerprints.add(fp)
   600→   600→            time.sleep(FETCH_DELAY * 2)  # Extra delay for Google
   601→   601→
   602→   602→        logger.info(f"Total new items after dedup: {len(all_new_items)}")
   603→   603→
   604→   604→        # ── Phase 3: Fetch article images ─────────────────────────────────────
   605→   605→        logger.info("Phase 3: Fetching article images...")
   606→   606→
   607→   607→        processed_items: List[Dict] = []
   608→   608→        for i, item in enumerate(all_new_items[:MAX_NEWS_PER_RUN]):
   609→   609→            article_images = []
   610→   610→
   611→   611→            # Try fetching images from the article page
   612→   612→            if item["url"]:
   613→   613→                article_images = fetch_article_images(client, item["url"])
   614→   614→                time.sleep(FETCH_DELAY)
   615→   615→
   616→   616→            # Combine: RSS images first, then article images
   617→   617→            all_images = item.get("rss_images", []) + article_images
   618→   618→            all_images = _dedupe_urls(all_images)
   619→   619→            all_images = all_images[:MAX_PHOTOS_PER_NEWS]
   620→   620→
   621→   621→            if not all_images:
   622→   622→                logger.info(f"  [{i+1}/{min(len(all_new_items), MAX_NEWS_PER_RUN)}] {item['title'][:60]}... — NO IMAGES")
   623→   623→            else:
   624→   624→                logger.info(f"  [{i+1}/{min(len(all_new_items), MAX_NEWS_PER_RUN)}] {item['title'][:60]}... — {len(all_images)} images")
   625→   625→
   626→   626→            news_item = {
   627→   627→                "title": item["title"],
   628→   628→                "url": item["url"],
   629→   629→                "summary": item.get("summary", ""),
   630→   630→                "published": item.get("published", ""),
   631→   631→                "source": item.get("source_name", ""),
   632→   632→                "lang": item.get("lang", ""),
   633→   633→                "images": all_images,
   634→   634→                "fetched_at": datetime.now(timezone.utc).isoformat(),
   635→   635→            }
   636→   636→            processed_items.append(news_item)
   637→   637→
   638→   638→    # ── Phase 4: Save results ─────────────────────────────────────────────────
   639→   639→    logger.info("Phase 4: Saving results...")
   640→   640→
   641→   641→    # Merge with existing news (newest first), keep max 1000
   642→   642→    merged = processed_items + existing_news
   643→   643→    merged = merged[:1000]
   644→   644→
   645→   645→    with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
   646→   646→        json.dump(merged, f, ensure_ascii=False, indent=2)
   647→   647→
   648→   648→    logger.info(f"Saved {len(processed_items)} new items, total {len(merged)} in {OUTPUT_FILE}")
   649→   649→
   650→   650→    # Save dedup state
   651→   651→    save_seen_urls(seen_urls)
   652→   652→
   653→   653→    # ── Summary ───────────────────────────────────────────────────────────────
   654→   654→    with_photos = sum(1 for item in processed_items if item.get("images"))
   655→   655→    without_photos = len(processed_items) - with_photos
   656→   656→    total_images = sum(len(item.get("images", [])) for item in processed_items)
   657→   657→
   658→   658→    logger.info("=" * 60)
   659→   659→    logger.info(f"PARSING COMPLETE")
   660→   660→    logger.info(f"  New items: {len(processed_items)}")
   661→   661→    logger.info(f"  With photos: {with_photos}")
   662→   662→    logger.info(f"  Without photos: {without_photos}")
   663→   663→    logger.info(f"  Total images: {total_images}")
   664→   664→    logger.info(f"  Total in JSON: {len(merged)}")
   665→   665→    logger.info("=" * 60)
   666→   666→
   667→   667→
   668→   668→if __name__ == "__main__":
   669→   669→    main()
