   119→def _extract_entities(title: str) -> str:
   120→    """Extract key entities from a news title for dedup.
   121→    
   122→    Returns a normalized entity key like "bmw_m5_reveal", "alonso_criticism", or "toyota_recalls".
   123→    This allows us to detect that "BMW M5 2027 revealed" and "BMW unveils new M5"
   124→    are about the SAME event and should not be posted twice.
   125→    Also detects notable people (F1 drivers, CEOs) and motorsport topics.
   126→    """
   127→    title_lower = title.lower()
   128→    
   129→    # Extract brand
   130→    brand = ""
   131→    for b in _AUTO_BRANDS:
   132→        if b.lower() in title_lower:
   133→            brand = b.lower().replace(" ", "_")
   134→            break
   135→    
   136→    # Extract notable person (F1 driver, CEO, etc.)
   137→    person = ""
   138→    for p in _NOTABLE_PEOPLE:
   139→        if p.lower() in title_lower:
   140→            person = p.lower().replace(" ", "_")
   141→            break
   142→    
   143→    # Extract motorsport team/series
   144→    team = ""
   145→    for t in _MOTORSPORT_TEAMS:
   146→        if t.lower() in title_lower:
   147→            team = t.lower().replace(" ", "_")
   148→            break
   149→    
   150→    # Extract model (alphanumeric after brand)
   151→    model = ""
   152→    if brand:
   153→        # Common model patterns: M3, M5, X5, Q7, A4, E-Class, 911, etc.
   154→        model_patterns = [
   155→            r'\b([mglxqsec]\d+)\b',  # M3, X5, Q7, A4, E300, C-Class
   156→            r'\b(\d{3,4}[ix]?)\b',   # 911, 330i, 540ix
   157→            r'\b(class|series|corolla|camry|civic|accord|model\s?[s3xy])\b',
   158→            r'\b(mustang|camaro|corvette|prius|rav4|highlander|pilot|tahoe)\b',
   159→            r'\b(supra|gr86|brz|miata|wrangler|bronco|range rover|defender)\b',
   160→            r'\b(taycan|macan|cayenne|panamera|911|718|boxster)\b',
   161→        ]
   162→        for pattern in model_patterns:
   163→            match = re.search(pattern, title_lower)
   164→            if match:
   165→                model = match.group(1).replace(" ", "_")
   166→                break
   167→    
   168→    # Extract event type
   169→    event = ""
   170→    for e in _EVENT_KEYWORDS:
   171→        if e in title_lower:
   172→            event = e
   173→            break
   174→    
   175→    parts = [p for p in [brand, person, team, model, event] if p]
   176→    entity_key = "_".join(parts) if parts else ""
   177→    
   178→    # ── BRAND-ONLY DEDUP KEY ──
   179→    # If a brand is mentioned without a specific model+event, also check brand-only coverage
   180→    # This prevents posting 3 different stories about the same brand in one day
   181→    
   182→    # ── PERSON-ONLY DEDUP KEY ──
   183→    # If a notable person is mentioned (e.g. "Alonso"), also register a person-only key
   184→    # This prevents posting 3 different stories about Alonso in the same day
   185→    if person and entity_key != person:
   186→        # Register person-only topic as well (e.g., "alonso" besides "alonso_criticism")
   187→        person_only_key = person
   188→        if _is_topic_covered(person_only_key) and not entity_key:
   189→            # Person was already posted about and we don't have a more specific key
   190→            # This helps prevent "Alonso 3x" duplicates
   191→            pass  # We still allow it if there's a more specific entity key
   192→        # We'll register person_only_key separately when posting
   193→    
   194→    return entity_key
   195→

   197→def _is_topic_covered(entity_key: str) -> bool:
   198→    """Check if this topic/entity was already posted about recently."""
   199→    if not entity_key:
   200→        return False
   201→    
   202→    entry = _topic_registry.get(entity_key)
   203→    if not entry:
   204→        return False
   205→    
   206→    # Check age
   207→    age_hours = (time.time() - entry["last_posted"]) / 3600
   208→    if age_hours > _REGISTRY_MAX_AGE_HOURS:
   209→        # Topic too old — allow again
   210→        del _topic_registry[entity_key]
   211→        return False
   212→    
   213→    # Topic was posted in last 72h — it's covered
   214→    return True

   217→def _register_topic(entity_key: str, title: str):
   218→    """Register that a topic was posted about.
   219→    
   220→    Also registers person-only dedup key to prevent "Alonso 3x" style duplicates.
   221→    Persists to DB so the registry survives restarts.
   222→    """
   223→    if not entity_key:
   224→        return
   225→    
   226→    now = time.time()
   227→    if entity_key in _topic_registry:
   228→        _topic_registry[entity_key]["post_count"] += 1
   229→        _topic_registry[entity_key]["last_posted"] = now
   230→        _topic_registry[entity_key]["titles"].append(title)
   231→    else:
   232→        _topic_registry[entity_key] = {
   233→            "first_seen": now,
   234→            "last_posted": now,
   235→            "post_count": 1,
   236→            "titles": [title],
   237→        }
   238→    
   239→    # ── Also register brand-only key for dedup ──
   240→    # Extract brand from entity_key (e.g., "bmw_x5_reveal" → "bmw")
   241→    for b in _AUTO_BRANDS:
   242→        b_key = b.lower().replace(" ", "_")
   243→        if b_key in entity_key:
   244→            if b_key not in _topic_registry:
   245→                _topic_registry[b_key] = {
   246→                    "first_seen": now,
   247→                    "last_posted": now,
   248→                    "post_count": 1,
   249→                    "titles": [f"[brand-dedup] {title}"],
   250→                }
   251→            else:
   252→                _topic_registry[b_key]["post_count"] += 1
   253→                _topic_registry[b_key]["last_posted"] = now
   254→            break
   255→    
   256→    # ── Also register person-only key for dedup ──
   257→    # Extract person from entity_key (e.g., "alonso_criticism" → "alonso")
   258→    for p in _NOTABLE_PEOPLE:
   259→        p_key = p.lower().replace(" ", "_")
   260→        if p_key in entity_key:
   261→            if p_key not in _topic_registry:
   262→                _topic_registry[p_key] = {
   263→                    "first_seen": now,
   264→                    "last_posted": now,
   265→                    "post_count": 1,
   266→                    "titles": [f"[person-dedup] {title}"],
   267→                }
   268→            else:
   269→                _topic_registry[p_key]["post_count"] += 1
   270→                _topic_registry[p_key]["last_posted"] = now
   271→            break
   272→    
   273→    # Persist to DB (async, fire-and-forget via background task)
   274→    try:
   275→        entry = _topic_registry[entity_key]
   276→        import asyncio
   277→        from bot.database import save_topic_to_registry
   278→        asyncio.create_task(save_topic_to_registry(
   279→            entity_key=entity_key,
   280→            first_seen=entry["first_seen"],
   281→            last_posted=entry["last_posted"],
   282→            post_count=entry["post_count"],
   283→            titles=entry["titles"],
   284→        ))
   285→    except Exception as e:
   286→        logger.debug(f"Could not persist topic to DB: {e}")

   300→            import asyncio
   301→            from bot.database import cleanup_topic_registry
   302→            asyncio.create_task(cleanup_topic_registry(_REGISTRY_MAX_AGE_HOURS))
   303→        except Exception:
   304→            pass

   381→def _score_interest(title: str, summary: str = "") -> float:
   382→    """Rate how interesting a news item is on a 0-1 scale.
   383→    
   384→    Higher scores = more interesting = more likely to engage readers.
   385→    """
   386→    text = f"{title} {summary}".lower()
   387→    score = 0.5  # Base score
   388→    
   389→    # Boost for high-interest keywords
   390→    for kw in _HIGH_INTEREST_KEYWORDS:
   391→        if kw.lower() in text:
   392→            score += 0.15
   393→            break  # Only count once
   394→    
   395→    # Smaller boost for medium-interest keywords
   396→    medium_count = sum(1 for kw in _MEDIUM_INTEREST_KEYWORDS if kw.lower() in text)
   397→    score += min(medium_count * 0.05, 0.15)
   398→    
   399→    # Penalty for low-interest keywords
   400→    for kw in _LOW_INTEREST_KEYWORDS:
   401→        if kw.lower() in text:
   402→            score -= 0.1
   403→            break
   404→    
   405→    # HEAVY penalty for boring Russian auto brands (in _LOW_INTEREST too, but need stronger effect)
   406→    _BORING_RUSSIAN_BRANDS = ["автоваз", "лада", "lada", "уаз", "uaz", "камаз", "kamaz",
   407→                              "соллерс", "vesta", "granta", "niva", "искра", "iskra"]
   408→    for kw in _BORING_RUSSIAN_BRANDS:
   409→        if kw.lower() in text:
   410→            score -= 0.4  # Heavy penalty — these topics are NEVER interesting
   411→            break
   412→    
   413→    # Penalty for very long/technical titles
   414→    if len(title) > 120:
   415→        score -= 0.1
   416→    
   417→    # Bonus for brand names (people search by brand)
   418→    # BUT: exclude Russian domestic brands — they get penalized in _LOW_INTEREST
   419→    _russian_brands = {"lada", "ваз", "газ", "уаз", "камаз", "соллерс", "niva", "vesta", "granta", "aurus", "аурус"}
   420→    for brand in _AUTO_BRANDS:
   421→        if brand.lower() in text:
   422→            if brand.lower() not in _russian_brands:
   423→                score += 0.05
   424→            break
   425→    
   426→    # Bonus for Russian market topics (but NOT domestic Russian car brands)
   427→    russian_market_kw = ["российск", "россия", "сочи", "краснодар",
   428→                         "импортозамещ", "параллельн импорт", "parallel import"]
   429→    for kw in russian_market_kw:
   430→        if kw in text:
   431→            score += 0.08
   432→            break
   433→    
   434→    # Bonus for EV/transition topics
   435→    ev_kw = ["электромобиль", "электрокар", "зарядн", "ev", "батарея",
   436→             "battery", "electric vehicle", "запас хода"]
   437→    for kw in ev_kw:
   438→        if kw in text:
   439→            score += 0.08
   440→            break
   441→    
   442→    return max(0.1, min(1.0, score))

   445→def _score_freshness(published_time: float) -> float:
   446→    """Score how fresh a news item is. STRICT: rejects items >24h old.

   711→def _extract_published_time_from_snippet(snippet: str) -> float:
   712→    """Try to extract actual publication time from search result snippet.
   713→    
   714→    Looks for patterns like "2 hours ago", "yesterday", "3 days ago",
   715→    dates like "Mar 5, 2026", etc. Returns a Unix timestamp or 0 if unknown.
   716→    """
   717→    if not snippet:
   718→        return 0
   719→    
   720→    snippet_lower = snippet.lower()
   721→    now = datetime.now(_MOSCOW_TZ)
   722→    
   723→    # ── Freshness penalty patterns — detect stale content ──
   724→    # These return timestamps far in the past so the item gets deprioritized
   725→    stale_patterns = [
   726→        (r'(\d+)\s+week', 7 * 24),      # "2 weeks ago"
   727→        (r'(\d+)\s+месяц', 30 * 24),    # "3 месяца назад"
   728→        (r'вчера', 24),                   # "yesterday"
   729→        (r'прошлый\s+недел', 7 * 24),    # "на прошлой неделе"
   730→        (r'прошлый\s+месяц', 30 * 24),   # "в прошлом месяце"
   731→        (r'прошлогод', 365 * 24),         # "прошлогодний"
   732→    ]
   733→    for pattern, hours_per_unit in stale_patterns:
   734→        match = re.search(pattern, snippet_lower)
   735→        if match:
   736→            try:
   737→                count = int(match.group(1)) if match.lastindex else 1
   738→            except (ValueError, IndexError):
   739→                count = 1
   740→            hours_ago = count * hours_per_unit
   741→            return now.timestamp() - (hours_ago * 3600)
   742→    
   743→    # ── Relative time patterns — try to compute actual time ──
   744→    rel_patterns = [
   745→        (r'(\d+)\s+hours?\s+ago', 1),      # "2 hours ago"
   746→        (r'(\d+)\s+минут', 1/60),           # "30 минут назад"
   747→        (r'(\d+)\s+час', 1),                # "2 часа назад"
   748→        (r'(\d+)\s+дн[еяь]', 24),           # "2 дня назад"
   749→        (r'(\d+)\s+days?\s+ago', 24),       # "3 days ago"
   750→        (r'сегодня|today', 0),               # "today"
   751→    ]
   752→    for pattern, hours_per_unit in rel_patterns:
   753→        match = re.search(pattern, snippet_lower)
   754→        if match:
   755→            try:
   756→                count = int(match.group(1)) if match.lastindex else 1
   757→            except (ValueError, IndexError):
   758→                count = 1
   759→            hours_ago = count * hours_per_unit
   760→            return now.timestamp() - (hours_ago * 3600)
   761→    
   762→    return 0  # Unknown — let _score_freshness handle it

  1621→# ── TONE ANALYSIS — Multi-step Chain of Thought ──────────────────────────────

  1623→from enum import Enum
  1624→from dataclasses import dataclass
  1625→from typing import Optional
  1626→
  1627→class NewsTone(Enum):
  1628→    """Classification of news tone for appropriate editorial response"""
  1629→    SERIOUS = "serious"      # ДТП, отзыв, трагедия, штрафы — NO jokes
  1630→    HYPE = "hype"            # Новый суперкар, рекорд, премьеры — fun jokes OK
  1631→    ROUTINE = "routine"      # Обновление цен, плановые новости — light jokes
  1632→    FUN = "fun"              # Забавные случаи, курьезы — funny jokes
  1633→    TECHNICAL = "technical"  # Технические характеристики — tech jokes
  1634→
  1635→@dataclass
  1636→class ExtractedFacts:
  1637→    """Structured facts extracted from news for validation"""
  1638→    brand: str
  1639→    model: str
  1640→    year: Optional[str]
  1641→    price: Optional[str]
  1642→    power: Optional[str]
  1643→    key_event: str
  1644→    tone: NewsTone
  1645→    is_partner: bool = False
  1646→
  1647→async def analyze_news_tone(title: str, summary: str, content: str = "") -> ExtractedFacts:
  1648→    """
  1649→    Multi-step Chain of Thought analysis:
  1650→    1. Extract key facts (brand, model, price, power)
  1651→    2. Determine tone (serious/hype/routine/fun/technical)
  1652→    3. Return structured data for post generation
  1653→    """
  1654→    text = f"{title} {summary} {content}".lower()
  1655→    
  1656→    # Extract brand
  1657→    brand = ""
  1658→    for b in _AUTO_BRANDS:
  1659→        if b.lower() in text:
  1660→            brand = b
  1661→            break
  1662→    
  1663→    # Extract model (simplified)
  1664→    model = ""
  1665→    if brand:
  1666→        model_patterns = [
  1667→            rf'{brand.lower()}\s+([a-z0-9]+)',
  1668→            rf'{brand.lower()}\s+([mglxqsec]\d+)',
  1669→        ]
  1670→        for pattern in model_patterns:
  1671→            match = re.search(pattern, text, re.IGNORECASE)
  1672→            if match:
  1673→                model = match.group(1).upper()
  1674→                break
  1675→    
  1676→    # Extract facts
  1677→    price = None
  1678→    power = None
  1679→    year = None
  1680→    
  1681→    # Price
  1682→    price_match = re.search(r'\$([\d,]+)', text)
  1683→    if price_match:
  1684→        price = price_match.group(1)
  1685→    else:
  1686→        price_match = re.search(r'([\d,]+)\s*(?:руб|рубл|rub|₽)', text, re.IGNORECASE)
  1687→        if price_match:
  1688→            price = price_match.group(1)
  1689→    
  1690→    # Power
  1691→    power_match = re.search(r'(\d+)\s*(?:л\.?с|hp|horsepower|лс)', text, re.IGNORECASE)
  1692→    if power_match:
  1693→        power = power_match.group(1)
  1694→    
  1695→    # Year
  1696→    year_match = re.search(r'\b(20[12]\d)\b', text)
  1697→    if year_match:
  1698→        year = year_match.group(1)
  1699→    
  1700→    # Determine tone
  1701→    tone = _determine_tone(text, title)
  1702→    
  1703→    return ExtractedFacts(
  1704→        brand=brand or "Неизвестно",
  1705→        model=model,
  1706→        year=year,
  1707→        price=price,
  1708→        power=power,
  1709→        key_event=title,
  1710→        tone=tone,
  1711→        is_partner=False
  1712→    )
  1713→
  1714→def _determine_tone(text: str, title: str) -> NewsTone:
  1715→    """Determine news tone based on keywords"""
  1716→    text_lower = text.lower()
  1717→    title_lower = title.lower()

  1811→def trim_to_telegram_limits(text: str, has_media: bool) -> str:
  1812→    """Trim text to Telegram limits"""
  1813→    MAX_CAPTION = 1024
  1814→    MAX_MESSAGE = 4096
  1815→    max_len = MAX_CAPTION if has_media else MAX_MESSAGE
  1816→    
  1817→    if len(text) <= max_len:
  1818→        return text
  1819→    
  1820→    lines = text.split('\n')
  1821→    trimmed = []
  1822→    current_len = 0
  1823→    
  1824→    for line in lines:
  1825→        if current_len + len(line) + 1 <= max_len:
  1826→            trimmed.append(line)
  1827→            current_len += len(line) + 1
  1828→        else:
  1829→            if current_len + 3 <= max_len:
  1830→                trimmed.append("...")

  1832→
