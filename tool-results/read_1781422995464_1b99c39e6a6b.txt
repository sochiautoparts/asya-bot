    97→class AIRouter:

   103→    - Level 2: Pollinations FREE API (no auth, rate-limited)
   104→    - Level 3: Cloudflare Workers AI (Mistral Small 3.1, 20K req/day)
   105→    - Last resort: Static fallback responses

   184→    async def chat(
   185→        self,
   186→        user_id: int,
   187→        message: str,
   188→        system_prompt: str = "",
   189→        model: str = "",
   190→        temperature: float = None,
   191→        max_tokens: int = None,
   192→        use_cache: bool = True,
   193→        save_history: bool = True,
   194→        extra_context: str = "",
   195→        route_type: str = "chat",
   196→    ) -> AIResponse:
   197→        """
   198→        Send a chat message through the AI router with CONCURRENT failover.
   199→
   200→        v9.0 CONCURRENT ROUTING (fast response, no sequential delays):
   201→        - "chat" (default): Try Local first (if available), then CONCURRENT Pollinations+Cloudflare
   202→        - "function": Try Pollinations key first, then CONCURRENT free+Cloudflare+Local
   203→        - "comment": Try Local first, then CONCURRENT Pollinations+Cloudflare
   204→
   205→        Key improvement: Instead of waiting for each level to timeout sequentially
   206→        (which could take 30+15+60 = 105 seconds worst case), we now launch
   207→        multiple providers CONCURRENTLY and return the FIRST successful response.
   208→        This means if Pollinations is slow but Cloudflare responds in 3s, the user
   209→        gets a response in ~3s instead of ~30s.
   210→        """
   211→        import asyncio as _asyncio
   212→        temperature = temperature or config.CHAT_TEMPERATURE
   213→        max_tokens = max_tokens or config.CHAT_MAX_TOKENS
   214→
   215→        # Build system prompt with time context
   216→        sys_prompt = self._build_system_prompt(system_prompt, extra_context)
   217→
   218→        # Check cache first
   219→        if use_cache:
   220→            cache_key = self._make_cache_key(sys_prompt, message)
   221→            cached = await get_ai_cached(cache_key)
   222→            if cached:
   223→                return AIResponse(
   224→                    text=cached,
   225→                    model="cached",
   226→                    provider="cache",
   227→                    cached=True,
   228→                )
   229→
   230→        # Get chat history for context
   231→        history = await get_chat_history(user_id)
   232→
   233→        # ── Select model based on route_type ──
   234→        if route_type == "comment":
   235→            model = model or "mistral"
   236→        elif route_type == "function":
   237→            model = model or "openai-large"
   238→        else:
   239→            model = model or ""
   240→
   241→        # ── LEVEL 0: Local Model (Qwen3-4B) — CHAT & COMMENT routes ──
   242→        # Local model is primary for simple chat and comments (saves cloud balance)
   243→        # Skip for function routes (need cloud quality) and when explicitly disabled
   244→        use_local_first = route_type in ("chat", "comment") and config.ENABLE_LOCAL_MODEL
   245→
   246→        if use_local_first:
   247→            # Quick check if local model is actually available (no timeout waste)
   248→            local_available = (
   249→                self._local is not None
   250→                and self._local._model_loaded
   251→                and self._local._llm is not None
   252→                and self._local._consecutive_errors < 5
   253→            )
   254→            if local_available:
   255→                response = await self._try_local(
   256→                    user_id, message, history, sys_prompt, temperature, max_tokens
   257→                )
   258→
   259→                if not response.error:
   260→                    self._level0_count += 1
   261→                    self._total_requests += 1
   262→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   263→
   264→                logger.debug(f"Level 0 (local) failed for route={route_type}: {response.error_message}")
   265→
   266→        # ── CONCURRENT FAILOVER: Launch multiple providers at once ──
   267→        # This is the key fix for the "Asya takes too long to respond" issue.
   268→        # Instead of trying providers one-by-one (each with 15-60s timeout),
   269→        # we launch them concurrently and return the FIRST successful response.
   270→
   271→        async def _safe_try_pollinations():
   272→            """Try Pollinations with key — return response or error."""
   273→            try:
   274→                return await self._try_pollinations(
   275→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   276→                )
   277→            except Exception as e:
   278→                logger.error(f"Pollinations key exception: {e}")
   279→                return AIResponse(text="", model=model, provider="pollinations", error=True, error_message=str(e))
   280→
   281→        async def _safe_try_pollinations_free():
   282→            """Try Pollinations free — return response or error."""
   283→            try:
   284→                return await self._try_pollinations_free(
   285→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   286→                )
   287→            except Exception as e:
   288→                logger.error(f"Pollinations free exception: {e}")
   289→                return AIResponse(text="", model=model, provider="pollinations-free", error=True, error_message=str(e))
   290→
   291→        async def _safe_try_cloudflare():
   292→            """Try Cloudflare — return response or error."""
   293→            try:
   294→                return await self._try_cloudflare(
   295→                    user_id, message, history, sys_prompt, temperature, max_tokens
   296→                )
   297→            except Exception as e:
   298→                logger.error(f"Cloudflare exception: {e}")
   299→                return AIResponse(text="", model="cloudflare", provider="cloudflare", error=True, error_message=str(e))
   300→
   301→        async def _safe_try_local_fallback():
   302→            """Try local model as fallback for function routes."""
   303→            try:
   304→                return await self._try_local(
   305→                    user_id, message, history, sys_prompt, temperature, max_tokens
   306→                )
   307→            except Exception as e:
   308→                logger.error(f"Local fallback exception: {e}")
   309→                return AIResponse(text="", model="local-qwen3-4b", provider="local", error=True, error_message=str(e))
   310→
   311→        # Build concurrent task list based on route type
   312→        # Priority ordering: we want the best quality response first.
   313→        # But we launch them ALL at once and take the first success.
   314→        concurrent_tasks = []
   315→
   316→        # Always try Pollinations with key (best quality when available)
   317→        if self._primary and self._primary._build_key_tier_list():
   318→            concurrent_tasks.append(("pollinations_key", _safe_try_pollinations()))
   319→
   320→        # Always try Cloudflare in parallel (fast, independent service)
   321→        if self._cloudflare and self._cloudflare._accounts:
   322→            concurrent_tasks.append(("cloudflare", _safe_try_cloudflare()))
   323→
   324→        # For function routes, also try local model as concurrent fallback
   325→        if route_type == "function" and config.ENABLE_LOCAL_MODEL and self._local and self._local._model_loaded:
   326→            concurrent_tasks.append(("local_fallback", _safe_try_local_fallback()))
   327→
   328→        if not concurrent_tasks:
   329→            # No providers available — try free API as last hope
   330→            concurrent_tasks.append(("pollinations_free", _safe_try_pollinations_free()))
   331→
   332→        # Execute all tasks concurrently, return FIRST successful response
   333→        # Use asyncio.wait with FIRST_COMPLETED to get the fastest response
   334→        task_names = [name for name, _ in concurrent_tasks]
   335→        task_coros = [coro for _, coro in concurrent_tasks]
   336→        tasks = [_asyncio.create_task(coro) for coro in task_coros]
   337→
   338→        try:
   339→            # Wait for the first task to complete
   340→            done, pending = await _asyncio.wait(tasks, return_when=_asyncio.FIRST_COMPLETED)
   341→
   342→            # Check if any completed task succeeded
   343→            for task in done:
   344→                response = task.result()
   345→                if not response.error and response.text:
   346→                    # Cancel remaining tasks (we have a winner!)
   347→                    for p in pending:
   348→                        p.cancel()
   349→                    # Also cancel other done tasks we don't need
   350→                    for t in tasks:
   351→                        if t is not task and not t.done():
   352→                            t.cancel()
   353→
   354→                    # Track which level succeeded
   355→                    task_idx = tasks.index(task)
   356→                    level_name = task_names[task_idx]
   357→                    if level_name == "pollinations_key":
   358→                        self._level1_count += 1
   359→                    elif level_name == "pollinations_free":
   360→                        self._level2_count += 1
   361→                    elif level_name == "cloudflare":
   362→                        self._level3_count += 1
   363→                    elif level_name == "local_fallback":
   364→                        self._level0_count += 1
   365→
   366→                    self._total_requests += 1
   367→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   368→
   369→            # First completed task(s) failed — wait for others
   370→            # If there are still pending tasks, wait for them
   371→            if pending:
   372→                done2, pending2 = await _asyncio.wait(pending, return_when=_asyncio.FIRST_COMPLETED)
   373→                done = done.union(done2)
   374→                pending = pending2
   375→
   376→                for task in done2:
   377→                    response = task.result()
   378→                    if not response.error and response.text:
   379→                        for p in pending2:
   380→                            p.cancel()
   381→                        for t in tasks:
   382→                            if t is not task and not t.done():
   383→                                t.cancel()
   384→
   385→                        task_idx = tasks.index(task)
   386→                        level_name = task_names[task_idx]
   387→                        if level_name == "pollinations_key":
   388→                            self._level1_count += 1
   389→                        elif level_name == "pollinations_free":
   390→                            self._level2_count += 1
   391→                        elif level_name == "cloudflare":
   392→                            self._level3_count += 1
   393→                        elif level_name == "local_fallback":
   394→                            self._level0_count += 1
   395→
   396→                        self._total_requests += 1
   397→                        return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   398→
   399→            # All concurrent tasks failed — try Pollinations free as sequential fallback
   400→            # (we didn't include it in concurrent tasks if we had paid keys)
   401→            if not any(n == "pollinations_free" for n in task_names):
   402→                logger.warning(f"All concurrent providers failed (route={route_type}), trying free API sequentially")
   403→                response = await self._try_pollinations_free(
   404→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   405→                )
   406→                if not response.error:
   407→                    self._level2_count += 1
   408→                    self._total_requests += 1
   409→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   410→
   411→            # ALL levels failed — collect error info from all tasks
   412→            errors = []
   413→            for task in done:
   414→                try:
   415→                    r = task.result()
   416→                    if r.error:
   417→                        errors.append(f"{r.provider}: {r.error_message}")
   418→                except Exception as e:
   419→                    errors.append(f"exception: {e}")
   420→
   421→            # Cancel any remaining tasks
   422→            for t in tasks:
   423→                if not t.done():
   424→                    t.cancel()
   425→
   426→        except Exception as e:
   427→            logger.error(f"Concurrent failover error: {e}")
   428→            # Cancel all tasks on unexpected error
   429→            for t in tasks:
   430→                if not t.done():
   431→                    t.cancel()
   432→
   433→            # Try sequential fallback as emergency path
   434→            try:
   435→                response = await self._try_pollinations_free(
   436→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   437→                )
   438→                if not response.error:
   439→                    self._level2_count += 1
   440→                    self._total_requests += 1
   441→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   442→            except Exception:
   443→                pass
   444→
   445→        # ── LAST RESORT: Static fallback ──
   446→        self._static_count += 1
   447→        self._total_fallbacks += 1
   448→        logger.error(
   449→            f"ALL LEVELS FAILED for route_type={route_type}. "
   450→            f"Level0(local)={self._level0_count}, Level1={self._level1_count}, "
   451→            f"Level2={self._level2_count}, Level3={self._level3_count}, Static={self._static_count}"
   452→        )
   453→        return AIResponse(
   454→            text=random.choice(FALLBACK_RESPONSES),
   455→            model="fallback",
   456→            provider="static",
   457→            tokens_used=0,
   458→        )

   460→    async def _save_response(
   461→        self, user_id: int, message: str, response: AIResponse,
   462→        sys_prompt: str, use_cache: bool, save_history: bool
   463→    ) -> AIResponse:
   464→        """Save response to history and cache."""
   465→        if save_history and response.text:
   466→            await add_chat_message(user_id, "user", message)
   467→            await add_chat_message(user_id, "assistant", response.text)
   468→
   469→            # Cache the response
   470→            if use_cache and response.text:
   471→                cache_key = self._make_cache_key(sys_prompt, message)
   472→                await set_ai_cached(cache_key, message, response.text, response.model)
   473→
   474→        return response

   476→    async def _try_pollinations(self, user_id: int, message: str, history: list,
   477→                                 sys_prompt: str, temperature: float, max_tokens: int,
   478→                                 model: str) -> AIResponse:
   479→        """Level 1: Try Pollinations with API key (KEY1 → KEY2 internally).
   480→        Also tries model-level fallbacks.
   481→        """
   482→        messages = self._primary.format_messages(sys_prompt, history, message)
   483→
   484→        # Try primary model (provider will try KEY1 → KEY2 internally)
   485→        response = await self._primary.chat(
   486→            messages=messages,
   487→            model=model,
   488→            temperature=temperature,
   489→            max_tokens=max_tokens,
   490→        )
   491→
   492→        # If primary failed, try fallback models
   493→        if response.error:
   494→            is_key_error = any(code in (response.error_message or "")
   495→                              for code in ["All API keys depleted", "401", "402", "unavailable", "cooldown"])
   496→
   497→            if is_key_error:
   498→                fallback_models = [m for m in FALLBACK_MODELS
   499→                                   if m != model and not self._primary._is_model_in_cooldown(m)][:1]
   500→                if fallback_models:
   501→                    logger.info(f"Key error, trying {len(fallback_models)} fallback model")
   502→            else:
   503→                fallback_models = [
   504→                    m for m in ["mistral-small", "nova-fast", "gemma"]
   505→                    if m != model and not self._primary._is_model_in_cooldown(m)
   506→                ][:1]
   507→                logger.info(f"Non-key error, trying fallback: {fallback_models}")
   508→
   509→            for fallback_model in fallback_models:
   510→                if fallback_model == model:
   511→                    continue
   512→                if self._primary._is_model_in_cooldown(fallback_model):
   513→                    continue
   514→                logger.info(f"Trying fallback model: {fallback_model}")
   515→                response = await self._primary.chat(
   516→                    messages=messages,
   517→                    model=fallback_model,
   518→                    temperature=temperature,
   519→                    max_tokens=max_tokens,
   520→                )
   521→                if not response.error:
   522→                    break
   523→
   524→        return response
   525→
   526→    async def _try_pollinations_free(self, user_id: int, message: str, history: list,
   527→                                      sys_prompt: str, temperature: float, max_tokens: int,
   528→                                      model: str) -> AIResponse:
   529→        """Level 2: Try Pollinations FREE API (no auth, text.pollinations.ai)."""

   579→    async def _try_local(self, user_id: int, message: str, history: list,
   580→                          sys_prompt: str, temperature: float, max_tokens: int) -> AIResponse:
   581→        """Level 0: Try Local Model (Qwen3-4B).
   582→
   583→        Uses ChatML format for Qwen3 with /no_think for fast responses.
   584→        Primary for CHAT and COMMENT routes, fallback for FUNCTION routes.
   585→        
   586→        CONTEXT WINDOW BUDGET (4096 tokens, ~1.3 chars/token for Russian):
   587→          - Output: 512 tokens (MODEL_MAX_TOKENS)
   588→          - Safety margin: 64 tokens
   589→          - Available for input: 3520 tokens (~4576 chars)
   590→          - System prompt (local_system_prompt v2): ~2340 chars (~1800 tokens)
   591→          - History: 3 turns × 200 chars = ~600 chars (~462 tokens)
   592→          - User message: up to ~1000 chars (~770 tokens)
   593→          - Total: ~3940 chars (~3032 tokens) — fits in 3520 with margin!
   594→        """
   595→        if not self._local:
   596→            return AIResponse(
   597→                text="",
   598→                model="local-qwen3-4b",
   599→                provider="local",
   600→                error=True,
   601→                error_message="Local provider not initialized",
   602→            )
   603→
   604→        if not config.ENABLE_LOCAL_MODEL:
   605→            return AIResponse(
   606→                text="",
   607→                model="local-qwen3-4b",
   608→                provider="local",
   609→                error=True,
   610→                error_message="Local model disabled (ENABLE_LOCAL_MODEL=false)",
   611→            )
   612→
   613→        # Use EXPANDED system prompt for local model (v2.0 — ~1800 tokens).
   614→        # The full 12K-char cloud prompt (~9000 tokens) still does NOT fit.
   615→        # But the expanded v2 prompt gives much better quality than the old compact one.
   616→        compact_prompt = persona.local_system_prompt
   617→
   618→        # Build messages for local model using its own ChatML format
   619→        messages = [{"role": "system", "content": compact_prompt}]
   620→
   621→        # Add limited history for local model (saves context window)
   622→        # With expanded prompt (~1800 tokens), we have ~1720 tokens left for history+message.
   623→        # 3 history turns × ~200 chars × 1.3 chars/token = ~462 tokens
   624→        # User message up to ~1000 chars = ~770 tokens
   625→        # Total: ~1232 tokens — well within budget.
   626→        history_limit = min(config.MODEL_HISTORY_LIMIT, 3)
   627→        limited_history = history[-history_limit:] if history else []
   628→        for msg in limited_history:
   629→            role = msg.get("role", "user")
   630→            content = msg.get("content", "")
   631→            # Truncate long history messages to save context tokens
   632→            if role in ("user", "assistant") and content:
   633→                messages.append({"role": role, "content": content[:200]})
   634→
   635→        # Truncate user message to prevent context overflow.
   636→        # With expanded system prompt, allow up to ~1000 chars for user message.
   637→        # The local_provider.chat() has its own safety truncation as well.
   638→        truncated_message = message[:1000] if len(message) > 1000 else message
   639→        if len(message) > 1000:
   640→            logger.debug(f"Truncated user message for local model: {len(message)} → 1000 chars")
   641→        messages.append({"role": "user", "content": truncated_message})
   642→
   643→        return await self._local.chat(
   644→            messages=messages,
   645→            temperature=temperature,
   646→            max_tokens=min(max_tokens, config.MODEL_MAX_TOKENS),
   647→        )

   649→    async def analyze_image(

   658→        Analyze an image with 3-level vision failover.

   660→        Level 1: Pollinations vision (with key)

   729→        logger.warning(f"Vision Level 2 failed, trying Cloudflare: {response.error_message}")

   975→    @staticmethod
   976→    def _router_smart_truncate(text: str, max_len: int) -> str:
   977→        """Smart truncation for AI router — cuts at sentence/paragraph boundary.
   978→        
   979→        Same logic as channel.py's _smart_truncate to avoid mid-word cuts.
   980→        """
   981→        if len(text) <= max_len:
   982→            return text
   983→        
   984→        target = max_len - 3
   985→        if target < 50:
   986→            return text[:target] + "..."
   987→        
   988→        search_zone = text[:target + 1]
   989→        
   990→        # 1. Paragraph break
   991→        last_para = search_zone.rfind("\n\n")
   992→        if last_para > target * 0.5:
   993→            return text[:last_para].rstrip() + "..."
   994→        
   995→        # 2. Sentence end
   996→        sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   997→        best_sentence_end = -1
   998→        for end_char in sentence_end_chars:
   999→            pos = search_zone.rfind(end_char)
  1000→            if pos > best_sentence_end and pos > target * 0.5:
  1001→                best_sentence_end = pos + len(end_char) - 1
  1002→        
  1003→        if best_sentence_end > target * 0.5:
  1004→            return text[:best_sentence_end + 1].rstrip() + "..."
  1005→        
  1006→        # 3. Newline
  1007→        last_newline = search_zone.rfind("\n")
  1008→        if last_newline > target * 0.5:
  1009→            return text[:last_newline].rstrip() + "..."
  1010→

  1188→    @staticmethod
  1189→    def _clean_ai_response(text: str) -> str:
  1190→        """Clean AI response artifacts (think tags, markdown, structured output, etc.)."""
  1191→        if not text:
  1192→            return ""
  1193→
  1194→        # Block structured YAML/MIDI/JSON output from text-to-music models
  1195→        import re
  1196→        structured_patterns = [
  1197→            r'^title:\s*.+?\n(duration|key|notation|pitch|velocity|tempo|bpm):',
  1198→            r'^---\s*\n.*?(title|duration|notation|pitch|velocity):',
  1199→            r'pitch,\s*time,\s*duration,\s*velocity',
  1200→        ]
  1201→        for pattern in structured_patterns:
  1202→            if re.search(pattern, text, re.DOTALL | re.IGNORECASE):
  1203→                return ""
  1204→
  1205→        # Strip think tags (Qwen3, reasoning models)
  1206→        text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1207→        text = re.sub(r'<thinking\b[^>]*>.*?</thinking\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1208→        text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
  1209→        text = re.sub(r'</?thinking[^>]*>', '', text, flags=re.IGNORECASE)
  1210→        text = re.sub(r'<think\b[^>]*$', '', text, flags=re.IGNORECASE)
  1211→
  1212→        # Strip /no_think prefix
  1213→        text = re.sub(r'^/no_think\s*', '', text)
  1214→
  1215→        # Strip prefixes
  1216→        for prefix in ["Ася:", "Asya:", "АСЯ:", "Assistant:", "Ответ Аси:"]:
  1217→            if text.startswith(prefix):
  1218→                text = text[len(prefix):].strip()
  1219→
  1220→        # Strip quotes
  1221→        if text.startswith('"') and text.endswith('"'):
  1222→            text = text[1:-1]
  1223→        if text.startswith("'") and text.endswith("'"):
  1224→            text = text[1:-1]
  1225→
  1226→        text = text.strip("*").strip()
  1227→
  1228→        # Strip markdown
  1229→        text = re.sub(r'\*\*([^*]+)\*\*', r'\1', text)
  1230→        text = re.sub(r'\*([^*]+)\*', r'\1', text)
  1231→        text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1232→
  1233→        # Clean up whitespace
  1234→        text = re.sub(r'\n{3,}', '\n\n', text)
  1235→        text = text.strip()
  1236→
  1237→        return text
