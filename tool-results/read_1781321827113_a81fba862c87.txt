   370→def lookup_obd2_code(code: str) -> Optional[str]:
   371→    """Look up an OBD-II error code description."""
   372→    code = code.upper().strip()
   373→    if code in OBD2_CODES:
   374→        return OBD2_CODES[code]
   375→    # Try matching pattern PXXXX
   376→    match = re.match(r"[PBUCE]\d{3,4}", code)
   377→    if match:
   378→        code_key = match.group(0)
   379→        if len(code_key) == 4:
   380→            code_key = code_key[0] + "0" + code_key[1:]
   381→        return OBD2_CODES.get(code_key)
   382→    return None

   434→def detect_symptoms(text: str) -> List[str]:

   437→    detected = []

   446→def detect_obd2_codes(text: str) -> List[str]:
   447→    """Extract OBD-II codes from text."""
   448→    pattern = r'[PBUCE]\d{3,4}'
   449→    matches = re.findall(pattern, text.upper())
   450→    # Normalize to PXXXX format
   451→    normalized = []
   452→    for m in matches:
   453→        if len(m) == 4:
   454→            m = m[0] + "0" + m[1:]
   455→        if m in OBD2_CODES:
   456→            normalized.append(m)
   457→    return normalized

   460→def is_part_number(text: str) -> bool:
   461→    """Check if text looks like a part/article number (OEM number)."""
   462→    text = text.strip()
   463→    # Common OEM patterns: XXXXXX-XXXXX, XXX XXX XXXX, etc.
   464→    patterns = [
   465→        r'^\d{4,10}$',                          # Just digits (4-10)
   466→        r'^[A-Z]{2,3}\d{4,8}$',                 # Letters+digits (e.g. WA6166)
   467→        r'^[A-Z]{1,3}\d{3,6}[A-Z]?\d?$',        # e.g. VAG, BMW part numbers
   468→        r'^\d{3,6}[A-Z]{1,3}\d{0,4}$',          # Digits+letters
   469→        r'^[A-Z0-9]{4,15}[-\s][A-Z0-9]{3,15}$', # With separator
   470→        r'^[A-Z]{2}\d{9,12}$',                   # VIN-like fragment
   471→    ]
   472→    for pattern in patterns:
   473→        if re.match(pattern, text.upper()):
   474→            return True
   475→    return False

   478→def extract_part_numbers(text: str) -> List[str]:
   479→    """Extract possible part numbers from text."""
   480→    words = text.replace(',', ' ').replace('.', ' ').split()
   481→    parts = []
   482→    for word in words:
   483→        if is_part_number(word):
   484→            parts.append(word.upper())
   485→    return parts

   493→def build_diagnostic_context(text: str) -> str:
   494→    """Build additional context for AI when user describes a car problem."""
   495→    context_parts = []
   496→
   497→    # Detect car brand
   498→    brand = identify_car_brand(text)
   499→    if brand:
   500→        info = get_brand_info(brand)
   501→        if info:
   502→            context_parts.append(f"Марка авто: {brand} ({info['country']}, холдинг: {info['parent']})")
   503→
   504→    # Detect symptoms
   505→    symptoms = detect_symptoms(text)
   506→    if symptoms:
   507→        for symptom_cat in symptoms:
   508→            data = SYMPTOM_DIAGNOSIS[symptom_cat]
   509→            context_parts.append(f"Категория проблемы: {symptom_cat}")
   510→            context_parts.append(f"Возможные причины: {', '.join(data['possible_causes'][:5])}")
   511→            context_parts.append(f"Первые шаги: {', '.join(data['first_steps'][:3])}")
   512→
   513→    # Detect OBD-II codes
   514→    codes = detect_obd2_codes(text)
   515→    if codes:
   516→        for code in codes:
   517→            desc = lookup_obd2_code(code)
   518→            if desc:
   519→                context_parts.append(f"Код ошибки {code}: {desc}")
   520→
   521→    # Detect part numbers
   522→    parts = extract_part_numbers(text)
   523→    if parts:
   524→        context_parts.append(f"Артикулы запчастей в запросе: {', '.join(parts)}")
   525→
   526→    if context_parts:
   527→        return "\n".join(context_parts)
   528→    return ""
