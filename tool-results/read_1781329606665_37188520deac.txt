    50→# ── VIN / Body number detection ───────────────────────────────────────────────
    51→
    52→_VIN_PATTERN = re.compile(r'\b[A-HJ-NPR-Z0-9]{17}\b', re.IGNORECASE)
    53→# Also match VINs with spaces or dashes (users often type them separated)
    54→_VIN_FLEX_PATTERN = re.compile(
    55→    r'(?:VIN[-:]?\s*|вин[-:]?\s*|вин-код[-:]?\s*)?([A-HJ-NPR-Z0-9](?:[A-HJ-NPR-Z0-9\s\-]{14,22})[A-HJ-NPR-Z0-9])',
    56→    re.IGNORECASE
    57→)
    58→_BODY_NUMBER_PATTERN = re.compile(
    59→    r'(?:номер\s+кузова|кузовн?ой\s+номер|body\s*number|кузов)\s*[:\s]*([A-Z0-9\-/]{5,20})',
    60→    re.IGNORECASE
    61→)
    62→
    63→
    64→def _detect_vin(text: str) -> Optional[str]:
    65→    """Detect a VIN code (17 chars) in text."""
    66→    match = _VIN_PATTERN.search(text.upper())
    67→    if match:
    68→        vin = match.group(0)
    69→        # Validate check digit position (9th char)
    70→        if len(vin) == 17 and vin[8] in '0123456789X':
    71→            return vin
    72→    return None
    73→
    74→
    75→def _detect_body_number(text: str) -> Optional[str]:
    76→    """Detect a body number reference in text."""
    77→    match = _BODY_NUMBER_PATTERN.search(text)
    78→    if match:
    79→        return match.group(1)
    80→    return None

   118→def _guess_gender(first_name: str) -> str:
   119→    """Guess gender from Russian first name. Returns 'male', 'female', or 'unknown'."""
   120→    if not first_name:
   121→        return "unknown"
   122→
   123→    name_lower = first_name.lower().strip()
   124→
   125→    if name_lower in COMMON_MALE_NAMES:
   126→        return "male"
   127→    if name_lower in COMMON_FEMALE_NAMES:
   128→        return "female"
   129→
   130→    if name_lower.endswith(FEMALE_NAME_ENDINGS):
   131→        if name_lower.endswith("ь"):
   132→            pass
   133→        else:
   134→            return "female"
   135→
   136→    if name_lower.endswith("й") or name_lower.endswith("ь"):
   137→        return "male"
   138→
   139→    return "unknown"

   999→def _smart_truncate(text: str, max_chars: int) -> str:
  1000→    """Truncate text to max_chars, trying not to cut in the middle of a sentence.
  1001→    
  1002→    Looks for the last sentence boundary (. ! ? …) within the limit.
  1003→    Falls back to last newline, then last space, then hard cut.
  1004→    Always appends "..." if truncated.
  1005→    """
  1006→    if len(text) <= max_chars:
  1007→        return text
  1008→
  1009→    # Reserve 3 chars for "..."
  1010→    limit = max_chars - 3
  1011→    if limit <= 0:
  1012→        return "..."
  1013→
  1014→    # Try to find last sentence boundary within limit
  1015→    sentence_endings = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
  1016→    best_pos = -1
  1017→    best_ending_len = 0
  1018→    for ending in sentence_endings:
  1019→        pos = text.rfind(ending, 0, limit + 1)
  1020→        if pos > best_pos:
  1021→            best_pos = pos
  1022→            best_ending_len = len(ending)
  1023→
  1024→    if best_pos > limit // 2:
  1025→        # Found a good sentence boundary — cut after the punctuation mark
  1026→        return text[:best_pos + best_ending_len].rstrip() + "..."
  1027→
  1028→    # Try last newline
  1029→    nl_pos = text.rfind('\n', 0, limit + 1)
  1030→    if nl_pos > limit // 2:
  1031→        return text[:nl_pos].rstrip() + "..."
  1032→
  1033→    # Try last space
  1034→    sp_pos = text.rfind(' ', 0, limit + 1)
  1035→    if sp_pos > limit // 2:
  1036→        return text[:sp_pos].rstrip() + "..."
  1037→
  1038→    # Hard cut
  1039→    return text[:limit].rstrip() + "..."

  1042→async def _send_response(message: Message, response, status_msg=None, partner_links=None):
  1043→    """Send AI response to user, handling errors and length limits.
  1044→    
  1045→    If partner_links is provided (list of (name, url) tuples), appends a cleanly
  1046→    formatted section with named links after the AI response text.
  1047→    """
  1048→    # Delete the "thinking" status message
  1049→    if status_msg:
  1050→        try:
  1051→            await status_msg.delete()
  1052→        except Exception:
  1053→            pass
  1054→
  1055→    if response.error:
  1056→        logger.error(f"AI error: {response.error_message}")
  1057→        await message.answer(
  1058→            "Ой, зависла немного 😅 Дай мне минутку и попробуй ещё раз!"
  1059→        )
  1060→        return
  1061→
  1062→    reply_text = response.text
  1063→
  1064→    # Ensure Asya doesn't use markdown formatting in chat
  1065→    reply_text = _clean_markdown(reply_text)
  1066→
  1067→    # CRITICAL: Replace any plain partner URLs (rossko.ru, autopiter.ru, etc.)
  1068→    # with affiliate goto_link equivalents from admitad_ads.json
  1069→    reply_text = _replace_plain_urls_with_affiliate(reply_text)
  1070→
  1071→    # Remove raw affiliate URLs that AI may have dumped into the response
  1072→    if partner_links:
  1073→        reply_text = _clean_raw_partner_urls(reply_text, partner_links)
  1074→
  1075→    # Append cleanly formatted partner links section
  1076→    if partner_links:
  1077→        partner_section = _format_partner_links_section(partner_links)
  1078→        if partner_section:
  1079→            reply_text = reply_text.rstrip() + "\n\n" + partner_section
  1080→
  1081→    # ── Enforce character limits based on chat type ──
  1082→    # Private chat: max CHAT_MAX_CHARS (1500) — AI asked for 500-1000, this is hard limit
  1083→    # Group/supergroup: max GROUP_MAX_CHARS (600) — AI asked for 300, this is hard limit
  1084→    # Comment in other group (not our channel): max COMMENT_MAX_CHARS (300)
  1085→    if message.chat.type == "private":
  1086→        if len(reply_text) > CHAT_MAX_CHARS:
  1087→            logger.info(f"Truncating private chat response: {len(reply_text)} → {CHAT_MAX_CHARS} chars")
  1088→            reply_text = _smart_truncate(reply_text, CHAT_MAX_CHARS)
  1089→    else:
  1090→        # Check if this is a comment in another group (not our channel)
  1091→        is_own_channel = (
  1092→            str(message.chat.id) == str(config.CHANNEL_ID)
  1093→            or message.chat.username == config.CHANNEL_ID.replace("@", "")
  1094→        )
  1095→        if not is_own_channel and message.chat.type in ("group", "supergroup"):
  1096→            if len(reply_text) > COMMENT_MAX_CHARS:
  1097→                logger.info(f"Truncating comment in other group: {len(reply_text)} → {COMMENT_MAX_CHARS} chars")
  1098→                reply_text = _smart_truncate(reply_text, COMMENT_MAX_CHARS)
  1099→        elif len(reply_text) > GROUP_MAX_CHARS:
  1100→            # Group or supergroup — keep it short!
  1101→            logger.info(f"Truncating group response: {len(reply_text)} → {GROUP_MAX_CHARS} chars")
  1102→            reply_text = _smart_truncate(reply_text, GROUP_MAX_CHARS)
  1103→
  1104→    # Split long messages (Telegram limit 4096 chars)
  1105→    # Plain text — no HTML parse mode needed since partner links are plain text
  1106→    if len(reply_text) <= config.TELEGRAM_TEXT_LIMIT:
  1107→        await message.answer(reply_text)
  1108→    else:
  1109→        chunks = _split_message(reply_text, max_length=config.TELEGRAM_TEXT_LIMIT)
  1110→        for chunk in chunks:
  1111→            await message.answer(chunk)

  1154→def _clean_raw_partner_urls(text: str, partner_links: list) -> str:
  1155→    """Remove raw affiliate URLs that AI may have dumped into the response.
  1156→    
  1157→    If the AI already included the full affiliate URLs in its response text,
  1158→    this function removes them to avoid duplication with the clean formatted
  1159→    section that will be appended by _format_partner_links_section.
  1160→    Also removes shop name + URL patterns like "Росско: https://...".
  1161→    """
  1162→    if not partner_links:
  1163→        return text
  1164→    
  1165→    for name, url in partner_links:
  1166→        url_escaped = re.escape(url)
  1167→        # Remove standalone URL lines
  1168→        text = re.sub(
  1169→            rf'^[\s\-—]*{url_escaped}[\s]*$',
  1170→            '',
  1171→            text,
  1172→            flags=re.MULTILINE
  1173→        )
  1174→        # Remove "name: URL" or "name — URL" patterns on their own lines
  1175→        text = re.sub(
  1176→            rf'^[\s\-—]*{re.escape(name)}\s*[:\-—]\s*{url_escaped}[\s]*$',
  1177→            '',
  1178→            text,
  1179→            flags=re.MULTILINE
  1180→        )
  1181→        # Remove any line containing the full raw affiliate URL (even mid-sentence)
  1182→        # This catches cases where AI writes: "Посмотри на Росско: https://ujhjj.com/g/..."
  1183→        text = re.sub(
  1184→            rf'[^\n]*{url_escaped}[^\n]*',
  1185→            '',
  1186→            text,
  1187→            flags=re.MULTILINE
  1188→        )
  1189→        # Remove bare URL anywhere in text (affiliate URLs are very long, easy to detect)
  1190→        if len(url) > 50:  # Only long tracking URLs
  1191→            # Match the base tracking domain pattern
  1192→            url_base = re.escape(url.split('?')[0])  # Before query params
  1193→            text = re.sub(
  1194→                rf'{url_base}\?[^\s\n"]+',
  1195→                '',
  1196→                text
  1197→            )
  1198→    
  1199→    text = re.sub(r'\n{3,}', '\n\n', text)
  1200→    return text.strip()

  1226→def _replace_plain_urls_with_affiliate(text: str) -> str:
  1227→    """Replace any plain partner domain URLs with affiliate goto_link equivalents.
  1228→    
  1229→    When the AI generates responses containing plain URLs like rossko.ru or 
  1230→    autopiter.ru instead of the affiliate tracking links from admitad_ads.json,
  1231→    this function detects them and replaces with the proper goto_link.
  1232→    
  1233→    This handles cases where:
  1234→    - AI ignores the system prompt and invents plain URLs
  1235→    - AI uses domain names without the affiliate wrapper
  1236→    - Photo handler bypasses the normal link injection pipeline
  1237→    """
  1238→    try:
  1239→        partner_manager.ensure_loaded()
  1240→    except Exception:
  1241→        return text
  1242→    
  1243→    for domain, display_name in _PARTNER_DOMAINS_MAP.items():
  1244→        prog = partner_manager.get_by_site(domain)
  1245→        if not prog or not prog.goto_link:
  1246→            continue
  1247→        
  1248→        affiliate_url = prog.goto_link
  1249→        
  1250→        # Pattern 1: Full URLs with paths — https://rossko.ru/search?text=abc
  1251→        # Replace the entire URL with the affiliate link
  1252→        pattern = rf'https?://{re.escape(domain)}[^\s<>)\]"\']*'
  1253→        matches = re.findall(pattern, text)
  1254→        for plain_url in matches:
  1255→            # Try to extract search query and build affiliate link with search
  1256→            search_query = ""
  1257→            if "search" in plain_url or "querystr" in plain_url or "q=" in plain_url:
  1258→                try:
  1259→                    from urllib.parse import urlparse, parse_qs
  1260→                    parsed = urlparse(plain_url)
  1261→                    params = parse_qs(parsed.query)
  1262→                    for key in ("text", "querystr", "q", "query", "SearchText", "keyword", "p"):
  1263→                        if key in params:
  1264→                            search_query = params[key][0]
  1265→                            break
  1266→                except Exception:
  1267→                    pass
  1268→            
  1269→            if search_query:
  1270→                replacement = prog.get_search_url(search_query)
  1271→            else:
  1272→                replacement = affiliate_url
  1273→            
  1274→            text = text.replace(plain_url, replacement)
  1275→        
  1276→        # Pattern 2: Bare domain mentions — rossko.ru or www.rossko.ru (not already part of a longer URL)
  1277→        # Only replace if it's NOT already part of an affiliate URL (which would be much longer)
  1278→        bare_pattern = rf'(?<![/\w.-])(?:www\.)?{re.escape(domain)}(?![/\w.-])'
  1279→        bare_matches = re.findall(bare_pattern, text)
  1280→        for bare_domain in bare_matches:
  1281→            # Check this isn't already inside an affiliate URL
  1282→            idx = text.find(bare_domain)
  1283→            if idx > 0:
  1284→                # Look back — if there's a tracking domain prefix, skip
  1285→                before = text[max(0, idx-50):idx]
  1286→                if any(tracking_domain in before for tracking_domain in ["ad.admitad.com", ".com/g/", "xmknb.com", "ujhjj.com", "rcpsj.com", "sgkaa.com"]):
  1287→                    continue
  1288→            text = text.replace(bare_domain, affiliate_url, 1)
  1289→    
  1290→    return text

  1293→def _clean_markdown(text: str) -> str:
  1294→    """Remove markdown formatting that Asya shouldn't use.
  1295→    
  1296→    Markdown links [text](url) are converted to plain text: text — url.
  1297→    Other markdown is stripped. No HTML — plain text for chat responses.
  1298→    """
  1299→    # Convert markdown links [text](url) → text — url (plain text format)
  1300→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1 — \2', text)
  1301→    
  1302→    # Remove bold (after escaping, ** won't be in &lt; form)
  1303→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
  1304→    # Remove italic
  1305→    text = re.sub(r'\*(.+?)\*', r'\1', text)
  1306→    # Remove code blocks
  1307→    text = re.sub(r'```[\s\S]*?```', lambda m: m.group(0).strip('`').strip(), text)
  1308→    # Remove inline code
  1309→    text = re.sub(r'`(.+?)`', r'\1', text)
  1310→    # Remove headers
  1311→    text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1312→    # Remove bullet points (convert to simple text)
  1313→    text = re.sub(r'^[-*]\s+', '— ', text, flags=re.MULTILINE)
  1314→    return text

  1317→def _split_message(text: str, max_length: int = 4096) -> list:
  1318→    """Split a long message into chunks at paragraph boundaries."""
  1319→    if len(text) <= max_length:
  1320→        return [text]
  1321→
  1322→    chunks = []
  1323→    while text:
  1324→        if len(text) <= max_length:
  1325→            chunks.append(text)
  1326→            break
  1327→
  1328→        # Try to split at newline near the limit
  1329→        split_pos = text.rfind('\n', 0, max_length)
  1330→        if split_pos < max_length // 2:
  1331→            # Try splitting at space
  1332→            split_pos = text.rfind(' ', 0, max_length)
  1333→        if split_pos < max_length // 2:
  1334→            # Hard split
  1335→            split_pos = max_length
  1336→
  1337→        chunks.append(text[:split_pos].strip())
  1338→        text = text[split_pos:].strip()
  1339→
  1340→    return chunks
