    96→class AIRouter:

    98→
    99→    v8.0 LOCAL-FIRST strategy:
   100→    - Level 0: Local Model (Qwen3-4B GGUF) — CHAT & COMMENT routes
   101→    - Level 1: Pollinations with API key (best quality, 60+ models)
   102→    - Level 2: Pollinations FREE API (no auth, rate-limited)
   103→    - Level 3: Cloudflare Workers AI (Mistral Small 3.1, 20K req/day)
   104→    - Last resort: Static fallback responses

   183→    async def chat(
   184→        self,
   185→        user_id: int,
   186→        message: str,
   187→        system_prompt: str = "",
   188→        model: str = "",
   189→        temperature: float = None,
   190→        max_tokens: int = None,
   191→        use_cache: bool = True,
   192→        save_history: bool = True,
   193→        extra_context: str = "",
   194→        route_type: str = "chat",
   195→    ) -> AIResponse:
   196→        """
   197→        Send a chat message through the AI router with 4-level failover.
   198→
   199→        v8.0 ROUTING via route_type:
   200→        - "chat" (default): Local → Pollinations key → free → Cloudflare → Static
   201→        - "function": Pollinations key → free → Cloudflare → Local(fallback) → Static
   202→        - "comment": Local → Pollinations key → free → Cloudflare → Static
   203→        """
   204→        temperature = temperature or config.CHAT_TEMPERATURE
   205→        max_tokens = max_tokens or config.CHAT_MAX_TOKENS
   206→
   207→        # Build system prompt with time context
   208→        sys_prompt = self._build_system_prompt(system_prompt, extra_context)
   209→
   210→        # Check cache first
   211→        if use_cache:
   212→            cache_key = self._make_cache_key(sys_prompt, message)
   213→            cached = await get_ai_cached(cache_key)
   214→            if cached:
   215→                return AIResponse(
   216→                    text=cached,
   217→                    model="cached",
   218→                    provider="cache",
   219→                    cached=True,
   220→                )
   221→
   222→        # Get chat history for context
   223→        history = await get_chat_history(user_id)
   224→
   225→        # ── Select model based on route_type ──
   226→        if route_type == "comment":
   227→            # Fast/cheap models for comments
   228→            model = model or "mistral"
   229→        elif route_type == "function":
   230→            # Best quality models for function routes
   231→            model = model or "openai-large"
   232→        else:
   233→            # CHAT (default): use provided model or let provider handle weighted selection
   234→            model = model or ""
   235→
   236→        # ── LEVEL 0: Local Model (Qwen3-4B) — CHAT & COMMENT routes ──
   237→        # Local model is primary for simple chat and comments (saves cloud balance)
   238→        # Skip for function routes (need cloud quality) and when explicitly disabled
   239→        use_local_first = route_type in ("chat", "comment") and config.ENABLE_LOCAL_MODEL
   240→
   241→        if use_local_first:
   242→            response = await self._try_local(
   243→                user_id, message, history, sys_prompt, temperature, max_tokens
   244→            )
   245→
   246→            if not response.error:
   247→                self._level0_count += 1
   248→                self._total_requests += 1
   249→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   250→
   251→            logger.debug(f"Level 0 (local) failed for route={route_type}: {response.error_message}")
   252→
   253→        # ── LEVEL 1: Pollinations with API key ──
   254→        response = await self._try_pollinations(
   255→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   256→        )
   257→
   258→        if not response.error:
   259→            self._level1_count += 1
   260→            self._total_requests += 1
   261→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   262→
   263→        # ── LEVEL 2: Pollinations FREE API ──
   264→        logger.warning(f"Level 1 failed (route={route_type}), trying Level 2 (free API): {response.error_message}")
   265→        response = await self._try_pollinations_free(
   266→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   267→        )
   268→
   269→        if not response.error:
   270→            self._level2_count += 1
   271→            self._total_requests += 1
   272→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   273→
   274→        # ── LEVEL 3: Cloudflare Workers AI ──
   275→        logger.warning(f"Level 2 failed (route={route_type}), trying Level 3 (Cloudflare): {response.error_message}")
   276→        response = await self._try_cloudflare(
   277→            user_id, message, history, sys_prompt, temperature, max_tokens
   278→        )
   279→
   280→        if not response.error:
   281→            self._level3_count += 1
   282→            self._total_requests += 1
   283→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   284→
   285→        # ── LEVEL 0 FALLBACK: Local model for FUNCTION routes ──
   286→        # If all cloud providers failed, try local model as last resort before static
   287→        if route_type == "function" and config.ENABLE_LOCAL_MODEL:
   288→            logger.warning(f"Level 3 failed (route=function), trying Level 0 fallback (local): {response.error_message}")
   289→            response = await self._try_local(
   290→                user_id, message, history, sys_prompt, temperature, max_tokens
   291→            )
   292→
   293→            if not response.error:
   294→                self._level0_count += 1
   295→                self._total_requests += 1
   296→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   297→
   298→        # ── LAST RESORT: Static fallback ──
   299→        self._static_count += 1
   300→        self._total_fallbacks += 1
   301→        logger.error(
   302→            f"ALL LEVELS FAILED for route_type={route_type}. "
   303→            f"Level0(local)={self._level0_count}, Level1={self._level1_count}, "
   304→            f"Level2={self._level2_count}, Level3={self._level3_count}, Static={self._static_count}"
   305→        )
   306→        return AIResponse(
   307→            text=random.choice(FALLBACK_RESPONSES),
   308→            model="fallback",
   309→            provider="static",
   310→            tokens_used=0,
   311→        )

   313→    async def _save_response(
   314→        self, user_id: int, message: str, response: AIResponse,
   315→        sys_prompt: str, use_cache: bool, save_history: bool
   316→    ) -> AIResponse:
   317→        """Save response to history and cache."""
   318→        if save_history and response.text:
   319→            await add_chat_message(user_id, "user", message)
   320→            await add_chat_message(user_id, "assistant", response.text)
   321→
   322→            # Cache the response
   323→            if use_cache and response.text:
   324→                cache_key = self._make_cache_key(sys_prompt, message)
   325→                await set_ai_cached(cache_key, message, response.text, response.model)
   326→
   327→        return response

   329→    async def _try_pollinations(self, user_id: int, message: str, history: list,
   330→                                 sys_prompt: str, temperature: float, max_tokens: int,
   331→                                 model: str) -> AIResponse:
   332→        """Level 1: Try Pollinations with API key (KEY1 → KEY2 internally).
   333→        Also tries model-level fallbacks.
   334→        """
   335→        messages = self._primary.format_messages(sys_prompt, history, message)
   336→
   337→        # Try primary model (provider will try KEY1 → KEY2 internally)
   338→        response = await self._primary.chat(
   339→            messages=messages,
   340→            model=model,
   341→            temperature=temperature,
   342→            max_tokens=max_tokens,
   343→        )
   344→
   345→        # If primary failed, try fallback models
   346→        if response.error:
   347→            is_key_error = any(code in (response.error_message or "")
   348→                              for code in ["All API keys depleted", "401", "402", "unavailable", "cooldown"])
   349→
   350→            if is_key_error:
   351→                fallback_models = [m for m in FALLBACK_MODELS
   352→                                   if m != model and not self._primary._is_model_in_cooldown(m)][:3]
   353→                if fallback_models:
   354→                    logger.info(f"Key error, trying {len(fallback_models)} fallback models")
   355→            else:
   356→                fallback_models = [
   357→                    m for m in ["mistral-small", "deepseek-v4", "llama-3.3", "nova-fast",
   358→                                "gemma", "qwen3-coder", "step-3.5-flash", "nova-micro"]
   359→                    if m != model and not self._primary._is_model_in_cooldown(m)
   360→                ][:3]
   361→                logger.info(f"Non-key error, trying fallbacks: {fallback_models}")
   362→
   363→            for fallback_model in fallback_models:
   364→                if fallback_model == model:
   365→                    continue
   366→                if self._primary._is_model_in_cooldown(fallback_model):
   367→                    continue
   368→                logger.info(f"Trying fallback model: {fallback_model}")
   369→                response = await self._primary.chat(
   370→                    messages=messages,
   371→                    model=fallback_model,
   372→                    temperature=temperature,
   373→                    max_tokens=max_tokens,
   374→                )
   375→                if not response.error:
   376→                    break
   377→
   378→        return response

   380→    async def _try_pollinations_free(self, user_id: int, message: str, history: list,
   381→                                      sys_prompt: str, temperature: float, max_tokens: int,
   382→                                      model: str) -> AIResponse:
   383→        """Level 2: Try Pollinations FREE API (no auth, text.pollinations.ai)."""
   384→        messages = self._primary.format_messages(sys_prompt, history, message)
   385→
   386→        # Use simpler models for free API — they're more reliable
   387→        free_models = ["openai", "mistral", "openai-large"]
   388→        if model and model not in free_models:
   389→            free_models.insert(0, model)
   390→
   391→        for free_model in free_models[:3]:
   392→            if self._primary._is_model_in_cooldown(free_model):
   393→                continue
   394→            result = await self._primary.chat_free(
   395→                messages=messages,
   396→                model=free_model,
   397→                temperature=temperature,
   398→                max_tokens=max_tokens,
   399→            )
   400→            if not result.error and result.text:
   401→                return result
   402→
   403→        return AIResponse(
   404→            text="",
   405→            model=model or "openai",
   406→            provider="pollinations-free",
   407→            error=True,
   408→            error_message="Free API failed for all models",
   409→        )

   433→    async def _try_local(self, user_id: int, message: str, history: list,
   434→                          sys_prompt: str, temperature: float, max_tokens: int) -> AIResponse:
   435→        """Level 0: Try Local Model (Qwen3-4B).
   436→
   437→        Uses ChatML format for Qwen3 with /no_think for fast responses.
   438→        Primary for CHAT and COMMENT routes, fallback for FUNCTION routes.
   439→        
   440→        CRITICAL: Uses COMPACT system prompt (persona.local_system_prompt) instead
   441→        of the full 12K-char prompt which exceeds the 4096 token context window.
   442→        The full prompt is ~9000 tokens — impossible to fit in 4096 ctx even alone.
   443→        """
   444→        if not self._local:
   445→            return AIResponse(
   446→                text="",
   447→                model="local-qwen3-4b",
   448→                provider="local",
   449→                error=True,
   450→                error_message="Local provider not initialized",
   451→            )
   452→
   453→        if not config.ENABLE_LOCAL_MODEL:
   454→            return AIResponse(
   455→                text="",
   456→                model="local-qwen3-4b",
   457→                provider="local",
   458→                error=True,
   459→                error_message="Local model disabled (ENABLE_LOCAL_MODEL=false)",
   460→            )
   461→
   462→        # Use COMPACT system prompt for local model — the full 12K-char prompt
   463→        # (~9000 tokens) does NOT fit in the 4096 token context window.
   464→        # Compact prompt is ~500 chars (~350 tokens) — leaves room for conversation.
   465→        compact_prompt = persona.local_system_prompt
   466→
   467→        # Build messages for local model using its own ChatML format
   468→        messages = [{"role": "system", "content": compact_prompt}]
   469→
   470→        # Add limited history for local model (saves context window)
   471→        # CRITICAL: With 4096 ctx and ~1.3 chars/token for Russian, we have
   472→        # ~5300 chars total. System prompt ~500 chars, max_tokens output ~665 chars,
   473→        # leaving ~4135 chars for history + message. That's ~6 history turns at 300 chars.
   474→        # Use 3 turns to be safe and leave room for the user message.
   475→        history_limit = min(config.MODEL_HISTORY_LIMIT, 3)
   476→        limited_history = history[-history_limit:] if history else []
   477→        for msg in limited_history:
   478→            role = msg.get("role", "user")
   479→            content = msg.get("content", "")
   480→            # Truncate long history messages to save context tokens
   481→            if role in ("user", "assistant") and content:
   482→                messages.append({"role": role, "content": content[:200]})
   483→
   484→        # CRITICAL: Truncate user message to prevent context overflow.
   485→        # With 4096 ctx, the user message should not exceed ~1500 chars
   486→        # (after system prompt + history are accounted for).
   487→        truncated_message = message[:1500] if len(message) > 1500 else message
   488→        if len(message) > 1500:
   489→            logger.debug(f"Truncated user message for local model: {len(message)} → 1500 chars")
   490→        messages.append({"role": "user", "content": truncated_message})
   491→
   492→        return await self._local.chat(
   493→            messages=messages,
   494→            temperature=temperature,
   495→            max_tokens=min(max_tokens, config.MODEL_MAX_TOKENS),
   496→        )

   498→    async def analyze_image(
   499→        self,
   500→        user_id: int,
   501→        image_url: str = "",
   502→        image_base64: str = "",
   503→        prompt: str = "",
   504→        extra_context: str = "",
   505→    ) -> AIResponse:
   506→        """
   507→        Analyze an image with 3-level vision failover.
   508→
   509→        Level 1: Pollinations vision (with key)
   510→        Level 2: Pollinations vision (free API)
   511→        Level 3: Cloudflare vision (Mistral Small 3.1 with image_url)
   512→        Last resort: Static fallback
   513→        """
   514→        # Build the vision prompt
   515→        if not prompt:
   516→            prompt = "Рассмотри это изображение внимательно и подробно опиши что на нём."
   517→
   518→        sys_prompt = self._build_system_prompt(
   519→            persona.system_prompt + persona.vision_prompt_suffix,
   520→            extra_context,
   521→        )
   522→
   523→        # Get chat history for context
   524→        history = await get_chat_history(user_id)
   525→
   526→        # Build context from history
   527→        context_summary = ""

   824→    @staticmethod
   825→    def _router_smart_truncate(text: str, max_len: int) -> str:
   826→        """Smart truncation for AI router — cuts at sentence/paragraph boundary.
   827→        
   828→        Same logic as channel.py's _smart_truncate to avoid mid-word cuts.
   829→        """
   830→        if len(text) <= max_len:
   831→            return text
   832→        
   833→        target = max_len - 3
   834→        if target < 50:
   835→            return text[:target] + "..."
   836→        
   837→        search_zone = text[:target + 1]
   838→        
   839→        # 1. Paragraph break
   840→        last_para = search_zone.rfind("\n\n")
   841→        if last_para > target * 0.5:
   842→            return text[:last_para].rstrip() + "..."
   843→        
   844→        # 2. Sentence end
   845→        sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   846→        best_sentence_end = -1
   847→        for end_char in sentence_end_chars:
   848→            pos = search_zone.rfind(end_char)
   849→            if pos > best_sentence_end and pos > target * 0.5:
   850→                best_sentence_end = pos + len(end_char) - 1
   851→        
   852→        if best_sentence_end > target * 0.5:
   853→            return text[:best_sentence_end + 1].rstrip() + "..."
   854→        
   855→        # 3. Newline
   856→        last_newline = search_zone.rfind("\n")
   857→        if last_newline > target * 0.5:
   858→            return text[:last_newline].rstrip() + "..."
   859→        
   860→        # 4. Space
   861→        last_space = search_zone.rfind(" ")
   862→        if last_space > target * 0.5:
   863→            return text[:last_space].rstrip() + "..."
   864→        
   865→        # 5. Hard cut
   866→        return text[:target].rstrip() + "..."

   870→    def _finalize_channel_post(self, response: AIResponse, has_media: bool) -> AIResponse:
   871→        """Finalize channel post: add footer, enforce limits."""
   872→        if response.text and not response.error:
   873→            text = response.text
   874→
   875→            # Clean markdown-style links
   876→            import re
   877→            text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1', text)
   878→            text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
   879→            text = re.sub(r'\*(.+?)\*', r'\1', text)
   880→
   881→            if "#sochiautoparts" not in text:
   882→                text = text.rstrip() + "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   883→            elif "@asiaexp_bot" not in text:
   884→                text = text.replace("@sochiautoparts", "Автор @asiaexp_bot\n@sochiautoparts")
   885→
   886→            # Enforce character limit (use smart truncation from channel.py)
   887→            footer = "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   888→            if has_media and len(text) > config.TELEGRAM_CAPTION_LIMIT:
   889→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   890→                    text = text.replace(foot_part, "")
   891→                text = text.rstrip()
   892→                max_content = config.TELEGRAM_CAPTION_LIMIT - len(footer)
   893→                if len(text) > max_content:
   894→                    # Smart truncation — find last sentence/paragraph boundary
   895→                    text = self._router_smart_truncate(text, max_content)
   896→                text += footer
   897→            elif not has_media and len(text) > config.TELEGRAM_TEXT_LIMIT:
   898→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   899→                    text = text.replace(foot_part, "")
   900→                text = text.rstrip()
   901→                max_content = config.TELEGRAM_TEXT_LIMIT - len(footer)
   902→                if len(text) > max_content:
   903→                    text = self._router_smart_truncate(text, max_content)
   904→                text += footer
   905→
   906→            response.text = text
   907→
   908→        return response

   978→    @staticmethod
   979→    def _clean_ai_response(text: str) -> str:
   980→        """Clean AI response artifacts (think tags, markdown, structured output, etc.)."""
   981→        if not text:
   982→            return ""
   983→
   984→        # Block structured YAML/MIDI/JSON output from text-to-music models
   985→        import re
   986→        structured_patterns = [
   987→            r'^title:\s*.+?\n(duration|key|notation|pitch|velocity|tempo|bpm):',
   988→            r'^---\s*\n.*?(title|duration|notation|pitch|velocity):',
   989→            r'pitch,\s*time,\s*duration,\s*velocity',
   990→        ]
   991→        for pattern in structured_patterns:
   992→            if re.search(pattern, text, re.DOTALL | re.IGNORECASE):
   993→                return ""
   994→
   995→        # Strip think tags (Qwen3, reasoning models)
   996→        text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
   997→        text = re.sub(r'<thinking\b[^>]*>.*?</thinking\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
   998→        text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
