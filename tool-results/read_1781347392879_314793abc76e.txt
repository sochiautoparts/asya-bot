    97→class AIRouter:

   103→    - Level 2: Pollinations FREE API (no auth, rate-limited)
   104→    - Level 3: Cloudflare Workers AI (Mistral Small 3.1, 20K req/day)
   105→    - Last resort: Static fallback responses

   184→    async def chat(
   185→        self,
   186→        user_id: int,
   187→        message: str,
   188→        system_prompt: str = "",
   189→        model: str = "",
   190→        temperature: float = None,
   191→        max_tokens: int = None,
   192→        use_cache: bool = True,
   193→        save_history: bool = True,
   194→        extra_context: str = "",
   195→        route_type: str = "chat",
   196→    ) -> AIResponse:
   197→        """
   198→        Send a chat message through the AI router with 4-level failover.
   199→
   200→        v8.0 ROUTING via route_type:
   201→        - "chat" (default): Local → Pollinations key → free → Cloudflare → Static
   202→        - "function": Pollinations key → free → Cloudflare → Local(fallback) → Static
   203→        - "comment": Local → Pollinations key → free → Cloudflare → Static
   204→        """
   205→        temperature = temperature or config.CHAT_TEMPERATURE
   206→        max_tokens = max_tokens or config.CHAT_MAX_TOKENS
   207→
   208→        # Build system prompt with time context
   209→        sys_prompt = self._build_system_prompt(system_prompt, extra_context)
   210→
   211→        # Check cache first
   212→        if use_cache:
   213→            cache_key = self._make_cache_key(sys_prompt, message)
   214→            cached = await get_ai_cached(cache_key)
   215→            if cached:
   216→                return AIResponse(
   217→                    text=cached,
   218→                    model="cached",
   219→                    provider="cache",
   220→                    cached=True,
   221→                )
   222→
   223→        # Get chat history for context
   224→        history = await get_chat_history(user_id)
   225→
   226→        # ── Select model based on route_type ──
   227→        if route_type == "comment":
   228→            # Fast/cheap models for comments
   229→            model = model or "mistral"
   230→        elif route_type == "function":
   231→            # Best quality models for function routes
   232→            model = model or "openai-large"
   233→        else:
   234→            # CHAT (default): use provided model or let provider handle weighted selection
   235→            model = model or ""
   236→
   237→        # ── LEVEL 0: Local Model (Qwen3-4B) — CHAT & COMMENT routes ──
   238→        # Local model is primary for simple chat and comments (saves cloud balance)
   239→        # Skip for function routes (need cloud quality) and when explicitly disabled
   240→        use_local_first = route_type in ("chat", "comment") and config.ENABLE_LOCAL_MODEL
   241→
   242→        if use_local_first:
   243→            response = await self._try_local(
   244→                user_id, message, history, sys_prompt, temperature, max_tokens
   245→            )
   246→
   247→            if not response.error:
   248→                self._level0_count += 1
   249→                self._total_requests += 1
   250→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   251→
   252→            logger.debug(f"Level 0 (local) failed for route={route_type}: {response.error_message}")
   253→
   254→        # ── LEVEL 1: Pollinations with API key ──
   255→        response = await self._try_pollinations(
   256→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   257→        )
   258→
   259→        if not response.error:
   260→            self._level1_count += 1
   261→            self._total_requests += 1
   262→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   263→
   264→        # ── LEVEL 2: Pollinations FREE API ──
   265→        logger.warning(f"Level 1 failed (route={route_type}), trying Level 2 (free API): {response.error_message}")
   266→        response = await self._try_pollinations_free(
   267→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   268→        )
   269→
   270→        if not response.error:
   271→            self._level2_count += 1
   272→            self._total_requests += 1
   273→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   274→
   275→        # ── LEVEL 3: Cloudflare Workers AI ──
   276→        logger.warning(f"Level 2 failed (route={route_type}), trying Level 3 (Cloudflare): {response.error_message}")
   277→        response = await self._try_cloudflare(
   278→            user_id, message, history, sys_prompt, temperature, max_tokens
   279→        )
   280→
   281→        if not response.error:
   282→            self._level3_count += 1
   283→            self._total_requests += 1
   284→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   285→
   286→        # ── LEVEL 0 FALLBACK: Local model for FUNCTION routes ──
   287→        # If all cloud providers failed, try local model as last resort before static
   288→        if route_type == "function" and config.ENABLE_LOCAL_MODEL:
   289→            logger.warning(f"Level 3 failed (route=function), trying Level 0 fallback (local): {response.error_message}")
   290→            response = await self._try_local(
   291→                user_id, message, history, sys_prompt, temperature, max_tokens
   292→            )
   293→
   294→            if not response.error:
   295→                self._level0_count += 1
   296→                self._total_requests += 1
   297→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   298→
   299→        # ── LAST RESORT: Static fallback ──
   300→        self._static_count += 1
   301→        self._total_fallbacks += 1
   302→        logger.error(
   303→            f"ALL LEVELS FAILED for route_type={route_type}. "
   304→            f"Level0(local)={self._level0_count}, Level1={self._level1_count}, "
   305→            f"Level2={self._level2_count}, Level3={self._level3_count}, Static={self._static_count}"
   306→        )
   307→        return AIResponse(
   308→            text=random.choice(FALLBACK_RESPONSES),
   309→            model="fallback",
   310→            provider="static",
   311→            tokens_used=0,
   312→        )

   314→    async def _save_response(
   315→        self, user_id: int, message: str, response: AIResponse,
   316→        sys_prompt: str, use_cache: bool, save_history: bool
   317→    ) -> AIResponse:
   318→        """Save response to history and cache."""
   319→        if save_history and response.text:
   320→            await add_chat_message(user_id, "user", message)
   321→            await add_chat_message(user_id, "assistant", response.text)
   322→
   323→            # Cache the response
   324→            if use_cache and response.text:
   325→                cache_key = self._make_cache_key(sys_prompt, message)
   326→                await set_ai_cached(cache_key, message, response.text, response.model)
   327→
   328→        return response

   330→    async def _try_pollinations(self, user_id: int, message: str, history: list,
   331→                                 sys_prompt: str, temperature: float, max_tokens: int,
   332→                                 model: str) -> AIResponse:
   333→        """Level 1: Try Pollinations with API key (KEY1 → KEY2 internally).
   334→        Also tries model-level fallbacks.
   335→        """
   336→        messages = self._primary.format_messages(sys_prompt, history, message)
   337→
   338→        # Try primary model (provider will try KEY1 → KEY2 internally)
   339→        response = await self._primary.chat(
   340→            messages=messages,
   341→            model=model,
   342→            temperature=temperature,
   343→            max_tokens=max_tokens,
   344→        )
   345→
   346→        # If primary failed, try fallback models
   347→        if response.error:
   348→            is_key_error = any(code in (response.error_message or "")
   349→                              for code in ["All API keys depleted", "401", "402", "unavailable", "cooldown"])
   350→
   351→            if is_key_error:
   352→                fallback_models = [m for m in FALLBACK_MODELS
   353→                                   if m != model and not self._primary._is_model_in_cooldown(m)][:3]
   354→                if fallback_models:
   355→                    logger.info(f"Key error, trying {len(fallback_models)} fallback models")
   356→            else:
   357→                fallback_models = [
   358→                    m for m in ["mistral-small", "deepseek-v4", "llama-3.3", "nova-fast",
   359→                                "gemma", "qwen3-coder", "step-3.5-flash", "nova-micro"]
   360→                    if m != model and not self._primary._is_model_in_cooldown(m)
   361→                ][:3]
   362→                logger.info(f"Non-key error, trying fallbacks: {fallback_models}")
   363→
   364→            for fallback_model in fallback_models:
   365→                if fallback_model == model:
   366→                    continue
   367→                if self._primary._is_model_in_cooldown(fallback_model):
   368→                    continue
   369→                logger.info(f"Trying fallback model: {fallback_model}")
   370→                response = await self._primary.chat(
   371→                    messages=messages,
   372→                    model=fallback_model,
   373→                    temperature=temperature,
   374→                    max_tokens=max_tokens,
   375→                )
   376→                if not response.error:
   377→                    break
   378→
   379→        return response

   381→    async def _try_pollinations_free(self, user_id: int, message: str, history: list,
   382→                                      sys_prompt: str, temperature: float, max_tokens: int,
   383→                                      model: str) -> AIResponse:
   384→        """Level 2: Try Pollinations FREE API (no auth, text.pollinations.ai)."""
   385→        messages = self._primary.format_messages(sys_prompt, history, message)
   386→
   387→        # Use simpler models for free API — they're more reliable
   388→        free_models = ["openai", "mistral", "openai-large"]
   389→        if model and model not in free_models:
   390→            free_models.insert(0, model)
   391→
   392→        for free_model in free_models[:3]:
   393→            if self._primary._is_model_in_cooldown(free_model):
   394→                continue
   395→            result = await self._primary.chat_free(
   396→                messages=messages,
   397→                model=free_model,
   398→                temperature=temperature,
   399→                max_tokens=max_tokens,
   400→            )
   401→            if not result.error and result.text:
   402→                return result
   403→
   404→        return AIResponse(
   405→            text="",
   406→            model=model or "openai",
   407→            provider="pollinations-free",
   408→            error=True,
   409→            error_message="Free API failed for all models",
   410→        )

   434→    async def _try_local(self, user_id: int, message: str, history: list,
   435→                          sys_prompt: str, temperature: float, max_tokens: int) -> AIResponse:
   436→        """Level 0: Try Local Model (Qwen3-4B).
   437→
   438→        Uses ChatML format for Qwen3 with /no_think for fast responses.
   439→        Primary for CHAT and COMMENT routes, fallback for FUNCTION routes.
   440→        
   441→        CONTEXT WINDOW BUDGET (4096 tokens, ~1.3 chars/token for Russian):
   442→          - Output: 512 tokens (MODEL_MAX_TOKENS)
   443→          - Safety margin: 64 tokens
   444→          - Available for input: 3520 tokens (~4576 chars)
   445→          - System prompt (local_system_prompt v2): ~2340 chars (~1800 tokens)
   446→          - History: 3 turns × 200 chars = ~600 chars (~462 tokens)
   447→          - User message: up to ~1000 chars (~770 tokens)
   448→          - Total: ~3940 chars (~3032 tokens) — fits in 3520 with margin!
   449→        """
   450→        if not self._local:
   451→            return AIResponse(
   452→                text="",
   453→                model="local-qwen3-4b",
   454→                provider="local",
   455→                error=True,
   456→                error_message="Local provider not initialized",
   457→            )
   458→
   459→        if not config.ENABLE_LOCAL_MODEL:
   460→            return AIResponse(
   461→                text="",
   462→                model="local-qwen3-4b",
   463→                provider="local",
   464→                error=True,
   465→                error_message="Local model disabled (ENABLE_LOCAL_MODEL=false)",
   466→            )
   467→
   468→        # Use EXPANDED system prompt for local model (v2.0 — ~1800 tokens).
   469→        # The full 12K-char cloud prompt (~9000 tokens) still does NOT fit.
   470→        # But the expanded v2 prompt gives much better quality than the old compact one.
   471→        compact_prompt = persona.local_system_prompt
   472→
   473→        # Build messages for local model using its own ChatML format
   474→        messages = [{"role": "system", "content": compact_prompt}]
   475→
   476→        # Add limited history for local model (saves context window)
   477→        # With expanded prompt (~1800 tokens), we have ~1720 tokens left for history+message.
   478→        # 3 history turns × ~200 chars × 1.3 chars/token = ~462 tokens
   479→        # User message up to ~1000 chars = ~770 tokens
   480→        # Total: ~1232 tokens — well within budget.
   481→        history_limit = min(config.MODEL_HISTORY_LIMIT, 3)
   482→        limited_history = history[-history_limit:] if history else []
   483→        for msg in limited_history:
   484→            role = msg.get("role", "user")
   485→            content = msg.get("content", "")
   486→            # Truncate long history messages to save context tokens
   487→            if role in ("user", "assistant") and content:
   488→                messages.append({"role": role, "content": content[:200]})
   489→
   490→        # Truncate user message to prevent context overflow.
   491→        # With expanded system prompt, allow up to ~1000 chars for user message.
   492→        # The local_provider.chat() has its own safety truncation as well.
   493→        truncated_message = message[:1000] if len(message) > 1000 else message
   494→        if len(message) > 1000:
   495→            logger.debug(f"Truncated user message for local model: {len(message)} → 1000 chars")
   496→        messages.append({"role": "user", "content": truncated_message})
   497→
   498→        return await self._local.chat(
   499→            messages=messages,
   500→            temperature=temperature,
   501→            max_tokens=min(max_tokens, config.MODEL_MAX_TOKENS),
   502→        )

   504→    async def analyze_image(
   505→        self,
   506→        user_id: int,
   507→        image_url: str = "",
   508→        image_base64: str = "",
   509→        prompt: str = "",
   510→        extra_context: str = "",
   511→    ) -> AIResponse:
   512→        """
   513→        Analyze an image with 3-level vision failover.
   514→
   515→        Level 1: Pollinations vision (with key)
   516→        Level 2: Pollinations vision (free API)
   517→        Level 3: Cloudflare vision (Mistral Small 3.1 with image_url)
   518→        Last resort: Static fallback
   519→        """
   520→        # Build the vision prompt
   521→        if not prompt:
   522→            prompt = "Рассмотри это изображение внимательно и подробно опиши что на нём."
   523→

   830→    @staticmethod
   831→    def _router_smart_truncate(text: str, max_len: int) -> str:
   832→        """Smart truncation for AI router — cuts at sentence/paragraph boundary.
   833→        
   834→        Same logic as channel.py's _smart_truncate to avoid mid-word cuts.
   835→        """
   836→        if len(text) <= max_len:
   837→            return text
   838→        
   839→        target = max_len - 3
   840→        if target < 50:
   841→            return text[:target] + "..."
   842→        
   843→        search_zone = text[:target + 1]
   844→        
   845→        # 1. Paragraph break
   846→        last_para = search_zone.rfind("\n\n")
   847→        if last_para > target * 0.5:
   848→            return text[:last_para].rstrip() + "..."
   849→        
   850→        # 2. Sentence end
   851→        sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   852→        best_sentence_end = -1
   853→        for end_char in sentence_end_chars:
   854→            pos = search_zone.rfind(end_char)
   855→            if pos > best_sentence_end and pos > target * 0.5:
   856→                best_sentence_end = pos + len(end_char) - 1
   857→        
   858→        if best_sentence_end > target * 0.5:
   859→            return text[:best_sentence_end + 1].rstrip() + "..."
   860→        
   861→        # 3. Newline
   862→        last_newline = search_zone.rfind("\n")
   863→        if last_newline > target * 0.5:
   864→            return text[:last_newline].rstrip() + "..."
   865→        
   866→        # 4. Space
   867→        last_space = search_zone.rfind(" ")
   868→        if last_space > target * 0.5:
   869→            return text[:last_space].rstrip() + "..."
   870→        
   871→        # 5. Hard cut
   872→        return text[:target].rstrip() + "..."

   876→    def _finalize_channel_post(self, response: AIResponse, has_media: bool) -> AIResponse:
   877→        """Finalize channel post: add footer, enforce limits."""
   878→        if response.text and not response.error:
   879→            text = response.text
   880→
   881→            # Clean markdown-style links
   882→            import re
   883→            text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1', text)
   884→            text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
   885→            text = re.sub(r'\*(.+?)\*', r'\1', text)
   886→
   887→            if "#sochiautoparts" not in text:
   888→                text = text.rstrip() + "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   889→            elif "@asiaexp_bot" not in text:
   890→                text = text.replace("@sochiautoparts", "Автор @asiaexp_bot\n@sochiautoparts")
   891→
   892→            # Enforce character limit (use smart truncation from channel.py)
   893→            footer = "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   894→            if has_media and len(text) > config.TELEGRAM_CAPTION_LIMIT:
   895→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   896→                    text = text.replace(foot_part, "")
   897→                text = text.rstrip()
   898→                max_content = config.TELEGRAM_CAPTION_LIMIT - len(footer)
   899→                if len(text) > max_content:
   900→                    # Smart truncation — find last sentence/paragraph boundary
   901→                    text = self._router_smart_truncate(text, max_content)
   902→                text += footer
   903→            elif not has_media and len(text) > config.TELEGRAM_TEXT_LIMIT:
   904→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   905→                    text = text.replace(foot_part, "")
   906→                text = text.rstrip()
   907→                max_content = config.TELEGRAM_TEXT_LIMIT - len(footer)
   908→                if len(text) > max_content:
   909→                    text = self._router_smart_truncate(text, max_content)
   910→                text += footer
   911→
   912→            response.text = text
   913→
   914→        return response

  1104→    @staticmethod
  1105→    def _clean_ai_response(text: str) -> str:
  1106→        """Clean AI response artifacts (think tags, markdown, structured output, etc.)."""
  1107→        if not text:
  1108→            return ""
  1109→
  1110→        # Block structured YAML/MIDI/JSON output from text-to-music models
  1111→        import re
  1112→        structured_patterns = [
  1113→            r'^title:\s*.+?\n(duration|key|notation|pitch|velocity|tempo|bpm):',
  1114→            r'^---\s*\n.*?(title|duration|notation|pitch|velocity):',
  1115→            r'pitch,\s*time,\s*duration,\s*velocity',
  1116→        ]
  1117→        for pattern in structured_patterns:
  1118→            if re.search(pattern, text, re.DOTALL | re.IGNORECASE):
  1119→                return ""
  1120→
  1121→        # Strip think tags (Qwen3, reasoning models)
  1122→        text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1123→        text = re.sub(r'<thinking\b[^>]*>.*?</thinking\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1124→        text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
  1125→        text = re.sub(r'</?thinking[^>]*>', '', text, flags=re.IGNORECASE)
  1126→        text = re.sub(r'<think\b[^>]*$', '', text, flags=re.IGNORECASE)
  1127→
  1128→        # Strip /no_think prefix
  1129→        text = re.sub(r'^/no_think\s*', '', text)
  1130→
  1131→        # Strip prefixes
  1132→        for prefix in ["Ася:", "Asya:", "АСЯ:", "Assistant:", "Ответ Аси:"]:
  1133→            if text.startswith(prefix):
  1134→                text = text[len(prefix):].strip()
  1135→
  1136→        # Strip quotes
  1137→        if text.startswith('"') and text.endswith('"'):
  1138→            text = text[1:-1]
  1139→        if text.startswith("'") and text.endswith("'"):
  1140→            text = text[1:-1]
  1141→
  1142→        text = text.strip("*").strip()
  1143→
  1144→        # Strip markdown
  1145→        text = re.sub(r'\*\*([^*]+)\*\*', r'\1', text)
  1146→        text = re.sub(r'\*([^*]+)\*', r'\1', text)
  1147→        text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1148→
  1149→        # Clean up whitespace
  1150→        text = re.sub(r'\n{3,}', '\n\n', text)
  1151→        text = text.strip()
  1152→
  1153→        return text
