    96→class AIRouter:

   102→    - Level 2: Pollinations FREE API (no auth, rate-limited)
   103→    - Level 3: Cloudflare Workers AI (Mistral Small 3.1, 20K req/day)
   104→    - Last resort: Static fallback responses

   183→    async def chat(
   184→        self,
   185→        user_id: int,
   186→        message: str,
   187→        system_prompt: str = "",
   188→        model: str = "",
   189→        temperature: float = None,
   190→        max_tokens: int = None,
   191→        use_cache: bool = True,
   192→        save_history: bool = True,
   193→        extra_context: str = "",
   194→        route_type: str = "chat",
   195→    ) -> AIResponse:
   196→        """
   197→        Send a chat message through the AI router with 4-level failover.
   198→
   199→        v8.0 ROUTING via route_type:
   200→        - "chat" (default): Local → Pollinations key → free → Cloudflare → Static
   201→        - "function": Pollinations key → free → Cloudflare → Local(fallback) → Static
   202→        - "comment": Local → Pollinations key → free → Cloudflare → Static
   203→        """
   204→        temperature = temperature or config.CHAT_TEMPERATURE
   205→        max_tokens = max_tokens or config.CHAT_MAX_TOKENS
   206→
   207→        # Build system prompt with time context
   208→        sys_prompt = self._build_system_prompt(system_prompt, extra_context)
   209→
   210→        # Check cache first
   211→        if use_cache:
   212→            cache_key = self._make_cache_key(sys_prompt, message)
   213→            cached = await get_ai_cached(cache_key)
   214→            if cached:
   215→                return AIResponse(
   216→                    text=cached,
   217→                    model="cached",
   218→                    provider="cache",
   219→                    cached=True,
   220→                )
   221→
   222→        # Get chat history for context
   223→        history = await get_chat_history(user_id)
   224→
   225→        # ── Select model based on route_type ──
   226→        if route_type == "comment":
   227→            # Fast/cheap models for comments
   228→            model = model or "mistral"
   229→        elif route_type == "function":
   230→            # Best quality models for function routes
   231→            model = model or "openai-large"
   232→        else:
   233→            # CHAT (default): use provided model or let provider handle weighted selection
   234→            model = model or ""
   235→
   236→        # ── LEVEL 0: Local Model (Qwen3-4B) — CHAT & COMMENT routes ──
   237→        # Local model is primary for simple chat and comments (saves cloud balance)
   238→        # Skip for function routes (need cloud quality) and when explicitly disabled
   239→        use_local_first = route_type in ("chat", "comment") and config.ENABLE_LOCAL_MODEL
   240→
   241→        if use_local_first:
   242→            response = await self._try_local(
   243→                user_id, message, history, sys_prompt, temperature, max_tokens
   244→            )
   245→
   246→            if not response.error:
   247→                self._level0_count += 1
   248→                self._total_requests += 1
   249→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   250→
   251→            logger.debug(f"Level 0 (local) failed for route={route_type}: {response.error_message}")
   252→
   253→        # ── LEVEL 1: Pollinations with API key ──
   254→        response = await self._try_pollinations(
   255→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   256→        )
   257→
   258→        if not response.error:
   259→            self._level1_count += 1
   260→            self._total_requests += 1
   261→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   262→
   263→        # ── LEVEL 2: Pollinations FREE API ──
   264→        logger.warning(f"Level 1 failed (route={route_type}), trying Level 2 (free API): {response.error_message}")
   265→        response = await self._try_pollinations_free(
   266→            user_id, message, history, sys_prompt, temperature, max_tokens, model
   267→        )
   268→
   269→        if not response.error:
   270→            self._level2_count += 1
   271→            self._total_requests += 1
   272→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   273→
   274→        # ── LEVEL 3: Cloudflare Workers AI ──
   275→        logger.warning(f"Level 2 failed (route={route_type}), trying Level 3 (Cloudflare): {response.error_message}")
   276→        response = await self._try_cloudflare(
   277→            user_id, message, history, sys_prompt, temperature, max_tokens
   278→        )
   279→
   280→        if not response.error:
   281→            self._level3_count += 1
   282→            self._total_requests += 1
   283→            return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   284→
   285→        # ── LEVEL 0 FALLBACK: Local model for FUNCTION routes ──
   286→        # If all cloud providers failed, try local model as last resort before static
   287→        if route_type == "function" and config.ENABLE_LOCAL_MODEL:
   288→            logger.warning(f"Level 3 failed (route=function), trying Level 0 fallback (local): {response.error_message}")
   289→            response = await self._try_local(
   290→                user_id, message, history, sys_prompt, temperature, max_tokens
   291→            )
   292→
   293→            if not response.error:
   294→                self._level0_count += 1
   295→                self._total_requests += 1
   296→                return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   297→
   298→        # ── LAST RESORT: Static fallback ──
   299→        self._static_count += 1
   300→        self._total_fallbacks += 1
   301→        logger.error(
   302→            f"ALL LEVELS FAILED for route_type={route_type}. "
   303→            f"Level0(local)={self._level0_count}, Level1={self._level1_count}, "
   304→            f"Level2={self._level2_count}, Level3={self._level3_count}, Static={self._static_count}"
   305→        )
   306→        return AIResponse(
   307→            text=random.choice(FALLBACK_RESPONSES),
   308→            model="fallback",
   309→            provider="static",
   310→            tokens_used=0,
   311→        )

   313→    async def _save_response(
   314→        self, user_id: int, message: str, response: AIResponse,
   315→        sys_prompt: str, use_cache: bool, save_history: bool
   316→    ) -> AIResponse:
   317→        """Save response to history and cache."""
   318→        if save_history and response.text:
   319→            await add_chat_message(user_id, "user", message)
   320→            await add_chat_message(user_id, "assistant", response.text)
   321→
   322→            # Cache the response
   323→            if use_cache and response.text:
   324→                cache_key = self._make_cache_key(sys_prompt, message)
   325→                await set_ai_cached(cache_key, message, response.text, response.model)
   326→
   327→        return response

   329→    async def _try_pollinations(self, user_id: int, message: str, history: list,
   330→                                 sys_prompt: str, temperature: float, max_tokens: int,
   331→                                 model: str) -> AIResponse:
   332→        """Level 1: Try Pollinations with API key (KEY1 → KEY2 internally).
   333→        Also tries model-level fallbacks.
   334→        """
   335→        messages = self._primary.format_messages(sys_prompt, history, message)
   336→
   337→        # Try primary model (provider will try KEY1 → KEY2 internally)
   338→        response = await self._primary.chat(
   339→            messages=messages,
   340→            model=model,
   341→            temperature=temperature,
   342→            max_tokens=max_tokens,
   343→        )
   344→
   345→        # If primary failed, try fallback models
   346→        if response.error:
   347→            is_key_error = any(code in (response.error_message or "")
   348→                              for code in ["All API keys depleted", "401", "402", "unavailable", "cooldown"])
   349→
   350→            if is_key_error:
   351→                fallback_models = [m for m in FALLBACK_MODELS
   352→                                   if m != model and not self._primary._is_model_in_cooldown(m)][:3]
   353→                if fallback_models:
   354→                    logger.info(f"Key error, trying {len(fallback_models)} fallback models")
   355→            else:
   356→                fallback_models = [
   357→                    m for m in ["mistral-small", "deepseek-v4", "llama-3.3", "nova-fast",
   358→                                "gemma", "qwen3-coder", "step-3.5-flash", "nova-micro"]
   359→                    if m != model and not self._primary._is_model_in_cooldown(m)
   360→                ][:3]
   361→                logger.info(f"Non-key error, trying fallbacks: {fallback_models}")
   362→
   363→            for fallback_model in fallback_models:
   364→                if fallback_model == model:
   365→                    continue
   366→                if self._primary._is_model_in_cooldown(fallback_model):
   367→                    continue
   368→                logger.info(f"Trying fallback model: {fallback_model}")
   369→                response = await self._primary.chat(
   370→                    messages=messages,
   371→                    model=fallback_model,
   372→                    temperature=temperature,
   373→                    max_tokens=max_tokens,
   374→                )
   375→                if not response.error:
   376→                    break
   377→
   378→        return response

   380→    async def _try_pollinations_free(self, user_id: int, message: str, history: list,
   381→                                      sys_prompt: str, temperature: float, max_tokens: int,
   382→                                      model: str) -> AIResponse:
   383→        """Level 2: Try Pollinations FREE API (no auth, text.pollinations.ai)."""
   384→        messages = self._primary.format_messages(sys_prompt, history, message)
   385→
   386→        # Use simpler models for free API — they're more reliable
   387→        free_models = ["openai", "mistral", "openai-large"]
   388→        if model and model not in free_models:
   389→            free_models.insert(0, model)
   390→
   391→        for free_model in free_models[:3]:
   392→            if self._primary._is_model_in_cooldown(free_model):
   393→                continue
   394→            result = await self._primary.chat_free(
   395→                messages=messages,
   396→                model=free_model,
   397→                temperature=temperature,
   398→                max_tokens=max_tokens,
   399→            )
   400→            if not result.error and result.text:
   401→                return result
   402→
   403→        return AIResponse(
   404→            text="",
   405→            model=model or "openai",
   406→            provider="pollinations-free",
   407→            error=True,
   408→            error_message="Free API failed for all models",
   409→        )

   433→    async def _try_local(self, user_id: int, message: str, history: list,
   434→                          sys_prompt: str, temperature: float, max_tokens: int) -> AIResponse:
   435→        """Level 0: Try Local Model (Qwen3-4B).
   436→
   437→        Uses ChatML format for Qwen3 with /no_think for fast responses.
   438→        Primary for CHAT and COMMENT routes, fallback for FUNCTION routes.
   439→        
   440→        CONTEXT WINDOW BUDGET (4096 tokens, ~1.3 chars/token for Russian):
   441→          - Output: 512 tokens (MODEL_MAX_TOKENS)
   442→          - Safety margin: 64 tokens
   443→          - Available for input: 3520 tokens (~4576 chars)
   444→          - System prompt (local_system_prompt v2): ~2340 chars (~1800 tokens)
   445→          - History: 3 turns × 200 chars = ~600 chars (~462 tokens)
   446→          - User message: up to ~1000 chars (~770 tokens)
   447→          - Total: ~3940 chars (~3032 tokens) — fits in 3520 with margin!
   448→        """
   449→        if not self._local:
   450→            return AIResponse(
   451→                text="",
   452→                model="local-qwen3-4b",
   453→                provider="local",
   454→                error=True,
   455→                error_message="Local provider not initialized",
   456→            )
   457→
   458→        if not config.ENABLE_LOCAL_MODEL:
   459→            return AIResponse(
   460→                text="",
   461→                model="local-qwen3-4b",
   462→                provider="local",
   463→                error=True,
   464→                error_message="Local model disabled (ENABLE_LOCAL_MODEL=false)",
   465→            )
   466→
   467→        # Use EXPANDED system prompt for local model (v2.0 — ~1800 tokens).
   468→        # The full 12K-char cloud prompt (~9000 tokens) still does NOT fit.
   469→        # But the expanded v2 prompt gives much better quality than the old compact one.
   470→        compact_prompt = persona.local_system_prompt
   471→
   472→        # Build messages for local model using its own ChatML format
   473→        messages = [{"role": "system", "content": compact_prompt}]
   474→
   475→        # Add limited history for local model (saves context window)
   476→        # With expanded prompt (~1800 tokens), we have ~1720 tokens left for history+message.
   477→        # 3 history turns × ~200 chars × 1.3 chars/token = ~462 tokens
   478→        # User message up to ~1000 chars = ~770 tokens
   479→        # Total: ~1232 tokens — well within budget.
   480→        history_limit = min(config.MODEL_HISTORY_LIMIT, 3)
   481→        limited_history = history[-history_limit:] if history else []
   482→        for msg in limited_history:
   483→            role = msg.get("role", "user")
   484→            content = msg.get("content", "")
   485→            # Truncate long history messages to save context tokens
   486→            if role in ("user", "assistant") and content:
   487→                messages.append({"role": role, "content": content[:200]})
   488→
   489→        # Truncate user message to prevent context overflow.
   490→        # With expanded system prompt, allow up to ~1000 chars for user message.
   491→        # The local_provider.chat() has its own safety truncation as well.
   492→        truncated_message = message[:1000] if len(message) > 1000 else message
   493→        if len(message) > 1000:
   494→            logger.debug(f"Truncated user message for local model: {len(message)} → 1000 chars")
   495→        messages.append({"role": "user", "content": truncated_message})
   496→
   497→        return await self._local.chat(
   498→            messages=messages,
   499→            temperature=temperature,
   500→            max_tokens=min(max_tokens, config.MODEL_MAX_TOKENS),
   501→        )

   503→    async def analyze_image(
   504→        self,
   505→        user_id: int,
   506→        image_url: str = "",
   507→        image_base64: str = "",
   508→        prompt: str = "",
   509→        extra_context: str = "",
   510→    ) -> AIResponse:
   511→        """
   512→        Analyze an image with 3-level vision failover.
   513→
   514→        Level 1: Pollinations vision (with key)
   515→        Level 2: Pollinations vision (free API)
   516→        Level 3: Cloudflare vision (Mistral Small 3.1 with image_url)
   517→        Last resort: Static fallback
   518→        """
   519→        # Build the vision prompt
   520→        if not prompt:
   521→            prompt = "Рассмотри это изображение внимательно и подробно опиши что на нём."
   522→
   523→        sys_prompt = self._build_system_prompt(

   829→    @staticmethod
   830→    def _router_smart_truncate(text: str, max_len: int) -> str:
   831→        """Smart truncation for AI router — cuts at sentence/paragraph boundary.
   832→        
   833→        Same logic as channel.py's _smart_truncate to avoid mid-word cuts.
   834→        """
   835→        if len(text) <= max_len:
   836→            return text
   837→        
   838→        target = max_len - 3
   839→        if target < 50:
   840→            return text[:target] + "..."
   841→        
   842→        search_zone = text[:target + 1]
   843→        
   844→        # 1. Paragraph break
   845→        last_para = search_zone.rfind("\n\n")
   846→        if last_para > target * 0.5:
   847→            return text[:last_para].rstrip() + "..."
   848→        
   849→        # 2. Sentence end
   850→        sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   851→        best_sentence_end = -1
   852→        for end_char in sentence_end_chars:
   853→            pos = search_zone.rfind(end_char)
   854→            if pos > best_sentence_end and pos > target * 0.5:
   855→                best_sentence_end = pos + len(end_char) - 1
   856→        
   857→        if best_sentence_end > target * 0.5:
   858→            return text[:best_sentence_end + 1].rstrip() + "..."
   859→        
   860→        # 3. Newline
   861→        last_newline = search_zone.rfind("\n")
   862→        if last_newline > target * 0.5:
   863→            return text[:last_newline].rstrip() + "..."
   864→        
   865→        # 4. Space
   866→        last_space = search_zone.rfind(" ")
   867→        if last_space > target * 0.5:
   868→            return text[:last_space].rstrip() + "..."
   869→        
   870→        # 5. Hard cut
   871→        return text[:target].rstrip() + "..."

   875→    def _finalize_channel_post(self, response: AIResponse, has_media: bool) -> AIResponse:
   876→        """Finalize channel post: add footer, enforce limits."""
   877→        if response.text and not response.error:
   878→            text = response.text
   879→
   880→            # Clean markdown-style links
   881→            import re
   882→            text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1', text)
   883→            text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
   884→            text = re.sub(r'\*(.+?)\*', r'\1', text)
   885→
   886→            if "#sochiautoparts" not in text:
   887→                text = text.rstrip() + "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   888→            elif "@asiaexp_bot" not in text:
   889→                text = text.replace("@sochiautoparts", "Автор @asiaexp_bot\n@sochiautoparts")
   890→
   891→            # Enforce character limit (use smart truncation from channel.py)
   892→            footer = "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   893→            if has_media and len(text) > config.TELEGRAM_CAPTION_LIMIT:
   894→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   895→                    text = text.replace(foot_part, "")
   896→                text = text.rstrip()
   897→                max_content = config.TELEGRAM_CAPTION_LIMIT - len(footer)
   898→                if len(text) > max_content:
   899→                    # Smart truncation — find last sentence/paragraph boundary
   900→                    text = self._router_smart_truncate(text, max_content)
   901→                text += footer
   902→            elif not has_media and len(text) > config.TELEGRAM_TEXT_LIMIT:
   903→                for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   904→                    text = text.replace(foot_part, "")
   905→                text = text.rstrip()
   906→                max_content = config.TELEGRAM_TEXT_LIMIT - len(footer)
   907→                if len(text) > max_content:
   908→                    text = self._router_smart_truncate(text, max_content)
   909→                text += footer
   910→
   911→            response.text = text
   912→
   913→        return response

  1044→    @staticmethod
  1045→    def _clean_ai_response(text: str) -> str:
  1046→        """Clean AI response artifacts (think tags, markdown, structured output, etc.)."""
  1047→        if not text:
  1048→            return ""
  1049→
  1050→        # Block structured YAML/MIDI/JSON output from text-to-music models
  1051→        import re
  1052→        structured_patterns = [
  1053→            r'^title:\s*.+?\n(duration|key|notation|pitch|velocity|tempo|bpm):',
  1054→            r'^---\s*\n.*?(title|duration|notation|pitch|velocity):',
  1055→            r'pitch,\s*time,\s*duration,\s*velocity',
  1056→        ]
  1057→        for pattern in structured_patterns:
  1058→            if re.search(pattern, text, re.DOTALL | re.IGNORECASE):
  1059→                return ""
  1060→
  1061→        # Strip think tags (Qwen3, reasoning models)
  1062→        text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1063→        text = re.sub(r'<thinking\b[^>]*>.*?</thinking\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
  1064→        text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
  1065→        text = re.sub(r'</?thinking[^>]*>', '', text, flags=re.IGNORECASE)
  1066→        text = re.sub(r'<think\b[^>]*$', '', text, flags=re.IGNORECASE)
  1067→
  1068→        # Strip /no_think prefix
  1069→        text = re.sub(r'^/no_think\s*', '', text)
  1070→
  1071→        # Strip prefixes
  1072→        for prefix in ["Ася:", "Asya:", "АСЯ:", "Assistant:", "Ответ Аси:"]:
  1073→            if text.startswith(prefix):
  1074→                text = text[len(prefix):].strip()
  1075→
  1076→        # Strip quotes
  1077→        if text.startswith('"') and text.endswith('"'):
  1078→            text = text[1:-1]
  1079→        if text.startswith("'") and text.endswith("'"):
  1080→            text = text[1:-1]
  1081→
  1082→        text = text.strip("*").strip()
  1083→
  1084→        # Strip markdown
  1085→        text = re.sub(r'\*\*([^*]+)\*\*', r'\1', text)
  1086→        text = re.sub(r'\*([^*]+)\*', r'\1', text)
  1087→        text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1088→
  1089→        # Clean up whitespace
  1090→        text = re.sub(r'\n{3,}', '\n\n', text)
  1091→        text = text.strip()
  1092→
  1093→        return text
