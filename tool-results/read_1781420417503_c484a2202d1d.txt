    64→    64→def _detect_vin(text: str) -> Optional[str]:
    65→    65→    """Detect a VIN code (17 chars) in text."""
    66→    66→    match = _VIN_PATTERN.search(text.upper())
    67→    67→    if match:
    68→    68→        vin = match.group(0)
    69→    69→        # Validate check digit position (9th char)
    70→    70→        if len(vin) == 17 and vin[8] in '0123456789X':
    71→    71→            return vin
    72→    72→    return None

   130→   130→    if name_lower.endswith(FEMALE_NAME_ENDINGS):
   131→   131→        if name_lower.endswith("ь"):
   132→   132→            pass
   133→   133→        else:

   135→   135→
   136→   136→    if name_lower.endswith("й") or name_lower.endswith("ь"):
   137→   137→        return "male"

  1091→  1091→def _smart_truncate(text: str, max_chars: int) -> str:
  1092→  1092→    """Truncate text to max_chars, trying not to cut in the middle of a sentence.
  1093→  1093→    
  1094→  1094→    Looks for the last sentence boundary (. ! ? …) within the limit.
  1095→  1095→    Falls back to last newline, then last space, then hard cut.
  1096→  1096→    Always appends "..." if truncated.
  1097→  1097→    """
  1098→  1098→    if len(text) <= max_chars:
  1099→  1099→        return text
  1100→  1100→
  1101→  1101→    # Reserve 3 chars for "..."
  1102→  1102→    limit = max_chars - 3
  1103→  1103→    if limit <= 0:
  1104→  1104→        return "..."
  1105→  1105→
  1106→  1106→    # Try to find last sentence boundary within limit
  1107→  1107→    sentence_endings = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
  1108→  1108→    best_pos = -1
  1109→  1109→    best_ending_len = 0
  1110→  1110→    for ending in sentence_endings:
  1111→  1111→        pos = text.rfind(ending, 0, limit + 1)
  1112→  1112→        if pos > best_pos:
  1113→  1113→            best_pos = pos
  1114→  1114→            best_ending_len = len(ending)
  1115→  1115→
  1116→  1116→    if best_pos > limit // 2:
  1117→  1117→        # Found a good sentence boundary — cut after the punctuation mark
  1118→  1118→        return text[:best_pos + best_ending_len].rstrip() + "..."
  1119→  1119→
  1120→  1120→    # Try last newline
  1121→  1121→    nl_pos = text.rfind('\n', 0, limit + 1)
  1122→  1122→    if nl_pos > limit // 2:
  1123→  1123→        return text[:nl_pos].rstrip() + "..."
  1124→  1124→
  1125→  1125→    # Try last space
  1126→  1126→    sp_pos = text.rfind(' ', 0, limit + 1)
  1127→  1127→    if sp_pos > limit // 2:
  1128→  1128→        return text[:sp_pos].rstrip() + "..."
  1129→  1129→
  1130→  1130→    # Hard cut
  1131→  1131→    return text[:limit].rstrip() + "..."

  1246→  1246→def _clean_raw_partner_urls(text: str, partner_links: list) -> str:
  1247→  1247→    """Remove raw affiliate URLs that AI may have dumped into the response.
  1248→  1248→    
  1249→  1249→    If the AI already included the full affiliate URLs in its response text,
  1250→  1250→    this function removes them to avoid duplication with the clean formatted
  1251→  1251→    section that will be appended by _format_partner_links_section.
  1252→  1252→    Also removes shop name + URL patterns like "Росско: https://...".
  1253→  1253→    """
  1254→  1254→    if not partner_links:
  1255→  1255→        return text
  1256→  1256→    
  1257→  1257→    for name, url in partner_links:
  1258→  1258→        url_escaped = re.escape(url)
  1259→  1259→        # Remove standalone URL lines
  1260→  1260→        text = re.sub(
  1261→  1261→            rf'^[\s\-—]*{url_escaped}[\s]*$',
  1262→  1262→            '',
  1263→  1263→            text,
  1264→  1264→            flags=re.MULTILINE
  1265→  1265→        )
  1266→  1266→        # Remove "name: URL" or "name — URL" patterns on their own lines
  1267→  1267→        text = re.sub(
  1268→  1268→            rf'^[\s\-—]*{re.escape(name)}\s*[:\-—]\s*{url_escaped}[\s]*$',
  1269→  1269→            '',
  1270→  1270→            text,
  1271→  1271→            flags=re.MULTILINE
  1272→  1272→        )
  1273→  1273→        # Remove any line containing the full raw affiliate URL (even mid-sentence)
  1274→  1274→        # This catches cases where AI writes: "Посмотри на Росско: https://ujhjj.com/g/..."
  1275→  1275→        text = re.sub(
  1276→  1276→            rf'[^\n]*{url_escaped}[^\n]*',
  1277→  1277→            '',
  1278→  1278→            text,
  1279→  1279→            flags=re.MULTILINE
  1280→  1280→        )
  1281→  1281→        # Remove bare URL anywhere in text (affiliate URLs are very long, easy to detect)
  1282→  1282→        if len(url) > 50:  # Only long tracking URLs
  1283→  1283→            # Match the base tracking domain pattern
  1284→  1284→            url_base = re.escape(url.split('?')[0])  # Before query params

  1287→  1287→                '',
  1288→  1288→                text
  1289→  1289→            )
  1290→  1290→    
  1291→  1291→    text = re.sub(r'\n{3,}', '\n\n', text)

  1318→  1318→def _replace_plain_urls_with_affiliate(text: str) -> str:
  1319→  1319→    """Replace any plain partner domain URLs with affiliate goto_link equivalents.
  1320→  1320→    
  1321→  1321→    When the AI generates responses containing plain URLs like rossko.ru or 
  1322→  1322→    autopiter.ru instead of the affiliate tracking links from admitad_ads.json,
  1323→  1323→    this function detects them and replaces with the proper goto_link.
  1324→  1324→    
  1325→  1325→    This handles cases where:
  1326→  1326→    - AI ignores the system prompt and invents plain URLs
  1327→  1327→    - AI uses domain names without the affiliate wrapper
  1328→  1328→    - Photo handler bypasses the normal link injection pipeline
  1329→  1329→    """
  1330→  1330→    try:
  1331→  1331→        partner_manager.ensure_loaded()
  1332→  1332→    except Exception:
  1333→  1333→        return text
  1334→  1334→    
  1335→  1335→    for domain, display_name in _PARTNER_DOMAINS_MAP.items():
  1336→  1336→        prog = partner_manager.get_by_site(domain)
  1337→  1337→        if not prog or not prog.goto_link:
  1338→  1338→            continue
  1339→  1339→        
  1340→  1340→        affiliate_url = prog.goto_link
  1341→  1341→        
  1342→  1342→        # Pattern 1: Full URLs with paths — https://rossko.ru/search?text=abc
  1343→  1343→        # Replace the entire URL with the affiliate link
  1344→  1344→        pattern = rf'https?://{re.escape(domain)}[^\s<>)\]"\']*'
  1345→  1345→        matches = re.findall(pattern, text)
  1346→  1346→        for plain_url in matches:
  1347→  1347→            # Try to extract search query and build affiliate link with search
  1348→  1348→            search_query = ""
  1349→  1349→            if "search" in plain_url or "querystr" in plain_url or "q=" in plain_url:
  1350→  1350→                try:
  1351→  1351→                    from urllib.parse import urlparse, parse_qs
  1352→  1352→                    parsed = urlparse(plain_url)
  1353→  1353→                    params = parse_qs(parsed.query)
  1354→  1354→                    for key in ("text", "querystr", "q", "query", "SearchText", "keyword", "p"):
  1355→  1355→                        if key in params:
  1356→  1356→                            search_query = params[key][0]
  1357→  1357→                            break
  1358→  1358→                except Exception:
  1359→  1359→                    pass
  1360→  1360→            
  1361→  1361→            if search_query:
  1362→  1362→                replacement = prog.get_search_url(search_query)
  1363→  1363→            else:
  1364→  1364→                replacement = affiliate_url
  1365→  1365→            
  1366→  1366→            text = text.replace(plain_url, replacement)
  1367→  1367→        
  1368→  1368→        # Pattern 2: Bare domain mentions — rossko.ru or www.rossko.ru (not already part of a longer URL)
  1369→  1369→        # Only replace if it's NOT already part of an affiliate URL (which would be much longer)
  1370→  1370→        bare_pattern = rf'(?<![/\w.-])(?:www\.)?{re.escape(domain)}(?![/\w.-])'
  1371→  1371→        bare_matches = re.findall(bare_pattern, text)
  1372→  1372→        for bare_domain in bare_matches:
  1373→  1373→            # Check this isn't already inside an affiliate URL
  1374→  1374→            idx = text.find(bare_domain)
  1375→  1375→            if idx > 0:
  1376→  1376→                # Look back — if there's a tracking domain prefix, skip
  1377→  1377→                before = text[max(0, idx-50):idx]
  1378→  1378→                if any(tracking_domain in before for tracking_domain in ["ad.admitad.com", ".com/g/", "xmknb.com", "ujhjj.com", "rcpsj.com", "sgkaa.com"]):
  1379→  1379→                    continue
  1380→  1380→            text = text.replace(bare_domain, affiliate_url, 1)
  1381→  1381→    
  1382→  1382→    return text

  1385→  1385→def _clean_markdown(text: str) -> str:
  1386→  1386→    """Remove markdown formatting that Asya shouldn't use.
  1387→  1387→    
  1388→  1388→    Markdown links [text](url) are converted to plain text: text — url.
  1389→  1389→    Other markdown is stripped. No HTML — plain text for chat responses.
  1390→  1390→    """
  1391→  1391→    # Convert markdown links [text](url) → text — url (plain text format)
  1392→  1392→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1 — \2', text)
  1393→  1393→    
  1394→  1394→    # Remove bold (after escaping, ** won't be in &lt; form)
  1395→  1395→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
  1396→  1396→    # Remove italic
  1397→  1397→    text = re.sub(r'\*(.+?)\*', r'\1', text)
  1398→  1398→    # Remove code blocks
  1399→  1399→    text = re.sub(r'```[\s\S]*?```', lambda m: m.group(0).strip('`').strip(), text)
  1400→  1400→    # Remove inline code
  1401→  1401→    text = re.sub(r'`(.+?)`', r'\1', text)
  1402→  1402→    # Remove headers
  1403→  1403→    text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1404→  1404→    # Remove bullet points (convert to simple text)
  1405→  1405→    text = re.sub(r'^[-*]\s+', '— ', text, flags=re.MULTILINE)
  1406→  1406→    return text

  1409→  1409→def _split_message(text: str, max_length: int = 4096) -> list:
  1410→  1410→    """Split a long message into chunks at paragraph boundaries."""
  1411→  1411→    if len(text) <= max_length:
  1412→  1412→        return [text]
  1413→  1413→
  1414→  1414→    chunks = []
  1415→  1415→    while text:
  1416→  1416→        if len(text) <= max_length:
  1417→  1417→            chunks.append(text)
  1418→  1418→            break
  1419→  1419→
  1420→  1420→        # Try to split at newline near the limit
  1421→  1421→        split_pos = text.rfind('\n', 0, max_length)
  1422→  1422→        if split_pos < max_length // 2:
  1423→  1423→            # Try splitting at space
  1424→  1424→            split_pos = text.rfind(' ', 0, max_length)
  1425→  1425→        if split_pos < max_length // 2:
  1426→  1426→            # Hard split
  1427→  1427→            split_pos = max_length
  1428→  1428→
  1429→  1429→        chunks.append(text[:split_pos].strip())
  1430→  1430→        text = text[split_pos:].strip()
  1431→  1431→
  1432→  1432→    return chunks
