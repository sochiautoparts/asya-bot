   170→   170→# ── Image Cache ───────────────────────────────────────────────────────────────
   171→   171→
   172→   172→class ImageCache:
   173→   173→    """File-based image cache with TTL. Key = entity/topic, Value = image data."""
   174→   174→
   175→   175→    def __init__(self, cache_dir: Path = IMAGE_CACHE_DIR, ttl_days: int = IMAGE_CACHE_TTL_DAYS):
   176→   176→        self.cache_dir = cache_dir
   177→   177→        self.ttl_days = ttl_days
   178→   178→        self._index_path = cache_dir / "index.json"
   179→   179→        self._index: Dict[str, Dict[str, Any]] = {}
   180→   180→        self._load_index()
   181→   181→
   182→   182→    def _load_index(self) -> None:
   183→   183→        try:
   184→   184→            if self._index_path.exists():
   185→   185→                with open(self._index_path, "r", encoding="utf-8") as f:
   186→   186→                    self._index = json.load(f)
   187→   187→        except Exception as e:
   188→   188→            logger.debug(f"Failed to load image cache index: {e}")
   189→   189→            self._index = {}
   190→   190→
   191→   191→    def _save_index(self) -> None:
   192→   192→        try:
   193→   193→            self.cache_dir.mkdir(parents=True, exist_ok=True)
   194→   194→            with open(self._index_path, "w", encoding="utf-8") as f:
   195→   195→                json.dump(self._index, f, ensure_ascii=False, indent=2)
   196→   196→        except Exception as e:
   197→   197→            logger.debug(f"Failed to save image cache index: {e}")
   198→   198→
   199→   199→    def _cache_key(self, topic: str) -> str:
   200→   200→        return hashlib.md5(topic.lower().strip().encode()).hexdigest()
   201→   201→
   202→   202→    def get(self, topic: str) -> Optional[List[bytes]]:
   203→   203→        """Get cached image bytes for a topic."""
   204→   204→        key = self._cache_key(topic)
   205→   205→        entry = self._index.get(key)
   206→   206→        if not entry:
   207→   207→            return None
   208→   208→
   209→   209→        cached_at = entry.get("cached_at", 0)
   210→   210→        age_days = (time.time() - cached_at) / 86400
   211→   211→        if age_days > self.ttl_days:
   212→   212→            self.delete(topic)
   213→   213→            return None
   214→   214→
   215→   215→        # Load all cached files for this topic
   216→   216→        results = []
   217→   217→        for filename in entry.get("filenames", []):
   218→   218→            file_path = self.cache_dir / filename
   219→   219→            if file_path.exists():
   220→   220→                try:
   221→   221→                    with open(file_path, "rb") as f:
   222→   222→                        results.append(f.read())
   223→   223→                except Exception:
   224→   224→                    continue
   225→   225→
   226→   226→        return results if results else None
   227→   227→
   228→   228→    def put(self, topic: str, images: List[bytes], source: str) -> None:
   229→   229→        """Store images in cache."""
   230→   230→        try:
   231→   231→            self.cache_dir.mkdir(parents=True, exist_ok=True)
   232→   232→            key = self._cache_key(topic)
   233→   233→            filenames = []
   234→   234→
   235→   235→            for i, img_bytes in enumerate(images):
   236→   236→                filename = f"{key}_{i}.jpg"
   237→   237→                file_path = self.cache_dir / filename
   238→   238→                with open(file_path, "wb") as f:
   239→   239→                    f.write(img_bytes)
   240→   240→                filenames.append(filename)
   241→   241→
   242→   242→            self._index[key] = {
   243→   243→                "topic": topic[:100],
   244→   244→                "filenames": filenames,
   245→   245→                "source": source,
   246→   246→                "cached_at": time.time(),
   247→   247→            }
   248→   248→            self._save_index()
   249→   249→        except Exception as e:
   250→   250→            logger.debug(f"Failed to cache images: {e}")
   251→   251→
   252→   252→    def delete(self, topic: str) -> None:
   253→   253→        key = self._cache_key(topic)
   254→   254→        entry = self._index.pop(key, None)
   255→   255→        if entry:
   256→   256→            for filename in entry.get("filenames", []):
   257→   257→                try:
   258→   258→                    file_path = self.cache_dir / filename
   259→   259→                    if file_path.exists():
   260→   260→                        file_path.unlink()
   261→   261→                except Exception:
   262→   262→                    pass
   263→   263→            self._save_index()

   268→   268→def _is_junk_url(url: str) -> bool:
   269→   269→    """Check if a URL is a junk/tracking/icon image.
   270→   270→    
   271→   271→    Consistent with channel.py _is_junk_image_url but enhanced.
   272→   272→    """
   273→   273→    url_lower = url.lower()
   274→   274→    parsed = urlparse(url_lower)
   275→   275→    hostname = parsed.hostname or ""
   276→   276→
   277→   277→    # Check junk domains
   278→   278→    for junk in JUNK_DOMAINS:
   279→   279→        if junk in hostname:
   280→   280→            return True
   281→   281→
   282→   282→    # Check junk keywords (from channel.py)
   283→   283→    for kw in JUNK_KEYWORDS:
   284→   284→        if kw in url_lower:
   285→   285→            return True
   286→   286→
   287→   287→    # Check junk patterns
   288→   288→    for pattern in JUNK_PATTERNS:
   289→   289→        if re.search(pattern, url_lower):
   290→   290→            return True
   291→   291→
   292→   292→    # Check junk extensions
   293→   293→    path = parsed.path.lower()
   294→   294→    for ext in JUNK_EXTENSIONS:
   295→   295→        if path.endswith(ext):
   296→   296→            return True
   297→   297→
   298→   298→    # Skip URLs with very small size indicators
   299→   299→    # NOTE: h=0 means "auto height" (proportional) — common in image CDNs
   300→   300→    # like TopSpeed (_800x0.webp = width 800, height auto). Do NOT block these.
   301→   301→    size_pattern = re.compile(r'[/=_x](\d{1,4})x(\d{1,4})[/._]')
   302→   302→    size_match = size_pattern.search(url_lower)
   303→   303→    if size_match:
   304→   304→        w, h = int(size_match.group(1)), int(size_match.group(2))
   305→   305→        # 0 = auto/proportional height — valid image size indicator
   306→   306→        if w < 100 or (0 < h < 100):
   307→   307→            return True
   308→   308→
   309→   309→    return False

   312→   312→def _is_content_image(image_data: bytes) -> bool:
   313→   313→    """Validate that image data represents a proper content photo.
   314→   314→    
   315→   315→    Uses PIL when available (checks dimensions, aspect ratio).
   316→   316→    Falls back to soft check when PIL is not available.
   317→   317→    """
   318→   318→    if len(image_data) < IMAGE_MIN_SIZE_BYTES:
   319→   319→        return False
   320→   320→
   321→   321→    try:
   322→   322→        from PIL import Image
   323→   323→        import io
   324→   324→
   325→   325→        img = Image.open(io.BytesIO(image_data))
   326→   326→        width, height = img.size
   327→   327→
   328→   328→        if width < IMAGE_MIN_WIDTH or height < IMAGE_MIN_HEIGHT:
   329→   329→            return False
   330→   330→        if width / max(height, 1) > 3.0:
   331→   331→            return False
   332→   332→        if height / max(width, 1) > 3.0:
   333→   333→            return False
   334→   334→        if width * height < 76800:  # 320*240 = 76800
   335→   335→            return False
   336→   336→
   337→   337→        return True
   338→   338→    except ImportError:
   339→   339→        logger.warning("PIL not available, accepting image without dimension check")
   340→   340→        return True
   341→   341→    except Exception:
   342→   342→        return True  # Soft check — accept if can't read

   345→   345→async def _validate_and_download(
   346→   346→    client: httpx.AsyncClient,
   347→   347→    url: str,
   348→   348→) -> Optional[bytes]:
   349→   349→    """Download and validate an image URL. Returns image bytes or None."""
   350→   350→    if _is_junk_url(url):
   351→   351→        return None
   352→   352→    
   353→   353→    # BLOCK: NSFW domain check — never download from porn/adult sites
   354→   354→    if _is_nsfw_image_url(url):
   355→   355→        return None
   356→   356→
   357→   357→    try:
   358→   358→        # Quick HEAD check
   359→   359→        try:
   360→   360→            head_resp = await client.head(url, timeout=6.0, follow_redirects=True)
   361→   361→            content_type = head_resp.headers.get("content-type", "").lower()
   362→   362→            content_length = int(head_resp.headers.get("content-length", "0"))
   363→   363→
   364→   364→            if content_type and not any(ct in content_type for ct in [
   365→   365→                "image/jpeg", "image/png", "image/webp", "image/jpg",
   366→   366→            ]):
   367→   367→                # Some servers don't return proper content-type on HEAD
   368→   368→                if not content_type.startswith("image/"):
   369→   369→                    return None
   370→   370→
   371→   371→            if 0 < content_length < IMAGE_MIN_SIZE_BYTES:
   372→   372→                return None
   373→   373→            if content_length > IMAGE_MAX_SIZE_BYTES:
   374→   374→                return None
   375→   375→        except Exception:
   376→   376→            pass
   377→   377→
   378→   378→        # Full GET
   379→   379→        resp = await client.get(url, timeout=IMAGE_FETCH_TIMEOUT, follow_redirects=True)
   380→   380→        if resp.status_code != 200:
   381→   381→            return None
   382→   382→
   383→   383→        img_bytes = resp.content
   384→   384→
   385→   385→        if len(img_bytes) < IMAGE_MIN_SIZE_BYTES:
   386→   386→            return None
   387→   387→        if len(img_bytes) > IMAGE_MAX_SIZE_BYTES:
   388→   388→            return None
   389→   389→
   390→   390→        # Content-type check
   391→   391→        content_type = resp.headers.get("content-type", "").lower()
   392→   392→        if content_type and not any(ct in content_type for ct in [
   393→   393→            "image/jpeg", "image/png", "image/webp", "image/jpg",
   394→   394→            "application/octet-stream",
   395→   395→        ]):
   396→   396→            return None
   397→   397→
   398→   398→        # Skip SVG
   399→   399→        if b'<svg' in img_bytes[:500] or 'svg' in content_type:
   400→   400→            return None
   401→   401→
   402→   402→        # Magic bytes check
   403→   403→        is_valid_format = (
   404→   404→            img_bytes[:3] == b'\xff\xd8\xff'  # JPEG
   405→   405→            or img_bytes[:4] == b'\x89PNG'     # PNG
   406→   406→            or img_bytes[:4] == b'RIFF'        # WebP
   407→   407→            or img_bytes[:6] in (b'GIF87a', b'GIF89a')  # GIF
   408→   408→        )
   409→   409→        if not is_valid_format:
   410→   410→            return None
   411→   411→
   412→   412→        # Dimension check (PIL when available)
   413→   413→        if not _is_content_image(img_bytes):
   414→   414→            return None
   415→   415→
   416→   416→        return img_bytes
   417→   417→
   418→   418→    except Exception as e:
   419→   419→        logger.debug(f"Image download failed for {url[:60]}: {e}")
   420→   420→        return None

   446→   446→
   447→   447→        async with httpx.AsyncClient(

   449→   449→            follow_redirects=True,
   450→   450→            max_redirects=5,
   451→   451→        ) as client:
   452→   452→            resp = await client.get(url, headers=_SCRAPE_HEADERS)
   453→   453→            if resp.status_code != 200:
   454→   454→                return images

   456→   456→            html = resp.text

   472→   472→                        seen.add(u)

   474→   474→

   476→   476→            for pattern in [
   477→   477→                r'<meta[^>]+name=["\x27]twitter:image["\x27][^>]+content=["\x27]([^"\x27]+)["\x27]',

   479→   479→            ]:

   487→   487→            jsonld_urls = _extract_jsonld_images(html)
   488→   488→            for u in jsonld_urls:

   494→   494→            picture_blocks = re.findall(r'<picture[^>]*>(.*?)</picture>', html, re.IGNORECASE | re.DOTALL)
   495→   495→            for block in picture_blocks:
   496→   496→                srcsets = re.findall(r'srcset=["\x27]([^"\x27]+)["\x27]', block, re.IGNORECASE)
   497→   497→                for srcset in srcsets:
   498→   498→                    for part in srcset.split(','):
   499→   499→                        u = part.strip().split()[0] if part.strip() else ''
   500→   500→                        if u and u not in seen and not _is_junk_url(u):
   501→   501→                            seen.add(u)

   524→   524→                    u = "https:" + u

   526→   526→                    seen.add(u)
   527→   527→                    candidate_urls.append(u)

   540→   540→    except Exception as e:
   541→   541→        logger.debug(f"Article image extraction failed for {url[:60]}: {e}")

   546→   546→def _extract_jsonld_images(html: str) -> List[str]:
   547→   547→    """Extract image URLs from JSON-LD structured data in HTML."""
   548→   548→    images = []
   549→   549→    try:
   550→   550→        jsonld_blocks = re.findall(
   551→   551→            r'<script[^>]+type=["\x27]application/ld\+json["\x27][^>]*>(.*?)</script>',
   552→   552→            html, re.IGNORECASE | re.DOTALL,
   553→   553→        )
   554→   554→        for block in jsonld_blocks:
   555→   555→            try:
   556→   556→                data = json.loads(block)
   557→   557→                items = data if isinstance(data, list) else [data]
   558→   558→                for item in items:
   559→   559→                    if not isinstance(item, dict):
   560→   560→                        continue
   561→   561→                    img_field = item.get("image") or item.get("images")
   562→   562→                    if not img_field:
   563→   563→                        continue
   564→   564→                    if isinstance(img_field, str):
   565→   565→                        images.append(img_field)
   566→   566→                    elif isinstance(img_field, dict):
   567→   567→                        url = img_field.get("url") or img_field.get("contentUrl") or img_field.get("@id", "")
   568→   568→                        if url:
   569→   569→                            images.append(url)
   570→   570→                    elif isinstance(img_field, list):
   571→   571→                        for img_item in img_field:
   572→   572→                            if isinstance(img_item, str):
   573→   573→                                images.append(img_item)
   574→   574→                            elif isinstance(img_item, dict):
   575→   575→                                url = img_item.get("url") or img_item.get("contentUrl") or img_item.get("@id", "")
   576→   576→                                if url:
   577→   577→                                    images.append(url)
   578→   578→            except Exception:
   579→   579→                continue
   580→   580→    except Exception:
   581→   581→        pass
   582→   582→    return images

   601→   601→    """Fallback RSS image extraction."""
   602→   602→    image_urls: List[str] = []
   603→   603→
   604→   604→    # enclosures

   614→   614→        url = mc.get("url", "")
   615→   615→        medium = mc.get("medium", "").lower()
   616→   616→        mc_type = mc.get("type", "").lower()
   617→   617→        if url and (medium == "image" or "image" in mc_type):
   618→   618→            if not _is_junk_url(url) and url not in image_urls:
   619→   619→                image_urls.append(url)
   620→   620→
   621→   621→    # media:thumbnail
   622→   622→    for mt in getattr(entry, "media_thumbnail", []) or []:
   623→   623→        url = mt.get("url", "")
   624→   624→        if url and not _is_junk_url(url) and url not in image_urls:
   625→   625→            image_urls.append(url)

   627→   627→    # <img> in content/summary
   628→   628→    for field_name in ("content", "summary", "description"):
   629→   629→        content_value = getattr(entry, field_name, None)
   630→   630→        if isinstance(content_value, list):
   631→   631→            content_value = content_value[0].get("value", "") if content_value else ""
   632→   632→        elif content_value is None:

   635→   635→            url = m.group(1).replace("&amp;", "&")
   636→   636→            if not _is_junk_url(url) and url not in image_urls:
   637→   637→                image_urls.append(url)
   638→   638→
   639→   639→    return image_urls[:10]

  1109→  1109→def _images_are_similar(img1: bytes, img2: bytes) -> bool:
  1110→  1110→    """Check if two images are identical or near-identical.
  1111→  1111→
  1112→  1112→    Uses SHA256 hash for exact match. For near-duplicates (resized/compressed
  1113→  1113→    versions of the same photo), uses a simple pixel sampling heuristic.
  1114→  1114→    Returns True if images are likely the same photo.
  1115→  1115→    """
  1116→  1116→    # Exact match — fastest check
  1117→  1117→    if img1 == img2:
  1118→  1118→        return True
  1119→  1119→
  1120→  1120→    h1, h2 = _image_hash(img1), _image_hash(img2)
  1121→  1121→    if h1 == h2:
  1122→  1122→        return True
  1123→  1123→
  1124→  1124→    # Size-based quick reject: if sizes differ by >3x, very unlikely same image
  1125→  1125→    ratio = len(img1) / max(len(img2), 1)
  1126→  1126→    if ratio < 0.3 or ratio > 3.3:
  1127→  1127→        return False
  1128→  1128→
  1129→  1129→    # Try PIL-based comparison for near-duplicates (resized/recompressed same photo)
  1130→  1130→    try:
  1131→  1131→        from PIL import Image
  1132→  1132→        import io
  1133→  1133→
  1134→  1134→        pil1 = Image.open(io.BytesIO(img1))
  1135→  1135→        pil2 = Image.open(io.BytesIO(img2))
  1136→  1136→
  1137→  1137→        # If dimensions are very different, probably different images
  1138→  1138→        w1, h1_dim = pil1.size
  1139→  1139→        w2, h2_dim = pil2.size
  1140→  1140→        if abs(w1 - w2) > max(w1, w2) * 0.3 or abs(h1_dim - h2_dim) > max(h1_dim, h2_dim) * 0.3:
  1141→  1141→            return False
  1142→  1142→
  1143→  1143→        # Resize both to tiny thumbnail and compare pixel values
  1144→  1144→        thumb_size = (16, 16)
  1145→  1145→        t1 = pil1.convert("L").resize(thumb_size)
  1146→  1146→        t2 = pil2.convert("L").resize(thumb_size)
  1147→  1147→
  1148→  1148→        pixels1 = list(t1.getdata())
  1149→  1149→        pixels2 = list(t2.getdata())
  1150→  1150→
  1151→  1151→        # Compare average pixel difference
  1152→  1152→        total_diff = sum(abs(p1 - p2) for p1, p2 in zip(pixels1, pixels2))
  1153→  1153→        avg_diff = total_diff / len(pixels1)
  1154→  1154→
  1155→  1155→        # Low average difference = same image with different compression/resize
  1156→  1156→        if avg_diff < 15:  # Threshold: very similar greyscale thumbnails
  1157→  1157→            return True
  1158→  1158→
  1159→  1159→    except Exception:
  1160→  1160→        pass  # If PIL comparison fails, only exact hash match counts
  1161→  1161→
  1162→  1162→    return False

  1165→  1165→def deduplicate_images(images: List[bytes]) -> List[bytes]:
  1166→  1166→    """Remove duplicate and near-duplicate images from a list.
  1167→  1167→
  1168→  1168→    Uses both exact SHA256 hash matching and PIL-based perceptual comparison
  1169→  1169→    to catch resized/compressed versions of the same photo.
  1170→  1170→    Returns deduplicated list preserving original order.
  1171→  1171→    """
  1172→  1172→    if not images:
  1173→  1173→        return images
  1174→  1174→
  1175→  1175→    unique: List[bytes] = []
  1176→  1176→    for img in images:
  1177→  1177→        is_dup = False
  1178→  1178→        for existing in unique:
  1179→  1179→            if _images_are_similar(img, existing):
  1180→  1180→                is_dup = True
  1181→  1181→                break
  1182→  1182→        if not is_dup:
  1183→  1183→            unique.append(img)
  1184→  1184→        else:
  1185→  1185→            logger.debug(f"Dedup: removed duplicate image ({len(img)} bytes)")
  1186→  1186→
  1187→  1187→    removed = len(images) - len(unique)
  1188→  1188→    if removed > 0:
  1189→  1189→        logger.info(f"Image dedup: {len(images)} → {len(unique)} (removed {removed} duplicates)")
  1190→  1190→
  1191→  1191→    return unique
