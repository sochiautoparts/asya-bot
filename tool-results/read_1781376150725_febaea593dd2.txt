    92→def _is_semantically_duplicate(title: str) -> bool:
    93→    """Check if 3+ significant words from title match a recently posted title.
    94→    
    95→    Uses a TWO-LEVEL check:
    96→    - Level 1: 3+ significant words overlap -> DUPLICATE
    97→    - Level 2: 2+ CORE words (brand + model/event) overlap -> DUPLICATE
    98→    
    99→    This catches both obvious and subtle duplicates like:
   100→    - "BMW X5 получил новый двигатель" vs "Новый мотор для BMW X5"
   101→    - "Tesla отзывает 10000 машин" vs "Tesla начала отзывную кампанию"
   102→    """
   103→    global _recent_post_keywords
   104→
   105→    # Extract significant words from the new title
   106→    words = re.findall(r'[a-zа-яё]{3,}', title.lower())
   107→    significant = [w for w in words if w not in _SEMANTIC_STOP_WORDS]
   108→
   109→    if len(significant) < 2:
   110→        return False
   111→
   112→    # Extract core words (brands, models, events) for Level 2 check
   113→    core_words = _extract_core_words_from_title(title)
   114→
   115→    for recent_words in _recent_post_keywords:
   116→        # Level 1: 3+ significant words overlap
   117→        matches = sum(1 for w in significant if w in recent_words)
   118→        if matches >= 3:
   119→            return True
   120→        
   121→        # Level 2: 2+ core words overlap (brand + model/event)
   122→        if len(core_words) >= 2:
   123→            recent_core = [w for w in recent_words if w in _ALL_CORE_WORDS_SET]
   124→            core_matches = sum(1 for w in core_words if w in recent_core)
   125→            if core_matches >= 2:
   126→                return True
   127→
   128→    return False

   217→def _clean_post_text(text: str) -> str:
   218→    """Clean post text: remove markdown links, formatting artifacts, SSE garbage,
   219→    AI meta-comments about duplicates, and other content that should not appear in posts."""
   220→    if not text:
   221→        return text
   222→
   223→    # Remove markdown links [text](url) → text (Telegram doesn't support them properly)
   224→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1', text)
   225→
   226→    # Remove bold/italic
   227→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
   228→    text = re.sub(r'\*(.+?)\*', r'\1', text)
   229→
   230→    # Remove SSE/streaming artifacts
   231→    text = re.sub(r'data:\s*\{[^}]*\}', '', text)
   232→    text = re.sub(r'\[DONE\]', '', text)
   233→
   234→    # Remove AI disclaimers
   235→    for phrase in ["As an AI", "Как AI", "Как искусственный интеллект",
   236→                   "powered by pollinations", "pollinations.ai"]:
   237→        text = re.sub(rf'.*{re.escape(phrase)}.*', '', text, flags=re.IGNORECASE)
   238→
   239→    # Remove think tags
   240→    text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
   241→    text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
   242→
   243→    # ── Remove AI meta-comments / editorial notes that leaked into post text ──
   244→    # Kept to ~15 essential patterns — removed common Russian words that match valid content
   245→    meta_comment_patterns = [
   246→        # Editorial leakage: "тему в канал не ставим", "не наш формат"
   247→        r'[^\n]*тему\s+в\s+канал\s+не\s+ставим[^\n]*',
   248→        r'[^\n]*в\s+канал\s+не\s+ставим[^\n]*',
   249→        r'[^\n]*не\s+наш\s+формат[^\n]*',
   250→        r'[^\n]*перепишу\s+тему[^\n]*',
   251→        r'[^\n]*напишу\s+готовый\s+пост[^\n]*',
   252→        r'[^\n]*я\s+сразу\s+предложу[^\n]*',
   253→        r'[^\n]*без\s+лишних\s+вопросов[^\n]*',
   254→        r'[^\n]*не\s+соответствует\s+(тематик|формат)[^\n]*',
   255→        r'[^\n]*по\s+вашим\s+(же\s+)?правилам[^\n]*',
   256→        # Duplicate/republish meta-comments
   257→        r'[^\n]*дубликат[^\n]*',
   258→        r'[^\n]*ЭТО\s+УЖЕ\s+ОПУБЛИКОВАНО[^\n]*',
   259→        # English variants
   260→        r'[^\n]*already\s+(posted|published|covered|wrote)[^\n]*',
   261→        r'[^\n]*do\s+not\s+(publish|post)[^\n]*',
   262→        # Generic catch-all for editorial notes in brackets
   263→        r'\[[^\]]*(?:не\s+публиков|не\s+став|не\s+наш|редакц|пропуск)[^\]]*\]',
   264→    ]
   265→    for pattern in meta_comment_patterns:
   266→        text = re.sub(pattern, '', text, flags=re.IGNORECASE)
   267→
   268→    # Remove AI/assistant name prefixes that leak into posts
   269→    for prefix in ["Ася:", "Asya:", "Assistant:"]:
   270→        if text.startswith(prefix):
   271→            text = text[len(prefix):].strip()
   272→
   273→    # Remove formal "Редакция сообщает" phrases — replace with nothing or informal version
   274→    formal_phrases = [
   275→        ("Редакция сообщает:", ""),
   276→        ("Редакция сообщает —", ""),
   277→        ("Редакция сообщает", ""),
   278→        ("Как стало известно редакции,", ""),
   279→        ("Как стало известно редакции", ""),
   280→        ("По данным нашей редакции,", ""),
   281→        ("По данным нашей редакции", ""),
   282→        ("Редакция @sochiautoparts сообщает:", ""),
   283→        ("Редакция @sochiautoparts сообщает", ""),
   284→    ]
   285→    for phrase, replacement in formal_phrases:
   286→        if phrase in text:
   287→            text = text.replace(phrase, replacement)
   288→
   289→    # ── Meta-line removal: remove lines that look like editorial notes ──
   290→    # Reduced to 10 core triggers that NEVER appear in valid automotive posts
   291→    _editorial_trigger_phrases = [
   292→        "не ставим", "не наш формат",
   293→        "перепишу тему", "напишу готовый",
   294→        "я сразу предложу", "без лишних вопросов",
   295→        "не для публикации", "внутренняя заметка",
   296→        "для редакции", "редакционная",
   297→    ]
   298→    lines = text.split('\n')
   299→    cleaned_lines = []
   300→    for line in lines:
   301→        line_lower = line.lower().strip()
   302→        is_editorial = False
   303→        for trigger in _editorial_trigger_phrases:
   304→            if trigger in line_lower:
   305→                is_editorial = True
   306→                break
   307→        # Also catch lines that are purely editorial in brackets [...]
   308→        if re.match(r'^\s*\[.*\]\s*$', line) and any(
   309→            kw in line.lower() for kw in ["не", "редакц", "пропуск", "не наш", "формат"]
   310→        ):
   311→            is_editorial = True
   312→        if not is_editorial:
   313→            cleaned_lines.append(line)
   314→    text = '\n'.join(cleaned_lines)
   315→
   316→    # Clean up whitespace
   317→    text = re.sub(r'\n{3,}', '\n\n', text)
   318→    text = text.strip()
   319→
   320→    return text

   323→def _validate_post_text(text: str) -> bool:
   324→    """Validate post text before sending to channel.
   325→    
   326→    Checks for:
   327→    - Empty text
   328→    - SSE/API artifacts
   329→    - AI meta-comments about duplicates ("already posted", "do not publish")
   330→    - Blocked topics (politics, war, Putin, etc.)
   331→    - Auto-relevance (must contain automotive keywords)
   332→    - Provider ad artifacts
   333→    - Raw JSON
   334→    - Duplicate indicators that leaked through cleaning
   335→    """
   336→    if not text or not text.strip():
   337→        return False
   338→
   339→    text_lower = text.lower()
   340→
   341→    # Block SSE artifacts
   342→    sse_patterns = [r'data:\s*\{', r'\[DONE\]', r'"type"\s*:\s*"start"', r'"type"\s*:\s*"error"']
   343→    for pattern in sse_patterns:
   344→        if re.search(pattern, text_lower):
   345→            return False
   346→
   347→    # Block API errors
   348→    error_patterns = ["authentication error", "no api key", "model not found",
   349→                      "rate limit", "internal server error", "bad request"]
   350→    for pattern in error_patterns:
   351→        if pattern in text_lower:
   352→            return False
   353→
   354→    # Block provider ad artifacts
   355→    ad_patterns = ["pollinations.ai", "powered by pollinations", "keep ai accessible"]
   356→    for pattern in ad_patterns:
   357→        if pattern in text_lower:
   358→            return False
   359→
   360→    # Block raw JSON
   361→    if text.strip().startswith(('{', '[', '```', 'data:')):
   362→        return False
   363→
   364→    # ── Block AI editorial notes that leaked through cleaning ──
   365→    # Reduced to ~20 truly editorial-specific phrases that NEVER appear in valid automotive posts.
   366→    # Removed common Russian words like "уже", "не стоит", "в канал", "отсеивать" etc.

   393→    #   - Removed "flood", "series", "championship", "election", "exam", "movie",
   394→    #     "drug", "theft", "school" — these are handled in news.py with compound phrases
   395→    blocked_keywords = [
   396→        "путин", "кремль", "госдума", "президент росс", "президент сша",
   397→        "сво ", "специальная военная", "мобилизац", "санкци",

   402→        # ── NSFW / Adult content — ABSOLUTE BLOCK ──
   403→        "порн", "секс", "эрот", "голая", "голые", "обнажён", "обнажен",
   404→        "интим", "проститут", "путан", "бордель",

   407→        "стриптиз", "камасутр", "ню фото",

   410→        "nude", "porn", "nsfw", "hentai", "milf",

   412→    # Block boring Russian auto brands — 50 years nothing interesting
   413→    blocked_auto_brands = [
   414→        "автоваз", "лада", "lada", "уаз", "uaz", "камаз", "kamaz",

   420→            return False

   530→def _enforce_char_limit(text: str, has_media: bool) -> str:
   531→    """Smart character limit enforcement — always preserves footer.
   532→    
   533→    Telegram limits:
   534→    - 1024 chars with media (caption)
   535→    - 4096 chars without media (text-only)
   536→    
   537→    This function:
   538→    1. Separates footer from content
   539→    2. Truncates content at sentence/paragraph boundary if needed
   540→    3. Re-attaches footer (always intact)
   541→    4. NEVER cuts mid-word or mid-sentence — always truncates at a natural break point
   542→    """
   543→    footer = "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   544→    char_limit = config.TELEGRAM_CAPTION_LIMIT if has_media else config.TELEGRAM_TEXT_LIMIT
   545→    
   546→    if len(text) <= char_limit:
   547→        return text
   548→    
   549→    # Strip existing footer parts to get pure content
   550→    content = text
   551→    for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   552→        content = content.replace(foot_part, "")
   553→    content = content.rstrip()
   554→    
   555→    # Calculate max content length (leave room for footer)
   556→    max_content = char_limit - len(footer)
   557→    if max_content < 100:
   558→        # Absolute minimum — just footer
   559→        return footer.lstrip('\n')
   560→    
   561→    if len(content) > max_content:
   562→        content = _smart_truncate(content, max_content)
   563→    
   564→    return content + footer
   565→

   567→def _smart_truncate(text: str, max_len: int) -> str:
   568→    """Truncate text at a natural sentence/paragraph boundary.
   569→    
   570→    Strategy (in priority order):
   571→    1. Find the last paragraph break (\n\n) before max_len
   572→    2. Find the last sentence end (. ! ? …) before max_len
   573→    3. Find the last newline (\n) before max_len
   574→    4. Find the last space before max_len (avoid mid-word cut)
   575→    5. Last resort: hard cut at max_len - 3 + "..."
   576→    
   577→    Always appends "..." to indicate truncation.
   578→    """
   579→    if len(text) <= max_len:
   580→        return text
   581→    
   582→    # We need room for "..." (3 chars)
   583→    target = max_len - 3
   584→    if target < 50:
   585→        return text[:target] + "..."
   586→    
   587→    # Look at the text up to target+50 chars — we want to find the BEST
   588→    # break point near the end, not just the very last one
   589→    search_zone = text[:target + 1]
   590→    
   591→    # 1. Try paragraph break (\n\n) — best break point
   592→    last_para = search_zone.rfind("\n\n")
   593→    if last_para > target * 0.5:  # Don't throw away more than half the text
   594→        return text[:last_para].rstrip() + "..."
   595→    
   596→    # 2. Try sentence end (. ! ? … followed by space or newline)
   597→    # Look for sentence endings in the last portion of the text
   598→    sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   599→    best_sentence_end = -1
   600→    for end_char in sentence_end_chars:
   601→        pos = search_zone.rfind(end_char)
   602→        if pos > best_sentence_end and pos > target * 0.5:
   603→            best_sentence_end = pos + len(end_char) - 1  # Include the punctuation
   604→    
   605→    if best_sentence_end > target * 0.5:
   606→        return text[:best_sentence_end + 1].rstrip() + "..."
   607→    
   608→    # 3. Try newline (\n)
   609→    last_newline = search_zone.rfind("\n")
   610→    if last_newline > target * 0.5:
   611→        return text[:last_newline].rstrip() + "..."
   612→    
   613→    # 4. Try space (avoid mid-word cut)
   614→    last_space = search_zone.rfind(" ")
   615→    if last_space > target * 0.5:
   616→        return text[:last_space].rstrip() + "..."
   617→    
   618→    # 5. Hard cut — very last resort
   619→    return text[:target].rstrip() + "..."

   677→    @staticmethod
   678→    def _ai_response_to_bytes(response) -> Optional[bytes]:
   679→        """Convert AIResponse from generate_image to raw bytes.
   680→

   690→        try:
   691→            from ai.providers.base import AIResponse
   692→            if isinstance(response, AIResponse):
   693→                if response.image_b64:
   694→                    import base64
   695→                    return base64.b64decode(response.image_b64)
   696→                if response.image_url:
   697→                    # Download the image from URL
   698→                    try:
   699→                        r = httpx.get(response.image_url, timeout=30.0, follow_redirects=True)
   700→                        if r.status_code == 200 and len(r.content) > 1000:
   701→                            return r.content
   702→                    except Exception:
   703→                        pass
   704→                return None
   705→        except ImportError:
   706→            pass
   707→        # Fallback: if it has image_b64 attribute
   708→        if hasattr(response, 'image_b64') and response.image_b64:
   709→            import base64
   710→            return base64.b64decode(response.image_b64)
   711→        return None

   729→
   730→        # Try image models — limited to 2 attempts to prevent OOM/timeout
   731→        _IMAGE_MODELS = ["flux", "flux-pro"]
   732→        attempts = 0

   750→
   751→                except asyncio.TimeoutError:

   754→                except Exception as e:

   760→                break

   809→    async def _download_partner_image(self, image_url: str) -> Optional[bytes]:
   810→        """Download a partner program image (logo/banner).
   811→        
   812→        Uses RELAXED validation for partner images — they're often smaller logos/banners
   813→        (5KB+), which is fine for partner posts. News image validation is stricter.
   814→        Handles SVG images by converting them to PNG using cairosvg.
   815→        """
   816→        if not image_url:
   817→            return None
   818→        try:
   819→            async with httpx.AsyncClient(timeout=20.0, follow_redirects=True) as client:
   820→                response = await client.get(image_url)
   821→                if response.status_code != 200:
   822→                    return None
   823→                content = response.content
   824→                content_type = response.headers.get("content-type", "")
   825→
   826→                # ── Handle SVG images — convert to PNG for Telegram ──
   827→                is_svg = (
   828→                    "svg" in content_type.lower() or
   829→                    b'<svg' in content[:1000] or
   830→                    image_url.lower().endswith('.svg')
   831→                )
   832→                if is_svg:
   833→                    try:
   834→                        import cairosvg
   835→                        import io
   836→                        # Convert SVG → PNG at 512px width (good for Telegram)
   837→                        png_data = cairosvg.svg2png(bytestring=content, output_width=512)
   838→                        if png_data and len(png_data) > 1000:
   839→                            logger.info(f"Converted partner SVG → PNG ({len(png_data)} bytes): {image_url[:60]}")
   840→                            content = png_data
   841→                            content_type = "image/png"
   842→                        else:
   843→                            logger.debug(f"SVG conversion produced tiny output, skipping")
   844→                            return None
   845→                    except ImportError:
   846→                        logger.debug("cairosvg not available, cannot convert SVG partner image")
   847→                        return None
   848→                    except Exception as e:
   849→                        logger.debug(f"SVG → PNG conversion failed: {e}")
   850→                        return None
   851→
   852→                # Relaxed size check for partner images — logos can be 5KB+
   853→                # (But after SVG→PNG conversion, image is larger)
   854→                if len(content) < 2000:
   855→                    logger.debug(f"Skipping tiny partner image ({len(content)} bytes)")
   856→                    return None
   857→
   858→                # Must be an image type
   859→                if not any(ft in content_type for ft in ["image/png", "image/jpeg", "image/gif", "image/webp"]):
   860→                    # Check magic bytes
   861→                    if not (content[:3] == b'\xff\xd8\xff' or content[:4] == b'\x89PNG' or
   862→                            (content[:4] == b'RIFF' and content[8:12] == b'WEBP') or
   863→                            content[:6] in (b'GIF87a', b'GIF89a')):
   864→                        logger.debug(f"Skipping partner image: not an image, content-type={content_type}")
   865→                        return None
   866→
   867→                # Relaxed dimension check for partner images — logos/banners are OK
   868→                try:
   869→                    from PIL import Image
   870→                    import io
   871→                    img = Image.open(io.BytesIO(content))
   872→                    width, height = img.size
   873→                    # Only skip extremely tiny images (under 50px)
   874→                    if width < 50 or height < 50:
   875→                        logger.debug(f"Partner image too small: {width}x{height}")
   876→                        return None
   877→                except ImportError:
   878→                    # No PIL — accept the image anyway for partner posts
   879→                    logger.debug("PIL not available, accepting partner image without dimension check")
   880→                except Exception:
   881→                    # Can't read — accept anyway for partner posts
   882→                    logger.debug("Can't read partner image dimensions, accepting as-is")
   883→
   884→                return content
   885→        except Exception as e:
   886→            logger.debug(f"Could not download partner image: {e}")
   887→        return None

  1779→    async def run_scheduled_post(self) -> bool:
  1780→        """
  1781→        Run a scheduled post — tries up to 3 different news items per cycle.
  1782→        
  1783→        SIMPLIFIED PIPELINE:
  1784→        1. Try partner content (20% chance if interval met)
  1785→        2. Try news — up to 3 attempts with DIFFERENT items
  1786→        3. Fallback: AI-generated "fun fact" if no web news works
  1787→        
  1788→        Each cycle is called 3 TIMES per 30-min interval (3 different posts).
  1789→        Dedup ensures each post is always different from the others.
  1790→        """
  1791→        logger.info("run_scheduled_post: called")
  1792→        now = time.time()
  1793→        partner_interval = config.PARTNER_POST_INTERVAL_HOURS * 3600
  1794→
  1795→        # Try partner content with 20% probability (if interval met)
  1796→        # Lower probability so news posts dominate — partner posts are supplementary
  1797→        if (now - self._last_partner_time >= partner_interval and
  1798→                partner_manager.should_post_partner()):
  1799→            if random.random() < 0.2:
  1800→                result = await self.post_partner_content()
  1801→                if result:
  1802→                    return True
  1803→                # Partner post failed/skipped — fall through to news
  1804→        
  1805→        # Primary: post NEWS — try up to 3 DIFFERENT items per cycle
  1806→        for attempt in range(3):
  1807→            result = await self.post_news()
  1808→            if result:
  1809→                return True
  1810→            logger.info(f"run_scheduled_post: attempt {attempt + 1}/3 failed, trying different item")
  1811→            # Brief pause between attempts to avoid hammering AI
  1812→            if attempt < 2:
  1813→                await asyncio.sleep(2)
  1814→
  1815→        # Fallback: generate a "fun fact" / "did you know" style post
  1816→        logger.info("All news attempts failed — trying fallback fun fact post")
  1817→        result = await self._post_fallback_fun_fact()
  1818→        if result:
  1819→            return True
  1820→
  1821→        # Final fallback: try partner content
  1822→        if partner_manager.should_post_partner():
  1823→            return await self.post_partner_content()
  1824→
  1825→        return False
  1826→
  1827→    async def _post_fallback_fun_fact(self) -> bool:
  1828→        """Generate a 'fun fact' / 'did you know' style post when no web news is available.
  1829→        
  1830→        This ensures the channel always has content even when web search fails
  1831→        or all results are dedup-filtered.
  1832→        """
  1833→        if not self._bot:
  1834→            return False
  1835→
  1836→        # Check daily/hourly limits
  1837→        today_count = await get_today_post_count()
  1838→        if today_count >= config.CHANNEL_MAX_POSTS_PER_DAY:
