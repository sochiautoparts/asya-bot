    97→    97→class AIRouter:
    98→    98→    """Routes AI requests through multiple providers with 4-level failover.
    99→    99→
   100→   100→    v8.0 LOCAL-FIRST strategy:
   101→   101→    - Level 0: Local Model (Qwen3-4B GGUF) — CHAT & COMMENT routes
   102→   102→    - Level 1: Pollinations with API key (best quality, 60+ models)
   103→   103→    - Level 2: Pollinations FREE API (no auth, rate-limited)
   104→   104→    - Level 3: Cloudflare Workers AI (Mistral Small 3.1, 20K req/day)
   105→   105→    - Last resort: Static fallback responses

   108→   108→    def __init__(self):

   184→   184→    async def chat(
   185→   185→        self,
   186→   186→        user_id: int,
   187→   187→        message: str,
   188→   188→        system_prompt: str = "",
   189→   189→        model: str = "",
   190→   190→        temperature: float = None,
   191→   191→        max_tokens: int = None,
   192→   192→        use_cache: bool = True,
   193→   193→        save_history: bool = True,
   194→   194→        extra_context: str = "",
   195→   195→        route_type: str = "chat",
   196→   196→    ) -> AIResponse:
   197→   197→        """
   198→   198→        Send a chat message through the AI router with CONCURRENT failover.
   199→   199→
   200→   200→        v9.0 CONCURRENT ROUTING (fast response, no sequential delays):
   201→   201→        - "chat" (default): Try Local first (if available), then CONCURRENT Pollinations+Cloudflare
   202→   202→        - "function": Try Pollinations key first, then CONCURRENT free+Cloudflare+Local
   203→   203→        - "comment": Try Local first, then CONCURRENT Pollinations+Cloudflare
   204→   204→
   205→   205→        Key improvement: Instead of waiting for each level to timeout sequentially
   206→   206→        (which could take 30+15+60 = 105 seconds worst case), we now launch
   207→   207→        multiple providers CONCURRENTLY and return the FIRST successful response.
   208→   208→        This means if Pollinations is slow but Cloudflare responds in 3s, the user
   209→   209→        gets a response in ~3s instead of ~30s.
   210→   210→        """
   211→   211→        import asyncio as _asyncio
   212→   212→        temperature = temperature or config.CHAT_TEMPERATURE
   213→   213→        max_tokens = max_tokens or config.CHAT_MAX_TOKENS
   214→   214→
   215→   215→        # Build system prompt with time context
   216→   216→        sys_prompt = self._build_system_prompt(system_prompt, extra_context)
   217→   217→
   218→   218→        # Check cache first
   219→   219→        if use_cache:
   220→   220→            cache_key = self._make_cache_key(sys_prompt, message)
   221→   221→            cached = await get_ai_cached(cache_key)
   222→   222→            if cached:
   223→   223→                return AIResponse(
   224→   224→                    text=cached,
   225→   225→                    model="cached",
   226→   226→                    provider="cache",
   227→   227→                    cached=True,
   228→   228→                )
   229→   229→
   230→   230→        # Get chat history for context
   231→   231→        history = await get_chat_history(user_id)
   232→   232→
   233→   233→        # ── Select model based on route_type ──
   234→   234→        if route_type == "comment":
   235→   235→            model = model or "mistral"
   236→   236→        elif route_type == "function":
   237→   237→            model = model or "openai-large"
   238→   238→        else:
   239→   239→            model = model or ""
   240→   240→
   241→   241→        # ── LEVEL 0: Local Model (Qwen3-4B) — CHAT & COMMENT routes ──
   242→   242→        # Local model is primary for simple chat and comments (saves cloud balance)
   243→   243→        # Skip for function routes (need cloud quality) and when explicitly disabled
   244→   244→        use_local_first = route_type in ("chat", "comment") and config.ENABLE_LOCAL_MODEL
   245→   245→
   246→   246→        if use_local_first:
   247→   247→            # Quick check if local model is actually available (no timeout waste)
   248→   248→            local_available = (
   249→   249→                self._local is not None
   250→   250→                and self._local._model_loaded
   251→   251→                and self._local._llm is not None
   252→   252→                and self._local._consecutive_errors < 5
   253→   253→            )
   254→   254→            if local_available:
   255→   255→                response = await self._try_local(
   256→   256→                    user_id, message, history, sys_prompt, temperature, max_tokens
   257→   257→                )
   258→   258→
   259→   259→                if not response.error:
   260→   260→                    self._level0_count += 1
   261→   261→                    self._total_requests += 1
   262→   262→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   263→   263→
   264→   264→                logger.debug(f"Level 0 (local) failed for route={route_type}: {response.error_message}")
   265→   265→
   266→   266→        # ── CONCURRENT FAILOVER: Launch multiple providers at once ──
   267→   267→        # This is the key fix for the "Asya takes too long to respond" issue.
   268→   268→        # Instead of trying providers one-by-one (each with 15-60s timeout),
   269→   269→        # we launch them concurrently and return the FIRST successful response.
   270→   270→
   271→   271→        async def _safe_try_pollinations():
   272→   272→            """Try Pollinations with key — return response or error."""
   273→   273→            try:
   274→   274→                return await self._try_pollinations(
   275→   275→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   276→   276→                )
   277→   277→            except Exception as e:
   278→   278→                logger.error(f"Pollinations key exception: {e}")
   279→   279→                return AIResponse(text="", model=model, provider="pollinations", error=True, error_message=str(e))
   280→   280→
   281→   281→        async def _safe_try_pollinations_free():
   282→   282→            """Try Pollinations free — return response or error."""
   283→   283→            try:
   284→   284→                return await self._try_pollinations_free(
   285→   285→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   286→   286→                )
   287→   287→            except Exception as e:
   288→   288→                logger.error(f"Pollinations free exception: {e}")
   289→   289→                return AIResponse(text="", model=model, provider="pollinations-free", error=True, error_message=str(e))
   290→   290→
   291→   291→        async def _safe_try_cloudflare():
   292→   292→            """Try Cloudflare — return response or error."""
   293→   293→            try:
   294→   294→                return await self._try_cloudflare(
   295→   295→                    user_id, message, history, sys_prompt, temperature, max_tokens
   296→   296→                )
   297→   297→            except Exception as e:
   298→   298→                logger.error(f"Cloudflare exception: {e}")
   299→   299→                return AIResponse(text="", model="cloudflare", provider="cloudflare", error=True, error_message=str(e))
   300→   300→
   301→   301→        async def _safe_try_local_fallback():
   302→   302→            """Try local model as fallback for function routes."""
   303→   303→            try:
   304→   304→                return await self._try_local(
   305→   305→                    user_id, message, history, sys_prompt, temperature, max_tokens
   306→   306→                )
   307→   307→            except Exception as e:
   308→   308→                logger.error(f"Local fallback exception: {e}")
   309→   309→                return AIResponse(text="", model="local-qwen3-4b", provider="local", error=True, error_message=str(e))
   310→   310→
   311→   311→        # Build concurrent task list based on route type
   312→   312→        # Priority ordering: we want the best quality response first.
   313→   313→        # But we launch them ALL at once and take the first success.
   314→   314→        concurrent_tasks = []
   315→   315→
   316→   316→        # Always try Pollinations with key (best quality when available)
   317→   317→        if self._primary and self._primary._build_key_tier_list():
   318→   318→            concurrent_tasks.append(("pollinations_key", _safe_try_pollinations()))
   319→   319→
   320→   320→        # Always try Cloudflare in parallel (fast, independent service)
   321→   321→        if self._cloudflare and self._cloudflare._accounts:
   322→   322→            concurrent_tasks.append(("cloudflare", _safe_try_cloudflare()))
   323→   323→
   324→   324→        # For function routes, also try local model as concurrent fallback
   325→   325→        if route_type == "function" and config.ENABLE_LOCAL_MODEL and self._local and self._local._model_loaded:
   326→   326→            concurrent_tasks.append(("local_fallback", _safe_try_local_fallback()))
   327→   327→
   328→   328→        if not concurrent_tasks:
   329→   329→            # No providers available — try free API as last hope
   330→   330→            concurrent_tasks.append(("pollinations_free", _safe_try_pollinations_free()))
   331→   331→
   332→   332→        # Execute all tasks concurrently, return FIRST successful response
   333→   333→        # Use asyncio.wait with FIRST_COMPLETED to get the fastest response
   334→   334→        task_names = [name for name, _ in concurrent_tasks]
   335→   335→        task_coros = [coro for _, coro in concurrent_tasks]
   336→   336→        tasks = [_asyncio.create_task(coro) for coro in task_coros]
   337→   337→
   338→   338→        try:
   339→   339→            # Wait for the first task to complete
   340→   340→            done, pending = await _asyncio.wait(tasks, return_when=_asyncio.FIRST_COMPLETED)
   341→   341→
   342→   342→            # Check if any completed task succeeded
   343→   343→            for task in done:
   344→   344→                response = task.result()
   345→   345→                if not response.error and response.text:
   346→   346→                    # Cancel remaining tasks (we have a winner!)
   347→   347→                    for p in pending:
   348→   348→                        p.cancel()
   349→   349→                    # Also cancel other done tasks we don't need
   350→   350→                    for t in tasks:
   351→   351→                        if t is not task and not t.done():
   352→   352→                            t.cancel()
   353→   353→
   354→   354→                    # Track which level succeeded
   355→   355→                    task_idx = tasks.index(task)
   356→   356→                    level_name = task_names[task_idx]
   357→   357→                    if level_name == "pollinations_key":
   358→   358→                        self._level1_count += 1
   359→   359→                    elif level_name == "pollinations_free":
   360→   360→                        self._level2_count += 1
   361→   361→                    elif level_name == "cloudflare":
   362→   362→                        self._level3_count += 1
   363→   363→                    elif level_name == "local_fallback":
   364→   364→                        self._level0_count += 1
   365→   365→
   366→   366→                    self._total_requests += 1
   367→   367→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   368→   368→
   369→   369→            # First completed task(s) failed — wait for others
   370→   370→            # If there are still pending tasks, wait for them
   371→   371→            if pending:
   372→   372→                done2, pending2 = await _asyncio.wait(pending, return_when=_asyncio.FIRST_COMPLETED)
   373→   373→                done = done.union(done2)
   374→   374→                pending = pending2
   375→   375→
   376→   376→                for task in done2:
   377→   377→                    response = task.result()
   378→   378→                    if not response.error and response.text:
   379→   379→                        for p in pending2:
   380→   380→                            p.cancel()
   381→   381→                        for t in tasks:
   382→   382→                            if t is not task and not t.done():
   383→   383→                                t.cancel()
   384→   384→
   385→   385→                        task_idx = tasks.index(task)
   386→   386→                        level_name = task_names[task_idx]
   387→   387→                        if level_name == "pollinations_key":
   388→   388→                            self._level1_count += 1
   389→   389→                        elif level_name == "pollinations_free":
   390→   390→                            self._level2_count += 1
   391→   391→                        elif level_name == "cloudflare":
   392→   392→                            self._level3_count += 1
   393→   393→                        elif level_name == "local_fallback":
   394→   394→                            self._level0_count += 1
   395→   395→
   396→   396→                        self._total_requests += 1
   397→   397→                        return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   398→   398→
   399→   399→            # All concurrent tasks failed — try Pollinations free as sequential fallback
   400→   400→            # (we didn't include it in concurrent tasks if we had paid keys)
   401→   401→            if not any(n == "pollinations_free" for n in task_names):
   402→   402→                logger.warning(f"All concurrent providers failed (route={route_type}), trying free API sequentially")
   403→   403→                response = await self._try_pollinations_free(
   404→   404→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   405→   405→                )
   406→   406→                if not response.error:
   407→   407→                    self._level2_count += 1
   408→   408→                    self._total_requests += 1
   409→   409→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   410→   410→
   411→   411→            # ALL levels failed — collect error info from all tasks
   412→   412→            errors = []
   413→   413→            for task in done:
   414→   414→                try:
   415→   415→                    r = task.result()
   416→   416→                    if r.error:
   417→   417→                        errors.append(f"{r.provider}: {r.error_message}")
   418→   418→                except Exception as e:
   419→   419→                    errors.append(f"exception: {e}")
   420→   420→
   421→   421→            # Cancel any remaining tasks
   422→   422→            for t in tasks:
   423→   423→                if not t.done():
   424→   424→                    t.cancel()
   425→   425→
   426→   426→        except Exception as e:
   427→   427→            logger.error(f"Concurrent failover error: {e}")
   428→   428→            # Cancel all tasks on unexpected error
   429→   429→            for t in tasks:
   430→   430→                if not t.done():
   431→   431→                    t.cancel()
   432→   432→
   433→   433→            # Try sequential fallback as emergency path
   434→   434→            try:
   435→   435→                response = await self._try_pollinations_free(
   436→   436→                    user_id, message, history, sys_prompt, temperature, max_tokens, model
   437→   437→                )
   438→   438→                if not response.error:
   439→   439→                    self._level2_count += 1
   440→   440→                    self._total_requests += 1
   441→   441→                    return await self._save_response(user_id, message, response, sys_prompt, use_cache, save_history)
   442→   442→            except Exception:
   443→   443→                pass

   476→   476→    async def _try_pollinations(self, user_id: int, message: str, history: list,
   477→   477→                                 sys_prompt: str, temperature: float, max_tokens: int,
   478→   478→                                 model: str) -> AIResponse:

   481→   481→        """

   483→   483→

   497→   497→            if is_key_error:

   518→   518→                    temperature=temperature,

   579→   579→    async def _try_local(self, user_id: int, message: str, history: list,
   580→   580→                          sys_prompt: str, temperature: float, max_tokens: int) -> AIResponse:
   581→   581→        """Level 0: Try Local Model (Qwen3-4B).
   582→   582→
   583→   583→        Uses ChatML format for Qwen3 with /no_think for fast responses.
   584→   584→        Primary for CHAT and COMMENT routes, fallback for FUNCTION routes.
   585→   585→        
   586→   586→        CONTEXT WINDOW BUDGET (4096 tokens, ~1.3 chars/token for Russian):
   587→   587→          - Output: 512 tokens (MODEL_MAX_TOKENS)
   588→   588→          - Safety margin: 64 tokens
   589→   589→          - Available for input: 3520 tokens (~4576 chars)
   590→   590→          - System prompt (local_system_prompt v2): ~2340 chars (~1800 tokens)
   591→   591→          - History: 3 turns × 200 chars = ~600 chars (~462 tokens)
   592→   592→          - User message: up to ~1000 chars (~770 tokens)
   593→   593→          - Total: ~3940 chars (~3032 tokens) — fits in 3520 with margin!
   594→   594→        """
   595→   595→        if not self._local:
   596→   596→            return AIResponse(
   597→   597→                text="",
   598→   598→                model="local-qwen3-4b",
   599→   599→                provider="local",
   600→   600→                error=True,
   601→   601→                error_message="Local provider not initialized",
   602→   602→            )
   603→   603→
   604→   604→        if not config.ENABLE_LOCAL_MODEL:
   605→   605→            return AIResponse(
   606→   606→                text="",
   607→   607→                model="local-qwen3-4b",
   608→   608→                provider="local",
   609→   609→                error=True,
   610→   610→                error_message="Local model disabled (ENABLE_LOCAL_MODEL=false)",
   611→   611→            )
   612→   612→
   613→   613→        # Use EXPANDED system prompt for local model (v2.0 — ~1800 tokens).
   614→   614→        # The full 12K-char cloud prompt (~9000 tokens) still does NOT fit.
   615→   615→        # But the expanded v2 prompt gives much better quality than the old compact one.
   616→   616→        compact_prompt = persona.local_system_prompt
   617→   617→
   618→   618→        # Build messages for local model using its own ChatML format
   619→   619→        messages = [{"role": "system", "content": compact_prompt}]
   620→   620→
   621→   621→        # Add limited history for local model (saves context window)
   622→   622→        # With expanded prompt (~1800 tokens), we have ~1720 tokens left for history+message.
   623→   623→        # 3 history turns × ~200 chars × 1.3 chars/token = ~462 tokens
   624→   624→        # User message up to ~1000 chars = ~770 tokens
   625→   625→        # Total: ~1232 tokens — well within budget.
   626→   626→        history_limit = min(config.MODEL_HISTORY_LIMIT, 3)
   627→   627→        limited_history = history[-history_limit:] if history else []
   628→   628→        for msg in limited_history:
   629→   629→            role = msg.get("role", "user")
   630→   630→            content = msg.get("content", "")
   631→   631→            # Truncate long history messages to save context tokens
   632→   632→            if role in ("user", "assistant") and content:
   633→   633→                messages.append({"role": role, "content": content[:200]})
   634→   634→
   635→   635→        # Truncate user message to prevent context overflow.
   636→   636→        # With expanded system prompt, allow up to ~1000 chars for user message.
   637→   637→        # The local_provider.chat() has its own safety truncation as well.
   638→   638→        truncated_message = message[:1000] if len(message) > 1000 else message
   639→   639→        if len(message) > 1000:
   640→   640→            logger.debug(f"Truncated user message for local model: {len(message)} → 1000 chars")
   641→   641→        messages.append({"role": "user", "content": truncated_message})
   642→   642→
   643→   643→        return await self._local.chat(
   644→   644→            messages=messages,
   645→   645→            temperature=temperature,
   646→   646→            max_tokens=min(max_tokens, config.MODEL_MAX_TOKENS),
   647→   647→        )

   976→   976→    def _router_smart_truncate(text: str, max_len: int) -> str:
   977→   977→        """Smart truncation for AI router — cuts at sentence/paragraph boundary.
   978→   978→        
   979→   979→        Same logic as channel.py's _smart_truncate to avoid mid-word cuts.
   980→   980→        """
   981→   981→        if len(text) <= max_len:
   982→   982→            return text
   983→   983→        
   984→   984→        target = max_len - 3
   985→   985→        if target < 50:
   986→   986→            return text[:target] + "..."
   987→   987→        
   988→   988→        search_zone = text[:target + 1]
   989→   989→        
   990→   990→        # 1. Paragraph break
   991→   991→        last_para = search_zone.rfind("\n\n")
   992→   992→        if last_para > target * 0.5:
   993→   993→            return text[:last_para].rstrip() + "..."
   994→   994→        
   995→   995→        # 2. Sentence end
   996→   996→        sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   997→   997→        best_sentence_end = -1
   998→   998→        for end_char in sentence_end_chars:
   999→   999→            pos = search_zone.rfind(end_char)
  1000→  1000→            if pos > best_sentence_end and pos > target * 0.5:
  1001→  1001→                best_sentence_end = pos + len(end_char) - 1
  1002→  1002→        
  1003→  1003→        if best_sentence_end > target * 0.5:
  1004→  1004→            return text[:best_sentence_end + 1].rstrip() + "..."
  1005→  1005→        
  1006→  1006→        # 3. Newline
  1007→  1007→        last_newline = search_zone.rfind("\n")
  1008→  1008→        if last_newline > target * 0.5:
  1009→  1009→            return text[:last_newline].rstrip() + "..."
  1010→  1010→        
  1011→  1011→        # 4. Space
  1012→  1012→        last_space = search_zone.rfind(" ")
  1013→  1013→        if last_space > target * 0.5:
  1014→  1014→            return text[:last_space].rstrip() + "..."
  1015→  1015→        
  1016→  1016→        # 5. Hard cut
  1017→  1017→        return text[:target].rstrip() + "..."
