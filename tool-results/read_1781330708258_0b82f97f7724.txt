    25→def _compute_fingerprint(title: str) -> str:
    26→    """Compute a simplified fingerprint for dedup.
    27→    Remove articles, lowercase, take first 5 words.
    28→    """
    29→    import re
    30→    # Remove common articles/prepositions
    31→    cleaned = re.sub(r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b', '', title.lower())
    32→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
    33→    words = cleaned.split()
    34→    return ' '.join(words[:5])

    37→def _fingerprint_matches_existing(fingerprint: str) -> bool:
    38→    """Check if a fingerprint matches any recently used fingerprint (edit distance on first 4 words)."""
    39→    fp_words = fingerprint.split()[:4]
    40→    for existing in _recent_fingerprints:
    41→        ex_words = existing.split()[:4]
    42→        # Count matching words
    43→        matches = sum(1 for w in fp_words if w in ex_words)
    44→        if matches >= 3:
    45→            return True
    46→    return False

   274→# ── RSS Fetcher ────────────────────────────────────────────────────────────────
   275→
   276→async def fetch_rss(source: NewsSource) -> List[Dict]:
   277→    """
   278→    Fetch and parse an RSS feed.
   279→    Returns list of news items.
   280→    """
   281→    items = []
   282→    try:
   283→        async with httpx.AsyncClient(
   284→            timeout=20.0,
   285→            follow_redirects=True,
   286→            headers={
   287→                "User-Agent": "Mozilla/5.0 (compatible; AsyaBot/1.0; +https://t.me/asiaexp_bot)",
   288→                "Accept": "application/rss+xml, application/xml, text/xml, application/atom+xml, */*",
   289→            },
   290→        ) as client:
   291→            response = await client.get(source.url)
   292→
   293→            if response.status_code != 200:
   294→                logger.warning(f"RSS {source.name} returned {response.status_code}")
   295→                return items
   296→
   297→            # feedparser can parse from string
   298→            feed = feedparser.parse(response.text)
   299→
   300→            for entry in feed.entries[:20]:  # Limit per source
   301→                title = entry.get("title", "").strip()
   302→                url = entry.get("link", "").strip()
   303→
   304→                if not title or not url:
   305→                    continue
   306→
   307→                # Get summary/description
   308→                summary = ""
   309→                if entry.get("summary"):
   310→                    summary = _clean_html(entry["summary"])
   311→                elif entry.get("description"):
   312→                    summary = _clean_html(entry["description"])
   313→
   314→                # Truncate summary
   315→                if len(summary) > 500:
   316→                    summary = summary[:500] + "..."
   317→
   318→                # Get publish date
   319→                published = 0.0
   320→                if entry.get("published_parsed"):
   321→                    try:
   322→                        published = time.mktime(entry["published_parsed"])
   323→                    except (ValueError, TypeError, OverflowError):
   324→                        published = time.time()
   325→                elif entry.get("updated_parsed"):
   326→                    try:
   327→                        published = time.mktime(entry["updated_parsed"])
   328→                    except (ValueError, TypeError, OverflowError):
   329→                        published = time.time()
   330→                else:
   331→                    published = time.time()
   332→
   333→                # Extract image URLs from RSS entry
   334→                image_urls = _extract_entry_images(entry)
   335→
   336→                items.append({
   337→                    "source": source.name,
   338→                    "title": title,
   339→                    "url": url,
   340→                    "summary": summary,
   341→                    "published": published,
   342→                    "category": source.category,
   343→                    "lang": source.lang,
   344→                    "image_urls": image_urls,
   345→                })
   346→
   347→            logger.info(f"Fetched {len(items)} items from {source.name}")
   348→
   349→    except httpx.TimeoutException:
   350→        logger.warning(f"RSS timeout: {source.name}")
   351→    except Exception as e:
   352→        logger.error(f"RSS fetch error for {source.name}: {e}")
   353→
   354→    return items

   359→def is_auto_relevant(item: Dict, lang: str = "ru") -> bool:
   360→    """Check if a news item is relevant to the auto topic."""
   361→    text = f"{item.get('title', '')} {item.get('summary', '')}".lower()
   362→
   363→    if lang == "ru" or lang == "":
   364→        keywords = AUTO_KEYWORDS_RU
   365→    elif lang == "en":
   366→        keywords = AUTO_KEYWORDS_EN
   367→    elif lang == "de":
   368→        keywords = AUTO_KEYWORDS_DE
   369→    else:
   370→        keywords = AUTO_KEYWORDS_RU + AUTO_KEYWORDS_EN
   371→
   372→    # Check if any keyword matches
   373→    for kw in keywords:
   374→        if kw.lower() in text:
   375→            return True
   376→
   377→    return False

   759→def _normalize_image_url(url: str) -> str:
   760→    """Normalize an image URL: decode HTML entities, fix scheme, clean up."""
   761→    if not url:
   762→        return ""
   763→    url = url.strip()
   764→    # Decode HTML entities: &amp; → &, &#038; → &, &lt; → <, etc.
   765→    url = _html_unescape(url)
   766→    # Fix scheme-relative URLs
   767→    if url.startswith("//"):
   768→        url = "https:" + url
   769→    return url

   772→def _is_junk_image_url(url: str) -> bool:
   773→    """Check if an image URL is junk: tracker, icon, tiny thumbnail, default image."""
   774→    from urllib.parse import urlparse, parse_qs
   775→    try:
   776→        url_lower = url.lower()
   777→        parsed = urlparse(url)
   778→        domain = parsed.netloc.lower()
   779→        path_lower = parsed.path.lower()
   780→
   781→        # Skip tracker/analytics domains
   782→        if any(skip in domain for skip in _SKIP_IMAGE_DOMAINS):
   783→            return True
   784→
   785→        # Skip icon/logo/badge/button filenames in path
   786→        if any(kw in path_lower for kw in [
   787→            "icon", "avatar", "logo", "favicon", "badge", "button", "banner",
   788→        ]):
   789→            return True
   790→
   791→        # Skip /feed/ URLs (CarScoops bug)
   792→        if _FEED_URL_PATTERN.search(url_lower):
   793→            return True
   794→
   795→        # Skip generic/default image filenames (also prefix-match: business_card-1)
   796→        stem = path_lower.rsplit("/", 1)[-1].split(".")[0].replace("-", "_")
   797→        if stem in _JUNK_IMAGE_STEMS:
   798→            return True
   799→        for junk_stem in _JUNK_IMAGE_STEMS:
   800→            if stem.startswith(junk_stem):
   801→                return True
   802→
   803→        # Skip tiny thumbnail dimensions in URL
   804→        for pattern in _TINY_SIZE_PATTERNS:
   805→            if re.search(pattern, url_lower):
   806→                return True
   807→
   808→    except Exception:
   809→        return True  # If we can't parse it, skip it
   810→    return False

   813→def _upgrade_thumbnail_url(url: str) -> str:
   814→    """Upgrade low-quality thumbnail URLs to higher resolution versions.
   815→
   816→    Known CDN patterns:
   817→    - BBC: /240/ → /640/ in path
   818→    - Autosport/Motorsport.com: /s6/ → /s12/ in path
   819→    - Reddit: width=140 → width=640 in query
   820→    - Autocar: /car_review_image_190/ → /body-image/ (190×125 → 900×600)
   821→    """
   822→    # BBC: /240/cpsprodpb/ → /640/cpsprodpb/
   823→    if "bbci.co.uk" in url or "bbc.co.uk" in url:
   824→        url = re.sub(r'/240/', '/640/', url, count=1)
   825→
   826→    # Autosport / Motorsport.com: /s6/ → /s12/
   827→    if "motorsport.com" in url or "autosport.com" in url:
   828→        url = re.sub(r'/s6/', '/s12/', url, count=1)
   829→
   830→    # Reddit preview: width=140 → width=640
   831→    if "preview.redd.it" in url or "external-preview.redd.it" in url:
   832→        url = re.sub(r'width=140', 'width=640', url, count=1)
   833→        url = re.sub(r'height=140', 'height=640', url, count=1)
   834→
   835→    # Autocar UK: /styles/car_review_image_190/ → /styles/body-image/
   836→    if "autocar.co.uk" in url:
   837→        url = url.replace('/styles/car_review_image_190/', '/styles/body-image/')
   838→
   839→    return url

   842→def _extract_entry_images(entry) -> List[str]:
   843→    """Extract image URLs from a feedparser entry.
   844→
   845→    Priority order (highest quality first):
   846→    1. media_content (Media RSS — editors pick these, usually hero images)
   847→    2. enclosures (often full-size article images)
   848→    3. media_thumbnail (small but from structured metadata)
   849→    4. links with image type
   850→    5. <img> tags in content/summary HTML (last — often thumbnails + junk)
   851→
   852→    All URLs are HTML-entity-decoded, deduplicated (normalized),
   853→    junk-filtered, and thumbnail-upgraded before returning.
   854→    Returns list of clean image URLs (up to 10).
   855→    """
   856→    images = []
   857→    seen_normalized = set()  # Dedup by normalized URL (after unescape)
   858→
   859→    def _add_image(raw_url: str):
   860→        """Add image URL if valid and not already seen (after normalization)."""
   861→        url = _normalize_image_url(raw_url)
   862→        if not url or len(url) < _MIN_IMAGE_URL_LEN:
   863→            return
   864→        if _is_junk_image_url(url):
   865→            return
   866→        # Upgrade low-quality thumbnails to bigger versions
   867→        url = _upgrade_thumbnail_url(url)
   868→        # Dedup by normalized form (lowercase for comparison)
   869→        norm_key = url.lower().rstrip("/")
   870→        if norm_key in seen_normalized:
   871→            return
   872→        seen_normalized.add(norm_key)
   873→        images.append(url)
   874→
   875→    # 1. media_content (Media RSS extension — most reliable, editor-chosen)
   876→    for media in entry.get("media_content", []) or []:
   877→        url = media.get("url", "")
   878→        media_type = media.get("type", "").lower()
   879→        medium = media.get("medium", "").lower()
   880→        if url and (medium == "image" or "image" in media_type or _has_image_ext(url)):
   881→            _add_image(url)
   882→
   883→    # 2. enclosures (often full-size article images)
   884→    for enclosure in entry.get("enclosures", []) or []:
   885→        url = enclosure.get("href", "") or enclosure.get("url", "")
   886→        enc_type = enclosure.get("type", "").lower()
   887→        if url and ("image" in enc_type or _has_image_ext(url)):
   888→            _add_image(url)
   889→
   890→    # 3. media_thumbnail (small but structured metadata)
   891→    for thumb in entry.get("media_thumbnail", []) or []:
   892→        url = thumb.get("url", "")
   893→        if url:
   894→            _add_image(url)
   895→
   896→    # 4. links with image rel or type
   897→    for link in entry.get("links", []) or []:
   898→        link_type = link.get("type", "").lower()
   899→        link_rel = link.get("rel", "").lower()
   900→        url = link.get("href", "")
   901→        if url and ("image" in link_type or link_rel == "enclosure"):
   902→            _add_image(url)
   903→
   904→    # 5. Extract <img> tags from content/summary HTML (last resort — noisy)
   905→    for field_name in ["content", "summary", "description"]:
   906→        content_value = entry.get(field_name)
   907→        if isinstance(content_value, list):
   908→            for item in content_value:
   909→                html = item.get("value", "") if isinstance(item, dict) else str(item)
   910→                _extract_img_urls(html, _add_image)
   911→        elif isinstance(content_value, str):
   912→            _extract_img_urls(content_value, _add_image)
   913→
   914→    return images[:10]

   917→def _has_image_ext(url: str) -> bool:
   918→    """Check if URL ends with an image extension."""
   919→    from urllib.parse import urlparse
   920→    try:
   921→        path = urlparse(url).path.lower().split("?")[0].split("#")[0]
   922→        return any(path.endswith(ext) for ext in _VALID_IMAGE_EXTENSIONS)
   923→    except Exception:
   924→        return False

   927→def _extract_img_urls(html: str, callback):
   928→    """Extract src attributes from <img> tags in HTML.
   929→
   930→    Also checks data-src and data-lazy-src for lazy-loaded images.
   931→    Decodes HTML entities before passing to callback.
   932→    """
   933→    if not html:
   934→        return
   935→    for attr in ["src", "data-src", "data-lazy-src"]:
   936→        for match in re.finditer(
   937→            rf'<img[^>]+{attr}=["\']([^"\']+)["\']', html, re.IGNORECASE
   938→        ):
   939→            url = match.group(1)
   940→            if url and len(url) >= _MIN_IMAGE_URL_LEN:
   941→                callback(url)
   942→

   945→
   946→def _clean_html(text: str) -> str:
   947→    """Remove HTML tags and decode entities."""
   948→    text = re.sub(r'<[^>]+>', '', text)
   949→    text = text.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>')
   950→    text = text.replace('&quot;', '"').replace('&#39;', "'").replace('&nbsp;', ' ')
   951→    text = re.sub(r'\s+', ' ', text).strip()
   952→    return text
