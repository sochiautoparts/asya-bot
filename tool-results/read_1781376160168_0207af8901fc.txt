    50→# ── VIN / Body number detection ───────────────────────────────────────────────
    51→
    52→_VIN_PATTERN = re.compile(r'\b[A-HJ-NPR-Z0-9]{17}\b', re.IGNORECASE)
    53→# Also match VINs with spaces or dashes (users often type them separated)
    54→_VIN_FLEX_PATTERN = re.compile(
    55→    r'(?:VIN[-:]?\s*|вин[-:]?\s*|вин-код[-:]?\s*)?([A-HJ-NPR-Z0-9](?:[A-HJ-NPR-Z0-9\s\-]{14,22})[A-HJ-NPR-Z0-9])',
    56→    re.IGNORECASE
    57→)
    58→_BODY_NUMBER_PATTERN = re.compile(
    59→    r'(?:номер\s+кузова|кузовн?ой\s+номер|body\s*number|кузов)\s*[:\s]*([A-Z0-9\-/]{5,20})',
    60→    re.IGNORECASE
    61→)
    62→
    63→
    64→def _detect_vin(text: str) -> Optional[str]:
    65→    """Detect a VIN code (17 chars) in text."""
    66→    match = _VIN_PATTERN.search(text.upper())
    67→    if match:
    68→        vin = match.group(0)
    69→        # Validate check digit position (9th char)
    70→        if len(vin) == 17 and vin[8] in '0123456789X':
    71→            return vin
    72→    return None
    73→
    74→
    75→def _detect_body_number(text: str) -> Optional[str]:
    76→    """Detect a body number reference in text."""
    77→    match = _BODY_NUMBER_PATTERN.search(text)
    78→    if match:
    79→        return match.group(1)
    80→    return None

   118→def _guess_gender(first_name: str) -> str:
   119→    """Guess gender from Russian first name. Returns 'male', 'female', or 'unknown'."""
   120→    if not first_name:
   121→        return "unknown"
   122→
   123→    name_lower = first_name.lower().strip()
   124→
   125→    if name_lower in COMMON_MALE_NAMES:
   126→        return "male"
   127→    if name_lower in COMMON_FEMALE_NAMES:
   128→        return "female"
   129→
   130→    if name_lower.endswith(FEMALE_NAME_ENDINGS):
   131→        if name_lower.endswith("ь"):
   132→            pass
   133→        else:
   134→            return "female"
   135→
   136→    if name_lower.endswith("й") or name_lower.endswith("ь"):
   137→        return "male"
   138→
   139→    return "unknown"

   486→    In GROUPS/SUPERGROUPS: use LOCAL-ONLY routing (comments mode).

   488→    """
   489→    if not await _check_user(message):
   490→        return
   491→
   492→    # Check if we're in a group/supergroup — use LOCAL-ONLY for comments
   493→    is_group = message.chat.type in ("group", "supergroup")
   494→
   495→    # In groups: Ася should just comment on local model ONLY, not use cloud vision
   496→    if is_group:
   497→        # LOCAL MODEL ONLY for group comments — no cloud API waste!
   498→        # Use singleton from ai_router — no reloading the model every time!
   499→        from ai.router import ai_router
   500→        local_provider = ai_router._local
   501→        if local_provider and await local_provider.is_available():

  1033→def _smart_truncate(text: str, max_chars: int) -> str:
  1034→    """Truncate text to max_chars, trying not to cut in the middle of a sentence.
  1035→    
  1036→    Looks for the last sentence boundary (. ! ? …) within the limit.
  1037→    Falls back to last newline, then last space, then hard cut.
  1038→    Always appends "..." if truncated.
  1039→    """
  1040→    if len(text) <= max_chars:
  1041→        return text
  1042→
  1043→    # Reserve 3 chars for "..."
  1044→    limit = max_chars - 3
  1045→    if limit <= 0:
  1046→        return "..."
  1047→
  1048→    # Try to find last sentence boundary within limit
  1049→    sentence_endings = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
  1050→    best_pos = -1
  1051→    best_ending_len = 0
  1052→    for ending in sentence_endings:
  1053→        pos = text.rfind(ending, 0, limit + 1)
  1054→        if pos > best_pos:
  1055→            best_pos = pos
  1056→            best_ending_len = len(ending)
  1057→
  1058→    if best_pos > limit // 2:
  1059→        # Found a good sentence boundary — cut after the punctuation mark
  1060→        return text[:best_pos + best_ending_len].rstrip() + "..."
  1061→
  1062→    # Try last newline
  1063→    nl_pos = text.rfind('\n', 0, limit + 1)
  1064→    if nl_pos > limit // 2:
  1065→        return text[:nl_pos].rstrip() + "..."
  1066→
  1067→    # Try last space
  1068→    sp_pos = text.rfind(' ', 0, limit + 1)
  1069→    if sp_pos > limit // 2:
  1070→        return text[:sp_pos].rstrip() + "..."
  1071→
  1072→    # Hard cut
  1073→    return text[:limit].rstrip() + "..."

  1188→def _clean_raw_partner_urls(text: str, partner_links: list) -> str:
  1189→    """Remove raw affiliate URLs that AI may have dumped into the response.
  1190→    
  1191→    If the AI already included the full affiliate URLs in its response text,
  1192→    this function removes them to avoid duplication with the clean formatted
  1193→    section that will be appended by _format_partner_links_section.
  1194→    Also removes shop name + URL patterns like "Росско: https://...".
  1195→    """
  1196→    if not partner_links:
  1197→        return text
  1198→    
  1199→    for name, url in partner_links:
  1200→        url_escaped = re.escape(url)
  1201→        # Remove standalone URL lines
  1202→        text = re.sub(
  1203→            rf'^[\s\-—]*{url_escaped}[\s]*$',
  1204→            '',
  1205→            text,
  1206→            flags=re.MULTILINE
  1207→        )
  1208→        # Remove "name: URL" or "name — URL" patterns on their own lines
  1209→        text = re.sub(
  1210→            rf'^[\s\-—]*{re.escape(name)}\s*[:\-—]\s*{url_escaped}[\s]*$',
  1211→            '',
  1212→            text,
  1213→            flags=re.MULTILINE
  1214→        )
  1215→        # Remove any line containing the full raw affiliate URL (even mid-sentence)
  1216→        # This catches cases where AI writes: "Посмотри на Росско: https://ujhjj.com/g/..."
  1217→        text = re.sub(
  1218→            rf'[^\n]*{url_escaped}[^\n]*',
  1219→            '',
  1220→            text,
  1221→            flags=re.MULTILINE
  1222→        )
  1223→        # Remove bare URL anywhere in text (affiliate URLs are very long, easy to detect)
  1224→        if len(url) > 50:  # Only long tracking URLs
  1225→            # Match the base tracking domain pattern
  1226→            url_base = re.escape(url.split('?')[0])  # Before query params
  1227→            text = re.sub(
  1228→                rf'{url_base}\?[^\s\n"]+',
  1229→                '',
  1230→                text
  1231→            )
  1232→    
  1233→    text = re.sub(r'\n{3,}', '\n\n', text)
  1234→    return text.strip()

  1260→def _replace_plain_urls_with_affiliate(text: str) -> str:
  1261→    """Replace any plain partner domain URLs with affiliate goto_link equivalents.
  1262→    
  1263→    When the AI generates responses containing plain URLs like rossko.ru or 
  1264→    autopiter.ru instead of the affiliate tracking links from admitad_ads.json,
  1265→    this function detects them and replaces with the proper goto_link.
  1266→    
  1267→    This handles cases where:
  1268→    - AI ignores the system prompt and invents plain URLs
  1269→    - AI uses domain names without the affiliate wrapper
  1270→    - Photo handler bypasses the normal link injection pipeline
  1271→    """
  1272→    try:
  1273→        partner_manager.ensure_loaded()
  1274→    except Exception:
  1275→        return text
  1276→    
  1277→    for domain, display_name in _PARTNER_DOMAINS_MAP.items():
  1278→        prog = partner_manager.get_by_site(domain)
  1279→        if not prog or not prog.goto_link:
  1280→            continue
  1281→        
  1282→        affiliate_url = prog.goto_link
  1283→        
  1284→        # Pattern 1: Full URLs with paths — https://rossko.ru/search?text=abc
  1285→        # Replace the entire URL with the affiliate link
  1286→        pattern = rf'https?://{re.escape(domain)}[^\s<>)\]"\']*'
  1287→        matches = re.findall(pattern, text)
  1288→        for plain_url in matches:
  1289→            # Try to extract search query and build affiliate link with search
  1290→            search_query = ""
  1291→            if "search" in plain_url or "querystr" in plain_url or "q=" in plain_url:
  1292→                try:
  1293→                    from urllib.parse import urlparse, parse_qs
  1294→                    parsed = urlparse(plain_url)
  1295→                    params = parse_qs(parsed.query)
  1296→                    for key in ("text", "querystr", "q", "query", "SearchText", "keyword", "p"):
  1297→                        if key in params:
  1298→                            search_query = params[key][0]
  1299→                            break
  1300→                except Exception:
  1301→                    pass
  1302→            
  1303→            if search_query:
  1304→                replacement = prog.get_search_url(search_query)
  1305→            else:
  1306→                replacement = affiliate_url
  1307→            
  1308→            text = text.replace(plain_url, replacement)
  1309→        
  1310→        # Pattern 2: Bare domain mentions — rossko.ru or www.rossko.ru (not already part of a longer URL)
  1311→        # Only replace if it's NOT already part of an affiliate URL (which would be much longer)
  1312→        bare_pattern = rf'(?<![/\w.-])(?:www\.)?{re.escape(domain)}(?![/\w.-])'
  1313→        bare_matches = re.findall(bare_pattern, text)
  1314→        for bare_domain in bare_matches:
  1315→            # Check this isn't already inside an affiliate URL
  1316→            idx = text.find(bare_domain)
  1317→            if idx > 0:
  1318→                # Look back — if there's a tracking domain prefix, skip
  1319→                before = text[max(0, idx-50):idx]
  1320→                if any(tracking_domain in before for tracking_domain in ["ad.admitad.com", ".com/g/", "xmknb.com", "ujhjj.com", "rcpsj.com", "sgkaa.com"]):
  1321→                    continue
  1322→            text = text.replace(bare_domain, affiliate_url, 1)
  1323→    
  1324→    return text

  1327→def _clean_markdown(text: str) -> str:
  1328→    """Remove markdown formatting that Asya shouldn't use.
  1329→    
  1330→    Markdown links [text](url) are converted to plain text: text — url.
  1331→    Other markdown is stripped. No HTML — plain text for chat responses.
  1332→    """
  1333→    # Convert markdown links [text](url) → text — url (plain text format)
  1334→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1 — \2', text)
  1335→    
  1336→    # Remove bold (after escaping, ** won't be in &lt; form)
  1337→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
  1338→    # Remove italic
  1339→    text = re.sub(r'\*(.+?)\*', r'\1', text)
  1340→    # Remove code blocks
  1341→    text = re.sub(r'```[\s\S]*?```', lambda m: m.group(0).strip('`').strip(), text)
  1342→    # Remove inline code
  1343→    text = re.sub(r'`(.+?)`', r'\1', text)
  1344→    # Remove headers
  1345→    text = re.sub(r'^#{1,6}\s+', '', text, flags=re.MULTILINE)
  1346→    # Remove bullet points (convert to simple text)
  1347→    text = re.sub(r'^[-*]\s+', '— ', text, flags=re.MULTILINE)
  1348→    return text

  1351→def _split_message(text: str, max_length: int = 4096) -> list:
  1352→    """Split a long message into chunks at paragraph boundaries."""
  1353→    if len(text) <= max_length:
  1354→        return [text]
  1355→
  1356→    chunks = []
  1357→    while text:
  1358→        if len(text) <= max_length:
  1359→            chunks.append(text)
  1360→            break
  1361→
  1362→        # Try to split at newline near the limit
  1363→        split_pos = text.rfind('\n', 0, max_length)
  1364→        if split_pos < max_length // 2:
  1365→            # Try splitting at space
  1366→            split_pos = text.rfind(' ', 0, max_length)
  1367→        if split_pos < max_length // 2:
  1368→            # Hard split
  1369→            split_pos = max_length
  1370→
  1371→        chunks.append(text[:split_pos].strip())
  1372→        text = text[split_pos:].strip()
  1373→
  1374→    return chunks
