   236→class PollinationsProvider(BaseAIProvider):
   237→    """Pollinations AI provider v5.0 — DUAL KEY + FREE API FALLBACK + 60+ MODELS!
   238→
   239→    Uses gen.pollinations.ai/v1/chat/completions (OpenAI-compatible).
   240→    Falls back to text.pollinations.ai (FREE, no auth) when keys are depleted.
   241→
   242→    FAILOVER CHAIN (within provider):
   243→      1. Try KEY1 first (config.POLLINATIONS_API_KEY)
   244→      2. On 402/401 from KEY1 -> switch to KEY2 (config.POLLINATIONS_API_KEY_2)
   245→      3. On 402/401 from KEY2 -> try FREE API (text.pollinations.ai, no auth)
   246→      4. Depleted keys auto-retry after 600 seconds cooldown
   247→      5. Free API always available but rate-limited and slower
   248→
   249→    IMPORTANT: Models are NEVER removed when unavailable.
   250→    Pollinations.ai rotates model availability — a failure today
   251→    doesn't mean the model is gone. Circuit breaking (5-min cooldown)
   252→    handles temporary failures.
   253→    """
   254→
   255→    def __init__(self):
   256→        super().__init__(
   257→            name="pollinations",
   258→            api_key=config.POLLINATIONS_API_KEY,
   259→            base_url=config.POLLINATIONS_BASE_URL,
   260→        )
   261→        # ── Dual API key storage ──
   262→        # KEY1 = primary key from config
   263→        self._api_key_1: str = config.POLLINATIONS_API_KEY
   264→        self._api_key_2: str = config.POLLINATIONS_API_KEY_2
   265→        # ── Per-key balance depletion tracking ──
   266→        # 0 = active/never depleted; >0 = timestamp when depleted
   267→        self._key1_depleted_at: float = 0.0
   268→        self._key2_depleted_at: float = 0.0
   269→
   270→        # ── Free API endpoints (no auth required) ──
   271→        self._free_text_url: str = config.POLLINATIONS_FREE_TEXT_URL
   272→        self._free_image_url: str = config.POLLINATIONS_FREE_IMAGE_URL
   273→        self._free_api_available: bool = True  # Assume available until proven otherwise
   274→        self._free_api_cooldown_until: float = 0.0  # Timestamp when free API cooldown ends
   275→
   276→    # ── API Key Management ──────────────────────────────────────
   277→
   278→    def _is_key_available(self, key_index: int) -> bool:
   279→        """Check if an API key is available (not depleted or cooldown expired)."""
   280→        if key_index == 1:
   281→            depleted_at = self._key1_depleted_at
   282→            key_val = self._api_key_1
   283→        elif key_index == 2:
   284→            depleted_at = self._key2_depleted_at
   285→            key_val = self._api_key_2
   286→        else:
   287→            return False  # Only KEY1 and KEY2 supported
   288→
   289→        if not key_val:
   290→            return False
   291→
   292→        if depleted_at == 0:
   293→            return True
   294→
   295→        elapsed = time.time() - depleted_at
   296→        if elapsed >= KEY_COOLDOWN:
   297→            if key_index == 1:
   298→                self._key1_depleted_at = 0
   299→            else:
   300→                self._key2_depleted_at = 0
   301→            logger.info(f"API KEY{key_index} cooldown expired after {elapsed:.0f}s — retrying")
   302→            return True
   303→
   304→        return False
   305→
   306→    def _mark_key_depleted(self, key_index: int) -> None:
   307→        """Mark an API key as depleted (balance exhausted)."""
   308→        if key_index == 1:
   309→            self._key1_depleted_at = time.time()
   310→        else:
   311→            self._key2_depleted_at = time.time()
   312→        logger.warning(
   313→            f"API KEY{key_index} depleted (402/401). "
   314→            f"Will auto-retry after {KEY_COOLDOWN}s cooldown."
   315→        )
   316→
   317→    def _get_active_key_tier(self) -> Tuple[str, int]:
   318→        """Determine which key/tier to use for the next request."""
   319→        if self._is_key_available(1):
   320→            return self._api_key_1, 1
   321→        if self._is_key_available(2):
   322→            return self._api_key_2, 2
   323→        return "", 0
   324→
   325→    def _get_key_status_summary(self) -> str:
   326→        """Get a human-readable summary of key statuses."""
   327→        parts = []
   328→        for idx, (key_val, depleted_at) in enumerate([
   329→            (self._api_key_1, self._key1_depleted_at),
   330→            (self._api_key_2, self._key2_depleted_at),
   331→        ], start=1):
   332→            if not key_val:
   333→                parts.append(f"KEY{idx}=not_set")
   334→            elif depleted_at == 0:
   335→                parts.append(f"KEY{idx}=active")
   336→            else:
   337→                remaining = KEY_COOLDOWN - (time.time() - depleted_at)
   338→                parts.append(f"KEY{idx}=depleted({remaining:.0f}s)")
   339→        return ", ".join(parts)
   340→
   341→    def _build_key_tier_list(self) -> List[Tuple[str, int]]:
   342→        """Build ordered list of available key tiers to try."""
   343→        tiers: List[Tuple[str, int]] = []
   344→        if self._is_key_available(1):
   345→            tiers.append((self._api_key_1, 1))
   346→        if self._is_key_available(2):
   347→            tiers.append((self._api_key_2, 2))
   348→        return tiers
   349→
   350→    async def chat(
   351→        self,
   352→        messages: List[Dict[str, str]],
   353→        model: str = "",
   354→        temperature: float = 0.7,
   355→        max_tokens: int = 2048,
   356→        **kwargs,
   357→    ) -> AIResponse:
   358→        """Send a chat completion request to Pollinations API with DUAL-KEY FAILOVER.
   359→
   360→        Strategy:
   361→        1. Try with KEY1 first (if balance available)
   362→        2. On 402/401 from KEY1 -> switch to KEY2
   363→        3. On 402/401 from KEY2 -> return error (router decides fallback)
   364→
   365→        Empty content handling:
   366→        - If a model returns empty content, check reasoning_content field
   367→        - If both are empty, mark as temporary failure (circuit break)
   368→        - Models in EMPTY_CONTENT_MODELS get special logging
   369→        - Models are NEVER removed from lists
   370→        """
   371→        model = model or DEFAULT_MODEL
   372→
   373→        # Check if model is in cooldown from recent failures
   374→        if self._is_model_in_cooldown(model):
   375→            alt = self._get_available_model()
   376→            if alt:
   377→                logger.info(f"Model {model} in cooldown, using {alt}")
   378→                model = alt
   379→
   380→        # Build key tier list: active keys first
   381→        tiers_to_try = self._build_key_tier_list()
   382→
   383→        # If no keys available at all, check cooldown
   384→        if not tiers_to_try:
   385→            return AIResponse(
   386→                text="",
   387→                model=model,
   388→                provider=self.name,
   389→                error=True,
   390→                error_message=f"All API keys depleted [{self._get_key_status_summary()}]",
   391→            )
   392→
   393→        payload = {
   394→            "model": model,
   395→            "messages": messages,
   396→            "temperature": temperature,
   397→            "max_tokens": max_tokens,
   398→        }
   399→
   400→        if kwargs.get("top_p"):
   401→            payload["top_p"] = kwargs["top_p"]
   402→        if kwargs.get("frequency_penalty"):
   403→            payload["frequency_penalty"] = kwargs["frequency_penalty"]
   404→        if kwargs.get("presence_penalty"):
   405→            payload["presence_penalty"] = kwargs["presence_penalty"]
   406→
   407→        last_error = None
   408→        for api_key, tier_index in tiers_to_try:
   409→            try:
   410→                headers = {"Content-Type": "application/json"}
   411→                if api_key:
   412→                    headers["Authorization"] = f"Bearer {api_key}"
   413→
   414→                async with httpx.AsyncClient(timeout=90.0) as client:
   415→                    start_time = time.time()
   416→                    url = f"{self.base_url}/v1/chat/completions"
   417→                    response = await client.post(url, headers=headers, json=payload)
   418→                    elapsed = time.time() - start_time
   419→
   420→                    if response.status_code == 200:
   421→                        data = response.json()
   422→                        text = ""
   423→                        choices = data.get("choices", [])
   424→                        if choices:
   425→                            msg = choices[0].get("message", {})
   426→                            # Primary content
   427→                            text = msg.get("content", "") or ""
   428→                            # Fallback: reasoning_content (some reasoning models
   429→                            # put their output here instead of content)
   430→                            if not text:
   431→                                text = msg.get("reasoning_content", "") or ""
   432→
   433→                        usage = data.get("usage", {})
   434→                        tokens_used = usage.get("total_tokens", 0)
   435→                        actual_model = data.get("model", model)
   436→                        tier_label = f"KEY{tier_index}"
   437→
   438→                        # Handle empty content gracefully
   439→                        if not text:
   440→                            # Check if this is a known empty-content model
   441→                            if model in EMPTY_CONTENT_MODELS:
   442→                                logger.warning(
   443→                                    f"Empty response from {model} (known empty-content model) "
   444→                                    f"via {tier_label} — marking as temporary failure"
   445→                                )
   446→                            else:
   447→                                logger.warning(
   448→                                    f"Empty response from {model} via {tier_label} — "
   449→                                    f"marking as temporary failure"
   450→                                )
   451→                            # Circuit break the model, but do NOT remove it
   452→                            _model_failures[model] = time.time()
   453→                            last_error = f"Empty response from {model}"
   454→                            continue  # Try next key tier
   455→
   456→                        logger.info(
   457→                            f"Pollinations response ({tier_label}): model={actual_model}, "
   458→                            f"tokens={tokens_used}, time={elapsed:.1f}s, "
   459→                            f"length={len(text)}"
   460→                        )
   461→
   462→                        return AIResponse(
   463→                            text=text,
   464→                            model=actual_model,
   465→                            provider=self.name,
   466→                            tokens_used=tokens_used,
   467→                        )
   468→
   469→                    elif response.status_code in (401, 402):
   470→                        # Balance depleted or unauthorized → switch key
   471→                        self._mark_key_depleted(tier_index)
   472→                        last_error = f"HTTP {response.status_code} from KEY{tier_index}"
   473→                        logger.warning(
   474→                            f"HTTP {response.status_code} from {model} via KEY{tier_index} — "
   475→                            f"switching to next key tier"
   476→                        )
   477→                        continue  # Try next key tier
   478→
   479→                    elif response.status_code == 429:
   480→                        # Rate limited — short cooldown
   481→                        logger.warning(f"Rate limited (429) on KEY{tier_index}, model={model}")
   482→                        _model_failures[model] = time.time()
   483→                        # Try next key tier
   484→                        continue
   485→
   486→                    else:
   487→                        error_text = response.text[:500]
   488→                        logger.error(
   489→                            f"Pollinations error: status={response.status_code}, "
   490→                            f"model={model}, error={error_text}"
   491→                        )
   492→                        _model_failures[model] = time.time()
   493→
   494→                        return AIResponse(
   495→                            text="",
   496→                            model=model,
   497→                            provider=self.name,
   498→                            error=True,
   499→                            error_message=f"HTTP {response.status_code}: {error_text}",
   500→                        )
   501→
   502→            except httpx.TimeoutException:
   503→                logger.error(f"Pollinations timeout: model={model}, tier=KEY{tier_index}")
   504→                _model_failures[model] = time.time()
   505→                last_error = f"Timeout from KEY{tier_index}"
   506→                continue  # Try next key tier
   507→
   508→            except Exception as e:
   509→                logger.error(f"Pollinations exception: model={model}, tier=KEY{tier_index}, error={e}")
   510→                _model_failures[model] = time.time()
   511→                last_error = str(e)
   512→                continue
   513→
   514→        # ── All key tiers failed → return error, let router decide ──
   515→        key_status = self._get_key_status_summary()
   516→        logger.error(
   517→            f"All API key tiers failed for model {model}. "
   518→            f"Key status: [{key_status}]. Last error: {last_error}. "
   519→            f"Router should decide fallback."
   520→        )
   521→
   522→        return AIResponse(
   523→            text="",
   524→            model=model,
   525→            provider=self.name,
   526→            error=True,
   527→            error_message=f"All API keys depleted/unavailable [{key_status}]. {last_error}",
   528→        )

   530→    async def generate_image(self, prompt: str, model: str = "flux") -> Optional[bytes]:
   531→        """Generate an image using Pollinations image API.
   532→        Uses dual-key failover: KEY1 -> KEY2.
   533→        Returns image bytes or None on failure.
   534→        """
   535→        # Build key tier list
   536→        tiers_to_try = self._build_key_tier_list()
   537→
   538→        if not tiers_to_try:
   539→            logger.warning("No API keys available for image generation")
   540→            return None
   541→
   542→        for api_key, tier_index in tiers_to_try:
   543→            try:
   544→                url = f"{self.base_url}/v1/images/generations"
   545→                headers = {
   546→                    "Content-Type": "application/json",
   547→                    "Authorization": f"Bearer {api_key}",
   548→                }
   549→                payload = {
   550→                    "model": model,
   551→                    "prompt": prompt,
   552→                    "size": "1344x768",  # Good for Telegram channel posts
   553→                }
   554→
   555→                async with httpx.AsyncClient(timeout=90.0) as client:
   556→                    response = await client.post(url, headers=headers, json=payload)
   557→
   558→                    if response.status_code == 200:
   559→                        data = response.json()
   560→                        for item in data.get("data", []):
   561→                            if item.get("b64_json"):
   562→                                return base64.b64decode(item["b64_json"])
   563→                            elif item.get("url"):
   564→                                img_resp = await client.get(item["url"])
   565→                                if img_resp.status_code == 200:
   566→                                    return img_resp.content
   567→                    elif response.status_code in (401, 402):
   568→                        self._mark_key_depleted(tier_index)
   569→                        logger.warning(f"Image gen HTTP {response.status_code} via KEY{tier_index}, trying next tier...")
   570→                        continue
   571→                    else:
   572→                        logger.error(f"Image generation error: {response.status_code} {response.text[:300]}")
   573→                        return None
   574→
   575→            except Exception as e:
   576→                logger.error(f"Image generation exception: {e}")
   577→                continue
   578→
   579→        return None

   581→    async def analyze_image(
   582→        self,
   583→        image_url: str = "",
   584→        image_base64: str = "",
   585→        prompt: str = "Опиши подробно что ты видишь на этом изображении.",
   586→        model: str = "openai",
   587→        system_prompt: str = "",
   588→        max_tokens: int = 600,
   589→        temperature: float = 0.7,
   590→    ) -> AIResponse:
   591→        """Analyze an image using vision-capable models.
   592→        Uses dual-key failover for each vision model attempt.
   593→        """
   594→        # Build the content array for vision
   595→        content = [
   596→            {"type": "text", "text": prompt}
   597→        ]
   598→
   599→        if image_url:
   600→            content.append({
   601→                "type": "image_url",
   602→                "image_url": {"url": image_url}
   603→            })
   604→        elif image_base64:
   605→            media_type = "image/jpeg"
   606→            if image_base64.startswith("data:"):
   607→                header, image_base64 = image_base64.split(",", 1)
   608→                if "png" in header:
   609→                    media_type = "image/png"
   610→                elif "webp" in header:
   611→                    media_type = "image/webp"
   612→                elif "gif" in header:
   613→                    media_type = "image/gif"
   614→
   615→            content.append({
   616→                "type": "image_url",
   617→                "image_url": {
   618→                    "url": f"data:{media_type};base64,{image_base64}"
   619→                }
   620→            })
   621→
   622→        messages = []
   623→        if system_prompt:
   624→            messages.append({"role": "system", "content": system_prompt})
   625→
   626→        messages.append({"role": "user", "content": content})
   627→
   628→        # Try primary vision model, then fallbacks
   629→        vision_models_to_try = [model]
   630→        for fallback in VISION_MODELS:
   631→            if fallback != model and not self._is_model_in_cooldown(fallback):
   632→                vision_models_to_try.append(fallback)
   633→
   634→        for vision_model in vision_models_to_try[:5]:  # Try up to 5 models
   635→            try:
   636→                result = await self.chat(
   637→                    messages=messages,
   638→                    model=vision_model,
   639→                    temperature=temperature,
   640→                    max_tokens=max_tokens,
   641→                )
   642→                if not result.error and result.text:
   643→                    return result
   644→                logger.warning(f"Vision model {vision_model} failed, trying next...")
   645→            except Exception as e:
   646→                logger.error(f"Vision error with {vision_model}: {e}")
   647→                continue
   648→
   649→        return AIResponse(
   650→            text="",
   651→            model=model,
   652→            provider=self.name,
   653→            error=True,
   654→            error_message="All vision models failed",
   655→        )

   688→    async def chat_free(
   689→        self,
   690→        messages: List[Dict[str, str]],
   691→        model: str = "openai",
   692→        temperature: float = 0.7,
   693→        max_tokens: int = 2048,
   694→    ) -> AIResponse:
   695→        """Send a chat completion request using FREE Pollinations API.
   696→
   697→        Uses text.pollinations.ai WITHOUT Authorization header.
   698→        This is the fallback when all API keys are depleted.
   699→        Free API may be rate-limited and slower.
   700→        """
   701→        if not self._is_free_api_available():
   702→            return AIResponse(
   703→                text="",
   704→                model=model,
   705→                provider=f"{self.name}-free",
   706→                error=True,
   707→                error_message="Free API unavailable or on cooldown",
   708→            )
   709→
   710→        model = model or DEFAULT_MODEL
   711→
   712→        payload = {
   713→            "model": model,
   714→            "messages": messages,
   715→            "temperature": temperature,
   716→            "max_tokens": max_tokens,
   717→        }
   718→
   719→        try:
   720→            headers = {"Content-Type": "application/json"}
   721→            # NO Authorization header — this is the free API
   722→
   723→            async with httpx.AsyncClient(timeout=90.0) as client:
   724→                start_time = time.time()
   725→                url = f"{self._free_text_url}/openai/chat/completions"
   726→                response = await client.post(url, headers=headers, json=payload)
   727→                elapsed = time.time() - start_time
   728→
   729→                if response.status_code == 200:
   730→                    data = response.json()
   731→                    text = ""
   732→                    choices = data.get("choices", [])
   733→                    if choices:
   734→                        msg = choices[0].get("message", {})
   735→                        text = msg.get("content", "") or ""
   736→                        if not text:
   737→                            text = msg.get("reasoning_content", "") or ""
   738→
   739→                    if not text:
   740→                        logger.warning(f"Free API: empty response from {model}")
   741→                        return AIResponse(
   742→                            text="",
   743→                            model=model,
   744→                            provider=f"{self.name}-free",
   745→                            error=True,
   746→                            error_message="Empty response from free API",
   747→                        )
   748→
   749→                    logger.info(
   750→                        f"Free API response: model={model}, "
   751→                        f"time={elapsed:.1f}s, length={len(text)}"
   752→                    )
   753→
   754→                    return AIResponse(
   755→                        text=text,
   756→                        model=model,
   757→                        provider=f"{self.name}-free",
   758→                    )
   759→
   760→                elif response.status_code == 429:
   761→                    logger.warning(f"Free API rate limited (429)")
   762→                    self._mark_free_api_cooldown(120.0)  # 2 min cooldown
   763→                    return AIResponse(
   764→                        text="",
   765→                        model=model,
   766→                        provider=f"{self.name}-free",
   767→                        error=True,
   768→                        error_message="Free API rate limited",
   769→                    )
   770→
   771→                else:
   772→                    error_text = response.text[:300]
   773→                    logger.error(f"Free API error: {response.status_code}: {error_text}")
   774→                    self._mark_free_api_cooldown(60.0)
   775→                    return AIResponse(
   776→                        text="",
   777→                        model=model,
   778→                        provider=f"{self.name}-free",
   779→                        error=True,
   780→                        error_message=f"Free API HTTP {response.status_code}: {error_text}",
   781→                    )
   782→
   783→        except httpx.TimeoutException:
   784→            logger.error(f"Free API timeout: model={model}")
   785→            self._mark_free_api_cooldown(30.0)
   786→            return AIResponse(
   787→                text="",
   788→                model=model,
   789→                provider=f"{self.name}-free",
   790→                error=True,
   791→                error_message="Free API timeout",
   792→            )
   793→
   794→        except Exception as e:
   795→            logger.error(f"Free API exception: {e}")
   796→            self._mark_free_api_cooldown(60.0)
   797→            return AIResponse(
   798→                text="",
   799→                model=model,
   800→                provider=f"{self.name}-free",
   801→                error=True,
   802→                error_message=f"Free API exception: {e}",
   803→            )

   907→    def _is_model_in_cooldown(self, model: str) -> bool:
   908→        """Check if a model has recently failed and is in cooldown.
   909→
   910→        IMPORTANT: We do NOT remove models from lists when they fail.
   911→        Pollinations.ai rotates model availability — a failure today
   912→        doesn't mean the model is permanently unavailable.
   913→        """
   914→        if model not in _model_failures:
   915→            return False
   916→        return time.time() - _model_failures[model] < _FAILURE_COOLDOWN

   918→    def _get_available_model(self) -> Optional[str]:
   919→        """Get an available model that's not in cooldown."""
   920→        if not self._is_model_in_cooldown(DEFAULT_MODEL):
   921→            return DEFAULT_MODEL
   922→        for model in FALLBACK_MODELS:
   923→            if not self._is_model_in_cooldown(model):
   924→                return model
   925→        if _model_failures:
   926→            oldest = min(_model_failures, key=_model_failures.get)
   927→            return oldest
   928→        return DEFAULT_MODEL
