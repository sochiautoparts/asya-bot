    91→def _is_semantically_duplicate(title: str) -> bool:
    92→    """Check if 3+ significant words from title match a recently posted title.
    93→    
    94→    Uses a TWO-LEVEL check:
    95→    - Level 1: 3+ significant words overlap -> DUPLICATE
    96→    - Level 2: 2+ CORE words (brand + model/event) overlap -> DUPLICATE
    97→    
    98→    This catches both obvious and subtle duplicates like:
    99→    - "BMW X5 получил новый двигатель" vs "Новый мотор для BMW X5"
   100→    - "Tesla отзывает 10000 машин" vs "Tesla начала отзывную кампанию"
   101→    """
   102→    global _recent_post_keywords
   103→
   104→    # Extract significant words from the new title
   105→    words = re.findall(r'[a-zа-яё]{3,}', title.lower())
   106→    significant = [w for w in words if w not in _SEMANTIC_STOP_WORDS]
   107→
   108→    if len(significant) < 2:
   109→        return False
   110→
   111→    # Extract core words (brands, models, events) for Level 2 check
   112→    core_words = _extract_core_words_from_title(title)
   113→
   114→    for recent_words in _recent_post_keywords:
   115→        # Level 1: 3+ significant words overlap
   116→        matches = sum(1 for w in significant if w in recent_words)
   117→        if matches >= 3:
   118→            return True
   119→        
   120→        # Level 2: 2+ core words overlap (brand + model/event)
   121→        if len(core_words) >= 2:
   122→            recent_core = [w for w in recent_words if w in _ALL_CORE_WORDS_SET]
   123→            core_matches = sum(1 for w in core_words if w in recent_core)
   124→            if core_matches >= 2:
   125→                return True
   126→
   127→    return False

   216→def _clean_post_text(text: str) -> str:
   217→    """Clean post text: remove markdown links, formatting artifacts, SSE garbage,
   218→    AI meta-comments about duplicates, and other content that should not appear in posts."""
   219→    if not text:
   220→        return text
   221→
   222→    # Remove markdown links [text](url) → text (Telegram doesn't support them properly)
   223→    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'\1', text)
   224→
   225→    # Remove bold/italic
   226→    text = re.sub(r'\*\*(.+?)\*\*', r'\1', text)
   227→    text = re.sub(r'\*(.+?)\*', r'\1', text)
   228→
   229→    # Remove SSE/streaming artifacts
   230→    text = re.sub(r'data:\s*\{[^}]*\}', '', text)
   231→    text = re.sub(r'\[DONE\]', '', text)
   232→
   233→    # Remove AI disclaimers
   234→    for phrase in ["As an AI", "Как AI", "Как искусственный интеллект",
   235→                   "powered by pollinations", "pollinations.ai"]:
   236→        text = re.sub(rf'.*{re.escape(phrase)}.*', '', text, flags=re.IGNORECASE)
   237→
   238→    # Remove think tags
   239→    text = re.sub(r'<think\b[^>]*>.*?</think\s*>', '', text, flags=re.DOTALL | re.IGNORECASE)
   240→    text = re.sub(r'</?think[^>]*>', '', text, flags=re.IGNORECASE)
   241→
   242→    # ── Remove AI meta-comments / editorial notes that leaked into post text ──
   243→    # Kept to ~15 essential patterns — removed common Russian words that match valid content
   244→    meta_comment_patterns = [
   245→        # Editorial leakage: "тему в канал не ставим", "не наш формат"
   246→        r'[^\n]*тему\s+в\s+канал\s+не\s+ставим[^\n]*',
   247→        r'[^\n]*в\s+канал\s+не\s+ставим[^\n]*',
   248→        r'[^\n]*не\s+наш\s+формат[^\n]*',
   249→        r'[^\n]*перепишу\s+тему[^\n]*',
   250→        r'[^\n]*напишу\s+готовый\s+пост[^\n]*',
   251→        r'[^\n]*я\s+сразу\s+предложу[^\n]*',
   252→        r'[^\n]*без\s+лишних\s+вопросов[^\n]*',
   253→        r'[^\n]*не\s+соответствует\s+(тематик|формат)[^\n]*',
   254→        r'[^\n]*по\s+вашим\s+(же\s+)?правилам[^\n]*',
   255→        # Duplicate/republish meta-comments
   256→        r'[^\n]*дубликат[^\n]*',
   257→        r'[^\n]*ЭТО\s+УЖЕ\s+ОПУБЛИКОВАНО[^\n]*',
   258→        # English variants
   259→        r'[^\n]*already\s+(posted|published|covered|wrote)[^\n]*',
   260→        r'[^\n]*do\s+not\s+(publish|post)[^\n]*',
   261→        # Generic catch-all for editorial notes in brackets
   262→        r'\[[^\]]*(?:не\s+публиков|не\s+став|не\s+наш|редакц|пропуск)[^\]]*\]',
   263→    ]
   264→    for pattern in meta_comment_patterns:
   265→        text = re.sub(pattern, '', text, flags=re.IGNORECASE)
   266→
   267→    # Remove AI/assistant name prefixes that leak into posts
   268→    for prefix in ["Ася:", "Asya:", "Assistant:"]:
   269→        if text.startswith(prefix):
   270→            text = text[len(prefix):].strip()
   271→
   272→    # Remove formal "Редакция сообщает" phrases — replace with nothing or informal version
   273→    formal_phrases = [
   274→        ("Редакция сообщает:", ""),
   275→        ("Редакция сообщает —", ""),
   276→        ("Редакция сообщает", ""),
   277→        ("Как стало известно редакции,", ""),
   278→        ("Как стало известно редакции", ""),
   279→        ("По данным нашей редакции,", ""),
   280→        ("По данным нашей редакции", ""),
   281→        ("Редакция @sochiautoparts сообщает:", ""),
   282→        ("Редакция @sochiautoparts сообщает", ""),
   283→    ]
   284→    for phrase, replacement in formal_phrases:
   285→        if phrase in text:
   286→            text = text.replace(phrase, replacement)
   287→
   288→    # ── Meta-line removal: remove lines that look like editorial notes ──
   289→    # Reduced to 10 core triggers that NEVER appear in valid automotive posts
   290→    _editorial_trigger_phrases = [
   291→        "не ставим", "не наш формат",
   292→        "перепишу тему", "напишу готовый",
   293→        "я сразу предложу", "без лишних вопросов",
   294→        "не для публикации", "внутренняя заметка",
   295→        "для редакции", "редакционная",
   296→    ]
   297→    lines = text.split('\n')
   298→    cleaned_lines = []
   299→    for line in lines:
   300→        line_lower = line.lower().strip()
   301→        is_editorial = False
   302→        for trigger in _editorial_trigger_phrases:
   303→            if trigger in line_lower:
   304→                is_editorial = True
   305→                break
   306→        # Also catch lines that are purely editorial in brackets [...]
   307→        if re.match(r'^\s*\[.*\]\s*$', line) and any(
   308→            kw in line.lower() for kw in ["не", "редакц", "пропуск", "не наш", "формат"]
   309→        ):
   310→            is_editorial = True
   311→        if not is_editorial:
   312→            cleaned_lines.append(line)
   313→    text = '\n'.join(cleaned_lines)
   314→
   315→    # Clean up whitespace
   316→    text = re.sub(r'\n{3,}', '\n\n', text)
   317→    text = text.strip()
   318→
   319→    return text

   322→def _validate_post_text(text: str) -> bool:
   323→    """Validate post text before sending to channel.
   324→    
   325→    Checks for:
   326→    - Empty text
   327→    - SSE/API artifacts
   328→    - AI meta-comments about duplicates ("already posted", "do not publish")
   329→    - Blocked topics (politics, war, Putin, etc.)
   330→    - Auto-relevance (must contain automotive keywords)
   331→    - Provider ad artifacts
   332→    - Raw JSON
   333→    - Duplicate indicators that leaked through cleaning
   334→    """
   335→    if not text or not text.strip():
   336→        return False
   337→
   338→    text_lower = text.lower()
   339→
   340→    # Block SSE artifacts
   341→    sse_patterns = [r'data:\s*\{', r'\[DONE\]', r'"type"\s*:\s*"start"', r'"type"\s*:\s*"error"']
   342→    for pattern in sse_patterns:
   343→        if re.search(pattern, text_lower):
   344→            return False
   345→
   346→    # Block API errors
   347→    error_patterns = ["authentication error", "no api key", "model not found",
   348→                      "rate limit", "internal server error", "bad request"]
   349→    for pattern in error_patterns:
   350→        if pattern in text_lower:
   351→            return False
   352→
   353→    # Block provider ad artifacts
   354→    ad_patterns = ["pollinations.ai", "powered by pollinations", "keep ai accessible"]
   355→    for pattern in ad_patterns:
   356→        if pattern in text_lower:
   357→            return False
   358→
   359→    # Block raw JSON
   360→    if text.strip().startswith(('{', '[', '```', 'data:')):
   361→        return False
   362→
   363→    # ── Block AI editorial notes that leaked through cleaning ──
   364→    # Reduced to ~20 truly editorial-specific phrases that NEVER appear in valid automotive posts.
   365→    # Removed common Russian words like "уже", "не стоит", "в канал", "отсеивать" etc.

   391→    #   - "выборы " (with space) instead of "выбор" which blocked partner posts about tire choices
   392→    #   - Removed "flood", "series", "championship", "election", "exam", "movie",
   393→    #     "drug", "theft", "school" — these are handled in news.py with compound phrases
   394→    blocked_keywords = [
   395→        "путин", "кремль", "госдума", "президент росс", "президент сша",
   396→        "сво ", "специальная военная", "мобилизац", "санкци",
   397→        "военные действ", "вооруженн", "министр оборон",

   400→        "политик", "депутат", "законопроект", "выборы ", "голосован",
   401→        # ── NSFW / Adult content — ABSOLUTE BLOCK ──
   402→        "порн", "секс", "эрот", "голая", "голые", "обнажён", "обнажен",
   403→        "интим", "проститут", "путан", "бордель",

   406→        "стриптиз", "камасутр", "ню фото",
   407→        "порно-", "секс-", "18+", "xxx",

   409→        "nude", "porn", "nsfw", "hentai", "milf",
   410→    ]
   411→    # Block boring Russian auto brands — 50 years nothing interesting
   412→    blocked_auto_brands = [
   413→        "автоваз", "лада", "lada", "уаз", "uaz", "камаз", "kamaz",
   414→        "соллерс", "vesta", "granta", "niva", "искра", "iskra",

   419→            return False

   480→    
   481→    Partner posts are about auto services/parts (Rossko, Autopiter, tire shops, etc.)

   529→def _enforce_char_limit(text: str, has_media: bool) -> str:
   530→    """Smart character limit enforcement — always preserves footer.
   531→    
   532→    Telegram limits:
   533→    - 1024 chars with media (caption)
   534→    - 4096 chars without media (text-only)
   535→    
   536→    This function:
   537→    1. Separates footer from content
   538→    2. Truncates content at sentence/paragraph boundary if needed
   539→    3. Re-attaches footer (always intact)
   540→    4. NEVER cuts mid-word or mid-sentence — always truncates at a natural break point
   541→    """
   542→    footer = "\n\nАвтор @asiaexp_bot\n@sochiautoparts\n#sochiautoparts"
   543→    char_limit = config.TELEGRAM_CAPTION_LIMIT if has_media else config.TELEGRAM_TEXT_LIMIT
   544→    
   545→    if len(text) <= char_limit:
   546→        return text
   547→    
   548→    # Strip existing footer parts to get pure content
   549→    content = text
   550→    for foot_part in ["\n\nАвтор @asiaexp_bot", "\n@sochiautoparts", "\n#sochiautoparts"]:
   551→        content = content.replace(foot_part, "")
   552→    content = content.rstrip()
   553→    
   554→    # Calculate max content length (leave room for footer)
   555→    max_content = char_limit - len(footer)
   556→    if max_content < 100:
   557→        # Absolute minimum — just footer
   558→        return footer.lstrip('\n')
   559→    
   560→    if len(content) > max_content:
   561→        content = _smart_truncate(content, max_content)
   562→    
   563→    return content + footer
   564→

   566→def _smart_truncate(text: str, max_len: int) -> str:
   567→    """Truncate text at a natural sentence/paragraph boundary.
   568→    
   569→    Strategy (in priority order):
   570→    1. Find the last paragraph break (\n\n) before max_len
   571→    2. Find the last sentence end (. ! ? …) before max_len
   572→    3. Find the last newline (\n) before max_len
   573→    4. Find the last space before max_len (avoid mid-word cut)
   574→    5. Last resort: hard cut at max_len - 3 + "..."
   575→    
   576→    Always appends "..." to indicate truncation.
   577→    """
   578→    if len(text) <= max_len:
   579→        return text
   580→    
   581→    # We need room for "..." (3 chars)
   582→    target = max_len - 3
   583→    if target < 50:
   584→        return text[:target] + "..."
   585→    
   586→    # Look at the text up to target+50 chars — we want to find the BEST
   587→    # break point near the end, not just the very last one
   588→    search_zone = text[:target + 1]
   589→    
   590→    # 1. Try paragraph break (\n\n) — best break point
   591→    last_para = search_zone.rfind("\n\n")
   592→    if last_para > target * 0.5:  # Don't throw away more than half the text
   593→        return text[:last_para].rstrip() + "..."
   594→    
   595→    # 2. Try sentence end (. ! ? … followed by space or newline)
   596→    # Look for sentence endings in the last portion of the text
   597→    sentence_end_chars = ['. ', '! ', '? ', '… ', '.\n', '!\n', '?\n', '…\n']
   598→    best_sentence_end = -1
   599→    for end_char in sentence_end_chars:
   600→        pos = search_zone.rfind(end_char)
   601→        if pos > best_sentence_end and pos > target * 0.5:
   602→            best_sentence_end = pos + len(end_char) - 1  # Include the punctuation
   603→    
   604→    if best_sentence_end > target * 0.5:
   605→        return text[:best_sentence_end + 1].rstrip() + "..."
   606→    
   607→    # 3. Try newline (\n)
   608→    last_newline = search_zone.rfind("\n")
   609→    if last_newline > target * 0.5:
   610→        return text[:last_newline].rstrip() + "..."
   611→    
   612→    # 4. Try space (avoid mid-word cut)
   613→    last_space = search_zone.rfind(" ")
   614→    if last_space > target * 0.5:
   615→        return text[:last_space].rstrip() + "..."
   616→    
   617→    # 5. Hard cut — very last resort
   618→    return text[:target].rstrip() + "..."

   693→                    import base64
   694→                    return base64.b64decode(response.image_b64)
   695→                if response.image_url:

   702→                        pass
   703→                return None
   704→        except ImportError:
   705→            pass
   706→        # Fallback: if it has image_b64 attribute
   707→        if hasattr(response, 'image_b64') and response.image_b64:
   708→            import base64
   709→            return base64.b64decode(response.image_b64)
   710→        return None

   728→

   769→    async def _get_post_images(self, news_item: Dict) -> tuple:
   770→        """Get images for a news post — download directly from JSON URLs.
   771→
   772→        v6.0: Images come pre-extracted from the external parser (news.json).
   773→        No RSS scraping, no article page crawling, no search engines.
   774→        Just download the image URLs from the JSON data.
   775→
   776→        Returns (image_list: List[bytes], source: str)
   777→        source is 'json', 'cache', or 'none'.
   778→        """
   779→        title = news_item.get("title", "")
   780→        image_urls = news_item.get("image_urls", [])
   781→
   782→        if not image_urls:
   783→            logger.info(f"No image URLs in JSON for '{title[:50]}' — text-only post")
   784→            return [], "none"
   785→
   786→        images = []
   787→        downloaded = 0
   788→        failed = 0
   789→
   790→        async with httpx.AsyncClient(timeout=15.0, follow_redirects=True) as client:
   791→            for url in image_urls[:MAX_IMAGES_PER_POST]:
   792→                try:
   793→                    # Skip obviously bad URLs
   794→                    url_lower = url.lower()
   795→                    bad_patterns = ['logo', 'icon', 'banner', 'ad.', 'pixel',
   796→                                    'tracking', '1x1', 'spacer', 'favicon',
   797→                                    'avatar', 'badge', 'newsletter', 'subscribe']
   798→                    if any(bp in url_lower for bp in bad_patterns):
   799→                        continue
   800→
   801→                    resp = await client.get(url)
   802→                    if resp.status_code != 200:
   803→                        failed += 1
   804→                        continue
   805→
   806→                    content_type = resp.headers.get("content-type", "")
   807→                    if not any(ft in content_type for ft in ["image/jpeg", "image/png", "image/webp", "image/jpg"]):
   808→                        # Check magic bytes if content-type is missing
   809→                        if not (resp.content[:3] == b'\xff\xd8\xff' or  # JPEG
   810→                                resp.content[:4] == b'\x89PNG' or        # PNG
   811→                                (resp.content[:4] == b'RIFF' and resp.content[8:12] == b'WEBP')):
   812→                            failed += 1
   813→                            continue
   814→
   815→                    # Size check — skip tiny (icons) and huge (might OOM) images
   816→                    size_kb = len(resp.content) / 1024
   817→                    if size_kb < 5:  # Too small — probably icon/tracking
   818→                        continue
   819→                    if size_kb > 5000:  # Too large — skip
   820→                        continue
   821→
   822→                    images.append(resp.content)
   823→                    downloaded += 1
   824→
   825→                except (httpx.TimeoutException, httpx.HTTPError) as e:
   826→                    failed += 1
   827→                    logger.debug(f"Image download failed for {url[:60]}: {e}")
   828→                    continue
   829→                except Exception as e:
   830→                    failed += 1
   831→                    logger.debug(f"Image error for {url[:60]}: {e}")
   832→                    continue
   833→
   834→        if images:
   835→            logger.info(f"Downloaded {downloaded}/{len(image_urls)} images for '{title[:50]}' ({failed} failed)")
   836→            return images, "json"
   837→        else:
   838→            logger.info(f"No valid images downloaded for '{title[:50]}' ({failed} failed) — text-only post")
   839→            return [], "none"

   841→    async def _download_partner_image(self, image_url: str) -> Optional[bytes]:
   842→        """Download a partner program image (logo/banner).
   843→        
   844→        Uses RELAXED validation for partner images — they're often smaller logos/banners
   845→        (5KB+), which is fine for partner posts. News image validation is stricter.
   846→        Handles SVG images by converting them to PNG using cairosvg.
   847→        """
   848→        if not image_url:
   849→            return None
   850→        try:
   851→            async with httpx.AsyncClient(timeout=20.0, follow_redirects=True) as client:
   852→                response = await client.get(image_url)
   853→                if response.status_code != 200:
   854→                    return None
   855→                content = response.content
   856→                content_type = response.headers.get("content-type", "")
   857→
   858→                # ── Handle SVG images — convert to PNG for Telegram ──
   859→                is_svg = (
   860→                    "svg" in content_type.lower() or
   861→                    b'<svg' in content[:1000] or
   862→                    image_url.lower().endswith('.svg')
   863→                )
   864→                if is_svg:
   865→                    try:
   866→                        import cairosvg
   867→                        import io
   868→                        # Convert SVG → PNG at 512px width (good for Telegram)
   869→                        png_data = cairosvg.svg2png(bytestring=content, output_width=512)
   870→                        if png_data and len(png_data) > 1000:
   871→                            logger.info(f"Converted partner SVG → PNG ({len(png_data)} bytes): {image_url[:60]}")
   872→                            content = png_data
   873→                            content_type = "image/png"
   874→                        else:
   875→                            logger.debug(f"SVG conversion produced tiny output, skipping")
   876→                            return None
   877→                    except ImportError:
   878→                        logger.debug("cairosvg not available, cannot convert SVG partner image")
   879→                        return None
   880→                    except Exception as e:
   881→                        logger.debug(f"SVG → PNG conversion failed: {e}")
   882→                        return None
   883→
   884→                # Relaxed size check for partner images — logos can be 5KB+
   885→                # (But after SVG→PNG conversion, image is larger)
   886→                if len(content) < 2000:
   887→                    logger.debug(f"Skipping tiny partner image ({len(content)} bytes)")
   888→                    return None
   889→
   890→                # Must be an image type
   891→                if not any(ft in content_type for ft in ["image/png", "image/jpeg", "image/gif", "image/webp"]):
   892→                    # Check magic bytes
   893→                    if not (content[:3] == b'\xff\xd8\xff' or content[:4] == b'\x89PNG' or
   894→                            (content[:4] == b'RIFF' and content[8:12] == b'WEBP') or
   895→                            content[:6] in (b'GIF87a', b'GIF89a')):
   896→                        logger.debug(f"Skipping partner image: not an image, content-type={content_type}")
   897→                        return None
   898→
   899→                # Relaxed dimension check for partner images — logos/banners are OK
   900→                try:
   901→                    from PIL import Image
   902→                    import io
   903→                    img = Image.open(io.BytesIO(content))

  1811→    async def run_scheduled_post(self) -> bool:
  1812→        """
  1813→        Run a scheduled post — tries up to 3 different news items per cycle.
  1814→        
  1815→        SIMPLIFIED PIPELINE:
  1816→        1. Try partner content (20% chance if interval met)
  1817→        2. Try news — up to 3 attempts with DIFFERENT items
  1818→        3. Fallback: AI-generated "fun fact" if no web news works
  1819→        
  1820→        Each cycle is called 3 TIMES per 30-min interval (3 different posts).
  1821→        Dedup ensures each post is always different from the others.
  1822→        """
  1823→        logger.info("run_scheduled_post: called")
  1824→        now = time.time()
  1825→        partner_interval = config.PARTNER_POST_INTERVAL_HOURS * 3600
  1826→
  1827→        # Try partner content with 20% probability (if interval met)
  1828→        # Lower probability so news posts dominate — partner posts are supplementary
  1829→        if (now - self._last_partner_time >= partner_interval and
  1830→                partner_manager.should_post_partner()):
  1831→            if random.random() < 0.2:
  1832→                result = await self.post_partner_content()
  1833→                if result:
  1834→                    return True
  1835→                # Partner post failed/skipped — fall through to news
  1836→        
  1837→        # Primary: post NEWS — try up to 3 DIFFERENT items per cycle
  1838→        for attempt in range(3):
  1839→            result = await self.post_news()
  1840→            if result:
  1841→                return True
  1842→            logger.info(f"run_scheduled_post: attempt {attempt + 1}/3 failed, trying different item")
