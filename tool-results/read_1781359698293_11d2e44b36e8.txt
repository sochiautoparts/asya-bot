    25→def _compute_fingerprint(title: str) -> str:
    26→    """Compute a simplified fingerprint for dedup.
    27→    Remove articles, lowercase, take first 5 words.
    28→    """
    29→    import re
    30→    # Remove common articles/prepositions
    31→    cleaned = re.sub(r'\b(в|на|с|о|у|по|из|за|от|до|к|не|и|но|а|что|как|это|тот|этот|для|при|через|между|после|перед|без|под|над|об|со|из-за|то|же|ли|бы|уже|ещё|еще|также|тоже|или|либо)\b', '', title.lower())
    32→    cleaned = re.sub(r'[^a-zа-яё0-9]', ' ', cleaned)
    33→    words = cleaned.split()
    34→    return ' '.join(words[:5])
    35→
    36→
    37→def _fingerprint_matches_existing(fingerprint: str) -> bool:
    38→    """Check if a fingerprint matches any recently used fingerprint (edit distance on first 4 words)."""
    39→    fp_words = fingerprint.split()[:4]
    40→    for existing in _recent_fingerprints:
    41→        ex_words = existing.split()[:4]
    42→        # Count matching words
    43→        matches = sum(1 for w in fp_words if w in ex_words)
    44→        if matches >= 3:
    45→            return True
    46→    return False

   274→# ── RSS Fetcher ────────────────────────────────────────────────────────────────
   275→
   276→async def fetch_rss(source: NewsSource) -> List[Dict]:
   277→    """
   278→    Fetch and parse an RSS feed.
   279→    Returns list of news items.
   280→    """
   281→    items = []
   282→    try:
   283→        async with httpx.AsyncClient(
   284→            timeout=20.0,
   285→            follow_redirects=True,
   286→            headers={
   287→                "User-Agent": "Mozilla/5.0 (compatible; AsyaBot/1.0; +https://t.me/asiaexp_bot)",
   288→                "Accept": "application/rss+xml, application/xml, text/xml, application/atom+xml, */*",
   289→            },
   290→        ) as client:
   291→            response = await client.get(source.url)
   292→
   293→            if response.status_code != 200:
   294→                logger.warning(f"RSS {source.name} returned {response.status_code}")
   295→                return items
   296→
   297→            # feedparser can parse from string
   298→            feed = feedparser.parse(response.text)
   299→
   300→            for entry in feed.entries[:20]:  # Limit per source
   301→                title = entry.get("title", "").strip()
   302→                url = entry.get("link", "").strip()
   303→
   304→                if not title or not url:
   305→                    continue
   306→
   307→                # Get summary/description
   308→                summary = ""
   309→                if entry.get("summary"):
   310→                    summary = _clean_html(entry["summary"])
   311→                elif entry.get("description"):
   312→                    summary = _clean_html(entry["description"])
   313→
   314→                # Truncate summary
   315→                if len(summary) > 500:
   316→                    summary = summary[:500] + "..."
   317→
   318→                # Get publish date
   319→                published = 0.0
   320→                if entry.get("published_parsed"):
   321→                    try:
   322→                        published = time.mktime(entry["published_parsed"])
   323→                    except (ValueError, TypeError, OverflowError):
   324→                        published = time.time()
   325→                elif entry.get("updated_parsed"):
   326→                    try:
   327→                        published = time.mktime(entry["updated_parsed"])
   328→                    except (ValueError, TypeError, OverflowError):
   329→                        published = time.time()
   330→                else:
   331→                    published = time.time()
   332→
   333→                # Extract image URLs from RSS entry
   334→                image_urls = _extract_entry_images(entry)
   335→
   336→                # Also extract full content text from RSS entry if available
   337→                # (many feeds include full article HTML in content field)
   338→                full_text = ""
   339→                content_value = entry.get("content") or entry.get("summary") or entry.get("description")
   340→                if content_value:
   341→                    if isinstance(content_value, list):
   342→                        full_text = _clean_html(content_value[0].get("value", "")) if content_value else ""
   343→                    elif isinstance(content_value, str):
   344→                        full_text = _clean_html(content_value)
   345→                    # Truncate to reasonable length
   346→                    if len(full_text) > 2000:
   347→                        full_text = full_text[:2000] + "..."
   348→
   349→                items.append({
   350→                    "source": source.name,
   351→                    "title": title,
   352→                    "url": url,
   353→                    "summary": summary,
   354→                    "full_text": full_text if len(full_text) > len(summary) else "",
   355→                    "published": published,
   356→                    "category": source.category,
   357→                    "lang": source.lang,
   358→                    "image_urls": image_urls,
   359→                })
   360→
   361→            logger.info(f"Fetched {len(items)} items from {source.name}")
   362→
   363→    except httpx.TimeoutException:
   364→        logger.warning(f"RSS timeout: {source.name}")
   365→    except Exception as e:
   366→        logger.error(f"RSS fetch error for {source.name}: {e}")
   367→
   368→    return items

   376→

   812→def _normalize_image_url(url: str) -> str:
   813→    """Normalize an image URL: decode HTML entities, fix scheme, clean up."""
   814→    if not url:
   815→        return ""
   816→    url = url.strip()
   817→    # Decode HTML entities: &amp; → &, &#038; → &, &lt; → <, etc.
   818→    url = _html_unescape(url)
   819→    # Fix scheme-relative URLs
   820→    if url.startswith("//"):
   821→        url = "https:" + url
   822→    return url

   825→def _is_junk_image_url(url: str) -> bool:
   826→    """Check if an image URL is junk: tracker, icon, tiny thumbnail, default image."""
   827→    from urllib.parse import urlparse, parse_qs
   828→    try:
   829→        url_lower = url.lower()
   830→        parsed = urlparse(url)
   831→        domain = parsed.netloc.lower()
   832→        path_lower = parsed.path.lower()
   833→
   834→        # Skip tracker/analytics domains
   835→        if any(skip in domain for skip in _SKIP_IMAGE_DOMAINS):
   836→            return True
   837→
   838→        # Skip icon/logo/badge/button/social/author/junk filenames in path
   839→        # v7.0: Expanded list to catch more junk image patterns
   840→        if any(kw in path_lower for kw in [
   841→            "icon", "avatar", "logo", "favicon", "badge", "button", "banner",
   842→            "gravatar", "userpic", "headshot", "mugshot", "portrait",
   843→            "author-photo", "author-img", "author-image", "profile-photo", "profile-img",
   844→            "share", "social", "facebook", "twitter", "telegram", "whatsapp",
   845→            "sidebar", "widget", "related-post", "related-article", "recommended",
   846→            "newsletter", "subscribe", "popup", "screenshot", "screen-shot",
   847→            "chart", "infographic", "diagram", "emoji", "sticker", "clipart",
   848→            "advert", "sponsor", "promo", "doubleclick", "adsense",
   849→        ]):
   850→            return True
   851→
   852→        # Skip /feed/ URLs (CarScoops bug)
   853→        if _FEED_URL_PATTERN.search(url_lower):
   854→            return True
   855→
   856→        # Skip generic/default image filenames (also prefix-match: business_card-1)
   857→        stem = path_lower.rsplit("/", 1)[-1].split(".")[0].replace("-", "_")
   858→        if stem in _JUNK_IMAGE_STEMS:
   859→            return True
   860→        for junk_stem in _JUNK_IMAGE_STEMS:
   861→            if stem.startswith(junk_stem):
   862→                return True
   863→
   864→        # Skip tiny thumbnail dimensions in URL
   865→        for pattern in _TINY_SIZE_PATTERNS:
   866→            if re.search(pattern, url_lower):
   867→                return True
   868→
   869→    except Exception:
   870→        return True  # If we can't parse it, skip it
   871→    return False

   874→def _upgrade_thumbnail_url(url: str) -> str:
   875→    """Upgrade low-quality thumbnail URLs to higher resolution versions.
   876→
   877→    Known CDN patterns:
   878→    - BBC: /240/ → /640/ in path
   879→    - Autosport/Motorsport.com: /s6/ → /s12/ in path
   880→    - Reddit: width=140 → width=640 in query
   881→    - Autocar: /car_review_image_190/ → /body-image/ (190×125 → 900×600)
   882→    - TASS: /w_320/ → /w_1080/ or remove size suffix
   883→    - Kommersant: small → large in path
   884→    - Mail.ru: preview → original
   885→    - 5Колесо: thumb → full
   886→    """
   887→    # BBC: /240/cpsprodpb/ → /640/cpsprodpb/
   888→    if "bbci.co.uk" in url or "bbc.co.uk" in url:
   889→        url = re.sub(r'/240/', '/640/', url, count=1)
   890→
   891→    # Autosport / Motorsport.com: /s6/ → /s12/
   892→    if "motorsport.com" in url or "autosport.com" in url:
   893→        url = re.sub(r'/s6/', '/s12/', url, count=1)
   894→
   895→    # Reddit preview: width=140 → width=640
   896→    if "preview.redd.it" in url or "external-preview.redd.it" in url:
   897→        url = re.sub(r'width=140', 'width=640', url, count=1)
   898→        url = re.sub(r'height=140', 'height=640', url, count=1)
   899→
   900→    # Autocar UK: /styles/car_review_image_190/ → /styles/body-image/
   901→    if "autocar.co.uk" in url:
   902→        url = url.replace('/styles/car_review_image_190/', '/styles/body-image/')
   903→
   904→    # TASS: upgrade small thumbnails to full-size
   905→    if "tass.ru" in url:
   906→        # /w_320/ → /w_1080/ or /w_800/
   907→        url = re.sub(r'/w_\d{2,3}/', '/w_1080/', url, count=1)
   908→        # Remove size parameters in query
   909→        url = re.sub(r'[?&](width|height|size)=\d+', '', url)
   910→
   911→    # Kommersant: upgrade thumbnails
   912→    if "kommersant.ru" in url:
   913→        url = url.replace('/small/', '/large/')
   914→        url = url.replace('/thumb/', '/full/')
   915→        url = re.sub(r'/\d+x\d+/', '/1200x800/', url, count=1)
   916→
   917→    # Mail.ru auto: upgrade previews
   918→    if "mail.ru" in url or "img.imgsmail.ru" in url:
   919→        url = url.replace('/preview/', '/original/')
   920→        url = url.replace('/thumb/', '/original/')
   921→        # Remove size restriction params
   922→        url = re.sub(r'[?&]w=\d{2,3}', '', url)
   923→
   924→    # 5Колесо: upgrade thumbnails
   925→    if "5koleso.ru" in url:
   926→        url = url.replace('/thumb/', '/full/')
   927→        url = url.replace('/preview/', '/full/')
   928→
   929→    return url

   932→def _extract_entry_images(entry) -> List[str]:
   933→    """Extract image URLs from a feedparser entry.
   934→
   935→    Priority order (highest quality first):
   936→    1. media_content (Media RSS — editors pick these, usually hero images)
   937→    2. enclosures (often full-size article images)
   938→    3. media_thumbnail (small but from structured metadata)
   939→    4. links with image type
   940→    5. <img> tags in content/summary HTML (last — often thumbnails + junk)
   941→
   942→    All URLs are HTML-entity-decoded, deduplicated (normalized),
   943→    junk-filtered, and thumbnail-upgraded before returning.
   944→    Returns list of clean image URLs (up to 10).
   945→    """
   946→    images = []
   947→    seen_normalized = set()  # Dedup by normalized URL (after unescape)
   948→
   949→    def _add_image(raw_url: str):
   950→        """Add image URL if valid and not already seen (after normalization)."""
   951→        url = _normalize_image_url(raw_url)
   952→        if not url or len(url) < _MIN_IMAGE_URL_LEN:
   953→            return
   954→        if _is_junk_image_url(url):
   955→            return
   956→        # Upgrade low-quality thumbnails to bigger versions
   957→        url = _upgrade_thumbnail_url(url)
   958→        # Dedup by normalized form (lowercase for comparison)
   959→        norm_key = url.lower().rstrip("/")
   960→        if norm_key in seen_normalized:
   961→            return
   962→        seen_normalized.add(norm_key)
   963→        images.append(url)
   964→
   965→    # 1. media_content (Media RSS extension — most reliable, editor-chosen)
   966→    for media in entry.get("media_content", []) or []:
   967→        url = media.get("url", "")
   968→        media_type = media.get("type", "").lower()
   969→        medium = media.get("medium", "").lower()
   970→        if url and (medium == "image" or "image" in media_type or _has_image_ext(url)):
   971→            _add_image(url)
   972→
   973→    # 2. enclosures (often full-size article images)
   974→    for enclosure in entry.get("enclosures", []) or []:
   975→        url = enclosure.get("href", "") or enclosure.get("url", "")
   976→        enc_type = enclosure.get("type", "").lower()
   977→        if url and ("image" in enc_type or _has_image_ext(url)):
   978→            _add_image(url)
   979→
   980→    # 3. media_thumbnail (small but structured metadata)
   981→    for thumb in entry.get("media_thumbnail", []) or []:
   982→        url = thumb.get("url", "")
   983→        if url:
   984→            _add_image(url)
   985→
   986→    # 4. links with image rel or type
   987→    for link in entry.get("links", []) or []:
   988→        link_type = link.get("type", "").lower()
   989→        link_rel = link.get("rel", "").lower()
   990→        url = link.get("href", "")
   991→        if url and ("image" in link_type or link_rel == "enclosure"):
   992→            _add_image(url)
   993→
   994→    # 5. Extract <img> tags from content/summary HTML (last resort — noisy)
   995→    for field_name in ["content", "summary", "description"]:
   996→        content_value = entry.get(field_name)
   997→        if isinstance(content_value, list):
   998→            for item in content_value:
   999→                html = item.get("value", "") if isinstance(item, dict) else str(item)
  1000→                _extract_img_urls(html, _add_image)
  1001→        elif isinstance(content_value, str):
  1002→            _extract_img_urls(content_value, _add_image)
  1003→
  1004→    return images[:10]

  1007→def _has_image_ext(url: str) -> bool:
  1008→    """Check if URL ends with an image extension."""
  1009→    from urllib.parse import urlparse
  1010→    try:
  1011→        path = urlparse(url).path.lower().split("?")[0].split("#")[0]
  1012→        return any(path.endswith(ext) for ext in _VALID_IMAGE_EXTENSIONS)
  1013→    except Exception:
  1014→        return False

  1017→def _extract_img_urls(html: str, callback):
  1018→    """Extract src attributes from <img> tags in HTML.
  1019→
  1020→    Also checks data-src and data-lazy-src for lazy-loaded images.
  1021→    Decodes HTML entities before passing to callback.
  1022→    """
  1023→    if not html:
  1024→        return
  1025→    for attr in ["src", "data-src", "data-lazy-src"]:
  1026→        for match in re.finditer(
  1027→            rf'<img[^>]+{attr}=["\']([^"\']+)["\']', html, re.IGNORECASE
  1028→        ):
  1029→            url = match.group(1)
  1030→            if url and len(url) >= _MIN_IMAGE_URL_LEN:
  1031→                callback(url)

  1036→def _clean_html(text: str) -> str:
  1037→    """Remove HTML tags and decode entities."""
  1038→    text = re.sub(r'<[^>]+>', '', text)
  1039→    text = text.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>')
  1040→    text = text.replace('&quot;', '"').replace('&#39;', "'").replace('&nbsp;', ' ')
  1041→    text = re.sub(r'\s+', ' ', text).strip()
  1042→    return text
